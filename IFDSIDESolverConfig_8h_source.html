30enum class SolverConfigOptions : uint32_t {

42/// \brief Configuration options for the solving process of IFDS/IDE problems
43class IFDSIDESolverConfig {
44public:
45 constexpr IFDSIDESolverConfig() noexcept = default;
46 constexpr IFDSIDESolverConfig(SolverConfigOptions Options) noexcept
47 : Options(Options) {}
48
49 /// Returns whether the solver should handle unbalanced returns (default:
50 /// false)
51 [[nodiscard]] constexpr bool followReturnsPastSeeds() const noexcept {
52 return hasFlag(Options, SolverConfigOptions::FollowReturnsPastSeeds);
53 }
54 /// Returns whether the solver should automatically insert an identityFlow
55 /// propagation for the special zero value (default: true)
56 [[nodiscard]] constexpr bool autoAddZero() const noexcept {
57 return hasFlag(Options, SolverConfigOptions::AutoAddZero);
58 }
59 /// Returns whether the IDE solver should perform IDE's phase 2 (default:
60 /// true). You may want to turn this off for IFDS analyses.
61 [[nodiscard]] constexpr bool computeValues() const noexcept {
62 return hasFlag(Options, SolverConfigOptions::ComputeValues);
63 }
64 /// Returns, whether the solver should record all ESG edges (default: false)
65 /// \note This option may severly hurt the solver's performance
66 [[nodiscard]] constexpr bool recordEdges() const noexcept {
67 return hasFlag(Options, SolverConfigOptions::RecordEdges);
68 }
69 /// Returns, whether the solver should emit the ESG as DOT graph on the
70 /// command-line (default: false)
71 [[nodiscard]] constexpr bool emitESG() const noexcept {
72 return hasFlag(Options, SolverConfigOptions::EmitESG);
73 }
74 /// Currently unused
75 [[nodiscard]] constexpr bool computePersistedSummaries() const noexcept {
76 return hasFlag(Options, SolverConfigOptions::ComputePersistedSummaries);
77 }
78
79 /// \see followReturnsPastSeeds
80 constexpr void setFollowReturnsPastSeeds(bool Set = true) {
81 setFlag(Options, SolverConfigOptions::FollowReturnsPastSeeds, Set);
82 }
83 /// \see autoAddZero
84 constexpr void setAutoAddZero(bool Set = true) {
85 setFlag(Options, SolverConfigOptions::AutoAddZero, Set);
86 }
87 /// \see computeValues
88 constexpr void setComputeValues(bool Set = true) {
89 setFlag(Options, SolverConfigOptions::ComputeValues, Set);
90 }
91 /// \see recordEdges
92 constexpr void setRecordEdges(bool Set = true) {
93 setFlag(Options, SolverConfigOptions::RecordEdges, Set);
94 }
95 /// \see emitESG
96 constexpr void setEmitESG(bool Set = true) {
97 setFlag(Options, SolverConfigOptions::EmitESG, Set);
98 }
99 /// \see computePersistedSummaries
100 constexpr void setComputePersistedSummaries(bool Set = true) {
101 setFlag(Options, SolverConfigOptions::ComputePersistedSummaries, Set);
102 }
103
104 constexpr void setConfig(SolverConfigOptions Opt) { Options = Opt; }
105
106 friend llvm::raw_ostream &operator<<(llvm::raw_ostream &OS,
107 IFDSIDESolverConfig SC);
108
109private:
110 SolverConfigOptions Options =
111 SolverConfigOptions::AutoAddZero | SolverConfigOptions::ComputeValues;
112};
