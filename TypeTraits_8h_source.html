301/// True if T can be relocated by copying its bytes (e.g. via memcpy) without
302/// invoking the move constructor or destructor on the source. Uses Clang's
303/// builtin trait when available (P1144 / P2786), which covers types like
304/// std::unique_ptr that are trivially relocatable but not trivially copyable.
305/// Falls back to std::is_trivially_copyable_v on other compilers.
306template <typename T>
307inline constexpr bool IsTriviallyRelocatable =
308#if defined(__has_builtin) && \
309 __has_builtin(__builtin_is_cpp_trivially_relocatable)
310 __builtin_is_cpp_trivially_relocatable(T);
311#elif defined(__has_builtin) && __has_builtin(__is_trivially_relocatable)
312 __is_trivially_relocatable(T);
313#else
314 std::is_trivially_copyable_v<T>;
315#endif
