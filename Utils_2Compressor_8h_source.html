30/// \brief A utility class that assigns a sequential Id to every inserted
31/// object.
32///
33/// This specialization handles types that can be efficiently passed by value
34template <typename T, IdType IdT>
35 requires(CanEfficientlyPassByValue<T> && has_llvm_dense_map_info<T>)
36class Compressor<T, IdT> {
37public:
38 void reserve(size_t Capacity) {
39 assert(Capacity <= UINT32_MAX);
40 ToInt.reserve(Capacity);
41 FromInt.reserve(Capacity);
42 }
43
44 IdT getOrInsert(T Elem) {
45 auto [It, Inserted] = ToInt.try_emplace(Elem, IdT(FromInt.size()));
46 if (Inserted) {
47 FromInt.push_back(Elem);
48 }
49 return It->second;
50 }
51
52 std::pair<IdT, bool> insert(T Elem) {
53 auto [It, Inserted] = ToInt.try_emplace(Elem, IdT(FromInt.size()));
54 if (Inserted) {
55 FromInt.push_back(Elem);
56 }
57 return {It->second, Inserted};
58 }
59
60 IdT insertDummy(T Elem) {
61 auto Ret = IdT(FromInt.size());
62 FromInt.push_back(Elem);
63 return Ret;
64 }
65
66 [[nodiscard]]
67 std::optional<IdT> getOrNull(T Elem) const {
68 if (auto It = ToInt.find(Elem); It != ToInt.end()) {
69 return It->second;
70 }
71 return std::nullopt;
72 }
73
74 [[nodiscard]] IdT get(T Elem) const {
75 auto It = ToInt.find(Elem);
76 assert(It != ToInt.end());
77 return It->second;
78 }
79
80 [[nodiscard]] bool inbounds(IdT Idx) const noexcept {
81 return FromInt.inbounds(Idx);
82 }
83
84 [[nodiscard]] T operator[](IdT Idx) const noexcept {
85 assert(inbounds(Idx));
86 return FromInt[Idx];
87 }
88
89 [[nodiscard]] size_t size() const noexcept { return FromInt.size(); }
90 [[nodiscard]] size_t capacity() const noexcept {
91 return FromInt.capacity() +
92 ToInt.getMemorySize() / sizeof(typename decltype(ToInt)::value_type);
93 }
94
95 [[nodiscard]] auto begin() const noexcept { return FromInt.begin(); }
96 [[nodiscard]] auto end() const noexcept { return FromInt.end(); }
97
98 [[nodiscard]] auto enumerate() const noexcept { return FromInt.enumerate(); }
99
100 void clear() noexcept {
101 ToInt.clear();
102 FromInt.clear();
103 }
104
105private:
106 llvm::DenseMap<T, IdT> ToInt;
107 TypedVector<IdT, T, 0> FromInt;
108};

110/// \brief A utility class that assigns a sequential Id to every inserted
111/// object.
112///
113/// This specialization handles types that cannot be efficiently passed by value
114template <typename T, IdType IdT>
115 requires(!CanEfficientlyPassByValue<T> || !has_llvm_dense_map_info<T>)
116class Compressor<T, IdT> {
117public:
118 void reserve(size_t Capacity) {
119 assert(Capacity <= UINT32_MAX);
120 ToInt.reserve(Capacity);
121 }
122
123 /// Returns the index of the given element in the compressors storage. If
124 /// the element isn't present yet, it will be added first and its index will
125 /// then be returned.
126 IdT getOrInsert(const T &Elem) {
127 if (auto It = ToInt.find(&Elem); It != ToInt.end()) {
128 return It->second;
129 }
130 auto Ret = IdT(FromInt.size());
131 auto *Ins = &FromInt.emplace_back(Elem);
132 ToInt[Ins] = Ret;
133 return Ret;
134 }
135
136 /// Returns the index of the given element in the compressors storage. If
137 /// the element isn't present yet, it will be added first and its index will
138 /// then be returned.
139 IdT getOrInsert(T &&Elem) {
140 if (auto It = ToInt.find(&Elem); It != ToInt.end()) {
141 return It->second;
142 }
143 auto Ret = IdT(FromInt.size());
144 auto *Ins = &FromInt.emplace_back(std::move(Elem));
145 ToInt[Ins] = Ret;
146 return Ret;
147 }
148
149 std::pair<IdT, bool> insert(const T &Elem) {
150 if (auto It = ToInt.find(&Elem); It != ToInt.end()) {
151 return {It->second, false};
152 }
153 auto Ret = Id(FromInt.size());
154 auto *Ins = &FromInt.emplace_back(Elem);
155 ToInt[Ins] = Ret;
156 return {Ret, true};
157 }
158
159 std::pair<IdT, bool> insert(T &&Elem) {
160 if (auto It = ToInt.find(&Elem); It != ToInt.end()) {
161 return {It->second, false};
162 }
163 auto Ret = Id(FromInt.size());
164 auto *Ins = &FromInt.emplace_back(std::move(Elem));
165 ToInt[Ins] = Ret;
166 return {Ret, true};
167 }
168
169 IdT insertDummy(std::convertible_to<T> auto &&Elem) {
170 auto Ret = Id(FromInt.size());
171 FromInt.emplace_back(PSR_FWD(Elem));
172 return Ret;
173 }
174
175 /// Returns the index of the given element in the compressors storage. If
176 /// the element isn't present, std::nullopt will be returned
177 [[nodiscard]] std::optional<IdT> getOrNull(const T &Elem) const {
178 if (auto It = ToInt.find(&Elem); It != ToInt.end()) {
179 return It->second;
180 }
181 return std::nullopt;
182 }
183
184 [[nodiscard]] IdT get(const T &Elem) const {
185 auto It = ToInt.find(&Elem);
186 assert(It != ToInt.end());
187 return It->second;
188 }
189
190 [[nodiscard]] bool inbounds(IdT Idx) const noexcept {
191 return size_t(Idx) < FromInt.size();
192 }
193
194 [[nodiscard]] const T &operator[](IdT Idx) const noexcept {
195 assert(inbounds(Idx));
196 return FromInt[size_t(Idx)];
197 }
198
199 [[nodiscard]] size_t size() const noexcept { return FromInt.size(); }
200 [[nodiscard]] size_t capacity() const noexcept {
201 return FromInt.size() +
202 ToInt.getMemorySize() / sizeof(typename decltype(ToInt)::value_type);
203 }
204
205 auto begin() const noexcept { return FromInt.begin(); }
206 auto end() const noexcept { return FromInt.end(); }
207
208 [[nodiscard]] auto enumerate() const noexcept {
209 return llvm::map_range(llvm::enumerate(FromInt),
210 [](const auto &IndexAndVal) {
211 return std::pair<IdT, const T &>{
212 IdT(IndexAndVal.index()), IndexAndVal.value()};
213 });
214 }
215
216 void clear() noexcept {
217 ToInt.clear();
218 FromInt.clear();
219 }
220
221private:
222 struct DSI : llvm::DenseMapInfo<const T *> {
223 static auto getHashValue(const T *Elem) noexcept {
224 assert(Elem != nullptr);
225 if constexpr (has_llvm_dense_map_info<T>) {
226 return llvm::DenseMapInfo<T>::getHashValue(*Elem);
227 } else {
228 return std::hash<T>{}(*Elem);
229 }
230 }
231 static auto isEqual(const T *LHS, const T *RHS) noexcept {
232 if (LHS == RHS) {
233 return true;
234 }
235 if (LHS == DSI::getEmptyKey() || LHS == DSI::getTombstoneKey() ||
236 RHS == DSI::getEmptyKey() || RHS == DSI::getTombstoneKey()) {
237 return false;
238 }
239 if constexpr (has_llvm_dense_map_info<T>) {
240 return llvm::DenseMapInfo<T>::isEqual(*LHS, *RHS);
241 } else {
242 return *LHS == *RHS;
243 }
244 }
245 };
246
247 std::deque<T> FromInt;
248 llvm::DenseMap<const T *, IdT, DSI> ToInt;
249};
