168 const SourceFactSet &getSourceFactsFor(auto &LocalProblem,
169 const DataFlowEnvironment<d_t> &In,
170 const FunctionSummary &CalleeSum,
171 SourceFactId CalleeSrc,
172 ByConstRef<n_t> CallInst) {
173 auto &Ret = Mapping[CalleeSrc];
174
175 if (ComputedMappings.tryInsert(CalleeSrc)) {
176 auto &&CSFacts = LocalProblem.invCallFlow(
177 CallInst, CalleeSum.SourceFactIds[CalleeSrc]);
178
179 for (const auto *Fact : CSFacts) {
180 if (const auto *FactSrc = getOrNull(In, Fact)) {
181 Ret.insertAllOf(*FactSrc);
182 }
183 }
184 }
185
186 return Ret;
187 }

257 void computeFixpointForSCC(SCCId<FunctionId> CurrSCC,
258 llvm::ArrayRef<FunctionId> CurrFuns) {
259
260 const size_t SCCSize = CurrFuns.size();
261 const bool InRecursion = SCCSize > 1;
262 IntermediateState IState(Problem, &PoolRes, CurrSCC, InRecursion);
263
264 const auto IterStrategy = Config.IterStrategy;
265 const bool UseTopoFixpointDriver = [=] {
266 if (IterStrategy == IterationStrategy::DedupFIFOQueue) {
267 return false;
268 }
269
270 if (IterStrategy == IterationStrategy::Hybrid) {
271 // return SCCSize < 20;
272 return SCCSize == 1;
273 }
274
275 return true;
276 }();
277
278 ControlFlowOrder<n_t> CFO;
279 if (UseTopoFixpointDriver) {
280 for (const auto &Fun : CurrFuns) {
281 computeCFGOrder(CFO, *ICF, (*Functions)[Fun]);
282 }
283 }
284
285 ArraySetDriver<n_t> DefaultDriver;
286 TopoFixpointDriver<n_t> TopoDriver;
287
288 const auto ComputeFixpointWithDriver =
289 [&](auto &Driver) LLVM_ATTRIBUTE_NOINLINE {
290 for (auto FunId : llvm::reverse(CurrFuns)) {
291 const auto *Fun = (*Functions)[FunId];
292 submitInitialSeeds(IState, Driver, Summaries[FunId].SourceFactIds,
293 Fun);
294 }
295 Driver.run([&](n_t BlockStart) {
296 analyzeBlock(IState, Driver, BlockStart);
297 });
298 assert(Driver.empty());
299
300 // llvm::errs() << '.';
301 };
302
303 const auto RepropagateInRecursion = [&](auto &Driver) {
304 rescheduleCalls(IState, Driver);
305 while (!Driver.empty()) {
306 Driver.run(
307 [&](n_t BlockStart) { analyzeBlock(IState, Driver, BlockStart); });
308 assert(Driver.empty());
309
310 rescheduleCalls(IState, Driver);
311 // llvm::errs() << '.';
312 }
313
314 assert(IState.HasNewSummary.empty() &&
315 "After repropagating, we should not have any summary "
316 "applications pending");
317 };
318 if (UseTopoFixpointDriver) {
319 ComputeFixpointWithDriver(TopoDriver);
320 } else {
321 ComputeFixpointWithDriver(DefaultDriver);
322 }
323
324 if (!Config.EagerReturnPropagation) {
325 if (IterStrategy == IterationStrategy::TopoPrioQueue) {
326 RepropagateInRecursion(TopoDriver);
327 } else {
328 RepropagateInRecursion(DefaultDriver);
329 }
330 }
331
332 repropagateLeaks(IState, CurrSCC);
333 }

358 void rescheduleCalls(IntermediateState &IState, auto &Driver) {
359 if (!IState.InRecursion) {
360 return;
361 }
362
363 const bool EnableEnvVersioning = Config.EnableEnvVersioning;
364
365 for (auto FunId : IState.HasNewSummary) {
366 IState.HasNewLeaks.erase(FunId);
367 const auto &Fun = (*Functions)[FunId];

371 if (auto CallerId = Functions->getOrNull(CSFun)) {

395 /// Continuation of RepropagateLeaks procedure in Algorithm 2
396 void handleLeaksForFun(IntermediateState &IState, SCCId<FunctionId> CurrSCC,
397 FunctionId CurrFun) {
398 const auto *Fun = (*Functions)[CurrFun];
399 const auto &Sum = Summaries[CurrFun];
400
401 Mapper M(Sum.SourceFactIds.size());
402
403 for (const auto &CS : ICF->getCallersOf(Fun)) {
404 auto CallerId = Functions->getOrNull(CS->getFunction());
405 if (!CallerId) {
406 continue;
407 }
408
409 auto CallerSCC = SCCs->SCCOfNode[*CallerId];
410 if (CallerSCC != CurrSCC) {
411 continue;
412 }

414 M.reset();
415

417
418 for (const auto &[CalleeLeak, LeakSrc] : Sum.LeakIf) {
419 const auto &CSSrc =

421 reportOrPropagateLeak(IState, *CallerId, CalleeLeak.first,

441 /// Procedure AnalyzeBlock (Lines 8-11+14 in Algorithm 4)
442 void analyzeBlockImpl(IntermediateState &IState, auto &Driver,
443 ByConstRef<n_t> BlockStart,
444 DataFlowEnvironment<d_t> LocalState) {
445
446 auto CurrFunId = Functions->get(BlockStart->getFunction());
447
448 Nullable<n_t> CurrInst = BlockStart;
449
450 do {
451 auto Last = unwrapNullable(CurrInst);
452
453 do {
454 auto Curr = unwrapNullable(CurrInst);
455 analyzeInstruction(IState, LocalState, CurrFunId, Curr);
456 Last = Curr;
457 if constexpr (IsBlockAwareControlFlow<i_t>) {
458 CurrInst = ICF->getUniqueSuccessor(Curr);
459 } else {
460 const auto &Succs = ICF->getSuccsOf(Curr);
461 if (Succs.size() == 1) {
462 CurrInst = Succs[0];
463 } else {
464 CurrInst = {};
465 }
466 }
467 } while (CurrInst);
468
469 Nullable<n_t> UniqueSucc{};
470
471 // We have at least one instruction, so we can safely unwrap here
472 const auto &Succs = ICF->getSuccsOf(Last);
473 const auto SuccSz = Succs.size();
474 const bool HasSingleSucc = SuccSz == 1;
475 for (const auto &Succ : Succs) {
476 bool HasSinglePred = [&]() {
477 if constexpr (IsBlockAwareControlFlow<i_t>) {
478 return ICF->hasUniquePredecessor(Succ);
479 }
480 return false;
481 }();
482
483 auto [SuccBBStateIt, Inserted] = IState.PathEdges.try_emplace(
484 Succ, lazy{[&] {
485 if (HasSingleSucc && !(HasSinglePred && !UniqueSucc)) {
486 return std::move(LocalState);
487 }
488
489 return LocalState;
490 }});
491
492 if (HasSinglePred) {
493 // Assign
494
495 if (Inserted || SuccBBStateIt->second != LocalState) {
496 if (!UniqueSucc) {
497 UniqueSucc = Succ;
498 if (!Inserted) {
499 // Note: Cannot move LocalState here, as we still
500 // need it in the next iteration
501 SuccBBStateIt->second = LocalState;
502 }
503
504 } else {
505 Driver.push(Succ);
506 if (!Inserted) {
507 if (HasSingleSucc) {
508 SuccBBStateIt->second = std::move(LocalState);
509 } else {
510 SuccBBStateIt->second = LocalState;
511 }
512 }
513 }
514
515 SuccBBStateIt->second.Version++;
516 }
517 continue;
518 }
519
520 // Merge
521 if (Inserted || tryMergeStates(SuccBBStateIt->second, LocalState)) {
522 SuccBBStateIt->second.Version++;
523 Driver.push(Succ);
524 }
525 }
526
527 if (SuccSz == 0 && Config.EagerReturnPropagation &&
528 ICF->isExitInst(Last)) {
529 if (IState.HasNewSummary.erase(CurrFunId)) {
530 rescheduleCallsAtExit(IState, Driver, CurrFunId);
531 }
532 }
533
534 CurrInst = UniqueSucc;
535 } while (CurrInst);
536 }

556 /// Procedure AnalyzeExit (Lines 35-38 in Algorithm 4, Line 39 is delayed to
557 /// rescheduleCalls())
558 void analyzeExitInst(IntermediateState &IState,
559 DataFlowEnvironment<d_t> &LocalState,
560 FunctionId CurrFunId, ByConstRef<n_t> Inst) {
561 const bool InRecursion = IState.InRecursion;

563
564 auto &Sum = Summaries[CurrFunId].EndSummary;
565
566 for (auto &&[ExitFact, ExitSrc] : LocalState) {

577 It->second.insertAllOf(std::move(ExitSrc));

579 }

582 IState.HasNewSummary.insert(CurrFunId);

586 /// Procedure AnalyzeCall (Lines 21-34 in Algorithm 4)
587 void analyzeCallInst(IntermediateState &IState,
588 DataFlowEnvironment<d_t> &LocalState,
589 FunctionId CurrFunId, ByConstRef<n_t> Inst) {
590
591 const auto &Callees = ICF->getCalleesOfCallAt(Inst);
592
593 auto CSInfo = handleCallSrcSinksAndMayRecurse(IState, LocalState, Callees,
594 CurrFunId, Inst);
595
596 if (CSInfo.MayRecurse) {
597 IState.InRecursion = true;
598 IState.PathEdges[Inst] = LocalState;
599 }
600
601 DataFlowEnvironment<d_t> CollectedSummary;
602
603 for (const auto &CalleeFun : Callees) {
604 // Collect all data-flows that need to be propagated. Don't update
605 // LocalState in-place
606
607 if (IState.LocalProblem.summaryFlow(std::as_const(LocalState),
608 CollectedSummary, Inst, CalleeFun)) {
609 continue;
610 }
611
612 auto CalleeId = Functions->get(CalleeFun);
613 if (ICF->getStartPointsOf(CalleeFun).empty()) {
614 CSInfo.CanCTR = false;
615 }
616 applySummary(IState, std::as_const(LocalState), CollectedSummary,
617 CalleeId, Inst, CurrFunId);
618 }
619 if (CSInfo.CanCTR) {
620 IState.LocalProblem.callToRetFlow(LocalState, Inst);
621 }
622
623 mergeStates(LocalState, std::move(CollectedSummary));
624 }

626 /// Lines 26-32 in Algorithm 4
627 void applySummary(IntermediateState &IState,
628 const DataFlowEnvironment<d_t> &In,
629 DataFlowEnvironment<d_t> &LocalState, FunctionId CalleeId,
630 n_t Inst, FunctionId CurrFunId) {
631 const auto &Sum = Summaries[CalleeId];
632 Mapper M(Sum.SourceFactIds.size());
633
634 for (const auto &[SumFact, SumSrc] : Sum.EndSummary) {
635 auto &&RetFacts = IState.LocalProblem.returnFlow(Inst, SumFact);
636 if (RetFacts.empty()) {
637 continue;
638 }
639
640 const auto &RetSrcFacts =
641 M.getAllSrcFactsFor(IState.LocalProblem, In, Sum, SumSrc, Inst);
642 if (RetSrcFacts.empty()) {
643 continue;
644 }
645
646 for (const auto *RetFact : RetFacts) {
647 LocalState[RetFact].insertAllOf(RetSrcFacts);
648 }
649 }
650
651 if (CalleeId != CurrFunId) { // Prevent self-insertion
652 for (const auto &[CalleeLeak, LeakSrc] : Sum.LeakIf) {
653 const auto &CSSrc =
654 M.getAllSrcFactsFor(IState.LocalProblem, In, Sum, LeakSrc, Inst);
655 reportOrPropagateLeak(IState, CurrFunId, CalleeLeak.first,
656 CalleeLeak.second, CSSrc);
657 }

666 [[nodiscard]] CallSiteInfo handleCallSrcSinksAndMayRecurse(
667 IntermediateState &IState, DataFlowEnvironment<d_t> &LocalState,
668 const auto &Callees, FunctionId CurrFunId, ByConstRef<n_t> Inst) {
669
670 const auto &SCCs = *this->SCCs;
671 const auto CurrSCC = IState.CurrSCC;
672
673 bool MayRecurse = false;
674 bool CanCTR = !Callees.empty();
675 for (f_t CalleeFun : Callees) {
676 auto CalleeId = Functions->get(CalleeFun);
677 auto CalleeSCC = SCCs.SCCOfNode[CalleeId];
678 if (CalleeSCC == CurrSCC) {
679 MayRecurse = true;
680 IState.Incoming[CalleeFun].insert(Inst);
681 }
682
683 PHASAR_LOG_LEVEL_CAT(DEBUG, LogCategory,
684 "[handleCallSrcSinksAndMayRecurse]: At call to "
685 << FToString(CalleeFun));
686
687 IState.LocalProblem.requestResultCallbackAtCallSite(
688 Inst, CalleeFun, [&](ByConstRef<d_t> LeakFact) {
689 PHASAR_LOG_LEVEL_CAT(
690 DEBUG, LogCategory,
691 "[handleCallSrcSinksAndMayRecurse]: LeakFact: "
692 << DToString(LeakFact));
693 if (const auto *LeakSrc = psr::getOrNull(LocalState, LeakFact)) {
694 reportOrPropagateLeak(IState, CurrFunId, Inst, LeakFact,
695 *LeakSrc);
696 }
697 });
698
699 // Generate taints from zero:
700 IState.LocalProblem.generateFactsAtCall(

748 // PropagateLeaks procedure in Algorithm 4
749 void reportOrPropagateLeak(IntermediateState &IState, FunctionId CurrFunId,
750 n_t LeakInst, d_t LeakFact, SourceFactSet From) {
751 PHASAR_LOG_LEVEL_CAT(DEBUG, LogCategory,
752 "[reportOrPropagateLeak]: " << DToString(LeakFact)
753 << " AT "
754 << NToString(LeakInst));
755 // The zero fact has always Id 0!
756 if (From.tryErase(SourceFactId(0))) {
757 if (Leaks[LeakInst].insert(LeakFact).second) {
758 IState.LocalProblem.onResult(LeakInst, LeakFact);
759 }
760 }
761
762 auto &CurrSum = Summaries[CurrFunId];
763
764 bool New =
765 CurrSum.LeakIf[{LeakInst, LeakFact}].tryMergeWith(std::move(From));
766
767 if (New && IState.InRecursion) {
768 IState.HasNewLeaks.insert(CurrFunId);
769 }
770 }

772 static void mergeStates(DataFlowEnvironment<d_t> &Into,
773 DataFlowEnvironment<d_t> &&From) {
774 if (Into.empty()) {
775 if (&Into != &From) {
776 Into = std::move(From);
777 }
778
779 return;
780 }
781
782 if (Into.size() < From.size()) {
783 std::swap(Into, From);
784 }
785
786 for (auto &[TgtFact, SrcFactIds] : From) {
787 auto [It, Inserted] = Into.try_emplace(TgtFact, std::move(SrcFactIds));
788 if (!Inserted) {
789 It->second.insertAllOf(std::move(SrcFactIds));
790 }
791 }
792 }
793
794 [[nodiscard]] static bool
795 tryMergeStates(DataFlowEnvironment<d_t> &Into,
796 const DataFlowEnvironment<d_t> &From) {
797 // TODO Handle phis
798
799 if (Into.empty()) {
800 Into = From;
801 return !From.empty();
802 }
803
804 bool Changed = false;
805 for (const auto &[TgtFact, SrcFactIds] : From) {
806 auto [It, Inserted] = Into.try_emplace(TgtFact, SrcFactIds);
807 Changed |= Inserted || It->second.tryMergeWith(SrcFactIds);
808 }
809
810 return Changed;
811 }
