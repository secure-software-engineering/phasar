29template <wpds::RuleProvider RuleProviderT, IsSemiRing SemiRingT>
30 requires std::same_as<typename RuleProviderT::weight_type,
31 typename SemiRingT::EdgeFunctionType>
32class WPDSSolver : private wpds::detail::SolverResultsData<
33 typename RuleProviderT::control_location_type,
34 typename RuleProviderT::stack_element_type,
35 typename RuleProviderT::weight_type> {
36 using base_t = wpds::detail::SolverResultsData<
37 typename RuleProviderT::control_location_type,
38 typename RuleProviderT::stack_element_type,
39 typename RuleProviderT::weight_type>;
40
41 using base_t::ESGNodeCompressor;
42 using base_t::Incoming;
43 using base_t::PathEdges;
44 using base_t::Seeds;
45
46public:
47 using cl_t = typename RuleProviderT::control_location_type;
48 using se_t = typename RuleProviderT::stack_element_type;
49 using weight_t = typename RuleProviderT::weight_type;
50 using path_edge_t = std::tuple<wpds::ESGNodeId, se_t, cl_t>;
51
52 static constexpr llvm::StringLiteral LogCategory = "RuleBasedSolver";
53
54 explicit WPDSSolver(RuleProviderT *RP, SemiRingT *SR) noexcept
55 : RP(RP), SR(SR) {}
56 explicit WPDSSolver(NonNullPtr<RuleProviderT> RP,
57 NonNullPtr<SemiRingT> SR) noexcept

59 WPDSSolver(RuleProviderT *RP, std::nullptr_t SR) = delete;

82private:
83 void solveImpl() {
84 submitInitialSeeds();
85
86 while (!WL.empty()) {
87 auto [CurrPE, CurrWeight] = WL.pop_back_val();
88 auto &&[CurrSrc, CurrTarget, CurrTgtFact] = CurrPE;
89
90 for (const auto &[SuccCL, SuccSE, SuccWeight] :
91 RP->getNormalRules(CurrTgtFact, CurrTarget)) {
92 propagate({CurrSrc, SuccSE, SuccCL},
93 SR->extend(CurrWeight, SuccWeight));
94 }
95
96 for (const auto &[SuccFact, RetSite, EntrySE, PushWeight] :
97 RP->getPushRules(CurrTgtFact, CurrTarget)) {
98
99 if constexpr (requires() { RP->makeSrcFacts(SuccFact); }) {
100 for (const auto &RealSuccFact : RP->makeSrcFacts(SuccFact)) {
101 auto Entry = compressNode(RealSuccFact, EntrySE);
102
103 propagate({Entry, EntrySE, SuccFact}, SR->identity());
104 auto IncWeight = SR->extend(CurrWeight, PushWeight);
105 addIncoming(Entry, CurrSrc, RetSite, IncWeight);
106 applyEndSummary(Entry, CurrSrc, RetSite, IncWeight);
107 }
108 } else {
109
110 auto Entry = compressNode(SuccFact, EntrySE);
111
112 propagate({Entry, EntrySE, SuccFact}, SR->identity());
113 auto IncWeight = SR->extend(CurrWeight, PushWeight);
114 addIncoming(Entry, CurrSrc, RetSite, IncWeight);
115 applyEndSummary(Entry, CurrSrc, RetSite, IncWeight);
116 }
117 }
118
119 if constexpr (wpds::CanInjectAdditionalPushEdges<RuleProviderT>) {
120 RP->injectAdditionalPushEdges(
121 CurrTgtFact, CurrTarget,
122 [&](llvm::ArrayRef<std::tuple<cl_t, se_t, se_t>> Edges,
123 weight_t W) {
124 if (!Edges.empty()) {
125 PHASAR_LOG_LEVEL_CAT(DEBUG, LogCategory,
126 "injectAdditionalPushEdges:");
127 PHASAR_LOG_LEVEL_CAT(
128 DEBUG, LogCategory,
129 "injectAdditionalPushEdges: > For CurrTgtFact: "
130 << DToString(CurrTgtFact) << " AT "
131 << NToString(CurrTarget));
132 }
133
134 if constexpr (requires() {
135 RP->makeSrcFacts(std::get<0>(Edges.front()));
136 }) {
137 // TODO: This is extremely ugly, but it works...
138 const auto &[FirstCL, FirstEntrySE, FirstRetSite] =
139 Edges.front();
140 for (const auto &RealCL : RP->makeSrcFacts(FirstCL)) {
141 auto TopmostSrc = CurrSrc;
142
143 PHASAR_LOG_LEVEL_CAT(
144 DEBUG, LogCategory,
145 " > CL" << DToString(RealCL)
146 << "; EntrySE: " << NToString(FirstEntrySE)
147 << "; RetSite: " << NToString(FirstRetSite));
148
149 auto Entry = compressNode(RealCL, FirstEntrySE);
150
151 propagate({Entry, FirstEntrySE, FirstCL}, SR->identity());
152 auto IncWeight =
153 SR->extend(CurrWeight, std::exchange(W, SR->identity()));
154 addIncoming(Entry, TopmostSrc, FirstRetSite, IncWeight);
155 applyEndSummary(Entry, TopmostSrc, FirstRetSite, IncWeight);
156
157 TopmostSrc = Entry;
158
159 for (auto &&[CL, EntrySE, RetSite] : Edges.drop_front()) {
160
161 PHASAR_LOG_LEVEL_CAT(
162 DEBUG, LogCategory,
163 " > CL" << DToString(CL)
164 << "; EntrySE: " << NToString(EntrySE)
165 << "; RetSite: " << NToString(RetSite));
166
167 auto Entry = compressNode(CL, EntrySE);
168
169 propagate({Entry, EntrySE, CL}, SR->identity());
170 auto IncWeight = SR->extend(
171 CurrWeight, std::exchange(W, SR->identity()));
172 addIncoming(Entry, TopmostSrc, RetSite, IncWeight);
173 applyEndSummary(Entry, TopmostSrc, RetSite, IncWeight);
174
175 TopmostSrc = Entry;
176 }
177 }
178 } else {
179 auto TopmostSrc = CurrSrc;
180 for (auto &&[CL, EntrySE, RetSite] : Edges) {
181
182 PHASAR_LOG_LEVEL_CAT(
183 DEBUG, LogCategory,
184 " > CL" << DToString(CL)
185 << "; EntrySE: " << NToString(EntrySE)
186 << "; RetSite: " << NToString(RetSite));
187
188 auto Entry = compressNode(CL, EntrySE);
189
190 propagate({Entry, EntrySE, CL}, SR->identity());
191 auto IncWeight =
192 SR->extend(CurrWeight, std::exchange(W, SR->identity()));
193 addIncoming(Entry, TopmostSrc, RetSite, IncWeight);
194 applyEndSummary(Entry, TopmostSrc, RetSite, IncWeight);
195
196 TopmostSrc = Entry;
197 }
198 }
199 });
200 }
201
202 if (RP->hasPopRules(CurrTgtFact, CurrTarget)) {
203
204 if (!addEndSummary(CurrSrc, CurrTgtFact, CurrTarget, CurrWeight)) {
205 continue;
206 }
207
208 const auto &[_, CurrSource] = ESGNodeCompressor[CurrSrc];
209 for (const auto &[CSNode, IncWeight] : Incoming[CurrSrc]) {
210 auto &&[CSSrc, RetSite] = CSNode;
211 auto SumExtWeight = SR->extend(IncWeight, CurrWeight);
212 for (const auto &[PopCL, PopWeight] :
213 RP->getPopRules(CurrTgtFact, CurrTarget, RetSite, CurrSource)) {
214 PHASAR_LOG_LEVEL_CAT(DEBUG, LogCategory, "Summary Edge:");
215 propagate({CSSrc, RetSite, PopCL},
216 SR->extend(SumExtWeight, PopWeight));
217 }
218 }
219 }
220 }
221 }

223 [[nodiscard]] wpds::ESGNodeId compressNode(cl_t CL, se_t SE) {
224 auto [Id, Inserted] =
225 ESGNodeCompressor.insert({std::move(CL), std::move(SE)});
226 if (Inserted) {
227 Incoming.emplace_back();
228 EndSummary.emplace_back();
229 }
230 return Id;
231 }
232
233 void submitInitialSeeds() {
234 for (const auto &[InitCL, InitSE, InitWeight] : RP->initialSeeds()) {
235 auto Init = compressNode(InitCL, InitSE);
236 propagate({Init, InitSE, InitCL}, SR->identity());
237
238 auto [It, Inserted] = Seeds.try_emplace(Init, InitWeight);
239 if (!Inserted) {
240 It->second = SR->combine(It->second, InitWeight);
241 }
242 }
243 }
244
245 void propagate(path_edge_t PE, weight_t Weight) {
246 auto &&[PESrc, SE, CL] = PE;
247
248 auto [It, Inserted] =
249 PathEdges[std::pair{CL, SE}].try_emplace(PESrc, Weight);
250 if (!Inserted) {
251 Weight = SR->combine(It->second, Weight);
252 if (It->second != Weight) {
253 It->second = Weight;
254 Inserted = true;
255
256 PHASAR_LOG_LEVEL_CAT(DEBUG, LogCategory,
257 "[propagate]: UPDATED ("
258 << NToString(ESGNodeCompressor[PESrc].second)
259 << ", "
260 << DToString(ESGNodeCompressor[PESrc].first)
261 << ")\t-->\t(" << NToString(SE) << ", "
262 << DToString(CL) << ") w/ " << It->second);
263 }
264 } else {
265 PHASAR_LOG_LEVEL_CAT(DEBUG, LogCategory,
266 "[propagate]: NEW ("
267 << NToString(ESGNodeCompressor[PESrc].second)
268 << ", "
269 << DToString(ESGNodeCompressor[PESrc].first)
270 << ")\t-->\t(" << NToString(SE) << ", "
271 << DToString(CL) << ") w/ " << It->second);
272 }
273
274 if (Inserted) {
275 WL.emplace_back(std::move(PE), std::move(Weight));
276 }
277 }
278
279 void addIncoming(wpds::ESGNodeId PushNod, wpds::ESGNodeId CSSrc, se_t RetSite,
280 weight_t PushWeight) {
281
282 PHASAR_LOG_LEVEL_CAT(
283 DEBUG, LogCategory,
284 "[addIncoming]: PushFact: "
285 << DToString(ESGNodeCompressor[PushNod].first)
286 << ", PushSE: " << NToString(ESGNodeCompressor[PushNod].second)
287 << "; CurrSource: " << NToString(ESGNodeCompressor[CSSrc].second)
288 << ", CSSrcFact: " << DToString(ESGNodeCompressor[CSSrc].first)
289 << "; RetSite: " << NToString(RetSite));
290
291 auto [It, Inserted] = Incoming[PushNod].try_emplace(
292 std::pair{CSSrc, std::move(RetSite)}, std::move(PushWeight));
293 if (!Inserted) {
294 It->second = SR->combine(std::move(It->second), std::move(PushWeight));
295 }
296 }
297
298 bool addEndSummary(wpds::ESGNodeId Src, cl_t TgtFact, se_t TgtSE,
299 weight_t TgtWeight) {
300 auto [It, Inserted] = EndSummary[Src].try_emplace(
301 std::pair{std::move(TgtFact), std::move(TgtSE)}, std::move(TgtWeight));
302 if (!Inserted) {
303 auto Merged = SR->combine(It->second, TgtWeight);
304 if (It->second != Merged) {
305 It->second = std::move(Merged);
306 Inserted = true;
307 }
308 }
309 return Inserted;
310 }
311
312 void applyEndSummary(wpds::ESGNodeId PushNod, wpds::ESGNodeId CSSrc,
313 se_t RetSite, weight_t PushWeight) {
314 const auto &Sum = EndSummary[PushNod];
315 const auto &PushSE = ESGNodeCompressor[PushNod].second;
316 for (const auto &[SumNode, SumWeight] : Sum) {
317 auto &&[SumCL, SumSE] = SumNode;
318 auto SumExtWeight = SR->extend(PushWeight, SumWeight);
319 for (const auto &[PopCL, PopWeight] :
320 RP->getPopRules(SumCL, SumSE, RetSite, PushSE)) {
321 propagate({CSSrc, RetSite, PopCL}, SR->extend(SumExtWeight, PopWeight));
322 }
323 }
324 }

326 NonNullPtr<RuleProviderT> RP{};
327 NonNullPtr<SemiRingT> SR{};
328
329 llvm::SmallVector<std::pair<path_edge_t, weight_t>> WL;
330
331 TypedVector<wpds::ESGNodeId,
332 llvm::SmallDenseMap<std::pair<cl_t, se_t>, weight_t>>
333 EndSummary;
334};
