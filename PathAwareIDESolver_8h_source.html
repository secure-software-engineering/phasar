20template <typename AnalysisDomainTy,
21 typename Container = std::set<typename AnalysisDomainTy::d_t>,
22 ICFG ICFGTy = typename AnalysisDomainTy::i_t>
23class PathAwareIDESolver
24 : public IDESolver<AnalysisDomainTy, Container, ICFGTy> {
25 using base_t = IDESolver<AnalysisDomainTy, Container, ICFGTy>;
26
27public:
28 using domain_t = AnalysisDomainTy;
29 using n_t = typename base_t::n_t;
30 using d_t = typename base_t::d_t;
31 using i_t = typename base_t::i_t;
32 using container_type = typename base_t::container_type;
33
34 explicit PathAwareIDESolver(
35 IDETabulationProblem<domain_t, container_type> *Problem, const i_t *ICF)
36 : base_t(Problem, ICF), ESG(Problem->getZeroValue()) {
37
38 if (Problem->getIFDSIDESolverConfig().autoAddZero()) {
39 PHASAR_LOG_LEVEL(
40 WARNING,
41 "The PathAwareIDESolver is initialized with the option 'autoAddZero' "
42 "being set. This might degrade the quality of the computed paths!");
43 }
44 }
45
46 explicit PathAwareIDESolver(
47 IDETabulationProblem<domain_t, container_type> &Problem, const i_t *ICF)
48 : PathAwareIDESolver(&Problem, ICF) {}
49
50 [[nodiscard]] const ExplodedSuperGraph<domain_t> &
51 getExplicitESG() const & noexcept {
52 return ESG;
53 }
54
55 [[nodiscard]] ExplodedSuperGraph<domain_t> &&getExplicitESG() && noexcept {
56 return std::move(ESG);
57 }
58
59private:
60 void saveEdges(n_t Curr, n_t Succ, d_t CurrNode,
61 const container_type &SuccNodes, ESGEdgeKind Kind) override {
62 ESG.saveEdges(std::move(Curr), std::move(CurrNode), std::move(Succ),
63 SuccNodes, Kind);
64 }
65
66 ExplodedSuperGraph<domain_t> ESG;
67};
