23template <typename ProblemTy, typename... ArgTys>
24ProblemTy createAnalysisProblem(HelperAnalyses &HA, ArgTys &&...Args) {
25 if constexpr (std::is_constructible_v<ProblemTy, HelperAnalyses &,
26 ArgTys...>) {
27 return ProblemTy(HA, std::forward<ArgTys>(Args)...);
28 } else if constexpr (std::is_constructible_v<
29 ProblemTy, const LLVMProjectIRDB *, ArgTys...>) {
30 return ProblemTy(&HA.getProjectIRDB(), std::forward<ArgTys>(Args)...);
31 } else if constexpr (std::is_constructible_v<
32 ProblemTy, const LLVMProjectIRDB *,
33 const LLVMBasedICFG *, ArgTys...>) {
34 return ProblemTy(&HA.getProjectIRDB(), &HA.getICFG(),
35 std::forward<ArgTys>(Args)...);
36 } else if constexpr (std::is_constructible_v<ProblemTy,
37 const LLVMProjectIRDB *,
38 LLVMAliasInfoRef, ArgTys...>) {
39 return ProblemTy(&HA.getProjectIRDB(), HA.getAliasInfo(),
40 std::forward<ArgTys>(Args)...);
41 } else if constexpr (std::is_constructible_v<ProblemTy,
42 const LLVMProjectIRDB *,
43 const LLVMBasedICFG *,
44 LLVMAliasInfoRef, ArgTys...>) {
45 return ProblemTy(&HA.getProjectIRDB(), &HA.getICFG(), HA.getAliasInfo(),
46 std::forward<ArgTys>(Args)...);
47 } else if constexpr (std::is_constructible_v<
48 ProblemTy, const LLVMProjectIRDB *,
49 const DIBasedTypeHierarchy *, const LLVMBasedCFG *,
50 LLVMAliasInfoRef, ArgTys...>) {
51 return ProblemTy(&HA.getProjectIRDB(), &HA.getTypeHierarchy(), &HA.getCFG(),
52 HA.getAliasInfo(), std::forward<ArgTys>(Args)...);
53 } else if constexpr (std::is_constructible_v<
54 ProblemTy, const LLVMProjectIRDB *,
55 const DIBasedTypeHierarchy *, const LLVMBasedICFG *,
56 LLVMAliasInfoRef, ArgTys...>) {
57 return ProblemTy(&HA.getProjectIRDB(), &HA.getTypeHierarchy(),
58 &HA.getICFG(), HA.getAliasInfo(),
59 std::forward<ArgTys>(Args)...);
60 } else if constexpr (std::is_constructible_v<
61 ProblemTy, const LLVMProjectIRDB *,
62 const LLVMBasedCFG *, LLVMAliasInfoRef, ArgTys...>) {
63 return ProblemTy(&HA.getProjectIRDB(), &HA.getCFG(), HA.getAliasInfo(),
64 std::forward<ArgTys>(Args)...);
65 } else {
66 static_assert(
67 std::is_constructible_v<ProblemTy, HelperAnalyses &, ArgTys...>,
68 "Cannot construct analysis problem from HelperAnalyses");
69 }
70}
