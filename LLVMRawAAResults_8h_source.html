22/// Returns a \c ValueId handler suitable for \c RawAliasSet::foreach() that
23/// maps each alias \c ValueId back to all of its underlying \c llvm::Value*
24/// (via \p VC), forwarding non-null values to \p Callback.
25constexpr std::invocable<ValueId> auto
26llvmRawAliasHandler(const ValueCompressor<PAGVariable> &VC,
27 std::invocable<const llvm::Value *> auto Callback) {
28 return [&VC, Callback{copyOrRef(Callback)}](ValueId Alias) {
29 for (auto V : VC.id2vars(Alias)) {
30 if (const auto *LLVMVar = V.valueOrNull()) [[likely]] {
31 std::invoke(Callback, LLVMVar);
32 }
33 }
34 };
35}

37/// CRTP mixin that adds the LLVM alias-iterator interface to a class that
38/// holds a \c RawAAResult and a \c ValueCompressor<PAGVariable>.
39///
40/// Provides \c forallAliasesOf(), \c mayAlias(), and \c alias() overloads
41/// accepting both \c llvm::Value* and \c ValueId arguments. Results are
42/// reported as \c llvm::Value* via the stored \c ValueCompressor.
43///
44/// The derived class must expose a \c VC member (pointer to a
45/// \c ValueCompressor<PAGVariable>).
46///
47/// \tparam Derived The CRTP derived class.
48/// \tparam AAResT A type satisfying \c RawAAResult.
49template <typename Derived, typename AAResT>
50 requires RawAAResult<std::remove_cvref_t<AAResT>>
51// NOLINTNEXTLINE(cppcoreguidelines-pro-type-member-init)
52struct LLVMRawAliasIteratorMixin {
53 [[no_unique_address]] AAResT AARes;
54
55 using v_t = const llvm::Value *;
56 using n_t = const llvm::Instruction *;
57
58 [[nodiscard]] decltype(auto) getRawAliasSet(ValueId ValId) const {
59 return AARes.getRawAliasSet(ValId);
60 }
61
62 [[nodiscard]] const auto &base() const noexcept { return AARes; }
63
64 void
65 forallAliasesOf(ValueId VId, const auto & /*Inst*/,
66 std::invocable<const llvm::Value *> auto Callback) const {
67 const auto &RawAliases = AARes.getRawAliasSet(VId);
68 RawAliases.foreach (llvmRawAliasHandler(*self().VC, copyOrRef(Callback)));
69 }
70
71 void
72 forallAliasesOf(const llvm::Value *Ptr, const auto &Inst,
73 std::invocable<const llvm::Value *> auto Callback) const {
74 if (auto ValId = self().VC->getOrNull(Ptr)) {
75 forallAliasesOf(*ValId, Inst, copyOrRef(Callback));
76 }
77 }
78
79 [[nodiscard]] bool mayAlias(ValueId Ptr1, ValueId Ptr2) const {
80 return AARes.mayAlias(Ptr1, Ptr2);
81 }
82
83 [[nodiscard]] bool mayAlias(ValueId Ptr1, ValueId Ptr2,
84 const auto & /*AtInstruction*/) const {
85 return AARes.mayAlias(Ptr1, Ptr2);
86 }
87
88 [[nodiscard]] bool mayAlias(const llvm::Value *Ptr1,
89 const llvm::Value *Ptr2) const {
90 auto ValId1 = self().VC->getOrNull(Ptr1);
91 auto ValId2 = self().VC->getOrNull(Ptr2);
92
93 return ValId1 && ValId2 && mayAlias(*ValId1, *ValId2);
94 }
95
96 [[nodiscard]] bool mayAlias(const llvm::Value *Ptr1, const llvm::Value *Ptr2,
97 const auto & /*AtInstruction*/) const {
98 return mayAlias(Ptr1, Ptr2);
99 }
100
101 [[nodiscard]] AliasResult alias(const llvm::Value *Ptr1,
102 const llvm::Value *Ptr2,
103 const auto &AtInstruction) const {
104 auto ValId1 = self().VC->getOrNull(Ptr1);
105 auto ValId2 = self().VC->getOrNull(Ptr2);
106 if (!ValId1 || !ValId2) {
107 return AliasResult::NoAlias;
108 }
109 if (*ValId1 == *ValId2) {
110 if (Ptr1 == Ptr2) {
111 return AliasResult::MustAlias;
112 }
113 return !llvm::isa<llvm::LoadInst>(Ptr1) &&
114 !llvm::isa<llvm::LoadInst>(Ptr2)
115 ? AliasResult::MustAlias
116 : AliasResult::MayAlias;
117 }
118 return mayAlias(*ValId1, *ValId2, AtInstruction) ? AliasResult::MayAlias
119 : AliasResult::NoAlias;
120 }
121
122 [[nodiscard]] constexpr const Derived &self() const noexcept {
123 return *static_cast<const Derived *>(this);
124 }
125};

160/// CRTP mixin adding a function-local view on top of a global
161/// \c RawAAResult.
162///
163/// Extends \c LLVMRawAliasIteratorMixin with \c forallAliasesOf()
164/// overloads that accept an \c llvm::Function* or \c llvm::Instruction*
165/// context. When a non-null context is provided, alias sets are intersected
166/// with the set of variables that are visible in that function (globals plus
167/// locals defined in that function), giving a function-local result even
168/// though the underlying analysis is interprocedural.
169template <typename Derived, typename AAResT>
170class LLVMLocalRawAliasIteratorMixin
171 : public detail::LLVMLocalRawAliasIteratorBase {
172public:
173 LLVMLocalRawAliasIteratorMixin(AAResT &&AARes,
174 const ValueCompressor<PAGVariable> &VC)
175 : detail::LLVMLocalRawAliasIteratorBase(VC), AARes(PSR_FWD(AARes)) {}
176
177 [[nodiscard]] decltype(auto) getRawAliasSet(ValueId ValId) const {
178 return AARes.getRawAliasSet(ValId);
179 }
180
181 [[nodiscard]] auto getRawAliasSet(ValueId ValId,
182 const llvm::Function *Context) const {
183 auto Vars = AARes.getRawAliasSet(ValId);
184 if (Context) {
185 Vars &= getOrDefault(GlobalsOrInFun, Context);
186 }
187 return Vars;
188 }
189
190 [[nodiscard]] const auto &base() const noexcept { return AARes; }
191
192 [[nodiscard]] auto getRawAliasSet(ValueId ValId,
193 const llvm::Instruction *Context) const {
194 return getRawAliasSet(ValId, getFunction(Context));
195 }
196
197 void forallAliasesOf(ValueId VId, const llvm::Function *Context,
198 std::invocable<const llvm::Value *> auto WithAlias) {
199 const auto AliasHandler =
200 llvmRawAliasHandler(*self().VC, copyOrRef(WithAlias));
201
202 auto &&RawVars = AARes.getRawAliasSet(VId);
203 if (Context) {
204 auto Vars = PSR_FWD(RawVars);
205 Vars &= getOrDefault(GlobalsOrInFun, Context);
206 Vars.foreach (AliasHandler);
207 } else {
208 RawVars.foreach (AliasHandler);
209 }
210 }
211
212 void forallAliasesOf(const llvm::Value *Val, const llvm::Function *Context,
213 std::invocable<const llvm::Value *> auto WithAlias) {
214 if (auto ValId = self().VC->getOrNull(Val)) {
215 forallAliasesOf(*ValId, Context, copyOrRef(WithAlias));
216 }
217 }

219 void forallAliasesOf(ValueId ValId, const llvm::Instruction *AtInstruction,
220 std::invocable<const llvm::Value *> auto WithAlias) {
221 forallAliasesOf(ValId, psr::getFunction(AtInstruction),
222 copyOrRef(WithAlias));
223 }

237 mayAlias(ValueId ValId1, ValueId ValId2,
