27/// Fixed-size array indexed by a strongly-typed id \p IdT instead of
28/// \c size_t.
29///
30/// \tparam IdT Index type. Must satisfy \c IdType (fits in \c size_t).
31/// \tparam ValueT Element type.
32/// \tparam N Compile-time size.
33template <IdType IdT, typename ValueT, unsigned N>
34class TypedArray : public std::array<ValueT, N> {
35 using Base = std::array<ValueT, N>;
36
37public:
38 using std::array<ValueT, N>::array;
39
40 explicit constexpr TypedArray(
41 generate_tag_t /*unused*/,
42 std::invocable<IdT> auto
43 Gen) noexcept(std::is_nothrow_invocable_v<decltype(Gen) &, IdT>)
44 : Base([Gen = copyOrRef(Gen)]<size_t... I>(
45 std::index_sequence<I...>) -> Base {
46 return {std::invoke(Gen, std::integral_constant<IdT, IdT(I)>())...};
47 }(std::make_index_sequence<N>())) {}
48
49 [[nodiscard]] constexpr bool inbounds(IdT Id) const noexcept {
50 return size_t(Id) < N;
51 }
52
53 [[nodiscard]] constexpr const ValueT &operator[](IdT Id) const & {
54 assert(inbounds(Id));
55 return this->Base::operator[](size_t(Id));
56 }
57
58 [[nodiscard]] constexpr ValueT &operator[](IdT Id) & {
59 assert(inbounds(Id));
60 return this->Base::operator[](size_t(Id));
61 }
62
63 [[nodiscard]] constexpr ValueT operator[](IdT Id) && {
64 assert(inbounds(Id));
65 return std::move(this->Base::operator[](size_t(Id)));
66 }
67
68 [[nodiscard]] llvm::ArrayRef<ValueT> asRef() const & noexcept {
69 return {this->Base::data(), N};
70 }
71 [[nodiscard]] llvm::ArrayRef<ValueT> asRef() && noexcept = delete;
72
73 [[nodiscard]] auto enumerate() const noexcept {
74 return llvm::map_range(llvm::enumerate(*this), [](const auto &IndexAndVal) {
75 return std::pair<IdT, ByConstRef<ValueT>>{IdT(IndexAndVal.index()),
76 IndexAndVal.value()};

78 }
79 [[nodiscard]] auto enumerate() noexcept {

84 }
85};
