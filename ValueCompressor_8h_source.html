37/// A utility-class that assigns sequential integer-like ids (psr::ValueId) to
38/// every inserted value.
39///
40/// In contrast to psr::Compressor, the ValueCompressor can optionally
41/// assign the same id to multiple values, and it can even produce ids without
42/// corresponding values.
43///
44/// \tparam T The type of inserted values. Expected to be mostly small and
45/// trivial.
46template <typename T> class ValueCompressor {
47public:
48 using value_type = T;
49 using id_type = ValueId;
50 using var_aliases_type = std::conditional_t<
51 detail::IsPointerWithAtleastOneFreeLowBit<T>, llvm::TinyPtrVector<T>,
52 // Note: The inline-buffer of SmallVector is pointer-aligned, so it would
53 // be wasteful to only allow one, e.g., int16
54 llvm::SmallVector<T, std::max(size_t(1), sizeof(void *) / sizeof(T))>>;
55
56 [[nodiscard]] constexpr const auto &
57 id2vars() const noexcept PSR_LIFETIMEBOUND {
58 return Id2Vars;
59 }
60 [[nodiscard]] constexpr const auto &
61 id2vars(ValueId Id) const noexcept PSR_LIFETIMEBOUND {
62 return Id2Vars[Id];
63 }
64 [[nodiscard]] constexpr const auto &
65 var2id() const noexcept PSR_LIFETIMEBOUND {
66 return Var2Id;
67 }
68
69 /// Inserts \p Var and assigns it a fresh \c ValueId, or returns the existing
70 /// id if \p Var was already present.
71 /// \returns {id, true} on first insertion; {id, false} if already present.
72 std::pair<ValueId, bool> insert(ByConstRef<T> Var) {
73 auto [It, Inserted] = Var2Id.try_emplace(Var, ValueId(Id2Vars.size()));
74 if (Inserted) {
75 Id2Vars.emplace_back().push_back(Var);
76 }
77 return {It->second, Inserted};
78 }
79
80 /// Registers \p Var as an additional name for the existing \p Id (aliasing).
81 /// \p Id must already exist. If \p Var already maps to some id, this is a
82 /// no-op and returns \c false; otherwise the mapping is added and \c true is
83 /// returned.
84 bool addAlias(ByConstRef<T> Var, ValueId Id) {
85 assert(Id2Vars.inbounds(Id) &&
86 "Can only add an alias to an already existing Id!");
87 if (Var2Id.try_emplace(Var, Id).second) {
88 Id2Vars[Id].push_back(Var);
89 return true;
90 }
91 return false;
92 }
93
94 /// Allocates a fresh \c ValueId with no associated variable (a placeholder
95 /// or sentinel slot).
96 [[nodiscard]] ValueId addDummy() {
97 auto Ret = ValueId(Id2Vars.size());
98 Id2Vars.emplace_back();
99 return Ret;
100 }
101
102 [[nodiscard]] std::optional<ValueId> getOrNull(ByConstRef<T> Var) const {
103 auto It = Var2Id.find(Var);
104 if (It == Var2Id.end()) {
105 return std::nullopt;
106 }
107 return It->second;
108 }
109
110 void reserve(size_t Capa) {
111 Var2Id.reserve(Capa);
112 Id2Vars.reserve(Capa);
113 }
114
115 [[nodiscard]] size_t size() const noexcept { return Id2Vars.size(); }
116 [[nodiscard]] bool empty() const noexcept { return Id2Vars.empty(); }
117
118private:
119 llvm::DenseMap<T, ValueId> Var2Id;
120 TypedVector<ValueId, var_aliases_type> Id2Vars;
121};
