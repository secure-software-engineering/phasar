24// Note: Use forward edges (i.e., cs->callee), such that the SCC-order reflects
25// the bottom-up iteration order.
26template <typename N, typename F, CFGOf<N, F> C>
27 requires InstructionClassifier<C>
28[[nodiscard]] SCCHolder<FunctionId>
29computeCGSCCs(const psr::CallGraph<N, F> &CG, const C &CF,
30 const FunctionCompressor<F> &Functions) {
31
32 SCCHolder<FunctionId> Ret{};
33
34 auto NumFuns = Functions.size();
35 if (!NumFuns) {
36 return Ret;
37 }
38
39 Ret.SCCOfNode.resize(NumFuns);
40
41 llvm::SmallVector<uint32_t, 128> Disc(NumFuns, UINT32_MAX);
42 llvm::SmallVector<uint32_t, 128> Low(NumFuns, UINT32_MAX);
43 BitSet<FunctionId> OnStack(NumFuns);
44 BitSet<FunctionId> Seen(NumFuns);
45
46 llvm::SmallVector<FunctionId> Stack;
47 uint32_t Time = 0;
48
49 constexpr auto SetMin = [](uint32_t &InOut, uint32_t Other) {
50 if (Other < InOut) {
51 InOut = Other;
52 }
53 };
54
55 const auto Dfs = [&](auto &&Dfs, FunctionId CurrNode) -> void {
56 auto CurrTime = Time++;
57 Disc[size_t(CurrNode)] = CurrTime;
58 Low[size_t(CurrNode)] = CurrTime;
59 Stack.push_back(CurrNode);
60 OnStack.insert(CurrNode);
61
62 const auto &CurrFun = Functions[CurrNode];
63 for (const auto &Inst : CF.getAllInstructionsOf(CurrFun)) {
64 if (!CF.isCallSite(Inst)) {
65 continue;
66 }
67
68 for (const auto &Succ : CG.getCalleesOfCallAt(Inst)) {
69 auto SuccNode = Functions.get(Succ);
70 if (Disc[size_t(SuccNode)] == UINT32_MAX) {
71 // Tree-edge: Not seen yet --> recurse
72
73 Dfs(Dfs, SuccNode);
74 SetMin(Low[size_t(CurrNode)], Low[size_t(SuccNode)]);
75 } else if (OnStack.contains(SuccNode)) {
76 // Back-edge --> circle!
77
78 SetMin(Low[size_t(CurrNode)], Disc[size_t(SuccNode)]);
79 }
80 }
81 }
82
83 if (Low[size_t(CurrNode)] == Disc[size_t(CurrNode)]) {
84 // Found SCC
85
86 auto SCCIdx = SCCId<FunctionId>(Ret.NodesInSCC.size());
87 auto &FunsInSCC = Ret.NodesInSCC.emplace_back();
88
89 assert(!Stack.empty());
90
91 while (Stack.back() != CurrNode) {
92 auto Fun = Stack.pop_back_val();
93 Ret.SCCOfNode[Fun] = SCCIdx;
94 OnStack.erase(Fun);
95 Seen.insert(Fun);
96 FunsInSCC.push_back(Fun);
97 }
98
99 auto Fun = Stack.pop_back_val();
100 Ret.SCCOfNode[Fun] = SCCIdx;
101 OnStack.erase(Fun);
102 Seen.insert(Fun);
103 FunsInSCC.push_back(Fun);
104 }
105 };
106
107 for (auto FunId : iota<FunctionId>(NumFuns)) {
108 if (!Seen.contains(FunId)) {
109 Dfs(Dfs, FunId);
110 }
111 }
112
113 return Ret;
114}

124template <typename N, typename F>
125[[nodiscard]] SCCDependencyGraph<FunctionId>
126computeCGSCCCallers(const psr::CallGraph<N, F> &CG, const CFGOf<N, F> auto &CF,
127 const FunctionCompressor<F> &Functions,
128 const SCCHolder<FunctionId> &SCCs) {
129 SCCDependencyGraph<FunctionId> Ret;
130 Ret.ChildrenOfSCC.resize(SCCs.size());
131
132 BitSet<SCCId<FunctionId>> Leaves(SCCs.size(), true);
133
134 for (auto [FunId, Fun] : Functions.enumerate()) {
135 auto SCC = SCCs.SCCOfNode[FunId];
136
137 for (const auto &CS : CG.getCallersOf(Fun)) {
138 const auto &CSFun = CF.getFunctionOf(CS);
139 if (auto CSFunId = Functions.getOrNull(CSFun)) {
140 auto CSFunSCC = SCCs.SCCOfNode[*CSFunId];
141 if (CSFunSCC != SCC) {
142 Ret.ChildrenOfSCC[SCC].insert(CSFunSCC);
143 Leaves.erase(CSFunSCC);
144 }
145 }
146 }
147 }
148
149 Ret.SCCRoots.reserve(Leaves.size());
150 Leaves.foreach (
151 [&](auto Leaf) { Ret.SCCRoots.push_back(SCCId<FunctionId>(Leaf)); });
152
153 return Ret;
154}
