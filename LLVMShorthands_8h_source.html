294/// Approximates, whether the given LLVM type may not contain a pointer.
295/// This check is designed to be extremely lightweight and is therefore not very
296/// precise.
297///
298/// \returns True, iff it can be proven that Ty does *not* contain a pointer
299[[nodiscard]] inline bool
300definitelyContainsNoPointer(const llvm::Type *Ty) noexcept {
301 if (definitelyContainsNoPointerFast(Ty)) {
302 return true;
303 }
304
305 return detail::definitelyContainsNoPointerImpl(Ty);
306}
307/// Approximates, whether the given LLVM value may not contain a pointer.
308/// This check is designed to be extremely lightweight and is therefore not very
309/// precise.
310///
311/// \returns True, iff it can be proven that Val does *not* contain a pointer
312[[nodiscard]] inline bool definitelyContainsNoPointer(const llvm::Value *Val) {
313 return llvm::isa<llvm::ConstantData>(Val) ||
314 definitelyContainsNoPointer(Val->getType());
315}

317/// Strips pointer-cast and alias wrappers from \p V, then invokes \p Handler
318/// for each concrete underlying value:
319/// - If \p V is not a ConstantExpr after stripping, Handler is called once
320/// with the stripped value.
321/// - If \p V is a ConstantExpr, the expression tree is walked and Handler
322/// is called for each GlobalObject leaf.
323template <std::invocable<const llvm::Value *> HandlerT>
324void forEachPointerOperand(const llvm::Value *V, HandlerT Handler) {
325 V = V->stripPointerCastsAndAliases();
326 const auto *CExpr = llvm::dyn_cast<llvm::ConstantExpr>(V);
327 if (!CExpr) [[likely]] {
328 std::invoke(Handler, V);
329 return;
330 }
331
332 llvm::SmallPtrSet<const llvm::Value *, 4> Seen = {V};
333 llvm::SmallVector<const llvm::User *> WL = {CExpr};
334 do {
335 const auto *Curr = WL.pop_back_val();
336 for (const auto *Op : Curr->operand_values()) {
337 if (definitelyContainsNoPointer(Op) || !Seen.insert(Op).second) {
338 continue;
339 }
340 if (const auto *GObj = llvm::dyn_cast<llvm::GlobalObject>(Op)) {
341 std::invoke(Handler, static_cast<const llvm::Value *>(GObj));
342 continue;
343 }
344 if (const auto *OpUser = llvm::dyn_cast<llvm::User>(Op)) {
345 WL.push_back(OpUser);
346 }
347 }
348 } while (!WL.empty());
349}

373inline const llvm::Function *getFunction(const llvm::Value *V) {
374 if (!V) {
375 return nullptr;
376 }
377 if (const auto *Inst = llvm::dyn_cast<llvm::Instruction>(V)) {
