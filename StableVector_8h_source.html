30/// A special-purpose container designed for providing the almost complete
31/// interface from std::vector, but with the guarantee that references to stored
32/// elements are not invalidated by insertion or deletion of other elements.
33///
34/// This container can be used as memory owner for objects of any size
35/// providing the allocator properties of std::pmr::monotonoc_buffer_resource
36/// but with the convenience of a standard container.
37///
38/// NOTE: This container performs slightly to very much better than std::deque
39/// depending on the task.
40template <typename T, typename Allocator = std::allocator<T>>
41class StableVector {
42
43 constexpr static size_t InitialLogCapacity = 5;
44 constexpr static size_t InitialCapacity = size_t(1) << InitialLogCapacity;

55 Iterator &operator++() noexcept {
56 if (LLVM_LIKELY(++It != ItEnd)) {
57/// Let the compiler optimize away the operator== call entirely
58#if __has_builtin(__builtin_assume)
59 __builtin_assume(It != nullptr);
60#endif
61 return *this;
62 }
63
64 if (Outer == OuterEnd - 1) {
65 // We are at the end of the tail loop
66 It = nullptr;
67 return *this;
68 }
69 if (++Outer == OuterEnd - 1) {
70 // We are at the end of the main loop => enter the tail loop now
71 It = *Outer;
72 ItEnd = Pos;
73#if __has_builtin(__builtin_assume)
74 __builtin_assume(It != nullptr);
75#endif
76 return *this;
77 }
78 // We are still in the main loop
79
80 It = *Outer;
81 ItEnd = It + Total;
82
83 Total <<= 1;
84#if __has_builtin(__builtin_assume)
85 __builtin_assume(It != nullptr);
86#endif
87 return *this;
88 }

92 ++*this;
93 return Ret;

128 Iterator(T *const *Outer, T *const *OuterEnd, T *Pos) noexcept
129 : Outer(Outer), OuterEnd(OuterEnd), Pos(Pos) {
130 if (OuterEnd == Outer) {
131 return;
132 }
133
134 It = *Outer;
135 if (OuterEnd - 1 == Outer) {
136 ItEnd = Pos;
137 } else {
138 ItEnd = It + InitialCapacity;
139 }
140 }

143 T *ItEnd = nullptr;
144 T *const *Outer = nullptr;
145 T *const *OuterEnd = nullptr;

176 explicit StableVector(const StableVector &Other)
177 : Size(Other.Size), BlockIdx(Other.BlockIdx),
178 Alloc(std::allocator_traits<allocator_type>::
179 select_on_container_copy_construction(Other.Alloc)) {
180 if (Other.empty()) {
181 return;
182 }
183 Blocks.reserve(BlockIdx + 1);
184 auto Cap = InitialCapacity;
185 auto Total = InitialCapacity;
186
187 for (size_t I = 0; I < BlockIdx; ++I) {
188 auto Blck =
189 (T *)std::allocator_traits<allocator_type>::allocate(Alloc, Cap);
190
191 std::uninitialized_copy_n(Other.Blocks[I], Cap, Blck);
192 Blocks.push_back(Blck);
193
194 Cap = Total;
195 Total <<= 1;
196 }
197
198 auto Blck =
199 (T *)std::allocator_traits<allocator_type>::allocate(Alloc, Cap);
200 std::uninitialized_copy(Other.Start, Other.Pos, Blck);
201 Blocks.push_back(Blck);
202
203 Start = Blck;
204 End = Blck + Cap;
205 Pos = Blck + (Other.Pos - Other.Start);
206
207 __asan_poison_memory_region(Pos, (End - Pos) * sizeof(T));
208 }

210 void swap(StableVector &Other) noexcept {
211 std::swap(Blocks, Other.Blocks);
212 std::swap(Start, Other.Start);
213 std::swap(Pos, Other.Pos);
214 std::swap(End, Other.End);
215 std::swap(Size, Other.Size);
216 std::swap(BlockIdx, Other.BlockIdx);
217
218 if constexpr (std::allocator_traits<
219 allocator_type>::propagate_on_container_swap::value) {
220 std::swap(Alloc, Other.Alloc);
221 } else {
222 assert(Alloc == Other.Alloc &&
223 "Do not swap two StableVectors with incompatible "
224 "allocators that do not propagate on swap!");
225 }
226 }

239 ~StableVector() {
240 auto Cap = InitialCapacity;
241 auto TotalSize = Cap;
242
243 for (size_t I = 0; I < BlockIdx; ++I) {
244 std::destroy_n(Blocks[I], Cap);
245
246 std::allocator_traits<allocator_type>::deallocate(Alloc, Blocks[I], Cap);
247
248 Cap = TotalSize;
249 TotalSize <<= 1;
250 }
251
252 std::destroy(Start, Pos);
253
254 for (size_t I = BlockIdx; I < Blocks.size(); ++I) {
255 __asan_unpoison_memory_region(Blocks[I], Cap * sizeof(T));
256 std::allocator_traits<allocator_type>::deallocate(Alloc, Blocks[I], Cap);
257
258 Cap = TotalSize;
259 TotalSize <<= 1;
260 }
261
262 Blocks.clear();
263 }

269 template <typename... ArgTys> T &emplace_back(ArgTys &&...Args) {
270 if (Pos == End) {
271 return growAndEmplace(std::forward<ArgTys>(Args)...);
272 }
273
274 auto Ret = Pos;
275 __asan_unpoison_memory_region(Ret, sizeof(T));
276 std::allocator_traits<allocator_type>::construct(
277 Alloc, Ret, std::forward<ArgTys>(Args)...);
278 ++Pos;
279 ++Size;
280 return *Ret;
281 }

322 [[nodiscard]] size_t max_size() const noexcept {
323 /// In theory, we could allocate as many blocks as necessary, such that the
324 /// accumulated size of them is exactly SIZE_MAX+1, but this will already
325 /// overflow the Size field and we still need to store the blocks and the
326 /// other metadata somewhere, so we will never be able to allocate
327 /// the last block (with size SIZE_MAX/2). Hence, the maximum number of
328 /// elements will be SIZE_MAX/2;
329 return (SIZE_MAX / 2) / sizeof(T);
330 };

360 void pop_back() noexcept {
361 assert(!empty() && "Do not call pop_back() on an empty StableVector!");
362
363 std::destroy_at(--Pos);
364 __asan_poison_memory_region(Pos, sizeof(T));
365
366 --Size;
367 if (Pos != Start) {
368 return;
369 }
370
371 if (BlockIdx) {
372 if (--BlockIdx) {
373 auto BlockSize = size_t(End - Start);
374 assert(llvm::isPowerOf2_64(BlockSize));
375 assert(BlockSize == Size);
376 Start = Blocks[BlockIdx];
377 End = Pos = Start + BlockSize / 2;
378 } else {
379 Start = Blocks[0];
380 End = Pos = Start + InitialCapacity;
381 }
382 }
383 }

385 [[nodiscard]] T pop_back_val() noexcept {
386 auto Ret = std::move(back());
387 pop_back();
388 return Ret;
389 }

391 void clear() noexcept {
392 auto Cap = InitialCapacity;
393 auto TotalSize = Cap;
394
395 for (size_t I = 0; I < BlockIdx; ++I) {
396 std::destroy_n(Blocks[I], Cap);
397 __asan_poison_memory_region(Blocks[I], Cap * sizeof(T));
398 Cap = TotalSize;
399 TotalSize += Cap;
400 }
401
402 std::destroy(Start, Pos);
403 __asan_poison_memory_region(Start, (Pos - Start) * sizeof(T));
404 BlockIdx = 0;
405 Size = 0;
406 if (!Blocks.empty()) {
407 Start = Pos = Blocks.front();
408 End = Start + InitialCapacity;
409 }
410 }

412 void pop_back_n(size_t N) noexcept {
413 assert(Size >= N && "Do not call pop_back() on an empty StableVector!");
414 if (!N) {
415 return;
416 }
417
418 do {
419 auto NumElementsInCurrBlock = size_t(Pos - Start);
420 if (NumElementsInCurrBlock > N) {
421 Pos -= N;
422 Size -= N;
423 std::destroy_n(Pos, N);
424 __asan_poison_memory_region(Pos, N * sizeof(T));
425 return;
426 }
427
428 std::destroy(Start, Pos);
429 __asan_poison_memory_region(Start, (Pos - Start) * sizeof(T));
430 Size -= NumElementsInCurrBlock;
431 N -= NumElementsInCurrBlock;
432
433 if (BlockIdx) {
434 if (--BlockIdx) {
435 auto BlockSize = size_t(End - Start);
436 Start = Blocks[BlockIdx];
437 Pos = End = Start + BlockSize / 2;
438 } else {
439 Start = Blocks.front();
440 Pos = End = Start + InitialCapacity;
441 }
442 } else {
443 Start = Pos = Blocks.front();
444 End = Start + InitialCapacity;
445 }
446 } while (N);
447 }

449 void shrink_to_fit() noexcept {
450 if (Blocks.empty() || BlockIdx == Blocks.size() - 1) {
451 return;
452 }
453
454 if (Size == 0) {
455 assert(BlockIdx == 0);
456 __asan_unpoison_memory_region(Blocks[0], InitialCapacity * sizeof(T));
457 std::allocator_traits<allocator_type>::deallocate(Alloc, Blocks[0],
458 InitialCapacity);
459 }
460
461 auto Cap = size_t(1) << (BlockIdx + InitialLogCapacity);
462
463 for (size_t I = BlockIdx + 1, BlocksEnd = Blocks.size(); I < BlocksEnd;
464 ++I) {
465 __asan_unpoison_memory_region(Blocks[I], Cap * sizeof(T));
466 std::allocator_traits<allocator_type>::deallocate(Alloc, Blocks[I], Cap);
467 Cap <<= 1;
468 }
469
470 Blocks.resize(BlockIdx + (Size != 0));
471 }

473 [[nodiscard]] friend bool operator==(const StableVector &LHS,
474 const StableVector &RHS) noexcept {
475 if (LHS.size() != RHS.size()) {
476 return false;
477 }
478 if (LHS.empty()) {
479 return true;
480 }
481
482 size_t Cap = InitialCapacity;
483 size_t Total = InitialCapacity;
484
485 for (size_t I = 0; I < LHS.BlockIdx; ++I) {
486 auto LIt = LHS.Blocks[I];
487 if (!std::equal(LIt, LIt + Cap, RHS.Blocks[I])) {
488 return false;
489 }
490
491 Cap = Total;
492 Total <<= 1;
493 }
494
495 return std::equal(LHS.Start, LHS.Pos, RHS.Start);
496 }

498 [[nodiscard]] friend bool operator!=(const StableVector &LHS,
499 const StableVector &RHS) noexcept {
500 return !(LHS == RHS);
501 }

509 template <typename... ArgTys>
510 [[nodiscard]] T &growAndEmplace(ArgTys &&...Args) {
511 auto makeBlock = [this](size_t N) {
512 auto *Ret = std::allocator_traits<allocator_type>::allocate(Alloc, N);
513 __asan_poison_memory_region(std::next(Ret), (N - 1) * sizeof(T));
514 return Ret;
515 };
516
517 if (Blocks.empty()) {
518 Blocks.push_back(makeBlock(InitialCapacity));
519 End = Blocks.back() + InitialCapacity;
520 } else if (BlockIdx < Blocks.size() - 1) {
521 assert(llvm::isPowerOf2_64(Size));
522 BlockIdx++;
523 End = Blocks[BlockIdx] + Size;
524 __asan_unpoison_memory_region(Blocks[BlockIdx], sizeof(T));
525 } else {
526 assert(llvm::isPowerOf2_64(Size));
527 BlockIdx++;
528 Blocks.push_back(makeBlock(Size));
529 End = Blocks.back() + Size;
530 }
531
532 auto *Ret = Blocks[BlockIdx];
533
534 Start = Ret;
535 Pos = Ret + 1;
536 ++Size;
537
538 std::allocator_traits<allocator_type>::construct(
539 Alloc, Ret, std::forward<ArgTys>(Args)...);
540 return *Ret;
541 }

543 [[nodiscard]] inline T &subscriptHelper(size_t Index) const noexcept {
544 if (Index < InitialCapacity) {
545 return Blocks[0][Index];
546 }
547
548 auto Log = llvm::Log2_64(Index);
549 auto LogIdx = Log - (InitialLogCapacity - 1);
550 auto Offset = Index - (size_t(1) << Log);
551 return Blocks[LogIdx][Offset];
552
553 // auto GreaterEqual32Mask = -(Index >= InitialCapacity);
554 // auto Log = llvm::Log2_64_Ceil(Index + 1);
555 // // Really wish to call @llvm.usub.sat here...
556 // auto LogIdx = (Log - InitialLogCapacity) & GreaterEqual32Mask;
557 // auto Offset = Index & (((size_t(1) << Log) >> 1) - 1);
558 // return Blocks[LogIdx][Offset];
559 }

563 T *Pos = nullptr;
564 T *End = nullptr;
565 size_t Size = 0;

567 [[no_unique_address]] allocator_type Alloc{};
