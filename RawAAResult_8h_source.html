17/// Base interface for results of a ValueId-based alias-analysis.
18template <typename T>
19concept RawAAResult = requires(const T &Result, ValueId Var) {
20 { T::isCached() } noexcept -> std::convertible_to<bool>;
21 { Result.getRawAliasSet(Var) } -> std::convertible_to<RawAliasSet<ValueId>>;
22 { Result.mayAlias(Var, Var) } -> std::convertible_to<bool>;
23 { Result.size() } noexcept -> std::convertible_to<size_t>;
24};
25
26/// The intersection of two (different) independent raw alias-analysis results.
27/// Use this to achieve better precision than feasible with a single analysis.
28template <RawAAResult FirstT, RawAAResult SecondT>
29struct RawAAResultIntersection {
30 [[nodiscard]] static constexpr bool isCached() noexcept {
31 // The set-intersection is not cached
32 return false;
33 }
34
35 [[nodiscard]] constexpr size_t size() const noexcept {
36 assert(
37 First.size() == Second.size() &&
38 "Only alias-results on the same ValueCompressor should be intersected");
39 return First.size();
40 }
41
42 [[nodiscard]] RawAliasSet<ValueId> getRawAliasSet(ValueId Var) const {
43 auto ResultSet = First.getRawAliasSet(Var);
44 ResultSet &= Second.getRawAliasSet(Var);
45 return ResultSet;
46 }
47
48 [[nodiscard]] constexpr bool mayAlias(ValueId Var1, ValueId Var2) const {
49 return First.mayAlias(Var1, Var2) && Second.mayAlias(Var1, Var2);
50 }
51
52 [[no_unique_address]] FirstT First;
53 [[no_unique_address]] SecondT Second;
54};
