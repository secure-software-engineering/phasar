42/// Solves an IFDS problem by materializing the realizable part of the exploded
43/// supergraph and answering reachability with DFS-interval labels, following
44/// "DFI: An Interprocedural Value-Flow Analysis Framework that Scales to Large
45/// Codebases" by Hsu, Hetzelt, and Franz (CGO'23). See
46/// docs/DFISolver-Design.md.
47///
48/// Computes the same results as the IFDS solvers, while evaluating each flow
49/// function only once per data-flow fact and instruction. Additionally, with
50/// StaticSolverConfigTy::BuildQueryIndex, getReachability() answers
51/// context-sensitive reachability queries between arbitrary
52/// (instruction, fact) pairs that hold.
53template <IFDSProblem ProblemTy,
54 typename StaticSolverConfigTy = DFISolverConfig,
55 ICFG ICFGTy = typename ProblemTy::ProblemAnalysisDomain::i_t>
56class DFISolver : public IDESolverAPIMixin<
57 DFISolver<ProblemTy, StaticSolverConfigTy, ICFGTy>> {
58 friend IDESolverAPIMixin<DFISolver<ProblemTy, StaticSolverConfigTy, ICFGTy>>;
59
60public:

65 using i_t = ICFGTy;
66 using config_t = StaticSolverConfigTy;

119
120 void doInitialize() {
121 for (const auto &Fun : ICF->getAllFunctions()) {
122 Funs.getOrInsert(Fun);
123 }
124 Graph.emplace(computeCGSCCs(*ICF, Funs),
125 SolverConfig.followReturnsPastSeeds());
126
127 auto ZeroId = Results->Facts.getOrInsert(Problem->getZeroValue());
128 assert(ZeroId == dfi::FactId::Zero);
129 (void)ZeroId;
130
131 auto Seeds = Problem->initialSeeds();
132 for (const auto &[Inst, SeedFacts] : Seeds.getSeeds()) {
133 auto Fun = Funs.get(ICF->getFunctionOf(Inst));
134 for (const auto &[Fact, Value] : SeedFacts) {
135 Graph->addSeed(getOrCreateVertex(Inst, Fact, Fun));
136 }
137 }
138 }
139
140 [[nodiscard]] bool doNext() {
141 auto Next = Graph->nextAction();
142 if (!Next) {
143 return false;
144 }
145
146 if (Next->Kind == detail::DFIExplodedSupergraph::ActionKind::Expand) {
147 expand(Next->Fun);
148 } else {
149 auto Fun = Next->Fun;
150 Graph->summarize(
151 Fun,
152 [this](dfi::VertexId CallVtx, dfi::EntryId Entry,
153 dfi::VertexId Exit) { applySummary(CallVtx, Entry, Exit); },
154 [this, Fun](dfi::VertexId Exit) { returnUnbalanced(Fun, Exit); });
155 }
156 return true;
157 }
158
159 auto doFinalize() & {
160 finalizeResults();
161 return getSolverResults();
162 }
163
164 auto doFinalize() && {
165 finalizeResults();
166 return consumeSolverResults();
167 }

171 dfi::VertexId getOrCreateVertex(ByConstRef<n_t> Inst, ByConstRef<d_t> Fact,
172 FunctionId Fun) {
173 return Graph->getOrCreateVertex(Results->Insts.getOrInsert(Inst),
174 Results->Facts.getOrInsert(Fact), Fun,
175 ICF->isExitInst(Inst));
176 }
177
178 /// Creates the vertex reached by an intra-procedural edge to Succ
179 dfi::VertexId getOrCreateSuccVertex(ByConstRef<n_t> Succ,
180 ByConstRef<d_t> Fact, FunctionId Fun) {
181 if constexpr (has_advanceToNextUser_v<ICFGTy, d_t>) {
182 return getOrCreateVertex(ICF->advanceToNextUser(Succ, Fact), Fact, Fun);
183 } else {
184 return getOrCreateVertex(Succ, Fact, Fun);
185 }
186 }

188 [[nodiscard]] n_t instOf(dfi::VertexId Vtx) const {
189 return Results->Insts[Graph->instOf(Vtx)];
190 }
191 [[nodiscard]] d_t factOf(dfi::VertexId Vtx) const {
192 return Results->Facts[Graph->factOf(Vtx)];
193 }

195 [[nodiscard]] auto computeTargets(const FlowFunctionPtrType &FF,
196 ByConstRef<d_t> Fact) {
197 auto Ret = (*FF).computeTargets(Fact);
198 if constexpr (StaticSolverConfigTy::AutoAddZero) {
199 if (Problem->isZeroValue(Fact)) {
200 Ret.insert(Problem->getZeroValue());
201 }
202 }
203 return Ret;
204 }

206 /// Adds the summary edges from the call vertex CallVtx that go through the
207 /// same-level path from Entry to the exit vertex Exit
208 void applySummary(dfi::VertexId CallVtx, dfi::EntryId Entry,
209 dfi::VertexId Exit) {
210 n_t CallSite = instOf(CallVtx);
211 n_t ExitInst = instOf(Exit);
212 d_t ExitFact = factOf(Exit);
213 f_t Callee = Funs[Graph->funOf(Entry)];
214 auto Caller = Graph->funOf(CallVtx);
215
216 for (const auto &RetSite : ICF->getReturnSitesOfCallAt(CallSite)) {
217 auto FF =
218 Problem->getRetFlowFunction(CallSite, Callee, ExitInst, RetSite);
219 for (const auto &Fact : computeTargets(FF, ExitFact)) {
220 Graph->addSummaryEdge(CallVtx,
221 getOrCreateSuccVertex(RetSite, Fact, Caller));
222 }
223 }
224 }
225
226 /// Returns the exit vertex Exit of Fun, which is reachable without calling
227 /// context, to all callers of Fun in the call-graph
228 void returnUnbalanced(FunctionId Fun, dfi::VertexId Exit) {
229 n_t ExitInst = instOf(Exit);
230 d_t ExitFact = factOf(Exit);
231 f_t Callee = Funs[Fun];
232
233 bool HasCallers = false;
234 for (const auto &CallSite : ICF->getCallersOf(Callee)) {
235 HasCallers = true;
236 auto Caller = Funs.get(ICF->getFunctionOf(CallSite));
237 for (const auto &RetSite : ICF->getReturnSitesOfCallAt(CallSite)) {
238 auto FF =
239 Problem->getRetFlowFunction(CallSite, Callee, ExitInst, RetSite);
240 for (const auto &Fact : computeTargets(FF, ExitFact)) {
241 Graph->addUnbalancedReturn(
242 Exit, getOrCreateSuccVertex(RetSite, Fact, Caller));
243 }
244 }
245 }
246
247 if constexpr (UnbalancedRetSideEffectProvider<ProblemTy>) {
248 if (!HasCallers) {
249 Problem->applyUnbalancedRetFlowFunctionSideEffects(Callee, ExitInst,
250 ExitFact);
251 }
252 }
253 }
254
255 /// Expands all pending vertices of Fun, batched per instruction, such that
256 /// each flow function is constructed once per batch.
257 void expand(FunctionId Fun) {
258 auto Batch = Graph->takeWorklist(Fun);
259 for (const auto *It = Batch.begin(), *End = Batch.end(); It != End;) {
260 auto Inst = Graph->instOf(*It);
261 const auto *RunEnd = std::find_if(It, End, [this, Inst](auto Vtx) {
262 return Graph->instOf(Vtx) != Inst;
263 });
264
265 n_t Stmt = Results->Insts[Inst];
266 llvm::ArrayRef<dfi::VertexId> Run(It, RunEnd);
267 if (ICF->isCallSite(Stmt)) {
268 expandCall(Stmt, Run, Fun);
269 } else {
270 expandNormal(Stmt, Run, Fun);
271 }
272 It = RunEnd;
273 }
274 }
275
276 void expandNormal(ByConstRef<n_t> Stmt, llvm::ArrayRef<dfi::VertexId> Run,
277 FunctionId Fun) {
278 llvm::SmallVector<std::pair<n_t, FlowFunctionPtrType>, 2> NormalFFs;
279 for (const auto &Succ : ICF->getSuccsOf(Stmt)) {
280 NormalFFs.emplace_back(Succ, Problem->getNormalFlowFunction(Stmt, Succ));
281 }
282
283 for (auto Vtx : Run) {
284 d_t Fact = factOf(Vtx);
285 Graph->beginEdges(Vtx);
286 for (const auto &[Succ, FF] : NormalFFs) {
287 for (const auto &Target : computeTargets(FF, Fact)) {
288 Graph->addEdge(getOrCreateSuccVertex(Succ, Target, Fun));
289 }
290 }
291 Graph->endEdges(Vtx);
292 }
293 }
294
295 void expandCall(ByConstRef<n_t> CallSite, llvm::ArrayRef<dfi::VertexId> Run,
296 FunctionId Fun) {
297 llvm::SmallVector<f_t> Callees;
298 for (const auto &Callee : ICF->getCalleesOfCallAt(CallSite)) {
299 Callees.push_back(Callee);
300 }
301
302 llvm::SmallVector<n_t, 2> RetSites;
303 llvm::SmallVector<FlowFunctionPtrType, 2> CallToRetFFs;
304 for (const auto &RetSite : ICF->getReturnSitesOfCallAt(CallSite)) {
305 RetSites.push_back(RetSite);
306 CallToRetFFs.push_back(
307 Problem->getCallToRetFlowFunction(CallSite, RetSite, Callees));
308 }
309
310 llvm::SmallVector<CalleeInfo, 1> CalleeInfos;
311 for (const auto &Callee : Callees) {
312 auto &Info = CalleeInfos.emplace_back();
313 Info.Callee = Funs.get(Callee);
314
315 if (auto SummaryFF = Problem->getSummaryFlowFunction(CallSite, Callee)) {
316 Info.FF = std::move(SummaryFF);
317 Info.IsSummaryFF = true;
318 continue;
319 }
320
321 Info.FF = Problem->getCallFlowFunction(CallSite, Callee);
322 for (const auto &StartPoint : ICF->getStartPointsOf(Callee)) {
323 Info.StartPoints.push_back(StartPoint);
324 }
325 }
326
327 for (auto Vtx : Run) {
328 d_t Fact = factOf(Vtx);
329
330 Graph->beginEdges(Vtx);
331 for (const auto &[RetSite, FF] : llvm::zip(RetSites, CallToRetFFs)) {
332 for (const auto &Target : computeTargets(FF, Fact)) {
333 Graph->addEdge(getOrCreateSuccVertex(RetSite, Target, Fun));
334 }
335 }
336 for (const auto &Info : CalleeInfos) {
337 if (!Info.IsSummaryFF) {
338 continue;
339 }
340 for (const auto &Target : computeTargets(Info.FF, Fact)) {
341 for (const auto &RetSite : RetSites) {
342 Graph->addEdge(getOrCreateSuccVertex(RetSite, Target, Fun));
343 }
344 }
345 }
346 Graph->endEdges(Vtx);
347
348 for (const auto &Info : CalleeInfos) {
349 if (Info.IsSummaryFF) {
350 continue;
351 }
352 for (const auto &CalleeFact : computeTargets(Info.FF, Fact)) {
353 for (const auto &StartPoint : Info.StartPoints) {
354 auto Entry = Graph->getOrCreateEntry(
355 getOrCreateVertex(StartPoint, CalleeFact, Info.Callee));
356 auto KnownExits = Graph->addCallEdge(Vtx, Entry);
357 for (auto Exit : KnownExits) {
358 applySummary(Vtx, Entry, Exit);
359 }
360 }
361 }
362 }
363 }
364 }

368 /// Indexes the reversed supergraph. Its roots are the sinks of the
369 /// supergraph; all other vertices follow, as vertices on cycles without a
370 /// path to a sink are not reachable from any sink.
371 [[nodiscard]] IntervalReachabilityIndex<dfi::VertexId> buildBackwardIndex() {
372 auto Reversed = CsrGraph<dfi::VertexId>::reversed(Graph->buildSupergraph());
373 auto Roots = std::move(Reversed).roots();
374 llvm::append_range(Roots, iota<dfi::VertexId>(Reversed.numVertices()));
375 return IntervalReachabilityBuilder<dfi::VertexId>().build(Reversed, Roots);
376 }
377
378 void finalizeResults() {
379 auto &R = *Results;
380 if constexpr (StaticSolverConfigTy::BuildQueryIndex) {
381 R.IndexDirection = StaticSolverConfigTy::IndexDirection;
382 R.Index.emplace(lazy{[&] {
383 if constexpr (StaticSolverConfigTy::IndexDirection ==
384 DFIIndexDirection::Backward) {
385 return buildBackwardIndex();
386 } else {
387 return IntervalReachabilityBuilder<dfi::VertexId>().build(
388 Graph->buildSupergraph(), Graph->seeds());
389 }
390 }});
391 }
392
393 R.VtxInst = Graph->takeVertexInsts();
394 R.VtxFact = Graph->takeVertexFacts();
395 auto NumVertices = R.VtxInst.size();
396
397 R.InstOffsets.assign(R.Insts.size() + 1, 0);
398 for (auto Inst : R.VtxInst) {
399 ++R.InstOffsets[Inst];
400 }
401 std::partial_sum(R.InstOffsets.begin(), R.InstOffsets.end(),
402 R.InstOffsets.begin());
403 R.VerticesAt.resize_for_overwrite(NumVertices);
404 for (auto Vtx : llvm::reverse(iota<dfi::VertexId>(NumVertices))) {
405 R.VerticesAt[--R.InstOffsets[R.VtxInst[Vtx]]] = Vtx;
406 }
407 for (auto Inst : iota<dfi::InstId>(R.Insts.size())) {
408 auto Vertices = csrSuccsMut(R.InstOffsets, R.VerticesAt, Inst);
409 llvm::sort(Vertices, [&R](dfi::VertexId Lhs, dfi::VertexId Rhs) {
410 return R.VtxFact[Lhs] < R.VtxFact[Rhs];
411 });
412 }
413
414 Graph.reset();
415 Funs = FunctionCompressor<f_t>();
416 }
