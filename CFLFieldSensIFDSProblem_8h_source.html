94/// Interns the Store- and Load field-strings
95class FieldStringManager {
96public:
97 static constexpr int32_t TopOffset = INT32_MIN;
98
99 FieldStringManager();
100
101 [[nodiscard]] FieldStringNodeId intern(FieldStringNode Nod) {
102 auto [Id, Inserted] = NodeCompressor.insert(Nod);
103
104 if (Inserted) {
105 Depth.push_back(Depth[Nod.Next] + 1);
106 }
107
108 return Id;
109 }
110
111 [[nodiscard]] FieldStringNodeId prepend(int32_t Head,
112 FieldStringNodeId Tail) {
113 auto Ret = intern(FieldStringNode{.Next = Tail, .Offset = Head});
114 PHASAR_LOG_LEVEL(DEBUG, "[prepend]: " << Head << " :: #" << uint32_t(Tail)
115 << " = #" << uint32_t(Ret));
116 return Ret;
117 }
118
119 [[nodiscard]] FieldStringNode operator[](FieldStringNodeId NId) const {
120 return NodeCompressor[NId];
121 }
122
123 [[nodiscard]] llvm::SmallVector<int32_t>
124 getFullFieldString(FieldStringNodeId NId) const;
125
126 [[nodiscard]] FieldStringNodeId
127 fromFullFieldString(llvm::ArrayRef<int32_t> FieldString);
128
129 [[nodiscard]] uint32_t depth(FieldStringNodeId NId) const {
130 return Depth[NId];
131 }
132
133 [[nodiscard]] KillSetId internKills(SmallArraySet<int32_t, 2> &&Kills) {
134 return KillsCompressor.getOrInsert(std::move(Kills));
135 }
136
137 [[nodiscard]] KillSetId addKill(KillSetId KS, int32_t Offs) {
138 if (Offs == TopOffset || KillsCompressor[KS].contains(Offs)) {
139 return KS;
140 }
141
142 auto Kills = KillsCompressor[KS];
143 Kills.insert(Offs);
144 return KillsCompressor.getOrInsert(std::move(Kills));
145 }
146
147 [[nodiscard]] bool isKilledBy(KillSetId KS, int32_t Offs) const {
148 if (Offs == TopOffset || KS == KillSetId::Empty) {
149 return false;
150 }
151 if (!KillsCompressor.inbounds(KS)) [[unlikely]] {
152 return false;
153 }
154
155 return KillsCompressor[KS].contains(Offs);
156 }
157
158 [[nodiscard]] const auto &kills(KillSetId KS) const {
159 return KillsCompressor[KS];
160 }
161
162 void reserve(size_t ExpectedCapacity) {
163 NodeCompressor.reserve(ExpectedCapacity);
164 Depth.reserve(ExpectedCapacity);
165 }
166
167private:
168 Compressor<FieldStringNode, FieldStringNodeId> NodeCompressor{};
169 TypedVector<FieldStringNodeId, uint32_t> Depth{};
170 Compressor<SmallArraySet<int32_t, 2>, KillSetId> KillsCompressor{};
171};

173/// A single CFL Field-Access String consisting of: gep, loads, kills, and
174/// stores
175struct AccessPath {
176 static constexpr int32_t TopOffset = FieldStringManager::TopOffset;
177
178 FieldStringNodeId Loads{};
179 FieldStringNodeId Stores{};
180 KillSetId Kills{};
181 // Add an offset for pending GEPs; INT32_MIN is Top
182 int32_t Offset{};
183
184 [[nodiscard]] bool empty() const noexcept {
185 return Loads == FieldStringNodeId::None &&
186 Stores == FieldStringNodeId::None && Kills == KillSetId::Empty &&
187 Offset == 0;
188 }
189
190 [[nodiscard]] constexpr bool
191 operator==(const AccessPath &Other) const noexcept = default;
192
193 friend constexpr size_t hash_value(const AccessPath &FieldString) noexcept {
194 size_t HC = 37;
195 HC = HC * 31 + size_t(FieldString.Loads);
196 HC = HC * 31 + size_t(FieldString.Stores);
197 HC = HC * 31 + size_t(FieldString.Kills);
198 HC = HC * 31 + size_t(FieldString.Offset);
199 return HC;
200 }
201
202 friend llvm::raw_ostream &operator<<(llvm::raw_ostream &OS,
203 const AccessPath &FieldString);
204
205 void print(llvm::raw_ostream &OS, const FieldStringManager &Mgr) const;
206};

226/// An edge-value consisting of a set if CFL field access strings.
227struct IFDSEdgeValue {
228 using container_type = llvm::SmallDenseSet<AccessPath, 2, AccessPathDMI>;
229
230 PSR_REQUIRE_EXPLICIT_INITIALIZATION FieldStringManager *Mgr{};
231 container_type Paths;
232
233 static constexpr llvm::StringLiteral LogCategory = "IFDSEdgeValue";
234
235 void applyTransforms(const IFDSEdgeValue &Txns, uint8_t DepthKLimit);
236
237 bool operator==(const IFDSEdgeValue &Other) const noexcept {
238 assert(Mgr == Other.Mgr);
239 assert(Mgr != nullptr);
240 return Paths == Other.Paths;
241 }
242 bool operator!=(const IFDSEdgeValue &Other) const noexcept {
243 return !(*this == Other);
244 }
245
246 [[nodiscard]] friend auto hash_value(const IFDSEdgeValue &EV) {
247 return llvm::hash_combine_range(EV.Paths.begin(), EV.Paths.end());
248 }
249
250 friend llvm::raw_ostream &operator<<(llvm::raw_ostream &OS,
251 const IFDSEdgeValue &EV);
252
253 [[nodiscard]] bool isEpsilon() const {
254 return Paths.size() == 1 && Paths.begin()->empty();
255 }
256
257 [[nodiscard]] static IFDSEdgeValue epsilon(FieldStringManager *Mgr) {
258 IFDSEdgeValue Ret{.Mgr = &assertNotNull(Mgr), .Paths = {}};
259 Ret.Paths.insert({}); // Not using initializer_list to prevent copying
260 return Ret;
261 }
262
263 // To be picked up via ADL by psr::join(LatticeDomain, LatticeDomain)
264 [[nodiscard]] friend auto join(const IFDSEdgeValue &L,
265 const IFDSEdgeValue &R) {
266 assert(L.Mgr == R.Mgr);
267 assert(L.Mgr != nullptr);
268 const bool LeftSmaller = L.Paths.size() < R.Paths.size();
269 auto Ret = LeftSmaller ? R : L;
270 const auto &Smaller = LeftSmaller ? L : R;
271 Ret.Paths.insert(Smaller.Paths.begin(), Smaller.Paths.end());
272 // XXX: k-limit num-paths: This may not be necessary, as join() is only
273 // called from IDE-Phase-II
274 return Ret;
275 }
276};

292/// Utility to strip off potential pointer-arithmetic from V and accumulating
293/// the byte-offset.
294[[nodiscard]] inline std::pair<const llvm::Value *, int32_t>
295getBaseAndOffset(const llvm::Value *V, const llvm::DataLayout &DL) {
296 llvm::APInt Offset(64, 0);
297 int32_t OffsVal = AccessPath::TopOffset;
298 const auto *Base = V->stripAndAccumulateConstantOffsets(DL, Offset, true);
299
300 if (llvm::isa<llvm::GEPOperator>(Base)) {
301 return {Base->stripPointerCastsAndAliases(), AccessPath::TopOffset};
302 }
303
304 auto RawOffsVal = Offset.getSExtValue();
305 if (RawOffsVal <= INT32_MAX && RawOffsVal >= INT32_MIN) {
306 OffsVal = int32_t(RawOffsVal);
307 }
308
309 return {Base->stripPointerCastsAndAliases(), OffsVal};
310}

312/// Checks whether Fact holds at Inst in a field-sensitive way
313template <bool AllowDeepTaints = true, typename ResultsT>
314[[nodiscard]] inline bool holdsFactAt(const ResultsT &Results,
315 IFDSDomain::n_t Inst,
316 IFDSDomain::d_t Fact) {
317 const auto &Fields = Results.resultAt(Inst, Fact);
318
319 if (Fields.isTop()) {
320 // Was not computed by the IDE Solver
321 return false;
322 }
323
324 if (const IFDSEdgeValue *FieldStrings = Fields.getValueOrNull()) {
325 if constexpr (!AllowDeepTaints) {
326 // whether Facts itself holds, not whether any fields of it may hold
327 return FieldStrings->isEpsilon();
328 }
329 if (FieldStrings->Paths.empty()) {
330 // has been killed entirely
331 return false;
332 }
333 }
334
335 return true;
336}

338/// Given a QueryMap of the form map<n_t, set<d_t>>, calls the Handler for all
339/// inst-fact pairs that hold in a field-sensitive way and filters out all
340/// others.
341///
342/// The Handler may opt into early exit by returning false. Returning void is
343/// permitted.
344template <bool AllowDeepTaints = true, typename ResultsT>
345bool filterFieldSensFacts(
346 const ResultsT &Results, const auto &QueryMap,
347 std::invocable<IFDSDomain::n_t, IFDSDomain::d_t> auto Handler) {
348 const IFDSDomain::l_t Top = psr::Top{};
349
350 for (const auto &[Inst, FactsAtInst] : QueryMap) {
351 const auto &Row = Results.row(Inst);
352 for (const auto &Fact : FactsAtInst) {
353 const auto &Fields = getOr(Row, Fact, Top);
354
355 if (Fields.isTop()) {
356 // Was not computed by the IDE Solver
357 continue;
358 }
359
360 if (const auto *FieldStrings = Fields.getValueOrNull()) {
361 if (!AllowDeepTaints && !FieldStrings->isEpsilon()) {
362 // Fact does not hold itself, but fields of Fact may hold. In
363 // aggressive mode, we ignore them
364 continue;
365 }
366 if (FieldStrings->Paths.empty()) {
367 // has been killed entirely
368 continue;
369 }
370 }
371
372 if constexpr (std::convertible_to<
373 std::invoke_result_t<decltype(Handler) &,
374 IFDSDomain::n_t, IFDSDomain::d_t>,
375 bool>) {
376 if (!std::invoke(Handler, Inst, Fact)) {
377 return false;
378 }
379 } else {
380 std::invoke(Handler, Inst, Fact);
381 }
382 }
383 }
384 return true;
385}

528/// An IFDS-Problem adaptor that makes any field-insensitive IFDS analysis
529/// field-sensitive. Just wrap your IFDS problem with CFLFieldSensIFDSProblem
530/// and use the IterativeIDESolver instead of the IFDSSolver.
531///
532/// The only thing to change in your usual IFDS problem is not to kill data-flow
533/// facts when only parts of the fields should be killed. This is now handled by
534/// the CFLFieldSensIFDSProblem. For that, provide a
535/// problem definition with a proper killsAt()
536/// implementation.
537template <IFDSProblem ProblemTy>
538 requires std::same_as<LLVMIFDSAnalysisDomainDefault,
539 typename ProblemTy::ProblemAnalysisDomain>
540class CFLFieldSensIFDSProblem
541 : public cfl_fieldsens::CFLFieldSensEdgeFunctions {
542 using Base = cfl_fieldsens::CFLFieldSensEdgeFunctions;
543
544 static decltype(cfl_fieldsens::IFDSProblemConfig::KillsAt)
545 deriveKillsAt(ProblemTy *UserProblem) {
546 assert(UserProblem != nullptr);
547 if constexpr (requires {
548 {
549 UserProblem->killsAt()
550 } -> psr::invocable_r<std::optional<int32_t>, n_t, d_t>;
551 }) {
552 return UserProblem->killsAt();
553 } else if constexpr (requires {
554 {
555 UserProblem->killsAt()
556 } -> std::invocable<n_t, d_t>;
557 }) {
558 // Intentionally leaving an unused variable, so that the compiler emits a
559 // warning here
560 auto KillsAtHasWrongReturnType = UserProblem->killsAt();
561 return nullptr;
562 } else {
563 return nullptr;
564 }
565 }
566
567public:
568 using typename Base::container_type;
569 using typename Base::d_t;
570 using typename Base::db_t;
571 using typename Base::f_t;
572 using typename Base::FlowFunctionPtrType;
573 using typename Base::i_t;
574 using typename Base::l_t;
575 using typename Base::n_t;
576 using typename Base::ProblemAnalysisDomain;
577 using typename Base::t_t;
578 using typename Base::v_t;
579
580 /// Constructs an IDETabulationProblem with the usual arguments, forwarded
581 /// from UserProblem
582 explicit CFLFieldSensIFDSProblem(
583 ProblemTy *UserProblem,
584 cfl_fieldsens::IFDSProblemConfig
585 Config) noexcept(std::is_nothrow_move_constructible_v<d_t>)
586 : Base(assertNotNull(UserProblem), std::move(Config)),
587 UserProblem(UserProblem) {}
588
589 /// Constructs an IDETabulationProblem with the usual arguments, forwarded
590 /// from UserProblem and tries to automatically derive the config from
591 /// additional functions specified by UserProblem
592
593 explicit CFLFieldSensIFDSProblem(ProblemTy *UserProblem)
594 : CFLFieldSensIFDSProblem(UserProblem,
595 cfl_fieldsens::IFDSProblemConfig{
596 .KillsAt = deriveKillsAt(UserProblem),
597 }) {}
598
599 CFLFieldSensIFDSProblem(std::nullptr_t,
600 cfl_fieldsens::IFDSProblemConfig Config) = delete;
601
602 CFLFieldSensIFDSProblem(std::nullptr_t) = delete;
603
604 [[nodiscard]] decltype(auto) getEntryPoints() const {
605 return UserProblem->getEntryPoints();
606 }

609 // offsets to the edge-functions (generating from zero currently always

615

617 return UserProblem->getNormalFlowFunction(Curr, Succ);
618 }

643 [[nodiscard]] const auto &base() const noexcept { return *UserProblem; }

645private:
