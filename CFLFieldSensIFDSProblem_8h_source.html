94/// Interns the Store- and Load field-strings
95class FieldStringManager {
96public:
97 static constexpr int32_t TopOffset = INT32_MIN;
98
99 FieldStringManager();
100
101 [[nodiscard]] FieldStringNodeId intern(FieldStringNode Nod) {
102 auto [Id, Inserted] = NodeCompressor.insert(Nod);
103
104 if (Inserted) {
105 Depth.push_back(Depth[Nod.Next] + 1);
106 }
107
108 return Id;
109 }
110
111 [[nodiscard]] FieldStringNodeId prepend(int32_t Head,
112 FieldStringNodeId Tail) {
113 auto Ret = intern(FieldStringNode{.Next = Tail, .Offset = Head});
114 PHASAR_LOG_LEVEL(DEBUG, "[prepend]: " << Head << " :: #" << uint32_t(Tail)
115 << " = #" << uint32_t(Ret));
116 return Ret;
117 }
118
119 [[nodiscard]] FieldStringNode operator[](FieldStringNodeId NId) const {
120 return NodeCompressor[NId];
121 }
122
123 [[nodiscard]] llvm::SmallVector<int32_t>
124 getFullFieldString(FieldStringNodeId NId) const;
125
126 [[nodiscard]] FieldStringNodeId
127 fromFullFieldString(llvm::ArrayRef<int32_t> FieldString);
128
129 [[nodiscard]] uint32_t depth(FieldStringNodeId NId) const {
130 return Depth[NId];
131 }
132
133 [[nodiscard]] KillSetId internKills(SmallArraySet<int32_t, 2> &&Kills) {
134 return KillsCompressor.getOrInsert(std::move(Kills));
135 }
136
137 [[nodiscard]] KillSetId addKill(KillSetId KS, int32_t Offs) {
138 if (Offs == TopOffset || KillsCompressor[KS].contains(Offs)) {
139 return KS;
140 }
141
142 auto Kills = KillsCompressor[KS];
143 Kills.insert(Offs);
144 return KillsCompressor.getOrInsert(std::move(Kills));
145 }
146
147 [[nodiscard]] bool isKilledBy(KillSetId KS, int32_t Offs) const {
148 if (Offs == TopOffset || KS == KillSetId::Empty) {
149 return false;
150 }
151 if (!KillsCompressor.inbounds(KS)) [[unlikely]] {
152 return false;
153 }
154
155 return KillsCompressor[KS].contains(Offs);
156 }
157
158 [[nodiscard]] const auto &kills(KillSetId KS) const {
159 return KillsCompressor[KS];
160 }
161
162 void reserve(size_t ExpectedCapacity) {
163 NodeCompressor.reserve(ExpectedCapacity);
164 Depth.reserve(ExpectedCapacity);
165 }
166
167private:
168 Compressor<FieldStringNode, FieldStringNodeId> NodeCompressor{};
169 TypedVector<FieldStringNodeId, uint32_t> Depth{};
170 Compressor<SmallArraySet<int32_t, 2>, KillSetId> KillsCompressor{};
171};

173/// A single CFL Field-Access String consisting of: gep, loads, kills, and
174/// stores
175struct AccessPath {
176 static constexpr int32_t TopOffset = FieldStringManager::TopOffset;
177
178 FieldStringNodeId Loads{};
179 FieldStringNodeId Stores{};
180 KillSetId Kills{};
181 // Add an offset for pending GEPs; INT32_MIN is Top
182 int32_t Offset{};
183
184 [[nodiscard]] bool empty() const noexcept {
185 return Loads == FieldStringNodeId::None &&
186 Stores == FieldStringNodeId::None && Kills == KillSetId::Empty &&
187 Offset == 0;
188 }
189
190 [[nodiscard]] constexpr bool
191 operator==(const AccessPath &Other) const noexcept = default;
192
193 friend constexpr size_t hash_value(const AccessPath &FieldString) noexcept {
194 size_t HC = 37;
195 HC = HC * 31 + size_t(FieldString.Loads);
196 HC = HC * 31 + size_t(FieldString.Stores);
197 HC = HC * 31 + size_t(FieldString.Kills);
198 HC = HC * 31 + size_t(FieldString.Offset);
199 return HC;
200 }
201
202 friend llvm::raw_ostream &operator<<(llvm::raw_ostream &OS,
203 const AccessPath &FieldString);
204
205 void print(llvm::raw_ostream &OS, const FieldStringManager &Mgr) const;
206};

226/// An edge-value consisting of a set if CFL field access strings.
227struct IFDSEdgeValue {
228 using container_type = llvm::SmallDenseSet<AccessPath, 2, AccessPathDMI>;
229
230 [[clang::require_explicit_initialization]] FieldStringManager *Mgr{};
231 container_type Paths;
232
233 static constexpr llvm::StringLiteral LogCategory = "IFDSEdgeValue";
234
235 void applyTransforms(const IFDSEdgeValue &Txns, uint8_t DepthKLimit);
236
237 bool operator==(const IFDSEdgeValue &Other) const noexcept {
238 assert(Mgr == Other.Mgr);
239 assert(Mgr != nullptr);
240 return Paths == Other.Paths;
241 }
242 bool operator!=(const IFDSEdgeValue &Other) const noexcept {
243 return !(*this == Other);
244 }
245
246 [[nodiscard]] friend auto hash_value(const IFDSEdgeValue &EV) {
247 return llvm::hash_combine_range(EV.Paths.begin(), EV.Paths.end());
248 }
249
250 friend llvm::raw_ostream &operator<<(llvm::raw_ostream &OS,
251 const IFDSEdgeValue &EV);
252
253 [[nodiscard]] bool isEpsilon() const {
254 return Paths.size() == 1 && Paths.begin()->empty();
255 }
256
257 [[nodiscard]] static IFDSEdgeValue epsilon(FieldStringManager *Mgr) {
258 IFDSEdgeValue Ret{.Mgr = &assertNotNull(Mgr), .Paths = {}};
259 Ret.Paths.insert({}); // Not using initializer_list to prevent copying
260 return Ret;
261 }
262
263 // To be picked up via ADL by psr::join(LatticeDomain, LatticeDomain)
264 [[nodiscard]] friend auto join(const IFDSEdgeValue &L,
265 const IFDSEdgeValue &R) {
266 assert(L.Mgr == R.Mgr);
267 assert(L.Mgr != nullptr);
268 const bool LeftSmaller = L.Paths.size() < R.Paths.size();
269 auto Ret = LeftSmaller ? R : L;
270 const auto &Smaller = LeftSmaller ? L : R;
271 Ret.Paths.insert(Smaller.Paths.begin(), Smaller.Paths.end());
272 // XXX: k-limit num-paths: This may not be necessary, as join() is only
273 // called from IDE-Phase-II
274 return Ret;
275 }
276};

300/// Utility to strip off potential pointer-arithmetic from V and accumulating
301/// the byte-offset.
302[[nodiscard]] inline std::pair<const llvm::Value *, int32_t>
303getBaseAndOffset(const llvm::Value *V, const llvm::DataLayout &DL) {
304 llvm::APInt Offset(64, 0);
305 int32_t OffsVal = AccessPath::TopOffset;
306 const auto *Base = V->stripAndAccumulateConstantOffsets(DL, Offset, true);
307
308 if (llvm::isa<llvm::GEPOperator>(Base)) {
309 return {Base->stripPointerCastsAndAliases(), AccessPath::TopOffset};
310 }
311
312 auto RawOffsVal = Offset.getSExtValue();
313 if (RawOffsVal <= INT32_MAX && RawOffsVal >= INT32_MIN) {
314 OffsVal = int32_t(RawOffsVal);
315 }
316
317 return {Base->stripPointerCastsAndAliases(), OffsVal};
318}

320/// Checks whether Fact holds at Inst in a field-sensitive way
321template <bool AllowDeepTaints = true, typename ResultsT>
322[[nodiscard]] inline bool holdsFactAt(const ResultsT &Results,
323 IFDSDomain::n_t Inst,
324 IFDSDomain::d_t Fact) {
325 const auto &Fields = Results.resultAt(Inst, Fact);
326
327 if (Fields.isTop()) {
328 // Was not computed by the IDE Solver
329 return false;
330 }
331
332 if (const IFDSEdgeValue *FieldStrings = Fields.getValueOrNull()) {
333 if constexpr (!AllowDeepTaints) {
334 // whether Facts itself holds, not whether any fields of it may hold
335 return FieldStrings->isEpsilon();
336 }
337 if (FieldStrings->Paths.empty()) {
338 // has been killed entirely
339 return false;
340 }
341 }
342
343 return true;
344}

346/// Given a QueryMap of the form map<n_t, set<d_t>>, calls the Handler for all
347/// inst-fact pairs that hold in a field-sensitive way and filters out all
348/// others.
349///
350/// The Handler may opt into early exit by returning false. Returning void is
351/// permitted.
352template <bool AllowDeepTaints = true, typename ResultsT>
353bool filterFieldSensFacts(
354 const ResultsT &Results, const auto &QueryMap,
355 std::invocable<IFDSDomain::n_t, IFDSDomain::d_t> auto Handler) {
356 const IFDSDomain::l_t Top = psr::Top{};
357
358 for (const auto &[Inst, FactsAtInst] : QueryMap) {
359 const auto &Row = Results.row(Inst);
360 for (const auto &Fact : FactsAtInst) {
361 const auto &Fields = getOr(Row, Fact, Top);
362
363 if (Fields.isTop()) {
364 // Was not computed by the IDE Solver
365 continue;
366 }
367
368 if (const auto *FieldStrings = Fields.getValueOrNull()) {
369 if (!AllowDeepTaints && !FieldStrings->isEpsilon()) {
370 // Fact does not hold itself, but fields of Fact may hold. In
371 // aggressive mode, we ignore them
372 continue;
373 }
374 if (FieldStrings->Paths.empty()) {
375 // has been killed entirely
376 continue;
377 }
378 }
379
380 if constexpr (std::convertible_to<
381 std::invoke_result_t<decltype(Handler) &,
382 IFDSDomain::n_t, IFDSDomain::d_t>,
383 bool>) {
384 if (!std::invoke(Handler, Inst, Fact)) {
385 return false;
386 }
387 } else {
388 std::invoke(Handler, Inst, Fact);
389 }
390 }
391 }
392 return true;
393}

459/// An IFDS-Problem adaptor that makes any field-insensitive IFDS analysis
460/// field-sensitive. Just wrap your IFDS problem with CFLFieldSensIFDSProblem
461/// and use the IterativeIDESolver instead of the IFDSSolver.
462///
463/// The only thing to change in your usual IFDS problem is not to kill data-flow
464/// facts when only parts of the fields should be killed. This is now handled by
465/// the CFLFieldSensIFDSProblem. For that, provide a CFLFieldSensIFDSProblem
466/// with a proper KillsAt implementation.
467class CFLFieldSensIFDSProblem
468 : public IDETabulationProblem<cfl_fieldsens::IFDSDomain> {
469 using Base = IDETabulationProblem<cfl_fieldsens::IFDSDomain>;
470
471 static decltype(cfl_fieldsens::IFDSProblemConfig::KillsAt)
472 deriveKillsAt(auto *UserProblem) {
473 assert(UserProblem != nullptr);
474 if constexpr (requires() {
475 {
476 UserProblem->killsAt()
477 } -> psr::invocable_r<std::optional<int32_t>, n_t, d_t>;
478 }) {
479 return UserProblem->killsAt();
480 } else if constexpr (requires() {
481 {
482 UserProblem->killsAt()
483 } -> std::invocable<n_t, d_t>;
484 }) {
485 // Intentionally leaving an unused variable, so that the compiler emits a
486 // warning here
487 auto KillsAtHasWrongReturnType = UserProblem->killsAt();
488 return nullptr;
489 } else {
490 return nullptr;
491 }
492 }
493
494public:
495 using typename Base::container_type;
496 using typename Base::d_t;
497 using typename Base::db_t;
498 using typename Base::f_t;
499 using typename Base::FlowFunctionPtrType;
500 using typename Base::i_t;
501 using typename Base::l_t;
502 using typename Base::n_t;
503 using typename Base::ProblemAnalysisDomain;
504 using typename Base::t_t;
505 using typename Base::v_t;
506
507 static constexpr llvm::StringLiteral LogCategory = "CFLFieldSensIFDSProblem";

509 /// Constructs an IDETabulationProblem with the usual arguments, forwarded
510 /// from UserProblem
511 explicit CFLFieldSensIFDSProblem(
512 IFDSTabulationProblem<LLVMIFDSAnalysisDomainDefault> *UserProblem,
513 cfl_fieldsens::IFDSProblemConfig
514 Config) noexcept(std::is_nothrow_move_constructible_v<d_t>)
515 : Base(assertNotNull(UserProblem).getProjectIRDB(),
516 assertNotNull(UserProblem).getEntryPoints(),
517 UserProblem->getZeroValue()),
518 UserProblem(UserProblem), Config(std::move(Config)) {
519 Mgr.reserve(UserProblem->getProjectIRDB()->getNumInstructions());
520 }

522 /// Constructs an IDETabulationProblem with the usual arguments, forwarded

524 /// additional functions specified by UserProblem
525 explicit CFLFieldSensIFDSProblem(

527 IFDSTabulationProblem<LLVMIFDSAnalysisDomainDefault>> auto
528 *UserProblem)
