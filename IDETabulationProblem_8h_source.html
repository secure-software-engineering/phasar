44/// \brief The analysis problem interface for IDE problems (solvable by the
45/// IDESolver). Create a subclass from this and override all pure-virtual
46/// functions to create your own IDE analysis.
47///
48/// For more information on how to write an IDE analysis, see [Writing an IDE
49/// Analysis](https://github.com/secure-software-engineering/phasar/wiki/Writing-an-IDE-analysis)
50template <IdeAnalysisDomain AnalysisDomainTy,
51 typename Container = std::set<typename AnalysisDomainTy::d_t>>
52class IDETabulationProblem : public FlowFunctions<AnalysisDomainTy, Container>,
53 public EdgeFunctions<AnalysisDomainTy>,
54 public JoinLattice<AnalysisDomainTy>,
55 public SemiRing<AnalysisDomainTy> {
56public:
57 using ProblemAnalysisDomain = AnalysisDomainTy;
58 using d_t = typename AnalysisDomainTy::d_t;
59 using n_t = typename AnalysisDomainTy::n_t;
60 using f_t = typename AnalysisDomainTy::f_t;
61 using t_t = typename AnalysisDomainTy::t_t;
62 using v_t = typename AnalysisDomainTy::v_t;
63 using l_t = typename AnalysisDomainTy::l_t;
64 using i_t = typename AnalysisDomainTy::i_t;
65 using db_t = typename AnalysisDomainTy::db_t;
66
67 using ConfigurationTy = HasNoConfigurationType;
68
69 /// Takes an IR database (IRDB) and collects information from it to create a
70 /// tabulation problem.
71 /// @param[in] IRDB The project IR database, that holds the code under
72 /// analysis
73 /// @param[in] EntryPoints The (mangled) names of all entry functions of the
74 /// target being analyzed, given as a vector of strings. An example would
75 /// simply be `{"main"}`. To set every function as entry point, pass
76 /// `"__ALL__"`
77 /// @param[in] ZeroValue Provides the special tautological zero value (aka.
78 /// Λ). If not provided here, you must set it via \link initializeZeroValue()
79 /// \endlink.
80 explicit IDETabulationProblem(
81 const db_t *IRDB, std::vector<std::string> EntryPoints,
82 std::optional<d_t>
83 ZeroValue) noexcept(std::is_nothrow_move_constructible_v<d_t>)
84 : IRDB(IRDB), EntryPoints(std::move(EntryPoints)),
85 ZeroValue(std::move(ZeroValue)),
86 Printer(std::make_unique<typename DefaultAnalysisPrinterSelector<
87 AnalysisDomainTy>::type>()) {
88 assert(IRDB != nullptr);
89
90 static_assert(IDEProblem<IDETabulationProblem>);
91 }
92
93 IDETabulationProblem(IDETabulationProblem &&) noexcept = default;
94 IDETabulationProblem &operator=(IDETabulationProblem &&) noexcept = default;
95
96 IDETabulationProblem(const IDETabulationProblem &) = delete;
97 IDETabulationProblem &operator=(const IDETabulationProblem &) = delete;
98
99 ~IDETabulationProblem() override = default;

101 void
102 setAnalysisPrinter(MaybeUniquePtr<AnalysisPrinterBase<AnalysisDomainTy>> P) {
103 if (P) {
104 Printer = std::move(P);
105 } else {
106 Printer = NullAnalysisPrinter<AnalysisDomainTy>::getInstance();
107 }
108 }
109
110 [[nodiscard]] constexpr AnalysisPrinterBase<AnalysisDomainTy> &
111 printer() noexcept {
112 assert(Printer != nullptr);
113 return *Printer;
114 }
115
116 [[nodiscard]] constexpr MaybeUniquePtr<AnalysisPrinterBase<AnalysisDomainTy>>
117 consumePrinter() noexcept {
118 assert(Printer != nullptr);
119 return std::exchange(Printer,
120 NullAnalysisPrinter<AnalysisDomainTy>::getInstance());
121 }

123 /// Checks if the given data-flow fact is the special tautological lambda (or
124 /// zero) fact.
125 [[nodiscard]] virtual bool isZeroValue(d_t FlowFact) const noexcept {
126 assert(ZeroValue.has_value());
127 return FlowFact == *ZeroValue;
128 }
129
130 /// Returns initial seeds to be used for the analysis. This is a mapping of
131 /// statements to initial analysis facts.
132 [[nodiscard]] virtual InitialSeeds<n_t, d_t, l_t> initialSeeds() = 0;
133
134 /// Returns the special tautological lambda (or zero) fact.
135 [[nodiscard]] ByConstRef<d_t> getZeroValue() const {
136 assert(ZeroValue.has_value());
137 return *ZeroValue;
138 }
139
140 void initializeZeroValue(d_t Zero) noexcept(
141 std::is_nothrow_assignable_v<std::optional<d_t> &, d_t &&>) {
142 assert(!ZeroValue.has_value());
143 ZeroValue = std::move(Zero);
144 }
145
146 /// Sets the configuration to be used by the IFDS/IDE solver.
147 void setIFDSIDESolverConfig(IFDSIDESolverConfig Config) noexcept {
148 SolverConfig = Config;
149 }
150
151 /// Returns the configuration of the IFDS/IDE solver.
152 [[nodiscard]] IFDSIDESolverConfig &getIFDSIDESolverConfig() noexcept {
153 return SolverConfig;
154 }
155
156 /// Generates a text report of the results that is written to the specified
157 /// output stream.
158 virtual void
159 emitTextReport([[maybe_unused]] GenericSolverResults<n_t, d_t, l_t> Results,
160 llvm::raw_ostream &OS = llvm::outs()) {
161 Printer->onFinalize(OS);
162 }
163
164 /// Generates a graphical report, e.g. in html or other markup languages, of
165 /// the results that is written to the specified output stream.
166 virtual void emitGraphicalReport(
167 [[maybe_unused]] GenericSolverResults<n_t, d_t, l_t> Results,
168 llvm::raw_ostream &OS = llvm::outs()) {
169 OS << "No graphical report available!\n";
170 }
171
172 /// Sets the level of soundness to be used by the analysis. Returns false if
173 /// the level of soundness is ignored. Otherwise, true.
174 virtual bool setSoundness(Soundness /*S*/) { return false; }
175
176 [[nodiscard]] const db_t *getProjectIRDB() const noexcept { return IRDB; }
177
178 [[nodiscard]] llvm::ArrayRef<std::string> getEntryPoints() const noexcept {
179 return EntryPoints;
180 }
181
182protected:
183 typename FlowFunctions<AnalysisDomainTy, Container>::FlowFunctionPtrType
184 generateFromZero(d_t FactToGenerate) {
185 return FlowFunctions<AnalysisDomainTy, Container>::generateFlow(
186 std::move(FactToGenerate), getZeroValue());
187 }
188
189 template <typename D = d_t, typename L = l_t>
190 void onResult(n_t Instr, D &&DfFact, L &&LatticeElement,
191 DataFlowAnalysisType AnalysisType) {
192 Printer->onResult(Instr, PSR_FWD(DfFact), PSR_FWD(LatticeElement),
193 AnalysisType);
194 }
195
196 template <typename D = d_t>
197 requires std::is_same_v<l_t, BinaryDomain>
198 void onResult(n_t Instr, D &&DfFact, DataFlowAnalysisType AnalysisType) {
199 Printer->onResult(Instr, PSR_FWD(DfFact), AnalysisType);
200 }
201
202 /// Seeds that just start with ZeroValue and bottomElement() at the starting
203 /// points of each EntryPoint function.
204 /// Takes the __ALL__ EntryPoint into account.
205 [[nodiscard]] InitialSeeds<n_t, d_t, l_t> createDefaultSeeds()
206 requires std::is_nothrow_default_constructible_v<
207 typename AnalysisDomainTy::c_t>
208 {
209 InitialSeeds<n_t, d_t, l_t> Seeds;
210 typename AnalysisDomainTy::c_t C{};
211
212 addSeedsForStartingPoints(EntryPoints, IRDB, C, Seeds, getZeroValue(),
213 this->bottomElement());
214
215 return Seeds;
216 }
217
218 const db_t *IRDB{};
219 std::vector<std::string> EntryPoints;
220 std::optional<d_t> ZeroValue;
221
222 IFDSIDESolverConfig SolverConfig{};
223
224 [[maybe_unused]] Soundness SF = Soundness::Soundy;
225
226 MaybeUniquePtr<AnalysisPrinterBase<AnalysisDomainTy>> Printer;
227};
