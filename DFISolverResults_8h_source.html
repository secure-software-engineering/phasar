47template <typename N, typename D> struct DFIResultsData {
48 Compressor<N, dfi::InstId> Insts;
49 Compressor<D, dfi::FactId> Facts;
50
51 TypedVector<dfi::VertexId, dfi::InstId> VtxInst;
52 TypedVector<dfi::VertexId, dfi::FactId> VtxFact;
53
54 /// The vertices at instruction I are
55 /// VerticesAt[InstOffsets[I] .. InstOffsets[I+1]), sorted by fact
56 TypedVector<dfi::InstId, uint32_t> InstOffsets{0};
57 llvm::SmallVector<dfi::VertexId, 0> VerticesAt;
58
59 std::optional<IntervalReachabilityIndex<dfi::VertexId>> Index;
60 DFIIndexDirection IndexDirection = DFIIndexDirection::Forward;
61};

63template <typename Derived, typename N, typename D> class DFISolverResultsBase {
64public:
65 using n_t = N;
66 using d_t = D;
67 using l_t = BinaryDomain;
68
69 /// BinaryDomain::BOTTOM if Fact holds at Stmt, BinaryDomain::TOP otherwise
70 [[nodiscard]] l_t resultAt(ByConstRef<n_t> Stmt, ByConstRef<d_t> Fact) const {
71 return findVertex(Stmt, Fact) ? BinaryDomain::BOTTOM : BinaryDomain::TOP;
72 }
73
74 [[nodiscard]] std::unordered_map<d_t, l_t>
75 resultsAt(ByConstRef<n_t> Stmt, bool StripZero = false) const {
76 std::unordered_map<d_t, l_t> Ret;
77 forEachFactAt(Stmt, StripZero, [&Ret](ByConstRef<d_t> Fact) {
78 Ret.try_emplace(Fact, BinaryDomain::BOTTOM);
79 });
80 return Ret;
81 }
82
83 [[nodiscard]] std::set<d_t> ifdsResultsAt(ByConstRef<n_t> Stmt) const {
84 std::set<d_t> Ret;
85 forEachFactAt(Stmt, /*StripZero*/ false,
86 [&Ret](ByConstRef<d_t> Fact) { Ret.insert(Fact); });
87 return Ret;
88 }
89
90 /// Calls Handler for each fact holding at Stmt
91 void forEachFactAt(ByConstRef<n_t> Stmt, bool StripZero,
92 std::invocable<ByConstRef<d_t>> auto Handler) const {
93 const auto &Data = data();
94 for (auto Vtx : verticesAt(Stmt)) {
95 auto Fact = Data.VtxFact[Vtx];
96 if (StripZero && Fact == dfi::FactId::Zero) {
97 continue;
98 }
99 std::invoke(Handler, Data.Facts[Fact]);
100 }
101 }
102
103 /// Number of instructions with results
104 [[nodiscard]] size_t size() const noexcept { return data().Insts.size(); }
105
106 [[nodiscard]] bool containsNode(ByConstRef<n_t> Stmt) const {
107 return data().Insts.getOrNull(Stmt).has_value();
108 }
109
110 void foreachResultEntry(
111 std::invocable<std::tuple<n_t, d_t, l_t>> auto Handler) const {
112 const auto &Data = data();
113 for (auto [Inst, Stmt] : Data.Insts.enumerate()) {
114 for (auto Vtx : verticesAt(Inst)) {
115 std::invoke(Handler,
116 std::make_tuple(Stmt, Data.Facts[Data.VtxFact[Vtx]],
117 BinaryDomain::BOTTOM));
118 }
119 }
120 }
121
122 void dumpResults(llvm::raw_ostream &OS = llvm::outs()) const {
123 const auto &Data = data();
124 OS << "\n***************************************************************\n"
125 << "* Raw DFISolver results *\n"
126 << "***************************************************************\n";
127 for (auto [Inst, Stmt] : Data.Insts.enumerate()) {
128 OS << "\nN: " << NToString(Stmt) << '\n';
129 for (auto Vtx : verticesAt(Inst)) {
130 OS << "\tD: " << DToString(Data.Facts[Data.VtxFact[Vtx]]) << '\n';
131 }
132 }
133 }
134
135 /// The vertex for Fact at Stmt, if Fact holds there
136 [[nodiscard]] std::optional<dfi::VertexId>
137 findVertex(ByConstRef<n_t> Stmt, ByConstRef<d_t> Fact) const {
138 const auto &Data = data();
139 auto FactId = Data.Facts.getOrNull(Fact);
140 if (!FactId) {
141 return std::nullopt;
142 }
143 auto Vertices = verticesAt(Stmt);
144 const auto *It =
145 std::lower_bound(Vertices.begin(), Vertices.end(), *FactId,
146 [&Data](dfi::VertexId Vtx, dfi::FactId F) {
147 return Data.VtxFact[Vtx] < F;
148 });
149 if (It == Vertices.end() || Data.VtxFact[*It] != *FactId) {
150 return std::nullopt;
151 }
152 return *It;
153 }
154
155 [[nodiscard]] llvm::ArrayRef<dfi::VertexId>
156 verticesAt(ByConstRef<n_t> Stmt) const {
157 auto Inst = data().Insts.getOrNull(Stmt);
158 if (!Inst) {
159 return {};
160 }
161 return verticesAt(*Inst);
162 }
163
164 [[nodiscard]] llvm::ArrayRef<dfi::VertexId>
165 verticesAt(dfi::InstId Inst) const {
166 return csrSuccs(data().InstOffsets, data().VerticesAt, Inst);
167 }
168
169 [[nodiscard]] n_t instOf(dfi::VertexId Vtx) const {
170 return data().Insts[data().VtxInst[Vtx]];
171 }
172 [[nodiscard]] d_t factOf(dfi::VertexId Vtx) const {
173 return data().Facts[data().VtxFact[Vtx]];
174 }
175
176private:
177 [[nodiscard]] const DFIResultsData<N, D> &data() const noexcept {
178 return static_cast<const Derived *>(this)->dataImpl();
179 }
180};
181} // namespace detail

236/// Context-sensitive reachability between the exploded-supergraph nodes that
237/// the DFISolver has materialized.
238///
239/// From reaches To iff there is a realizable path from From to To that may
240/// descend into callees, but does not return beyond the function of From.
241/// Every node reaches itself. The nodes reachable from a node are its
242/// descendants, the nodes reaching it are its ancestors; both include the node
243/// itself.
244///
245/// Point queries cost O(log k) in both index directions. Enumerating the
246/// descendants (forward index) or ancestors (backward index) is linear in the
247/// output; the other enumeration scans all nodes.
248template <typename N, typename D> class DFIReachability {
249public:
250 using n_t = N;
251 using d_t = D;
252
253 explicit DFIReachability(const detail::DFIResultsData<N, D> *Data) noexcept
254 : Results(Data), Index(&*Data->Index), Direction(Data->IndexDirection) {}
255
256 /// The node <Stmt, Fact>, if it has been materialized, i.e., if Fact holds
257 /// at Stmt.
258 [[nodiscard]] std::optional<dfi::VertexId> find(ByConstRef<n_t> Stmt,
259 ByConstRef<d_t> Fact) const {
260 return Results.findVertex(Stmt, Fact);
261 }
262
263 [[nodiscard]] bool reaches(dfi::VertexId From,
264 dfi::VertexId To) const noexcept {
265 if (Direction == DFIIndexDirection::Backward) {
266 return Index->reaches(To, From);
267 }
268 return Index->reaches(From, To);
269 }
270
271 [[nodiscard]] bool reaches(ByConstRef<n_t> FromStmt, ByConstRef<d_t> FromFact,
272 ByConstRef<n_t> ToStmt,
273 ByConstRef<d_t> ToFact) const {
274 auto From = find(FromStmt, FromFact);
275 auto To = find(ToStmt, ToFact);
276 return From && To && reaches(*From, *To);
277 }
278
279 /// Calls Handler for each fact at At that is reachable from From
280 void
281 forEachDescendantFactAt(dfi::VertexId From, ByConstRef<n_t> At,
282 std::invocable<ByConstRef<d_t>> auto Handler) const {
283 for (auto To : Results.verticesAt(At)) {
284 if (reaches(From, To)) {
285 std::invoke(Handler, Results.factOf(To));
286 }
287 }
288 }
289
290 /// Calls Handler for each fact at At that reaches To
291 void
292 forEachAncestorFactAt(dfi::VertexId To, ByConstRef<n_t> At,
293 std::invocable<ByConstRef<d_t>> auto Handler) const {
294 for (auto From : Results.verticesAt(At)) {
295 if (reaches(From, To)) {
296 std::invoke(Handler, Results.factOf(From));
297 }
298 }
299 }
300
301 /// Calls Handler for each node reachable from From
302 void forEachDescendant(
303 dfi::VertexId From,
304 std::invocable<ByConstRef<n_t>, ByConstRef<d_t>> auto Handler) const {
305 forEachRelated(From, DFIIndexDirection::Forward, Handler);
306 }
307
308 /// Calls Handler for each node that reaches To
309 void forEachAncestor(
310 dfi::VertexId To,
311 std::invocable<ByConstRef<n_t>, ByConstRef<d_t>> auto Handler) const {
312 forEachRelated(To, DFIIndexDirection::Backward, Handler);
313 }
314
315 [[nodiscard]] n_t instOf(dfi::VertexId Vtx) const {
316 return Results.instOf(Vtx);
317 }
318 [[nodiscard]] d_t factOf(dfi::VertexId Vtx) const {
319 return Results.factOf(Vtx);
320 }
321
322 [[nodiscard]] const IntervalReachabilityIndex<dfi::VertexId> &
323 index() const noexcept {
324 return *Index;
325 }
326
327 [[nodiscard]] DFIIndexDirection indexDirection() const noexcept {
328 return Direction;
329 }
330
331private:
332 /// Calls Handler for each node related to Vtx in direction Dir
333 void forEachRelated(
334 dfi::VertexId Vtx, DFIIndexDirection Dir,
335 std::invocable<ByConstRef<n_t>, ByConstRef<d_t>> auto &Handler) const {
336 const auto Report = [&](dfi::VertexId Related) {
337 std::invoke(Handler, Results.instOf(Related), Results.factOf(Related));
338 };
339
340 if (Dir == Direction) {
341 Index->forEachReachableTarget(Vtx, Report);
342 return;
343 }
344
345 // The index points the other way, so the related nodes are not contiguous
346 for (auto Other : iota<dfi::VertexId>(Index->numVertices())) {
347 if (Index->reaches(Other, Vtx)) {
348 Report(Other);
349 }
350 }
351 }
352
353 DFISolverResults<N, D> Results;
354 const IntervalReachabilityIndex<dfi::VertexId> *Index{};
355 DFIIndexDirection Direction{};
356};
