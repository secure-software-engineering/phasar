20template <typename AnalysisDomainTy> class WithAnalysisPrinterMixin {
21 // NOTE: Don't add type-aliases for n_t,d_t, ... here; this will break the
22 // compilation with gcc; gcc apparently ignores private visibility when
23 // checking for ambiguous declarations. clang instead behaves as expected.
24
25public:
26 WithAnalysisPrinterMixin()
27 : Printer(std::make_unique<typename DefaultAnalysisPrinterSelector<
28 AnalysisDomainTy>::type>()) {}
29
30 WithAnalysisPrinterMixin(
31 MaybeUniquePtr<AnalysisPrinterBase<AnalysisDomainTy>> Printer) noexcept
32 : Printer(std::move(Printer)) {}
33
34 constexpr auto setAnalysisPrinter(
35 MaybeUniquePtr<AnalysisPrinterBase<AnalysisDomainTy>> P) noexcept {
36 if (P) {
37 return std::exchange(Printer, std::move(P));
38 }
39
40 return std::exchange(Printer,
41 NullAnalysisPrinter<AnalysisDomainTy>::getInstance());
42 }
43
44 [[nodiscard]] constexpr auto consumePrinter() noexcept {
45 return setAnalysisPrinter(nullptr);
46 }
47
48 [[nodiscard]] constexpr AnalysisPrinterBase<AnalysisDomainTy> &
49 printer() noexcept {
50 assert(Printer != nullptr);
51 return *Printer;
52 }
53
54 void emitTextReport(
55 [[maybe_unused]] GenericSolverResultsFor<AnalysisDomainTy> Results,
56 llvm::raw_ostream &OS = llvm::outs()) {
57 Printer->onFinalize(OS);
58 }
59
60protected:
61 template <typename D = AnalysisDomainTy::d_t,
62 typename L = detail::ValueDomainAdder<AnalysisDomainTy>::l_t>
63 void onResult(AnalysisDomainTy::n_t Instr, D &&DfFact, L &&LatticeElement,
64 DataFlowAnalysisType AnalysisType) {
65 Printer->onResult(Instr, PSR_FWD(DfFact), PSR_FWD(LatticeElement),
66 AnalysisType);
67 }
68
69 template <typename D = AnalysisDomainTy::d_t>
70 requires std::is_same_v<
71 typename detail::ValueDomainAdder<AnalysisDomainTy>::l_t, BinaryDomain>
72 void onResult(AnalysisDomainTy::n_t Instr, D &&DfFact,
73 DataFlowAnalysisType AnalysisType) {
74 Printer->onResult(Instr, PSR_FWD(DfFact), AnalysisType);
75 }
76
77private:
78 MaybeUniquePtr<AnalysisPrinterBase<AnalysisDomainTy>> Printer;
79};
