33/// \brief Holds the SCCs of a given graph. Each SCC is assigned a unique
34/// sequential id.
35template <typename GraphNodeId> struct SCCHolder {
36 TypedVector<GraphNodeId, SCCId<GraphNodeId>> SCCOfNode;
37 TypedVector<SCCId<GraphNodeId>, llvm::SmallVector<GraphNodeId, 1>>
38 NodesInSCC{};
39
40 /// Number of SCCs
41 [[nodiscard]] size_t size() const noexcept { return NodesInSCC.size(); }
42 [[nodiscard]] bool empty() const noexcept { return NodesInSCC.empty(); }
43
44 /// \brief Prints the given Graph as dot, highlighting the SCCs in the graph.
45 ///
46 /// \param Graph The graph to print
47 /// \param OS The output-stream, where to print into
48 /// \param Name The name of the graph
49 /// \param NodeToString If the graph has node-labels, convert a node-label to
50 /// string
51 template <is_const_graph G, typename NodeTransform = DefaultNodeTransform>
52 requires std::is_same_v<typename GraphTraits<G>::vertex_t, GraphNodeId>
53 void print(const G &Graph, llvm::raw_ostream &OS, llvm::StringRef Name = "",
54 NodeTransform NodeToString = {}) const {
55 OS << "digraph \"";
56 OS.write_escaped(Name) << "\" {\n";
57 psr::scope_exit CloseBrace = [&] { OS << "}\n"; };
58
59 using GTraits = psr::GraphTraits<G>;
60
61 for (const auto &[SCCId, SCC] : NodesInSCC.enumerate()) {
62 OS << " subgraph cluster_" << +SCCId
63 << "{\n node [style=filled]; color=blue; label=\"SCC " << +SCCId
64 << "\";\n";
65 psr::scope_exit CloseSCC = [&] { OS << " }\n"; };
66
67 for (auto Nod : SCC) {
68 OS << " " << size_t(Nod);
69 if constexpr (!std::is_empty_v<typename GTraits::value_type>) {
70 OS << "[label=\"";
71 OS.write_escaped(
72 std::invoke(NodeToString, GTraits::node(Graph, Nod)));
73 OS << "\"]";
74 }
75 OS << ";\n";
76 }
77 }
78
79 for (auto FromVtx : GTraits::vertices(Graph)) {
80 for (const auto &Succ : GTraits::outEdges(Graph, FromVtx)) {
81 OS << " " << size_t(FromVtx) << "->";
82 if constexpr (is_llvm_printable_v<decltype(Succ)>) {
83 // to print the edge-weight as well, if possible
84 OS << Succ;
85 } else {
86 OS << size_t(GTraits::target(Succ));
87 }
88 OS << ";\n";
89 }
90 }
91 }
92};

103template <typename GraphNodeId>

168/// \brief Creates a graph based on the given input Graph, collapsing all SCCs
169/// to single nodes. The resulting graph is always a DAG, i.e., it contains no
170/// cycles
171template <typename G>
172// TODO: is_const_graph is too restrictive!
173// #if __cplusplus >= 202002L
174// requires is_const_graph<G>
175// #endif
176SCCDependencyGraph<typename GraphTraits<G>::vertex_t> computeSCCDependencies(
177 const G &Graph, const SCCHolder<typename GraphTraits<G>::vertex_t> &SCCs) {
178
179 using GTraits = GraphTraits<G>;
180 using GraphNodeId = typename GraphTraits<G>::vertex_t;
181
182 SCCDependencyGraph<GraphNodeId> Ret;
183 Ret.ChildrenOfSCC.resize(SCCs.size());
184
185 BitSet<SCCId<GraphNodeId>> Roots(SCCs.size(), true);
186
187 for (auto NodeId : GTraits::vertices(Graph)) {
188 auto SrcSCC = SCCs.SCCOfNode[NodeId];
189
190 for (const auto &Edge : GTraits::outEdges(Graph, NodeId)) {
191 auto Succ = GTraits::target(Edge);
192 auto SuccSCC = SCCs.SCCOfNode[Succ];
193 if (SuccSCC != SrcSCC) {
194 Ret.ChildrenOfSCC[SrcSCC].insert(SuccSCC);
195 Roots.erase(SuccSCC);
196 }
197 }
198 }
199
200 Ret.SCCRoots.reserve(Roots.size());
201 Roots.foreach ([&](auto Rt) { Ret.SCCRoots.push_back(Rt); });
202
203 return Ret;
204}

206/// \brief Computes a topological order of the nodes in the given
207/// dependency-graph.
208///
209/// Uses a simple, recursive postorder-DFS search to find a topological
210/// ordering.
211template <typename GraphNodeId>
212[[nodiscard]] SCCOrder<GraphNodeId>
213computeSCCOrder(const SCCHolder<GraphNodeId> &SCCs,
214 const SCCDependencyGraph<GraphNodeId> &Callers) {
215 SCCOrder<GraphNodeId> Ret;
216 Ret.SCCIds.reserve(SCCs.size());
217
218 BitSet<SCCId<GraphNodeId>> Seen;
219 Seen.reserve(SCCs.size());
220
221 auto Dfs = [&](auto &Dfs, SCCId<GraphNodeId> CurrSCC) -> void {
222 Seen.insert(CurrSCC);
223 for (auto Caller : Callers.ChildrenOfSCC[CurrSCC]) {
224 if (!Seen.contains(Caller)) {
225 Dfs(Dfs, Caller);
226 }
227 }
228 Ret.SCCIds.push_back(CurrSCC);
229 };
230
231 for (auto Leaf : Callers.SCCRoots) {
232 if (!Seen.contains(Leaf)) {
233 Dfs(Dfs, Leaf);
234 }
235 }
236
237 std::reverse(Ret.SCCIds.begin(), Ret.SCCIds.end());
238
239 return Ret;
240}

242/// State of Pearce's SCC algorithm. Can be reused across multiple calls to
243/// visitSCCsFrom() on the same graph and, after reset(), on other graphs to
244/// avoid repeated allocations.
245template <typename GraphNodeId> struct PearceSCCData {
246 TypedVector<GraphNodeId, uint32_t> RIndex; // only per-vertex array
247 BitSet<GraphNodeId> Root; // root[v] in Algorithm 4
248 uint32_t Index = 1; // DFS counter
249 uint32_t C{}; // SCC id counter
250 llvm::SmallVector<GraphNodeId> Stack;
251
252 PearceSCCData() noexcept = default;
253 explicit PearceSCCData(size_t Num) { reset(Num); }
254
255 /// Prepares the state for a fresh search on a graph with Num vertices.
256 void reset(size_t Num) {
257 RIndex.assign(Num, 0);
258 Root.clear();
259 Root.reserve(Num);
260 Index = 1;
261 C = Num ? uint32_t(Num - 1) : 0;
262 Stack.clear();
263 }
264
265 [[nodiscard]] bool isVisited(GraphNodeId Vtx) const noexcept {
266 return RIndex[Vtx] != 0;
267 }
268};

272/// Completes the SCC rooted at Root. The members are the stack elements with
273/// an rindex not smaller than the root's.
274template <typename GraphNodeId, typename OnSCCFn>
275void pearce4FinishSCC(PearceSCCData<GraphNodeId> &Data, GraphNodeId Root,
276 OnSCCFn &OnSCC) {
277 auto RootIndex = Data.RIndex[Root];
278 size_t NumMembers = 1;
279 while (NumMembers <= Data.Stack.size() &&
280 RootIndex <= Data.RIndex[Data.Stack[Data.Stack.size() - NumMembers]]) {
281 ++NumMembers;
282 }
283 Data.Stack.push_back(Root);
284
285 auto Members = llvm::ArrayRef<GraphNodeId>(Data.Stack).take_back(NumMembers);
286 auto NewSCC = SCCId<GraphNodeId>(uint32_t(Data.RIndex.size() - 1 - Data.C));
287 std::invoke(OnSCC, NewSCC, Members);
288
289 for (auto W : Members) {
290 Data.RIndex[W] = Data.C;
291 }
292 Data.Index -= uint32_t(NumMembers);
293 Data.C--;
294 Data.Stack.truncate(Data.Stack.size() - NumMembers);
295}

297// Recursive variant of Pearce's algorithm (based on Algo 3 in the paper)
298template <typename G, typename OnDiscoverFn, typename OnSCCFn>
299static void
300pearce4VisitRec(const G &Graph, typename GraphTraits<G>::vertex_t V,
301 PearceSCCData<typename GraphTraits<G>::vertex_t> &Data,
302 OnDiscoverFn &OnDiscover, OnSCCFn &OnSCC) {
303 using GTraits = psr::GraphTraits<G>;
304
305 bool Root = true;
306 Data.RIndex[V] = Data.Index++;
307 std::invoke(OnDiscover, V);
308
309 for (const auto &Edge : GTraits::outEdges(Graph, V)) {
310 auto W = GTraits::target(Edge);
311 if (Data.RIndex[W] == 0) {
312 pearce4VisitRec(Graph, W, Data, OnDiscover, OnSCC);
313 }
314 if (Data.RIndex[W] < Data.RIndex[V]) {
315 Data.RIndex[V] = Data.RIndex[W];
316 Root = false;
317 }
318 }
319
320 if (Root) {
321 pearce4FinishSCC(Data, V, OnSCC);
322 } else {
323 Data.Stack.push_back(V);
324 }
325}

327// Iterative variant of Pearce's algorithm (adapted from on Algo 4 in the paper)
328template <typename G, typename OnDiscoverFn, typename OnSCCFn>
329static void
330pearce4VisitIt(const G &Graph, typename GraphTraits<G>::vertex_t Start,
331 PearceSCCData<typename GraphTraits<G>::vertex_t> &Data,
332 OnDiscoverFn &OnDiscover, OnSCCFn &OnSCC) {
333 using GTraits = psr::GraphTraits<G>;
334 using Vertex = typename GTraits::vertex_t;
335
336 using OutEdgeRange =
337 decltype(GTraits::outEdges(Graph, std::declval<Vertex>()));
338 using OutEdgeIterator =
339 decltype(llvm::adl_begin(std::declval<OutEdgeRange &>()));
340 using OutEdgeSentinel =
341 decltype(llvm::adl_end(std::declval<OutEdgeRange &>()));
342
343 struct DfsFrame {
344 Vertex CurrVtx;
345 OutEdgeIterator It;
346 [[no_unique_address]] OutEdgeSentinel End;
347 };
348
349 llvm::SmallVector<DfsFrame> CallStack;
350
351 const auto PushFrames = [&](Vertex V, DfsFrame *Frame) {
352 if (Frame->It == Frame->End) {
353 return false;
354 }
355 // Recurse into children until reaching the bottom
356 do {
357 auto W = GTraits::target(*Frame->It);
358
359 if (Data.RIndex[W] != 0) {
360 // Already pushed the children of W
361 break;
362 }
363
364 Data.RIndex[W] = Data.Index++;
365 Data.Root.insert(W);
366 std::invoke(OnDiscover, W);
367
368 auto &&OutEdges = GTraits::outEdges(Graph, W);
369 Frame = &CallStack.emplace_back(
370 DfsFrame{W, llvm::adl_begin(OutEdges), llvm::adl_end(OutEdges)});
371 V = W;
372
373 } while (Frame->It != Frame->End);
374
375 return true;
376 };
377
378 const auto VisitLoop = [&](Vertex V, DfsFrame &Frame) {
379 // Finish visiting the current child and advance to the next child
380 if (Frame.It != Frame.End) {
381 auto W = GTraits::target(*Frame.It);
382 if (Data.RIndex[W] < Data.RIndex[V]) {
383 Data.RIndex[V] = Data.RIndex[W];
384 Data.Root.erase(V);
385 }
386
387 ++Frame.It;
388 }
389 };
390
391 const auto FinishFrame = [&](Vertex V) {
392 // finish visiting V and backtrack to the parent
393
394 if (Data.Root.contains(V)) {
395 pearce4FinishSCC(Data, V, OnSCC);
396 } else {
397 Data.Stack.push_back(V);
398 }
399
400 CallStack.pop_back();
401 };
402
403 // Initialize the callstack by pushing the initial frame
404 Data.RIndex[Start] = Data.Index++;
405 Data.Root.insert(Start);
406 std::invoke(OnDiscover, Start);
407 {
408 auto &&OutEdges = GTraits::outEdges(Graph, Start);
409 static_assert(
410 std::is_lvalue_reference_v<decltype(OutEdges)> ||
411 std::is_trivially_destructible_v<std::decay_t<decltype(OutEdges)>>,
412 "We assume that outEdges gives either a reference or a view into "
413 "the out-edges, but never an owning container by value. Otherwise, "
414 "the DFSFrame iterators may be dangling");
415 CallStack.emplace_back(
416 DfsFrame{Start, llvm::adl_begin(OutEdges), llvm::adl_end(OutEdges)});
417 }
418
419 // Simulate the recursion
420
421 PushFrames(Start, &CallStack.back());
422 while (true) {
423 auto &Frame = CallStack.back();
424 Vertex V = Frame.CurrVtx;
425 VisitLoop(V, Frame);
426 if (PushFrames(V, &Frame)) {
427 continue; // we don't pop from the callstack here
428 }
429
430 FinishFrame(V);
431 if (CallStack.empty()) {
432 break;
433 }
434 }
435}

439/// Runs Pearce's SCC algorithm from Start, visiting all vertices reachable
440/// from Start that have not yet been visited with the same Data. Call it with
441/// multiple start vertices to cover more of the graph.
442///
443/// \param OnDiscover Called with each vertex when it is visited first, i.e.,
444/// in DFS pre-order.
445/// \param OnSCC Called with each SCC when it is complete, in reverse
446/// topological order. SCC ids are sequential, starting at 0 after
447/// Data.reset(). Members.back() is the first discovered member, i.e., the root
448/// of the SCC in the DFS tree. At this point, all SCCs reachable from this SCC
449/// are complete and the DFS subtree of the root is fully discovered.
450template <typename G, bool Iterative = true>
451void visitSCCsFrom(
452 const G &Graph, typename GraphTraits<G>::vertex_t Start,
453 PearceSCCData<typename GraphTraits<G>::vertex_t> &Data,
454 std::invocable<typename GraphTraits<G>::vertex_t> auto OnDiscover,
455 std::invocable<SCCId<typename GraphTraits<G>::vertex_t>,
456 llvm::ArrayRef<typename GraphTraits<G>::vertex_t>> auto
457 OnSCC,
458 std::bool_constant<Iterative> /*Iterative*/ = {}) {
459 assert(!Data.isVisited(Start));
460 if constexpr (Iterative) {
461 detail::pearce4VisitIt(Graph, Start, Data, OnDiscover, OnSCC);
462 } else {
463 detail::pearce4VisitRec(Graph, Start, Data, OnDiscover, OnSCC);
464 }
465 assert(Data.Stack.empty() && "The start vertex must be an SCC root");
466}

468/// Compute SCCs adapted from the paper "A Space-Efficient Algorithm for Finding
469/// Strongly Connected Components", Pearce 2015, DOI:
470/// <https://doi.org/10.1016/j.ipl.2015.08.010>
471///
472/// \tparam G The graph-type
473/// \tparam Iterative Whether to use the iterative or recursive variant of the
474/// algorithm (default: true)
475/// \param Graph The graph for with to compute SCCs and topological ordering
476template <typename G, bool Iterative = true>
477[[nodiscard]] SCCHolder<typename GraphTraits<G>::vertex_t>
478computeSCCs(const G &Graph, std::bool_constant<Iterative> /*Iterative*/ = {}) {
479 using GTraits = psr::GraphTraits<G>;
480 using Vertex = typename GTraits::vertex_t;
481
482 SCCHolder<Vertex> Ret;
483 auto N = GTraits::size(Graph);
484 if (!N) {
485 return Ret;
486 }
487
488 Ret.SCCOfNode.resize(N);
489
490 PearceSCCData<Vertex> Data(N);
491
492 const auto OnSCC = [&Ret](SCCId<Vertex> SCC, llvm::ArrayRef<Vertex> Members) {
493 assert(size_t(SCC) == Ret.NodesInSCC.size());
494 Ret.NodesInSCC.emplace_back(Members.begin(), Members.end());
495 for (auto Member : Members) {
496 Ret.SCCOfNode[Member] = SCC;
497 }
498 };
499
500 // for all v in V do if rindex[v]==0 then visit(v)
501 for (auto V : GTraits::vertices(Graph)) {
502 if (!Data.isVisited(V)) {
503 visitSCCsFrom(
504 Graph, V, Data, [](Vertex /*Vtx*/) {}, OnSCC,
505 std::bool_constant<Iterative>{});
506 }
507 }
508
509 return Ret;
510}

512/// Compute SCCs and a topological ordering on the SCCs, adapted from the paper
513/// "A Space-Efficient Algorithm for Finding Strongly Connected Components",
514/// Pearce 2015, DOI: <https://doi.org/10.1016/j.ipl.2015.08.010>
515///
516/// \tparam G The graph-type \tparam Iterative Whether to use the iterative
517/// or recursive variant of the algorithm (default: true) \param Graph The graph
518/// for with to compute SCCs and topological ordering
519template <typename G, bool Iterative = true>
520[[nodiscard]] std::pair<SCCHolder<typename GraphTraits<G>::vertex_t>,
521 SCCOrder<typename GraphTraits<G>::vertex_t>>
522computeSCCsAndTopologicalOrder(
523 const G &Graph, std::bool_constant<Iterative> /*Iterative*/ = {}) {
524 using Vertex = typename GraphTraits<G>::vertex_t;
525
526 std::pair<SCCHolder<Vertex>, SCCOrder<Vertex>> Ret = {
527 computeSCCs(Graph, std::bool_constant<Iterative>{}),
528 {},
529 };
530
531 // Pearce's algorithm produces SCCs in reverse topological order
532 auto Ids = llvm::reverse(psr::iota<SCCId<Vertex>>(Ret.first.size()));
533 Ret.second.SCCIds.append(Ids.begin(), Ids.end());
534
535 return Ret;
536}
