32/// \brief Holds the SCCs of a given graph. Each SCC is assigned a unique
33/// sequential id.
34template <typename GraphNodeId> struct SCCHolder {
35 TypedVector<GraphNodeId, SCCId<GraphNodeId>> SCCOfNode;
36 TypedVector<SCCId<GraphNodeId>, llvm::SmallVector<GraphNodeId, 1>>
37 NodesInSCC{};
38
39 /// Number of SCCs
40 [[nodiscard]] size_t size() const noexcept { return NodesInSCC.size(); }
41 [[nodiscard]] bool empty() const noexcept { return NodesInSCC.empty(); }

43 /// \brief Prints the given Graph as dot, highlighting the SCCs in the graph.
44 ///
45 /// \param Graph The graph to print
46 /// \param OS The output-stream, where to print into
47 /// \param Name The name of the graph
48 /// \param NodeToString If the graph has node-labels, convert a node-label to
49 /// string
50 template <is_const_graph G, typename NodeTransform = DefaultNodeTransform>
51 requires std::is_same_v<typename GraphTraits<G>::vertex_t, GraphNodeId>
52 void print(const G &Graph, llvm::raw_ostream &OS, llvm::StringRef Name = "",
53 NodeTransform NodeToString = {}) const {
54 OS << "digraph \"";
55 OS.write_escaped(Name) << "\" {\n";
56 psr::scope_exit CloseBrace = [&] { OS << "}\n"; };
57
58 using GTraits = psr::GraphTraits<G>;
59
60 for (const auto &[SCCId, SCC] : NodesInSCC.enumerate()) {
61 OS << " subgraph cluster_" << +SCCId
62 << "{\n node [style=filled]; color=blue; label=\"SCC " << +SCCId
63 << "\";\n";
64 psr::scope_exit CloseSCC = [&] { OS << " }\n"; };
65
66 for (auto Nod : SCC) {
67 OS << " " << size_t(Nod);
68 if constexpr (!std::is_empty_v<typename GTraits::value_type>) {
69 OS << "[label=\"";
70 OS.write_escaped(
71 std::invoke(NodeToString, GTraits::node(Graph, Nod)));
72 OS << "\"]";
73 }
74 OS << ";\n";
75 }
76 }
77
78 for (auto FromVtx : GTraits::vertices(Graph)) {
79 for (const auto &Succ : GTraits::outEdges(Graph, FromVtx)) {
80 OS << " " << size_t(FromVtx) << "->";
81 if constexpr (is_llvm_printable_v<decltype(Succ)>) {
82 // to print the edge-weight as well, if possible
83 OS << Succ;
84 } else {
85 OS << size_t(GTraits::target(Succ));
86 }
87 OS << ";\n";
88 }
89 }
90 }
91};

167/// \brief Creates a graph based on the given input Graph, collapsing all SCCs
168/// to single nodes. The resulting graph is always a DAG, i.e., it contains no
169/// cycles
170template <typename G>
171#if __cplusplus >= 202002L
172 requires is_const_graph<G>
173#endif
174SCCDependencyGraph<typename GraphTraits<G>::vertex_t> computeSCCDependencies(
175 const G &Graph, const SCCHolder<typename GraphTraits<G>::vertex_t> &SCCs) {
176
177 using GTraits = GraphTraits<G>;
178 using GraphNodeId = typename GraphTraits<G>::vertex_t;
179
180 SCCDependencyGraph<GraphNodeId> Ret;
181 Ret.ChildrenOfSCC.resize(SCCs.size());
182
183 BitSet<SCCId<GraphNodeId>> Roots(SCCs.size(), true);
184
185 for (auto NodeId : GTraits::vertices(Graph)) {
186 auto SrcSCC = SCCs.SCCOfNode[NodeId];
187
188 for (const auto &Edge : GTraits::outEdges(Graph, NodeId)) {
189 auto Succ = GTraits::target(Edge);
190 auto SuccSCC = SCCs.SCCOfNode[Succ];
191 if (SuccSCC != SrcSCC) {
192 Ret.ChildrenOfSCC[SrcSCC].insert(SuccSCC);
193 Roots.erase(SuccSCC);
194 }
195 }
196 }
197
198 Ret.SCCRoots.reserve(Roots.size());
199 Roots.foreach ([&](auto Rt) { Ret.SCCRoots.push_back(Rt); });
200
201 return Ret;
202}

204/// \brief Computes a topological order of the nodes in the given
205/// dependency-graph.
206///
207/// Uses a simple, recursive postorder-DFS search to find a topological
208/// ordering.
209template <typename GraphNodeId>
210[[nodiscard]] SCCOrder<GraphNodeId>
211computeSCCOrder(const SCCHolder<GraphNodeId> &SCCs,
212 const SCCDependencyGraph<GraphNodeId> &Callers) {
213 SCCOrder<GraphNodeId> Ret;
214 Ret.SCCIds.reserve(SCCs.size());
215
216 BitSet<SCCId<GraphNodeId>> Seen;
217 Seen.reserve(SCCs.size());
218
219 auto Dfs = [&](auto &Dfs, SCCId<GraphNodeId> CurrSCC) -> void {
220 Seen.insert(CurrSCC);
221 for (auto Caller : Callers.ChildrenOfSCC[CurrSCC]) {
222 if (!Seen.contains(Caller)) {
223 Dfs(Dfs, Caller);
224 }
225 }
226 Ret.SCCIds.push_back(CurrSCC);
227 };
228
229 for (auto Leaf : Callers.SCCRoots) {
230 if (!Seen.contains(Leaf)) {
231 Dfs(Dfs, Leaf);
232 }
233 }
234
235 std::reverse(Ret.SCCIds.begin(), Ret.SCCIds.end());
236
237 return Ret;
238}

241/// Data for Pearce's Algorithm.
242template <typename GraphNodeId> struct Pearce4Data {
243 TypedVector<GraphNodeId, uint32_t> RIndex; // only per-vertex array
244 BitSet<GraphNodeId> Root; // root[v] in Algorithm 4
245 uint32_t Index = 1; // DFS counter
246 uint32_t C; // SCC id counter
247 llvm::SmallVector<GraphNodeId> Stack;
248
249 explicit Pearce4Data(size_t Num)
250 : RIndex(Num, 0), Root(Num), C(Num ? Num - 1 : 0) {}
251};

253// Recursive variant of Pearce's algorithm (based on Algo 3 in the paper)
254template <typename G>
255static void
256pearce4VisitRec(const G &Graph, typename GraphTraits<G>::vertex_t V,
257 Pearce4Data<typename GraphTraits<G>::vertex_t> &Data,
258 SCCHolder<typename GraphTraits<G>::vertex_t> &Holder) {
259 using GTraits = psr::GraphTraits<G>;
260 using Vertex = typename GTraits::vertex_t;
261 using SCCId = psr::SCCId<Vertex>;
262
263 bool Root = true;
264 Data.RIndex[V] = Data.Index++;
265
266 for (const auto &Edge : GTraits::outEdges(Graph, V)) {
267 auto W = GTraits::target(Edge);
268 if (Data.RIndex[W] == 0) {
269 pearce4VisitRec(Graph, W, Data, Holder);
270 }
271 if (Data.RIndex[W] < Data.RIndex[V]) {
272 Data.RIndex[V] = Data.RIndex[W];
273 Root = false;
274 }
275 }
276
277 if (Root) {
278 Data.Index--;
279 auto NewSCC = SCCId(Holder.NodesInSCC.size());
280 auto &Nodes = Holder.NodesInSCC.emplace_back();
281
282 while (!Data.Stack.empty() &&
283 Data.RIndex[V] <= Data.RIndex[Data.Stack.back()]) {
284 auto W = Data.Stack.pop_back_val();
285 Data.RIndex[W] = Data.C;
286 Data.Index--;
287
288 Holder.SCCOfNode[W] = NewSCC;
289 Nodes.push_back(W);
290 }
291 Nodes.push_back(V);
292 Holder.SCCOfNode[V] = NewSCC;
293 Data.RIndex[V] = Data.C;
294 Data.C--;
295 } else {
296 Data.Stack.push_back(V);
297 }
298}

300// Iterative variant of Pearce's algorithm (adapted from on Algo 4 in the paper)
301template <typename G>
302static void
303pearce4VisitIt(const G &Graph, typename GraphTraits<G>::vertex_t Start,
304 Pearce4Data<typename GraphTraits<G>::vertex_t> &Data,
305 SCCHolder<typename GraphTraits<G>::vertex_t> &Holder) {
306 using GTraits = psr::GraphTraits<G>;
307 using Vertex = typename GTraits::vertex_t;
308 using SCCId = psr::SCCId<Vertex>;
309
310 using OutEdgeRange =
311 decltype(GTraits::outEdges(Graph, std::declval<Vertex>()));
312 using OutEdgeIterator =
313 decltype(llvm::adl_begin(std::declval<OutEdgeRange &>()));
314 using OutEdgeSentinel =
315 decltype(llvm::adl_end(std::declval<OutEdgeRange &>()));
316
317 struct DfsFrame {
318 Vertex CurrVtx;
319 OutEdgeIterator It;
320 [[no_unique_address]] OutEdgeSentinel End;
321 };
322
323 llvm::SmallVector<DfsFrame> CallStack;
324
325 const auto PushFrames = [&](Vertex V, DfsFrame *Frame) {
326 if (Frame->It == Frame->End) {
327 return false;
328 }
329 // Recurse into children until reaching the bottom
330 do {
331 auto W = GTraits::target(*Frame->It);
332
333 if (Data.RIndex[W] != 0) {
334 // Already pushed the children of W
335 break;
336 }
337
338 Data.RIndex[W] = Data.Index++;
339 Data.Root.insert(W);
340
341 auto &&OutEdges = GTraits::outEdges(Graph, W);
342 Frame = &CallStack.emplace_back(
343 DfsFrame{W, llvm::adl_begin(OutEdges), llvm::adl_end(OutEdges)});
344 V = W;
345
346 } while (Frame->It != Frame->End);
347
348 return true;
349 };
350
351 const auto VisitLoop = [&](Vertex V, DfsFrame &Frame) {
352 // Finish visiting the current child and advance to the next child
353 if (Frame.It != Frame.End) {
354 auto W = GTraits::target(*Frame.It);
355 if (Data.RIndex[W] < Data.RIndex[V]) {
356 Data.RIndex[V] = Data.RIndex[W];
357 Data.Root.erase(V);
358 }
359
360 ++Frame.It;
361 }
362 };
363
364 const auto FinishFrame = [&](Vertex V) {
365 // finish visiting V and backtrack to the parent
366
367 if (Data.Root.contains(V)) {
368 // Found a SCC
369
370 Data.Index--;
371 auto NewSCC = SCCId(Holder.NodesInSCC.size());
372 auto &Nodes = Holder.NodesInSCC.emplace_back();
373 while (!Data.Stack.empty() &&
374 Data.RIndex[V] <= Data.RIndex[Data.Stack.back()]) {
375 auto W = Data.Stack.pop_back_val();
376 Data.RIndex[W] = Data.C;
377 Data.Index--;
378
379 Holder.SCCOfNode[W] = NewSCC;
380 Nodes.push_back(W);
381 }
382 Nodes.push_back(V);
383 Holder.SCCOfNode[V] = NewSCC;
384 Data.RIndex[V] = Data.C;
385 Data.C--;
386 } else {
387 Data.Stack.push_back(V);
388 }
389
390 CallStack.pop_back();
391 };
392
393 // Initialize the callstack by pushing the initial frame
394 Data.RIndex[Start] = Data.Index++;
395 Data.Root.insert(Start);
396 {
397 auto &&OutEdges = GTraits::outEdges(Graph, Start);
398 static_assert(
399 std::is_lvalue_reference_v<decltype(OutEdges)> ||
400 std::is_trivially_destructible_v<std::decay_t<decltype(OutEdges)>>,
401 "We assume that outEdges gives either a reference or a view into "
402 "the out-edges, but never an owning container by value. Otherwise, "
403 "the DFSFrame iterators may be dangling");
404 CallStack.emplace_back(
405 DfsFrame{Start, llvm::adl_begin(OutEdges), llvm::adl_end(OutEdges)});
406 }
407
408 // Simulate the recursion
409
410 PushFrames(Start, &CallStack.back());
411 while (true) {
412 auto &Frame = CallStack.back();
413 Vertex V = Frame.CurrVtx;
414 VisitLoop(V, Frame);
415 if (PushFrames(V, &Frame)) {
416 continue; // we don't pop from the callstack here
417 }
418
419 FinishFrame(V);
420 if (CallStack.empty()) {
421 break;
422 }
423 }
424}

428/// Compute SCCs adapted from the paper "A Space-Efficient Algorithm for Finding
429/// Strongly Connected Components", Pearce 2015, DOI:
430/// <https://doi.org/10.1016/j.ipl.2015.08.010>
431///
432/// \tparam G The graph-type
433/// \tparam Iterative Whether to use the iterative or recursive variant of the
434/// algorithm (default: true)
435/// \param Graph The graph for with to compute SCCs and topological ordering
436template <typename G, bool Iterative = true>
437[[nodiscard]] SCCHolder<typename GraphTraits<G>::vertex_t>
438computeSCCs(const G &Graph, std::bool_constant<Iterative> /*Iterative*/ = {}) {
439 using GTraits = psr::GraphTraits<G>;
440 using Vertex = typename GTraits::vertex_t;
441
442 SCCHolder<Vertex> Ret;
443 auto N = GTraits::size(Graph);
444 if (!N) {
445 return Ret;
446 }
447
448 Ret.SCCOfNode.resize(N);
449
450 detail::Pearce4Data<Vertex> Data(N);
451
452 // for all v ∈ V do if rindex[v]==0 then visit(v)
453 for (auto V : GTraits::vertices(Graph)) {
454 if (Data.RIndex[V] == 0) {
455 if constexpr (Iterative) {
456 detail::pearce4VisitIt(Graph, V, Data, Ret);
457 } else {
458 detail::pearce4VisitRec(Graph, V, Data, Ret);
459 }
460
461 if (!Data.Stack.empty()) {
462 auto NewSCC = SCCId<Vertex>(Ret.NodesInSCC.size());
463 auto &Nodes = Ret.NodesInSCC.emplace_back();
464 Nodes.reserve(Data.Stack.size());
465 for (auto Vtx : Data.Stack) {
466 Nodes.push_back(Vtx);
467 Ret.SCCOfNode[Vtx] = NewSCC;
468 }
469 Data.Stack.clear();
470 }
471 }
472 }
473
474 return Ret;
475}

477/// Compute SCCs and a topological ordering on the SCCs, adapted from the paper
478/// "A Space-Efficient Algorithm for Finding Strongly Connected Components",
479/// Pearce 2015, DOI: <https://doi.org/10.1016/j.ipl.2015.08.010>
480///
481/// \tparam G The graph-type \tparam Iterative Whether to use the iterative
482/// or recursive variant of the algorithm (default: true) \param Graph The graph
483/// for with to compute SCCs and topological ordering
484template <typename G, bool Iterative = true>
485[[nodiscard]] std::pair<SCCHolder<typename GraphTraits<G>::vertex_t>,
486 SCCOrder<typename GraphTraits<G>::vertex_t>>
487computeSCCsAndTopologicalOrder(
488 const G &Graph, std::bool_constant<Iterative> /*Iterative*/ = {}) {
489 using Vertex = typename GraphTraits<G>::vertex_t;
490
491 std::pair<SCCHolder<Vertex>, SCCOrder<Vertex>> Ret = {
492 computeSCCs(Graph, std::bool_constant<Iterative>{}),
493 {},
494 };
495
496 // Pearce's algorithm produces SCCs in reverse topological order
497 auto Ids = llvm::reverse(psr::iota<SCCId<Vertex>>(Ret.first.size()));
498 Ret.second.SCCIds.append(Ids.begin(), Ids.end());
499
500 return Ret;
501}
