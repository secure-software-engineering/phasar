32/// Represents the infimum of the lattice:
33/// Top is the greatest element that is less than or equal to all elements of
34/// the lattice.
35struct Top {
36 friend constexpr bool operator==(Top /*unused*/, Top /*unused*/) noexcept {
37 return true;
38 }
39 friend constexpr bool operator!=(Top /*unused*/, Top /*unused*/) noexcept {
40 return false;
41 }
42 friend llvm::raw_ostream &operator<<(llvm::raw_ostream &OS, Top /*unused*/) {
43 return OS << "Top";
44 }
45};

65/// A easy shorthand to construct a complete lattice of L.
66template <typename L>
67struct LatticeDomain : public std::variant<Top, L, Bottom> {
68 using std::variant<Top, L, Bottom>::variant;
69
70 [[nodiscard]] constexpr bool isBottom() const noexcept {
71 return std::holds_alternative<Bottom>(*this);
72 }
73 [[nodiscard]] constexpr bool isTop() const noexcept {
74 return std::holds_alternative<Top>(*this);
75 }
76
77 [[nodiscard]] constexpr L *getValueOrNull() noexcept {
78 return std::get_if<L>(this);
79 }
80 [[nodiscard]] constexpr const L *getValueOrNull() const noexcept {
81 return std::get_if<L>(this);
82 }
83
84 friend llvm::hash_code hash_value(const LatticeDomain &LD) noexcept
85 requires is_llvm_hashable_v<L>
86 { // NOLINT
87 if (LD.isBottom()) {
88 return llvm::hash_value(INTPTR_MAX);
89 }
90 if (LD.isTop()) {
91 return llvm::hash_value(INTPTR_MIN);
92 }
93 return hash_value(std::get<L>(LD));
94 }
95
96 [[nodiscard]] constexpr L &assertGetValue() noexcept {
97 assert(std::holds_alternative<L>(*this));
98 return std::get<L>(*this);
99 }
100 [[nodiscard]] constexpr const L &assertGetValue() const noexcept {
101 assert(std::holds_alternative<L>(*this));
102 return std::get<L>(*this);
103 }
104
105 template <typename TransformFn, typename... ArgsT>
106 constexpr void onValue(TransformFn Transform, ArgsT &&...Args) {
107 if (auto *Val = getValueOrNull()) {
108 std::invoke(std::move(Transform), *Val, PSR_FWD(Args)...);
109 }
110 }
111
112 friend llvm::raw_ostream &operator<<(llvm::raw_ostream &OS,
113 const LatticeDomain &LD) {
114 if (LD.isBottom()) {
115 return OS << "Bottom";
116 }
117 if (LD.isTop()) {
118 return OS << "Top";
119 }
120
121 const auto *Val = LD.getValueOrNull();
122 assert(Val && "Only alternative remaining is L");
123 if constexpr (is_llvm_printable_v<L>) {
124 return OS << *Val;
125 } else {
126 return OS << PrettyPrinter{*Val};
127 }
128 }
129
130 friend std::ostream &operator<<(std::ostream &OS, const LatticeDomain &LD) {
131 llvm::raw_os_ostream ROS(OS);
132 ROS << LD;
133 return OS;
134 }
135
136 constexpr bool operator==(const LatticeDomain &Rhs) const {
137 if (this->index() != Rhs.index()) {
138 return false;
139 }
140 if (auto LhsPtr = this->getValueOrNull()) {
141 /// No need to check whether Rhs is an L; the indices are already the same
142 return *LhsPtr == *Rhs.getValueOrNull();
143 }
144 return true;
145 }
146
147 template <typename LL>
148 requires AreEqualityComparable<L, LL>
149 constexpr bool operator==(const LL &Rhs) const {
150 if (auto LVal = this->getValueOrNull()) {
151 return *LVal == Rhs;
152 }
153 return false;
154 }
155
156 constexpr bool operator==(Bottom /*Rhs*/) const noexcept {
157 return this->isBottom();
158 }
159
160 constexpr bool operator==(Top /*Rhs*/) const noexcept {
161 return this->isTop();
162 }
163
164 constexpr bool operator<(const LatticeDomain &Rhs) const {
165 /// Top < (Lhs::L < Rhs::L) < Bottom
166 if (Rhs.isTop()) {
167 return false;
168 }
169 if (this->isTop()) {
170 return true;
171 }
172 if (auto LhsPtr = this->getValueOrNull()) {
173 if (auto RhsPtr = Rhs.getValueOrNull()) {
174 return *LhsPtr < *RhsPtr;
175 }
176 } else if (this->isBottom()) {
177 return false;
178 }
179 if (Rhs.isBottom()) {
180 return true;
181 }
182 llvm_unreachable("All comparison cases should be handled above.");
183 }
184};

186template <typename L> struct JoinLatticeTraits<LatticeDomain<L>> {
187 using l_t = L;
188 static constexpr Bottom bottom() noexcept { return {}; }
189 static constexpr Top top() noexcept { return {}; }
190 static constexpr LatticeDomain<L> join(ByConstRef<LatticeDomain<l_t>> LHS,
191 ByConstRef<LatticeDomain<l_t>> RHS) {
192 // Top < (Lhs::l_t < Rhs::l_t) < Bottom
193 if (LHS.isTop() || LHS == RHS) {
194 return RHS;
195 }
196
197 if (RHS.isTop()) {
198 return LHS;
199 }
200
201 if constexpr (has_adl_join<l_t>) {
202 if (auto LhsPtr = LHS.getValueOrNull()) {
203 if (auto RhsPtr = RHS.getValueOrNull()) {
204 return psr::adl_join(*LhsPtr, *RhsPtr);
205 }
206 }
207 }
208
209 return Bottom{};
210 }
211};

213/// If we know that a stored L value is never Top or Bottom, we don't need to
214/// store the discriminator of the std::variant.
215template <typename L>
216 requires(std::is_nothrow_constructible_v<LatticeDomain<L>, const L &> ||
217 !std::is_nothrow_copy_constructible_v<LatticeDomain<L>>)
218struct NonTopBotValue<LatticeDomain<L>> {
219 using type = L;
220
221 constexpr static L unwrap(LatticeDomain<L> Value) noexcept(
222 std::is_nothrow_move_constructible_v<L>) {
223 return std::get<L>(std::move(Value));
224 }

230template <typename L> struct hash<psr::LatticeDomain<L>> {
231 constexpr size_t operator()(const psr::LatticeDomain<L> &LD) noexcept {
232 if (LD.isBottom()) {
233 return SIZE_MAX;
234 }
235 if (LD.isTop()) {
236 return SIZE_MAX - 1;
237 }
238 assert(LD.getValueOrNull() != nullptr);
239 if constexpr (psr::is_std_hashable_v<L>) {
240 return std::hash<L>{}(*LD.getValueOrNull());
241 } else {
242 using llvm::hash_value;
243 return hash_value(*LD.getValueOrNull());
244 }
245 }
246};
