39/// Half-open interval [Lo, Hi) of ranks.
40struct RankInterval {
41 [[nodiscard]] constexpr bool empty() const noexcept { return Lo >= Hi; }
42 [[nodiscard]] constexpr size_t size() const noexcept {
43 return empty() ? 0 : to_underlying(Hi) - to_underlying(Lo);
44 }
45 [[nodiscard]] constexpr bool contains(ReachRank Rank) const noexcept {
46 return Lo <= Rank && Rank < Hi;
47 }
48
49 [[nodiscard]] friend constexpr bool
50 operator==(RankInterval Lhs, RankInterval Rhs) noexcept = default;
51
52 ReachRank Lo{};
53 ReachRank Hi{};
54};

58/// Reachability index over a directed graph: answers whether a vertex reaches
59/// a target vertex in O(log k), where k is the number of rank intervals of the
60/// source (usually small).
61///
62/// Built by IntervalReachabilityBuilder
63template <SmallIdType VtxId> class IntervalReachabilityIndex {
64 friend class IntervalReachabilityBuilder<VtxId>;
65
66public:
67 using SCCIdTy = SCCId<VtxId>;
68
69 static constexpr SCCIdTy NoSCC = SCCIdTy(UINT32_MAX);
70
71 [[nodiscard]] size_t numVertices() const noexcept { return SCCOf.size(); }
72 [[nodiscard]] size_t numTargets() const noexcept {
73 return RankToVertex.size();
74 }
75 [[nodiscard]] size_t numSCCs() const noexcept { return SCCRange.size(); }
76
77 /// Whether Vtx was reachable from the roots at build time.
78 [[nodiscard]] bool isIndexed(VtxId Vtx) const noexcept {
79 return SCCOf[Vtx] != NoSCC;
80 }
81
82 // A sink node
83 [[nodiscard]] bool isTarget(VtxId Vtx) const noexcept {
84 return RankOf[Vtx] != ReachRank::None;
85 }
86
87 /// The rank of Vtx, or ReachRank::None if Vtx is not a target.
88 [[nodiscard]] ReachRank rankOf(VtxId Vtx) const noexcept {
89 return RankOf[Vtx];
90 }
91
92 [[nodiscard]] VtxId vertexOfRank(ReachRank Rank) const noexcept {
93 return RankToVertex[Rank];
94 }
95
96 /// The targets covered by the given rank interval.
97 [[nodiscard]] llvm::ArrayRef<VtxId>
98 targetsIn(RankInterval Interval) const noexcept {
99 if (Interval.empty()) {
100 return {};
101 }
102 return {&RankToVertex[Interval.Lo], Interval.size()};
103 }
104
105 /// The strongly connected component of Vtx, or NoSCC if Vtx is not indexed.
106 /// SCC ids are in reverse topological order: an SCC only reaches SCCs with
107 /// smaller or equal id.
108 [[nodiscard]] SCCIdTy sccOf(VtxId Vtx) const noexcept { return SCCOf[Vtx]; }
109
110 /// Whether From reaches the target To. False, if From is not indexed or To
111 /// is not a target.
112 [[nodiscard]] bool reaches(VtxId From, VtxId To) const noexcept {
113 auto SCC = SCCOf[From];
114 auto Rank = RankOf[To];
115 if (SCC == NoSCC || Rank == ReachRank::None) {
116 return false;
117 }
118 return sccReaches(SCC, Rank);
119 }
120
121 /// Whether the given SCC reaches the target with the given rank.
122 [[nodiscard]] bool sccReaches(SCCIdTy SCC, ReachRank Rank) const noexcept {
123 if (SCCRange[SCC].contains(Rank)) {
124 return true;
125 }
126
127 auto Extras = extrasOf(SCC);
128 const auto *It = std::upper_bound(
129 Extras.begin(), Extras.end(), Rank,
130 [](ReachRank R, RankInterval Interval) { return R < Interval.Lo; });
131 return It != Extras.begin() && Rank < std::prev(It)->Hi;
132 }
133
134 /// Calls Handler for each interval of target ranks reachable from From, in
135 /// ascending order. The intervals are disjoint and non-empty.
136 void forEachReachableRange(VtxId From,
137 std::invocable<RankInterval> auto Handler) const {
138 auto SCC = SCCOf[From];
139 if (SCC == NoSCC) {
140 return;
141 }
142 for (auto Interval : extrasOf(SCC)) {
143 std::invoke(Handler, Interval);
144 }
145 if (!SCCRange[SCC].empty()) {
146 std::invoke(Handler, SCCRange[SCC]);
147 }
148 }
149
150 /// Calls Handler for each target reachable from From, in rank order.
151 void forEachReachableTarget(VtxId From,
152 std::invocable<VtxId> auto Handler) const {
153 forEachReachableRange(From, [&](RankInterval Interval) {
154 for (auto Vtx : targetsIn(Interval)) {
155 std::invoke(Handler, Vtx);
156 }
157 });
158 }
159
160 /// Number of rank intervals describing the targets reachable from SCC.
161 [[nodiscard]] size_t numIntervalsOf(SCCIdTy SCC) const noexcept {
162 return extrasOf(SCC).size() + size_t(!SCCRange[SCC].empty());
163 }
164
165 [[nodiscard]] size_t getApproxSizeInBytes() const noexcept {
166 return SCCOf.capacity() * sizeof(SCCIdTy) +
167 RankOf.capacity() * sizeof(ReachRank) +
168 RankToVertex.capacity() * sizeof(VtxId) +
169 SCCRange.capacity() * sizeof(RankInterval) +
170 SCCExtras.capacity() * sizeof(IntervalSetId) +
171 IntervalPool.capacity_in_bytes() +
172 SetOffsets.capacity() * sizeof(uint32_t);
173 }
174
175private:
176 [[nodiscard]] llvm::ArrayRef<RankInterval>
177 extrasOf(SCCIdTy SCC) const noexcept {
178 return csrSuccs(SetOffsets, IntervalPool, SCCExtras[SCC]);
179 }
180
181 void reset(size_t NumVertices) {
182 SCCOf.assign(NumVertices, NoSCC);
183 RankOf.assign(NumVertices, ReachRank::None);
184 RankToVertex.clear();
185 SCCRange.clear();
186 SCCExtras.clear();
187 IntervalPool.clear();
188 SetOffsets.clear();
189 // IntervalSetId::Empty
190 SetOffsets.push_back(0);
191 SetOffsets.push_back(0);
192 }
193
194 TypedVector<VtxId, SCCIdTy> SCCOf;
195 TypedVector<VtxId, ReachRank> RankOf;
196 TypedVector<ReachRank, VtxId> RankToVertex;
197
198 /// Ranks of the targets in the DFS subtree of the SCC's root
199 TypedVector<SCCIdTy, RankInterval> SCCRange;
200 /// Targets reachable via non-tree edges. Their intervals all lie below
201 /// SCCRange[SCC].Lo: targets with higher ranks are discovered after the SCC
202 /// is complete, so they are not reachable from it.
203 TypedVector<SCCIdTy, IntervalSetId> SCCExtras;
204
205 /// Interval sets in CSR format
206 llvm::SmallVector<RankInterval, 0> IntervalPool;
207 TypedVector<IntervalSetId, uint32_t> SetOffsets;
208};

210/// Computes an IntervalReachabilityIndex during a single DFS (Pearce's SCC
211/// algorithm, see visitSCCsFrom()).
212///
213/// Owns scratch buffers that are reused across build() calls; keep the builder
214/// alive when indexing many (small) graphs.
215///
216/// Complexity: O(V + E + total size of the merged interval sets).
217template <SmallIdType VtxId> class IntervalReachabilityBuilder {
218public:
219 using IndexTy = IntervalReachabilityIndex<VtxId>;
220 using SCCIdTy = typename IndexTy::SCCIdTy;
221
222 /// Indexes all vertices of Graph reachable from Roots, all of them as
223 /// targets. The ranks are then DFS pre-order numbers.
224 template <is_const_graph G>
225 requires std::same_as<typename GraphTraits<G>::vertex_t, VtxId>
226 [[nodiscard]] IndexTy build(const G &Graph, llvm::ArrayRef<VtxId> Roots) {
227 return build(Graph, Roots, /*IsTarget=*/psr::TrueFn{});
228 }
229
230 template <is_const_graph G>
231 requires std::same_as<typename GraphTraits<G>::vertex_t, VtxId>
232 void build(const G &Graph, llvm::ArrayRef<VtxId> Roots, IndexTy &Out) {
233 build(Graph, Roots, Out, /*IsTarget=*/psr::TrueFn{});
234 }
235
236 /// Indexes all vertices of Graph reachable from Roots, only those
237 /// satisfying IsTarget as targets.
238 ///
239 /// Select few targets if only reachability to those is of interest: the
240 /// interval sets shrink accordingly.
241 template <is_const_graph G>
242 requires std::same_as<typename GraphTraits<G>::vertex_t, VtxId>
243 void build(const G &Graph, llvm::ArrayRef<VtxId> Roots, IndexTy &Out,
244 std::predicate<VtxId> auto IsTarget) {
245 auto NumVertices = GraphTraits<G>::size(Graph);
246 Out.reset(NumVertices);
247 SCCData.reset(NumVertices);
248 RankBegin.resize(NumVertices);
249 SeenBy.clear();
250
251 const auto OnDiscover = [&](VtxId Vtx) {
252 auto Rank = ReachRank(Out.RankToVertex.size());
253 RankBegin[Vtx] = Rank;
254 if (std::invoke(IsTarget, Vtx)) {
255 Out.RankOf[Vtx] = Rank;
256 Out.RankToVertex.push_back(Vtx);
257 }
258 };
259 const auto OnSCC = [&](SCCIdTy SCC, llvm::ArrayRef<VtxId> Members) {
260 completeSCC(Graph, SCC, Members, Out);
261 };
262
263 for (auto Root : Roots) {
264 if (!SCCData.isVisited(Root)) {
265 visitSCCsFrom(Graph, Root, SCCData, OnDiscover, OnSCC);
266 }
267 }
268 }
269
270 /// Indexes all vertices of Graph reachable from Roots, only those
271 /// satisfying IsTarget as targets.
272 template <is_const_graph G>
273 requires std::same_as<typename GraphTraits<G>::vertex_t, VtxId>
274 [[nodiscard]] IndexTy build(const G &Graph, llvm::ArrayRef<VtxId> Roots,
275 std::predicate<VtxId> auto IsTarget) {
276 IndexTy Out{};
277 build(Graph, Roots, Out, copyOrRef(IsTarget));
278 return Out;
279 }
280
281private:
282 template <typename G>
283 void completeSCC(const G &Graph, SCCIdTy SCC, llvm::ArrayRef<VtxId> Members,
284 IndexTy &Out) {
285 assert(size_t(SCC) == Out.SCCRange.size());
286
287 // Members.back() is the root of the SCC in the DFS tree. Its subtree is
288 // completely discovered.
289 auto Range = RankInterval{
290 .Lo = RankBegin[Members.back()],
291 .Hi = ReachRank(Out.RankToVertex.size()),
292 };
293
294 for (auto Member : Members) {
295 Out.SCCOf[Member] = SCC;
296 }
297 Out.SCCRange.push_back(Range);
298 SeenBy.push_back(SCC);
299
300 // All successor SCCs are complete, as SCCs complete in reverse
301 // topological order
302 SuccSCCs.clear();
303 for (auto Member : Members) {
304 for (const auto &Edge : GraphTraits<G>::outEdges(Graph, Member)) {
305 auto SuccSCC = Out.SCCOf[GraphTraits<G>::target(Edge)];
306 if (SuccSCC == SCC || SeenBy[SuccSCC] == SCC) {
307 continue;
308 }
309 SeenBy[SuccSCC] = SCC;
310 SuccSCCs.push_back(SuccSCC);
311 }
312 }
313
314 Out.SCCExtras.push_back(mergeExtras(Range.Lo, Out));
315 }
316
317 /// Computes the extras-set of an SCC whose range starts at Lo from the
318 /// interval sets of its successor SCCs.
319 ///
320 /// The range of a successor SCC lies either inside the own range (DFS-tree
321 /// descendant) or entirely below Lo (cross edge). Successor extras at or
322 /// above Lo are covered by the own range, so they are clipped at Lo.
323 [[nodiscard]] IntervalSetId mergeExtras(ReachRank Lo, IndexTy &Out) {
324 const auto IsCrossRange = [Lo](RankInterval Range) {
325 return !Range.empty() && Range.Hi <= Lo;
326 };
327
328 // Fast path for chains and loop bodies: a single contributing descendant,
329 // whose extras need no clipping
330 size_t NumContributing = 0;
331 std::optional<IntervalSetId> Reusable;
332 for (auto SuccSCC : SuccSCCs) {
333 bool IsCross = IsCrossRange(Out.SCCRange[SuccSCC]);
334 auto Extras = Out.extrasOf(SuccSCC);
335 if (!IsCross && Extras.empty()) {
336 continue;
337 }
338
339 ++NumContributing;
340 Reusable = std::nullopt;
341 if (!IsCross && Extras.back().Hi <= Lo) {
342 Reusable = Out.SCCExtras[SuccSCC];
343 }
344 }
345 if (NumContributing == 0) {
346 return IntervalSetId::Empty;
347 }
348 if (NumContributing == 1 && Reusable) {
349 return *Reusable;
350 }
351
352 Candidates.clear();
353 for (auto SuccSCC : SuccSCCs) {
354 auto Range = Out.SCCRange[SuccSCC];
355 if (IsCrossRange(Range)) {
356 Candidates.push_back(Range);
357 }
358 for (auto Interval : Out.extrasOf(SuccSCC)) {
359 if (Interval.Lo >= Lo) {
360 break;
361 }
362 Candidates.push_back({Interval.Lo, std::min(Interval.Hi, Lo)});
363 }
364 }
365
366 llvm::sort(Candidates, [](RankInterval Lhs, RankInterval Rhs) {
367 return Lhs.Lo < Rhs.Lo;
368 });
369
370 auto Ret = IntervalSetId(Out.SetOffsets.size() - 1);
371 for (auto Interval : Candidates) {
372 // Coalesce overlapping and adjacent intervals
373 if (Out.IntervalPool.size() != Out.SetOffsets.asRef().back() &&
374 Interval.Lo <= Out.IntervalPool.back().Hi) {
375 Out.IntervalPool.back().Hi =
376 std::max(Out.IntervalPool.back().Hi, Interval.Hi);
377 continue;
378 }
379 Out.IntervalPool.push_back(Interval);
380 }
381 Out.SetOffsets.push_back(uint32_t(Out.IntervalPool.size()));
382 return Ret;
383 }
384
385 PearceSCCData<VtxId> SCCData;
386 TypedVector<VtxId, ReachRank> RankBegin;
387
388 /// The last SCC that has seen this SCC as successor
389 TypedVector<SCCIdTy, SCCIdTy> SeenBy;
390
391 llvm::SmallVector<SCCIdTy, 0> SuccSCCs;
392 llvm::SmallVector<RankInterval, 0> Candidates;
393};
