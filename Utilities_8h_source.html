41template <typename T>
42std::set<std::set<T>> computePowerSet(const std::set<T> &S) {
43 // compute all subsets of {a, b, c, d}
44 // bit-pattern - {d, c, b, a}
45 // 0000 {}
46 // 0001 {a}
47 // 0010 {b}
48 // 0011 {a, b}
49 // 0100 {c}
50 // 0101 {a, c}
51 // 0110 {b, c}
52 // 0111 {a, b, c}
53 // 1000 {d}
54 // 1001 {a, d}
55 // 1010 {b, d}
56 // 1011 {a, b, d}
57 // 1100 {c, d}
58 // 1101 {a, c, d}
59 // 1110 {b, c, d}
60 // 1111 {a, b, c, d}
61 std::set<std::set<T>> Powerset;
62 for (std::size_t I = 0; I < (1 << S.size()); ++I) {
63 std::set<T> Subset;
64 for (std::size_t J = 0; J < S.size(); ++J) {
65 if ((I & (1 << J)) > 0) {
66 auto It = S.begin();
67 advance(It, J);
68 Subset.insert(*It);
69 }
70 Powerset.insert(Subset);
71 }
72 }
73 return Powerset;
74}

76/// \brief Computes the set-intersection of the potentially unordered sets
77/// Dest and Src and stores the result back in Dest.
78///
79/// This function should work on all types of sets as long as they provide the
80/// type value_type specifying, which type their elements have.
81/// By-reference iteration is required, but the elements do not have any
82/// requirements, although the performance is probably higher for small
83/// elements that are trivially copyable.
84template <typename ContainerTy, typename OtherContainerTy>
85 requires(!has_erase_iterator_v<ContainerTy>)
86void intersectWith(ContainerTy &Dest, const OtherContainerTy &Src) {
87 static_assert(std::is_same_v<typename ContainerTy::value_type,
88 typename OtherContainerTy::value_type>,
89 "The containers Src and Dest must be compatible");
90 using ValueTy = typename ContainerTy::value_type;
91 using ElementTy =
92 std::conditional_t<std::is_trivially_copy_constructible_v<ValueTy> &&
93 sizeof(ValueTy) <= sizeof(void *),
94 ValueTy, ValueTy *>;
95
96 auto removeFrom = [](auto &Dst, auto &&Elem) { // NOLINT
97 if constexpr (std::is_same_v<ValueTy, ElementTy>) {
98 Dst.erase(Elem);
99 } else {
100 Dst.erase(*Elem);
101 }
102 };
103
104 /// This whole functionality is only for computing the set-intersection of
105 /// Dest and Src storing the result in-place in Dest. It would be a big win,
106 /// if the STL would provide us with in-place set operations and those that
107 /// do not require the sets to be sorted...
108
109 llvm::SmallVector<ElementTy, 16> Buffer;
110
111 if constexpr (std::is_same_v<ValueTy, ElementTy>) {
112 for (auto &&Elem : Dest) {
113 if (!Src.count(Elem)) {
114 Buffer.push_back(Elem);
115 }
116 }
117 } else {
118 for (auto &Elem : Dest) {
119 if (!Src.count(Elem)) {
120 Buffer.insert(&Elem);
121 }
122 }
123 }
124
125 for (auto &&Elem : Buffer) {
126 removeFrom(Dest, Elem);
127 }
128}

130template <has_erase_iterator_v ContainerTy, typename OtherContainerTy>
131void intersectWith(ContainerTy &Dest, const OtherContainerTy &Src) {
132 static_assert(std::is_same_v<typename ContainerTy::value_type,
133 typename OtherContainerTy::value_type>,
134 "The containers Src and Dest must be compatible");
135
136 for (auto It = Dest.begin(), End = Dest.end(); It != End;) {
137 if (Src.count(*It)) {
138 ++It;
139 } else {
140 if constexpr (std::is_void_v<decltype(Dest.erase(It))>) {
141 auto OldIt = It++;
142 Dest.erase(OldIt);
143 } else {
144 It = Dest.erase(It);
145 }
146 }
147 }
148}

186/// Based on the reference implementation of std::remove_if
187/// <https://en.cppreference.com/w/cpp/algorithm/remove> and optimized for the
188/// case that a sorted list of indices is given instead of an unary predicate
189/// specifying the elements to be removed.
190template <typename It, typename EndIt, typename IdxIt,
191 typename IdxEndIt> // NOLINTNEXTLINE(readability-identifier-naming)
192It remove_by_index(It First, EndIt Last, IdxIt FirstIndex, IdxEndIt LastIndex) {
193 if (FirstIndex == LastIndex || First == Last) {
194 return Last;
195 }
196
197 if constexpr (std::is_same_v<It, EndIt> &&
198 std::is_same_v<
199 std::random_access_iterator_tag,
200 typename std::iterator_traits<It>::iterator_category>) {
201 /// Random-access version inspired from
202 /// "https://codereview.stackexchange.com/a/207056" and slightly changed
203
204 auto Bounds = size_t(std::distance(First, Last));
205 auto Out = std::next(First, *FirstIndex);
206 auto In = std::next(Out);
207 while (++FirstIndex != LastIndex) {
208 auto CurrIdx = *FirstIndex;
209 if (*std::prev(FirstIndex) + 1 == CurrIdx) {
210 ++In;
211 continue;
212 }
213 if (LLVM_UNLIKELY(CurrIdx >= Bounds)) {
214 break;
215 }
216
217 auto Tar = std::next(First, CurrIdx);
218 Out = std::move(In, Tar, Out);
219 In = std::next(Tar);
220 }
221 return std::move(In, Last, Out);
222 } else {
223 auto CurrIdx = *FirstIndex;
224 First = std::next(First, CurrIdx);
225 if (First == Last) {
226 return First;
227 }
228 for (auto I = First; I != Last; ++CurrIdx, ++I) {
229 if (CurrIdx != *FirstIndex) {
230 *First++ = std::move(*I);
231 if (++FirstIndex == LastIndex) {
232 return std::move(std::next(I), Last, First);
233 }
234 }
235 }
236 return First;
237 }
238}

262forward_like(U &&X) noexcept { // NOLINT
263 // NOLINTNEXTLINE
264 constexpr bool is_adding_const = std::is_const_v<std::remove_reference_t<T>>;
265 if constexpr (std::is_lvalue_reference_v<T &&>) {
266 if constexpr (is_adding_const) {

347template <typename CompareFn = std::less<>>
348constexpr void sortUnique(auto &Range, CompareFn Cmp = {}) {
349 auto It = llvm::adl_begin(Range);
350 auto End = llvm::adl_end(Range);
351 std::sort(It, End, std::move(Cmp));
352 Range.erase(std::unique(It, End), End);
353}
