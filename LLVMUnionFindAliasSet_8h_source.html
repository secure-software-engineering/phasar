44/// Concrete \c IsAliasInfo implementation backed by a union-find alias
45/// analysis.
46///
47/// Constructed once from an \c LLVMProjectIRDB and a call graph; the chosen
48/// analysis variant is determined by \c Config. Alias sets are materialized
49/// lazily on first query and cached per \c ValueId in \c AliasSets.
50///
51/// \note When \c AnalysisLocality::FunctionLocal is selected, alias sets are
52/// filtered to variables visible in the function that contains the query
53/// instruction. The per-\c ValueId cache does **not** account for the
54/// instruction context, so the first caller's function wins — do not mix
55/// queries to local pointers from different functions for the same value in
56/// local mode.
57class LLVMUnionFindAliasSet
58 : public AnalysisPropertiesMixin<LLVMUnionFindAliasSet> {
59public:
60 using traits_t = AliasInfoTraits<LLVMUnionFindAliasSet>;
61 using n_t = traits_t::n_t;
62 using v_t = traits_t::v_t;
63 using AliasSetTy = traits_t::AliasSetTy;
64 using AliasSetPtrTy = traits_t::AliasSetPtrTy;
65 using AllocationSiteSetPtrTy = traits_t::AllocationSiteSetPtrTy;
66
67 /// Whether alias sets are reported globally or filtered to the function
68 /// containing the query instruction.
69 enum class AnalysisLocality : uint8_t {
70 /// All aliases across all functions are reported.
71 Global,
72 /// Aliases are intersected with variables visible in the querying function
73 /// (globals + function-local values).
74 FunctionLocal,
75 };
76
77 struct Config {
78 /// The specific union-find analysis variant to run (default:
79 /// \c BotCtxIndSens — bottom-up, context- and indirection-sensitive).
80 UnionFindAliasAnalysisType AType =
81 UnionFindAliasAnalysisType::BotCtxIndSens;
82 /// Controls whether alias sets are scoped to the querying function.
83 AnalysisLocality ALocality = AnalysisLocality::Global;
84 };

122 [[nodiscard]] AliasSetPtrTy getAliasSet(v_t V, n_t I) {
123 assert(isValid());
124 auto ValId = AARes->VC->getOrNull(V);
125 if (!ValId) {
126 return getEmptyAliasSet();
127 }
128
129 assert(AliasSets.inbounds(*ValId));
130 if (!AliasSets[*ValId]) [[unlikely]] {
131 AliasSets[*ValId] = AARes->constructAliasSet(*ValId, I, Owner);
132 }
133
134 return AliasSets[*ValId];
135 }
136
137 [[nodiscard]] AllocationSiteSetPtrTy
138 getReachableAllocationSites(v_t V, bool IntraProcOnly, n_t I) const {
139 assert(isValid());
140 auto ValId = AARes->VC->getOrNull(V);
141 if (!ValId) {
142 return std::make_unique<AliasSetTy>();
143 }
144
145 return AARes->constructReachableAllocSites(V, *ValId, IntraProcOnly, I);
146 }
147
148 [[nodiscard]] bool isInReachableAllocationSites(
149 const llvm::Value *V, const llvm::Value *PotentialValue,
150 bool IntraProcOnly, const llvm::Instruction *I) const {
151 assert(isValid());
152 if (!psr::isInterestingPointer(V)) {
153 return false;
154 }
155
156 if (!psr::isInReachableAllocationSitesTy(V, PotentialValue,
157 IntraProcOnly)) {
158 return false;
159 }
160
161 return alias(V, PotentialValue, I) != AliasResult::NoAlias;
162 }
