97 /// \brief Calls Handler for all operands of Inst (maybe including Inst
98 /// itself) that are generated unconditionally as tainted.
99 ///
100 /// If Inst is a function-call, the Callee function should be specified
101 /// explicitly.
102 template <typename HandlerFn>
103 void forAllGeneratedValuesAt(n_t Inst, Nullable<f_t> Callee,
104 HandlerFn &&Handler) const {
105 self().forAllGeneratedValuesAtImpl(std::move(Inst), std::move(Callee),
106 std::forward<HandlerFn>(Handler));
107 }
108
109 /// \brief Calls Handler for all operands of Inst that may generate a leak
110 /// when they are tainted.
111 ///
112 /// If Inst is a function-call, the Callee function should be specified
113 /// explicitly.
114 template <typename HandlerFn>
115 void forAllLeakCandidatesAt(n_t Inst, Nullable<f_t> Callee,
116 HandlerFn &&Handler) const {
117 self().forAllLeakCandidatesAtImpl(std::move(Inst), std::move(Callee),
118 std::forward<HandlerFn>(Handler));
119 }
120
121 /// \brief Calls Handler for all operands of Inst that become sanitized after
122 /// the instruction is completed.
123 ///
124 /// If Inst is a function-call, the Callee function should be specified
125 /// explicitly.
126 template <typename HandlerFn>
127 void forAllSanitizedValuesAt(n_t Inst, Nullable<f_t> Callee,
128 HandlerFn &&Handler) const {
129 self().forAllSanitizedValuesAtImpl(std::move(Inst), std::move(Callee),
130 std::forward<HandlerFn>(Handler));
131 }
