32/// \brief A set-type that can compactly store sets of sequential integer-like
33/// types.
34///
35/// Use this type for sequential (unsigned) integers and ids that can convert
36/// from and to uint32_t.
37///
38/// \tparam IdT The type of elements to store in this set. Must be losslessly
39/// convertible from and to uint32_t.
40/// \tparam BitVectorTy The underlying bit-vector to use. Must be either
41/// llvm::BitVector or llvm::SmallBitVector.
42template <SmallIdType IdT, typename BitVectorTy = llvm::BitVector>
43class BitSet {
44 static llvm::ArrayRef<uintptr_t> getWords(const llvm::BitVector &BV,
45 uintptr_t & /*Store*/) {
46 // getData() accesses Bits[0] unconditionally; guard against empty vector.
47 if (BV.empty()) {
48 return {};
49 }
50 return BV.getData();
51 }
52 static llvm::ArrayRef<uintptr_t> getWords(const llvm::SmallBitVector &BV,
53 uintptr_t &Store) {
54 if (BV.empty()) {
55 return {};
56 }
57 return BV.getData(Store);

173 [[nodiscard]] friend bool operator==(const BitSet &Lhs,
174 const BitSet &Rhs) noexcept {
175 // Check, whether Lhs and Rhs actually have the same bits set and not
176 // whether their internal representation is exactly identitcal
177 uintptr_t LhsStore{};
178 uintptr_t RhsStore{};
179
180 auto LhsWords = getWords(Lhs.Bits, LhsStore);
181 auto RhsWords = getWords(Rhs.Bits, RhsStore);
182 if (LhsWords.size() == RhsWords.size()) {
183 return LhsWords == RhsWords;
184 }
185 auto MinSize = std::min(LhsWords.size(), RhsWords.size());
186 if (LhsWords.slice(0, MinSize) != RhsWords.slice(0, MinSize)) {
187 return false;
188 }
189 auto Rest = (LhsWords.size() > RhsWords.size() ? LhsWords : RhsWords)
190 .slice(MinSize);
191 return std::all_of(Rest.begin(), Rest.end(),
192 [](auto Word) { return Word == 0; });
193 }

220 /// Calls the given handler function for each set bit in the bitset.
221 ///
222 /// This is likely faster than using iterators.
223 template <std::invocable<IdT> HandlerFn>
224 bool foreach (HandlerFn Handler) const
225 noexcept(std::is_nothrow_invocable_v<HandlerFn &, IdT>) {
226 uintptr_t Store{};
227 auto Words = getWords(Bits, Store);
228 uint32_t Offset = 0;
229 for (auto W : Words) {
230 while (W) {
231 auto Curr = std::countr_zero(W) + Offset;
232 W &= W - 1;
233 if (!invokeControlFlow(Handler, IdT(Curr))) {
234 return false;
235 }
236 }
237
238 Offset += sizeof(W) * CHAR_BIT;
239 }
240 return true;
241 }

266 [[nodiscard]] bool isSubsetOf(const BitSet &Of) const {
267 uintptr_t Buf = 0;
268 uintptr_t OfBuf = 0;
269
270 auto Words = getWords(Bits, Buf);
271 auto OfWords = getWords(Of.Bits, OfBuf);
272 if (Words.size() > OfWords.size()) {
273 if (llvm::any_of(Words.drop_front(OfWords.size()),
274 [](uintptr_t W) { return W != 0; })) {
275 return false;
276 }
277 }
278
279 for (auto [W, OfW] : llvm::zip(Words, OfWords)) {
280 if ((W & ~OfW) != 0) {
281 return false;
282 }
283 }
284
285 return true;
286 }

288 [[nodiscard]] bool isSupersetOf(const BitSet &Of) const {
289 return Of.isSubsetOf(*this);
