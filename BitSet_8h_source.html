30/// \brief A set-type that can compactly store sets of sequential integer-like
31/// types.
32///
33/// Use this type for sequential (unsigned) integers and ids that can convert
34/// from and to uint32_t.
35///
36/// \tparam IdT The type of elements to store in this set. Must be losslessly
37/// convertible from and to uint32_t.
38/// \tparam BitVectorTy The underlying bit-vector to use. Must be either
39/// llvm::BitVector or llvm::SmallBitVector.
40template <SmallIdType IdT, typename BitVectorTy = llvm::BitVector>
41class BitSet {
42 static llvm::ArrayRef<uintptr_t> getWords(const llvm::BitVector &BV,
43 uintptr_t & /*Store*/) {
44 // getData() accesses Bits[0] unconditionally; guard against empty vector.
45 if (BV.empty()) {
46 return {};
47 }
48 return BV.getData();
49 }
50 static llvm::ArrayRef<uintptr_t> getWords(const llvm::SmallBitVector &BV,
51 uintptr_t &Store) {
52 if (BV.empty()) {
53 return {};
54 }
55 return BV.getData(Store);
56 }

158 [[nodiscard]] friend bool operator==(const BitSet &Lhs,
159 const BitSet &Rhs) noexcept {
160 // Check, whether Lhs and Rhs actually have the same bits set and not
161 // whether their internal representation is exactly identitcal
162 uintptr_t LhsStore{};
163 uintptr_t RhsStore{};
164
165 auto LhsWords = getWords(Lhs.Bits, LhsStore);
166 auto RhsWords = getWords(Rhs.Bits, RhsStore);
167 if (LhsWords.size() == RhsWords.size()) {
168 return LhsWords == RhsWords;
169 }
170 auto MinSize = std::min(LhsWords.size(), RhsWords.size());
171 if (LhsWords.slice(0, MinSize) != RhsWords.slice(0, MinSize)) {
172 return false;
173 }
174 auto Rest = (LhsWords.size() > RhsWords.size() ? LhsWords : RhsWords)
175 .slice(MinSize);
176 return std::all_of(Rest.begin(), Rest.end(),
177 [](auto Word) { return Word == 0; });
178 }

205 /// Calls the given handler function for each set bit in the bitset.
206 ///
207 /// This is likely faster than using iterators.
208 template <std::invocable<IdT> HandlerFn>
209 void foreach (HandlerFn Handler) const
210 noexcept(std::is_nothrow_invocable_v<HandlerFn &, IdT>) {
211 uintptr_t Store{};
212 auto Words = getWords(Bits, Store);
213 uint32_t Offset = 0;
214 for (auto W : Words) {
215 while (W) {
216 auto Curr = std::countr_zero(W) + Offset;
217 W &= W - 1;
218 std::invoke(Handler, IdT(Curr));
219 }
220
221 Offset += sizeof(W) * CHAR_BIT;
222 }
223 }

248 [[nodiscard]] bool isSubsetOf(const BitSet &Of) const {
249 uintptr_t Buf = 0;
250 uintptr_t OfBuf = 0;
251
252 auto Words = getWords(Bits, Buf);
253 auto OfWords = getWords(Of.Bits, OfBuf);
254 if (Words.size() > OfWords.size()) {
255 if (llvm::any_of(Words.drop_front(OfWords.size()),
256 [](uintptr_t W) { return W != 0; })) {
257 return false;
258 }
259 }
260
261 for (auto [W, OfW] : llvm::zip(Words, OfWords)) {
262 if ((W & ~OfW) != 0) {
263 return false;
264 }
265 }
266
267 return true;
268 }
269
270 [[nodiscard]] bool isSupersetOf(const BitSet &Of) const {
271 return Of.isSubsetOf(*this);
272 }
