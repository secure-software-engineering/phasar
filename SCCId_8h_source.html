21// Unfortunately, `enum class` cannot be templated, but we want type-safety for
22// SCC-IDs...
23struct SCCIdBase {
24 uint32_t Value{};
25
26 constexpr SCCIdBase() noexcept = default;
27
28 explicit constexpr SCCIdBase(uint32_t Val) noexcept : Value(Val) {}
29
30 explicit constexpr operator uint32_t() const noexcept { return Value; }
31 template <typename T = size_t>
32 explicit constexpr operator size_t() const noexcept
33 requires(!std::is_same_v<uint32_t, T>)
34 {
35 return Value;
36 }
37
38 explicit constexpr operator ptrdiff_t() const noexcept {
39 return ptrdiff_t(Value);
40 }
41
42 constexpr uint32_t operator+() const noexcept { return Value; }
43
44 friend constexpr bool operator==(SCCIdBase L, SCCIdBase R) noexcept {
45 return L.Value == R.Value;
46 }
47 friend constexpr bool operator!=(SCCIdBase L, SCCIdBase R) noexcept {
48 return !(L == R);
49 }
50};
51} // namespace detail
