38/// A flow function that serves as default-implementation for the call-to-return
39/// flow function.
40/// For more details on the use-case of call-to-return flow functions, see the
41/// documentation of FlowFunction::getCallToRetFlowFunction().
42///
43/// Propagates all dataflow facts unchanged, except for global variables and
44/// arguments of the specified call-site.
45/// Global variables are propagated only if PropagateGlobals is set to true; for
46/// the argument values the PropagateArgs function defines on a per-arg basis
47/// whether the respective argument should be propagated or killed.
48/// By default (when not specifying PropagateArgs), all parameters are
49/// propagated unchanged.
50///
51/// For analyses that propagate values via reference parameters in the
52/// return flow function, it is useful to kill the respective arguments here to
53/// enable strong updates.
54///
55template <typename D = const llvm::Value *, typename Container = std::set<D>,
56 typename Fn = TrueFn, typename DCtor = DefaultConstruct<D>,
57 typename = std::enable_if_t<
58 std::is_invocable_r_v<bool, Fn, const llvm::Value *>>>
59auto mapFactsAlongsideCallSite(const llvm::CallBase *CallSite,
60 Fn &&PropagateArgs = {},
61 bool PropagateGlobals = true,
62 DCtor &&FactConstructor = {}) {
63 struct Mapper : public FlowFunction<D, Container> {
64
65 Mapper(const llvm::CallBase *CS, bool PropagateGlobals, Fn &&PropArgs,
66 DCtor &&FactConstructor)
67 : CSAndPropGlob(CS, PropagateGlobals),
68 PropArgs(std::forward<Fn>(PropArgs)),
69 FactConstructor(std::forward<DCtor>(FactConstructor)) {}
70
71 Container computeTargets(D Source) override {
72 // Pass ZeroValue as is
73 if (LLVMZeroValue::isLLVMZeroValue(Source)) {
74 return {Source};
75 }
76 // Pass global variables as is, if desired
77 // Need llvm::Constant here to cover also ConstantExpr and
78 // ConstantAggregate
79 if (llvm::isa<llvm::Constant>(Source)) {
80 if (CSAndPropGlob.getInt()) {
81 return {Source};
82 }
83 return {};
84 }
85
86 for (const auto &Arg : CSAndPropGlob.getPointer()->args()) {
87 if (Arg.get() == Source) {
88 if (std::invoke(PropArgs, Arg.get())) {
89 return {std::invoke(FactConstructor, Arg.get())};
90 }
91 return {};
92 }
93 }
94
95 return {Source};
96 }
97
98 llvm::PointerIntPair<const llvm::CallBase *, 1, bool> CSAndPropGlob;
99 [[no_unique_address]] std::decay_t<Fn> PropArgs;
100 [[no_unique_address]] std::decay_t<DCtor> FactConstructor;
101 };
102
103 return std::make_shared<Mapper>(CallSite, PropagateGlobals,
104 std::forward<Fn>(PropagateArgs),
105 std::forward<DCtor>(FactConstructor));
106}

108/// A flow function that serves as default implementation of the
109/// call flow function. For more information about call flow functions, see
110/// FlowFunctions::getCallFlowFunction().
111///
112/// Propagates the arguments of the specified call-site into the callee function
113/// if the function PropagateArgumentWithSource evaluates to true when invoked
114/// with the argument and a currently holding dataflow fact.
115/// Global variables are propagated into the callee function only, if the flag
116/// PropagateGlobals is set to true.
117/// The special zero (Λ) value gets propagated into the callee function if
118/// PropagateZeroToCallee is true. For most analyses it makes sense to propagate
119/// Λ everywhere.
120///
121/// Given a call-site cs: r = fun(..., ax, ...) and a function prototype
122/// fun(..., px, ...). Further let f = mapFactsToCallee(cs, fun, ...). Then for
123/// any dataflow fact x:
124/// f(Λ) = {Λ} if PropagateZeroToCallee else {},
125/// f(ax) = {px} if PropagateArgumentWithSource(ax, ax) else {},
126/// f(g) = {g} if g is GlobalVariable && PropagateGlobals else {},
127/// f(x) = {px} if PropagateArgumentWithSource(ax, x) else {}.
128///
129/// \note Unlike the old version, this one is only meant for forward-analyses
130template <typename D = const llvm::Value *, typename Container = std::set<D>,
131 typename Fn = std::equal_to<D>, typename DCtor = DefaultConstruct<D>,
132 typename = std::enable_if_t<
133 std::is_invocable_r_v<bool, Fn, const llvm::Value *, D>>>
134FlowFunctionPtrType<D, Container>
135mapFactsToCallee(const llvm::CallBase *CallSite, const llvm::Function *DestFun,
136 Fn &&PropagateArgumentWithSource = {},
137 DCtor &&FactConstructor = {}, bool PropagateGlobals = true,
138 bool PropagateZeroToCallee = true) {
139 struct Mapper : public FlowFunction<D, Container> {
140
141 Mapper(const llvm::CallBase *CS, const llvm::Function *DestFun,
142 bool PropagateGlobals, bool PropagateZeroToCallee, Fn &&PropArg,
143 DCtor &&FactConstructor)
144 : CSAndPropGlob(CS, PropagateGlobals),
145 DestFunAndPropZero(DestFun, PropagateZeroToCallee),
146 PropArg(std::forward<Fn>(PropArg)),
147 FactConstructor(std::forward<DCtor>(FactConstructor)) {}
148
149 Container computeTargets(D Source) override {
150 // If DestFun is a declaration we cannot follow this call, we thus need to
151 // kill everything
152 if (DestFunAndPropZero.getPointer()->isDeclaration()) {
153 return {};
154 }
155
156 Container Res;
157 if (DestFunAndPropZero.getInt() &&
158 LLVMZeroValue::isLLVMZeroValue(Source)) {
159 Res.insert(std::move(Source));
160 } else if (CSAndPropGlob.getInt() &&
161 !LLVMZeroValue::isLLVMZeroValue(Source) &&
162 llvm::isa<llvm::Constant>(Source)) {
163 // Pass global variables as is, if desired
164 // Globals could also be actual arguments, then the formal argument
165 // needs to be generated below. Need llvm::Constant here to cover also
166 // ConstantExpr and ConstantAggregate
167 Res.insert(std::move(Source));
168 }
169
170 const auto *CS = CSAndPropGlob.getPointer();
171 const auto *DestFun = DestFunAndPropZero.getPointer();
172 assert(CS->arg_size() >= DestFun->arg_size());
173 assert(CS->arg_size() == DestFun->arg_size() || DestFun->isVarArg());
174
175 llvm::CallBase::const_op_iterator ArgIt = CS->arg_begin();
176 llvm::CallBase::const_op_iterator ArgEnd = CS->arg_end();
177 llvm::Function::const_arg_iterator ParamIt = DestFun->arg_begin();
178 llvm::Function::const_arg_iterator ParamEnd = DestFun->arg_end();
179
180 if (ParamIt != ParamEnd && (*ParamIt).hasStructRetAttr()) {
181 // sret parameters are writeonly
182 ++ParamIt;
183 ++ArgIt;
184 }
185
186 for (; ParamIt != ParamEnd; ++ParamIt, ++ArgIt) {
187 if (std::invoke(PropArg, ArgIt->get(), Source)) {
188 Res.insert(std::invoke(FactConstructor, &*ParamIt));
189 }
190 }
191
192 if (ArgIt != ArgEnd &&
193 std::any_of(
194 ArgIt, ArgEnd,
195 std::bind(std::ref(PropArg), std::placeholders::_1, Source))) {
196 if (const auto *VaListAlloca = getVaListTagOrNull(*DestFun)) {
197 Res.insert(std::invoke(FactConstructor, VaListAlloca));
198 }
199 }
200
201 return Res;
202 }
203
204 llvm::PointerIntPair<const llvm::CallBase *, 1, bool> CSAndPropGlob;
205 llvm::PointerIntPair<const llvm::Function *, 1, bool> DestFunAndPropZero;
206 [[no_unique_address]] std::decay_t<Fn> PropArg;
207 [[no_unique_address]] std::decay_t<DCtor> FactConstructor;
208 };
209
210 return std::make_shared<Mapper>(CallSite, DestFun, PropagateGlobals,
211 PropagateZeroToCallee,
212 std::forward<Fn>(PropagateArgumentWithSource),
213 std::forward<DCtor>(FactConstructor));
214}

216/// A flow function that serves as default-implementation of the return flow
217/// function. For more information about return flow functions, see
218/// FlowFunctions::getRetFlowFunction().
219///
220/// Propagates the return value back to the call-site and based on the
221/// PropagateParameter predicate propagates back parameters holding as dataflow
222/// facts. The resulting out-set of dataflow facts can be post-processed if
223/// necessary.
224///
225/// Let a call-site cs: r = fun(..., ax, ...) a function prototype fun(...,
226/// px, ...) and an exit statement exit: return rv.
227/// Further given a flow function f = mapFactsToCaller(cs, exit, ...). Then for
228/// all dataflow facts x holding at exit:
229/// f(rv) = {r} if PropagateRet(rv, rv) else {},
230/// f(px) = {ax} if PropagateParameter(px, px) else {},
231/// f(g) = {g} if PropagateGlobals else {},
232/// f(Λ) = {Λ} if PropagateZeroToCaller else {},
233/// f(x) = ({ax} if PropagateParameter(ax, x) else {}) union ({r} if
234/// PropagateRet(rv, x) else {}).
235///
236template <
237 typename D = const llvm::Value *, typename Container = std::set<D>,
238 typename FnParam = std::equal_to<D>, typename FnRet = std::equal_to<D>,
239 typename DCtor = DefaultConstruct<D>, typename PostProcessFn = IgnoreArgs,
240 typename = std::enable_if_t<
241 std::is_invocable_r_v<bool, FnParam, const llvm::Value *, D> &&
242 std::is_invocable_r_v<bool, FnRet, const llvm::Value *, D>>>
243FlowFunctionPtrType<D, Container> mapFactsToCaller(
244 const llvm::CallBase *CallSite, const llvm::Instruction *ExitInst,
245 FnParam &&PropagateParameter = {}, FnRet &&PropagateRet = {},
246 DCtor &&FactConstructor = {}, bool PropagateGlobals = true,
247 bool PropagateZeroToCaller = true, PostProcessFn &&PostProcess = {}) {
248
249 struct Mapper : public FlowFunction<D, Container> {
250 Mapper(const llvm::CallBase *CallSite, const llvm::Instruction *ExitInst,
251 bool PropagateGlobals, FnParam &&PropagateParameter,
252 FnRet &&PropagateRet, DCtor &&FactConstructor,
253 bool PropagateZeroToCaller, PostProcessFn &&PostProcess)
254 : CSAndPropGlob(CallSite, PropagateGlobals),
255 ExitInstAndPropZero(ExitInst, PropagateZeroToCaller),
256 PropArg(std::forward<FnParam>(PropagateParameter)),
257 PropRet(std::forward<FnRet>(PropagateRet)),
258 FactConstructor(std::forward<DCtor>(FactConstructor)),
259 PostProcess(std::forward<PostProcessFn>(PostProcess)) {}
260
261 Container computeTargets(D Source) override {
262 Container Res;
263 if (ExitInstAndPropZero.getInt() &&
264 LLVMZeroValue::isLLVMZeroValue(Source)) {
265 Res.insert(Source);
266 } else if (CSAndPropGlob.getInt() && llvm::isa<llvm::Constant>(Source)) {
267 // Pass global variables as is, if desired
268 // Globals could also be actual arguments, then the formal argument
269 // needs to be generated below. Need llvm::Constant here to cover also
270 // ConstantExpr and ConstantAggregate
271 Res.insert(Source);
272 }
273
274 const auto *CS = CSAndPropGlob.getPointer();
275 const auto *DestFun = ExitInstAndPropZero.getPointer()->getFunction();
276 assert(CS->arg_size() >= DestFun->arg_size());
277 assert(CS->arg_size() == DestFun->arg_size() || DestFun->isVarArg());
278
279 llvm::CallBase::const_op_iterator ArgIt = CS->arg_begin();
280 llvm::CallBase::const_op_iterator ArgEnd = CS->arg_end();
281 llvm::Function::const_arg_iterator ParamIt = DestFun->arg_begin();
282 llvm::Function::const_arg_iterator ParamEnd = DestFun->arg_end();
283
284 for (; ParamIt != ParamEnd; ++ParamIt, ++ArgIt) {
285 if (std::invoke(PropArg, &*ParamIt, Source)) {
286 Res.insert(std::invoke(FactConstructor, ArgIt->get()));
287 }
288 }
289
290 if (ArgIt != ArgEnd) {
291 if (const auto *VaListAlloca = getVaListTagOrNull(*DestFun)) {
292 if (std::invoke(PropArg, VaListAlloca, Source)) {
293 std::transform(ArgIt, ArgEnd, std::inserter(Res, Res.end()),
294 FactConstructor);
295 }
296 }
297 }
298
299 if (const auto *RetInst = llvm::dyn_cast<llvm::ReturnInst>(
300 ExitInstAndPropZero.getPointer());
301 RetInst && RetInst->getReturnValue()) {
302 if (std::invoke(PropRet, RetInst->getReturnValue(), Source)) {
303 Res.insert(std::invoke(FactConstructor, CS));
304 }
305 }
306
307 std::invoke(PostProcess, Res);
308
309 return Res;
310 }
311
312 llvm::PointerIntPair<const llvm::CallBase *, 1, bool> CSAndPropGlob;
313 llvm::PointerIntPair<const llvm::Instruction *, 1, bool>
314 ExitInstAndPropZero;
315 [[no_unique_address]] std::decay_t<FnParam> PropArg;
316 [[no_unique_address]] std::decay_t<FnRet> PropRet;
317 [[no_unique_address]] std::decay_t<DCtor> FactConstructor;
318 [[no_unique_address]] std::decay_t<PostProcessFn> PostProcess;
319 };
320
321 return std::make_shared<Mapper>(
322 CallSite, ExitInst, PropagateGlobals,
323 std::forward<FnParam>(PropagateParameter),
324 std::forward<FnRet>(PropagateRet), std::forward<DCtor>(FactConstructor),
325 PropagateZeroToCaller, std::forward<PostProcessFn>(PostProcess));
326}

331/// A flow function that models a strong update on a memory location modified by
332/// a store instruction
333///
334/// Given a flow function f = strongUpdateStore(store a to b, pred), for all
335/// holding dataflow facts x:
336/// f(b) = {},
337/// f(x) = {x, b} if pred(x) else {x}.
338///
339template <typename Fn, typename Container = std::set<const llvm::Value *>,
340 typename = std::enable_if_t<
341 std::is_invocable_r_v<bool, Fn, const llvm::Value *>>>
342FlowFunctionPtrType<const llvm::Value *, Container>
343strongUpdateStore(const llvm::StoreInst *Store, Fn &&GeneratePointerOpIf) {
344 // Here we cheat a bit and "look through" the GetElementPtrInst to the
345 // targeted memory location.
346 const auto *BasePtrOp = Store->getPointerOperand()->stripPointerCasts();
347 if (BasePtrOp != Store->getPointerOperand()) {
348 struct StrongUpdateFlow
349 : public FlowFunction<const llvm::Value *, Container> {
350
351 StrongUpdateFlow(const llvm::Value *PointerOp,
352 const llvm::Value *BasePtrOp, Fn &&GeneratePointerOpIf)
353 : PointerOp(PointerOp), BasePtrOp(BasePtrOp),
354 Pred(std::forward<Fn>(GeneratePointerOpIf)) {}
355
356 Container computeTargets(const llvm::Value *Source) override {
357 if (Source == PointerOp || Source == BasePtrOp) {
358 return {};
359 }
360 if (std::invoke(Pred, Source)) {
361 return {Source, PointerOp, BasePtrOp};
362 }
363 return {Source};
364 }
365
366 const llvm::Value *PointerOp;
367 const llvm::Value *BasePtrOp;
368 [[no_unique_address]] std::decay_t<Fn> Pred;
369 };
370
371 return std::make_shared<StrongUpdateFlow>(
372 Store->getPointerOperand(), BasePtrOp,
373 std::forward<Fn>(GeneratePointerOpIf));
374 }

400/// A flow function that models a strong update on a memory location modified by
401/// a store instruction. Similar to transferFlow.
402///
403/// Given a flow function f = strongUpdateStore(store a to b), for all
404/// holding dataflow facts x:
405/// f(b) = {},
406/// f(a) = {a, b},
407/// f(x) = {x}.
408///
409/// In the exploded supergraph it may look as follows:
410///
411/// x a b ...
412/// | |\ |
413/// | | \ ...
414/// store a to b | | \ ...
415/// v v v
416/// x a b ...
417///
418template <typename Container = std::set<const llvm::Value *>>
419FlowFunctionPtrType<const llvm::Value *, Container>
420strongUpdateStore(const llvm::StoreInst *Store) {
421 // Here we cheat a bit and "look through" the GetElementPtrInst to the
422 // targeted memory location.
423 const auto *BasePtrOp = Store->getPointerOperand()->stripPointerCasts();
424 if (BasePtrOp != Store->getPointerOperand()) {
425 struct StrongUpdateFlow
426 : public FlowFunction<const llvm::Value *, Container> {
427
428 StrongUpdateFlow(const llvm::StoreInst *Store,
429 const llvm::Value *BasePtrOp) noexcept
430 : Store(Store), BasePtrOp(BasePtrOp) {}
431
432 Container computeTargets(const llvm::Value *Source) override {
433 if (Source == Store->getPointerOperand() || Source == BasePtrOp) {
434 return {};
435 }
436 if (Source == Store->getValueOperand()) {
437 return {Source, Store->getPointerOperand(), BasePtrOp};
438 }
439 return {Source};
440 }
441
442 const llvm::StoreInst *Store;
443 const llvm::Value *BasePtrOp;
444 };
445
446 return std::make_shared<StrongUpdateFlow>(Store, BasePtrOp);
447 }
