38template <typename ProblemT, typename ICFGTy, bool ComputeWeights>
39class IfdsIdeRuleProvider {
40public:
41 using control_location_type = typename ProblemT::d_t;
42 using stack_element_type = typename ProblemT::n_t;
43 using weight_type =
44 typename detail::WeightTypeOf<ComputeWeights, ProblemT>::type;

46 static constexpr auto LogCategory =
47 ComputeWeights ? llvm::StringLiteral("IDERuleProvider")
48 : llvm::StringLiteral("IFDSRuleProvider");

57 [[nodiscard]] auto getNormalRules(ByConstRef<control_location_type> CL,
58 ByConstRef<stack_element_type> SE) {
59 llvm::SmallVector<
60 std::tuple<control_location_type, stack_element_type, weight_type>>
61 Outs;
62
63 PHASAR_LOG_LEVEL_CAT(DEBUG, LogCategory,
64 "[getNormalRules]: CL=" << DToString(CL)
65 << "; SE=" << NToString(SE));
66
67 if (ICF->isCallSite(SE)) {
68 auto Callees = ICF->getCalleesOfCallAt(SE);
69 auto RetSites = ICF->getReturnSitesOfCallAt(SE);
70 for (const auto &Succ : RetSites) {
71 auto Facts = FECache.getCallToRetFlowFunction(SE, Succ, Callees)
72 ->computeTargets(CL);
73 for (auto &Fct : Facts) {
74 const auto FctSucc = factSucc(Succ, Fct);
75 if constexpr (ComputeWeights) {
76 auto W =
77 FECache.getCallToRetEdgeFunction(SE, CL, Succ, Fct, Callees);
78 Outs.emplace_back(std::move(Fct), FctSucc, std::move(W));
79 } else {
80 Outs.emplace_back(std::move(Fct), FctSucc, weight_type{});
81 }
82 }
83 }
84
85 for (const auto &DestFun : Callees) {
86 if (auto SumFF = FECache.getSummaryFlowFunction(SE, DestFun)) {
87 auto Facts = SumFF->computeTargets(CL);
88 for (const auto &Succ : RetSites) {
89 for (auto &Fct : Facts) {
90 const auto FctSucc = factSucc(Succ, Fct);
91 if constexpr (ComputeWeights) {
92 auto W = FECache.getSummaryEdgeFunction(SE, CL, Succ, Fct);
93 Outs.emplace_back(Fct, FctSucc, std::move(W));
94 } else {
95 Outs.emplace_back(Fct, FctSucc, weight_type{});
96 }
97 }
98 }
99 }
100 }
101
102 } else {
103 for (const auto &Succ : ICF->getSuccsOf(SE)) {
104 auto Facts =
105 FECache.getNormalFlowFunction(SE, Succ)->computeTargets(CL);
106 for (auto &Fct : Facts) {
107 const auto FctSucc = factSucc(Succ, Fct);
108 if constexpr (ComputeWeights) {
109 auto W = FECache.getNormalEdgeFunction(SE, CL, Succ, Fct);
110 Outs.emplace_back(std::move(Fct), FctSucc, std::move(W));
111 } else {
112 Outs.emplace_back(std::move(Fct), FctSucc, weight_type{});
113 }
114 }
115 }
116 }
117
118 return Outs;
119 }

121 [[nodiscard]] auto getPushRules(ByConstRef<control_location_type> CL,
122 ByConstRef<stack_element_type> SE) {
123 llvm::SmallVector<std::tuple<control_location_type, stack_element_type,
124 stack_element_type, weight_type>>
125 Outs;
126 if (!ICF->isCallSite(SE)) {
127 return Outs;
128 }
129
130 PHASAR_LOG_LEVEL_CAT(DEBUG, LogCategory,
131 "[getPushRules]: CL=" << DToString(CL)
132 << "; SE=" << NToString(SE));
133
134 auto Callees = ICF->getCalleesOfCallAt(SE);
135 auto RetSites = ICF->getReturnSitesOfCallAt(SE);
136 for (const auto &DestFun : Callees) {
137 if (FECache.getSummaryFlowFunction(SE, DestFun)) {
138 // Handled in getNormalRules()
139 continue;
140 }
141
142 auto Facts = FECache.getCallFlowFunction(SE, DestFun)->computeTargets(CL);
143 auto EntrySEs = ICF->getStartPointsOf(DestFun);
144 for (auto &&Fct : Facts) {
145 auto W = [&] {
146 if constexpr (ComputeWeights) {
147 return FECache.getCallEdgeFunction(SE, CL, DestFun, Fct);
148 } else {
149 return weight_type{};
150 }
151 }();
152 for (const auto &Succ : RetSites) {
153 const auto FctSucc = factSucc(Succ, Fct);
154 for (const auto &EntrySE : EntrySEs) {
155 Outs.emplace_back(Fct, FctSucc, EntrySE, W);
156 }
157 }
158 }
159 }
160
161 return Outs;
162 }

171 [[nodiscard]] auto getPopRules(ByConstRef<control_location_type> CL,
172 ByConstRef<stack_element_type> ExitSE,
173 ByConstRef<stack_element_type> RetSiteSE,
174 ByConstRef<stack_element_type> /*EntrySE*/
175 ) {
176 llvm::SmallVector<std::tuple<control_location_type, weight_type>> Outs;
177
178 PHASAR_LOG_LEVEL_CAT(DEBUG, LogCategory,
179 "[getPopRules]: CL="
180 << DToString(CL)
181 << "; ExitSE=" << NToString(ExitSE)
182 << "; RetSiteSE=" << NToString(RetSiteSE));
183
184 auto DestFun = ICF->getFunctionOf(ExitSE);
185
186 for (const auto &CS : ICF->getPredsOf(RetSiteSE)) {
187 if (!ICF->isCallSite(CS)) {
188 continue;
189 }
190
191 auto Facts = FECache.getRetFlowFunction(CS, DestFun, ExitSE, RetSiteSE)
192 ->computeTargets(CL);
193 for (auto &Fct : Facts) {
194 if constexpr (ComputeWeights) {
195 auto W = FECache.getReturnEdgeFunction(CS, DestFun, ExitSE, CL,
196 RetSiteSE, Fct);
197 Outs.emplace_back(std::move(Fct), std::move(W));
198 } else {
199 Outs.emplace_back(std::move(Fct), weight_type{});
200 }
201 }
202 }
203
204 return Outs;
205 }

207 [[nodiscard]] auto initialSeeds() {
208 llvm::SmallVector<
209 std::tuple<control_location_type, stack_element_type, weight_type>>
210 Outs;
211
212 for (const auto &[Inst, Facts] : Problem->initialSeeds().getSeeds()) {
213 for (const auto &[Fact, Val] : Facts) {
214 if constexpr (ComputeWeights) {
215 using l_t = typename ProblemT::l_t;
216 if (Val == Problem->topElement()) {
217 continue;
218 }
219 if (Val == Problem->bottomElement()) {
220 if constexpr (HasJoinLatticeTraits<l_t>) {
221 Outs.emplace_back(Fact, Inst, AllBottom<l_t>{});
222 } else {
223 Outs.emplace_back(Fact, Inst, AllBottom<l_t>{Val});
224 }
225 } else {
226 Outs.emplace_back(
227 Fact, Inst,
228 ConstantEdgeFunction<l_t>{NonTopBotValue<l_t>::unwrap(Val)});
229 }
230 } else {
231 Outs.emplace_back(Fact, Inst, weight_type{});
232 }
233 }
234 }
235
236 return Outs;
237 }

242 [[nodiscard]] auto factSucc(stack_element_type Succ,
243 ByConstRef<control_location_type> CL) {
244 if constexpr (has_advanceToNextUser_v<ICFGTy, control_location_type>) {
245 return ICF->advancetoNextUser(Succ, CL);
246 } else {
247 return Succ;
248 }
249 }
