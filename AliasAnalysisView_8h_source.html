28class FunctionAliasView {
29public:

32 const llvm::Value *,
33 const llvm::DataLayout &);

35 [[nodiscard]] AliasResult alias(const llvm::Value *V, const llvm::Value *Rep,
36 const llvm::DataLayout &DL) {
37 return Alias(Context, V, Rep, DL);
38 }
39
40 template <typename T, typename AliasFn>
41 requires(std::is_empty_v<AliasFn> &&
42 std::is_default_constructible_v<AliasFn>)
43 constexpr FunctionAliasView(T *Context, AliasFn /*Alias*/) noexcept
44 : Context(Context), Alias(&callAlias<T, AliasFn>) {}
45
46private:
47 template <typename T, typename AliasFn>
48 static AliasResult callAlias(void *Context, const llvm::Value *V1,
49 const llvm::Value *V2,
50 const llvm::DataLayout &DL) {
51 return AliasFn{}(static_cast<T *>(Context), V1, V2, DL);
52 }
53
54 void *Context{};
55 AliasCallbackTy<void> Alias{};
56};
