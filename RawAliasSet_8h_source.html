25/// Concept for alias-set containers used inside the union-find analyses.
26/// The container must support set insertion, membership test, bitwise
27/// union/intersection/difference, a \c foreach() visitor, and comparison.
28template <typename ASet>
29concept IsRawAliasSet = requires(ASet &MutSet, const ASet &ConstSet,
30 typename ASet::value_type ValId) {
31 ASet();
32 ASet(ConstSet);
33 ASet(std::move(MutSet));
34 MutSet = ConstSet;
35 MutSet = std::move(MutSet);
36 MutSet.insert(ValId);
37 { MutSet.tryInsert(ValId) } -> std::convertible_to<bool>;
38 { ConstSet.contains(ValId) } -> std::convertible_to<bool>;
39 // ConstSet.begin();
40 // ConstSet.end();
41
42 /// Iteration must be in ascending order
43 ConstSet.foreach (DummyFn<typename ASet::value_type>{});
44 MutSet |= ConstSet;
45 MutSet &= ConstSet;
46 MutSet -= ConstSet;
47 { ConstSet - ConstSet } -> std::convertible_to<ASet>;
48 { ConstSet == ConstSet } noexcept -> std::convertible_to<bool>;
49 { ConstSet != ConstSet } noexcept -> std::convertible_to<bool>;
50 { MutSet.tryMergeWith(ConstSet) } -> std::convertible_to<bool>;
51 { MutSet.clear() } noexcept;
52 { ConstSet.empty() } noexcept -> std::convertible_to<bool>;
53 { ConstSet.size() } noexcept -> std::convertible_to<size_t>;
54 {
55 // Merges the ConstSet into MutSet, as with tryMergeWith, but invokes a
56 // callback for each element that was newly inserted.The Diff will be
57 // materialized and merged into that out-param
58 MutSet.mergeWithDiff(ConstSet, DummyFn<typename ASet::value_type>{}, MutSet)
59 } -> std::convertible_to<bool>;
60 {
61 // Merges the ConstSet into MutSet, as with tryMergeWith, but invokes a
62 // callback for each element that was newly inserted.
63 MutSet.mergeWithDiff(ConstSet, DummyFn<typename ASet::value_type>{})
64 } -> std::convertible_to<bool>;
65};

67/// Sparse bit-set used to represent alias sets in union-find analyses.
68///
69/// Currently backed by \c llvm::SparseBitVector for compact storage when ids
70/// are scattered in a large range and dense storage when ids are clustered.
71/// Satisfies \c IsRawAliasSet.
72///
73/// \tparam IdT Integer-like id type (e.g., \c ValueId).
74template <SmallIdType IdT> class LLVMRawAliasSet {
75public:
76 using value_type = IdT;
77
78 LLVMRawAliasSet() = default;
79
80 void insert(IdT Id) { Bits.set(uint32_t(Id)); }
81
82 [[nodiscard]] bool tryInsert(IdT Id) {
83 return Bits.test_and_set(uint32_t(Id));
84 }
85
86 [[nodiscard]] bool contains(IdT Id) const { return Bits.test(uint32_t(Id)); }
87
88 template <std::invocable<IdT> HandlerFn>
89 LLVM_ATTRIBUTE_ALWAYS_INLINE void foreach (HandlerFn Handler) const {
90 for (auto Bit : Bits) {
91 if constexpr (std::convertible_to<std::invoke_result_t<HandlerFn &, IdT>,
92 bool>) {
93 if (!std::invoke(Handler, IdT(Bit))) {
94 break;
95 }
96 } else {
97 std::invoke(Handler, IdT(Bit));
98 }
99 }
100 }
101
102 void operator|=(const LLVMRawAliasSet &Other) { Bits |= Other.Bits; }
103 void operator&=(const LLVMRawAliasSet &Other) { Bits &= Other.Bits; }
104 void operator-=(const LLVMRawAliasSet &Other) {
105 Bits.intersectWithComplement(Other.Bits);
106 }
107
108 [[nodiscard]] LLVMRawAliasSet operator-(const LLVMRawAliasSet &Other) const {
109 LLVMRawAliasSet Ret;
110 Ret.Bits = Bits - Other.Bits;
111 return Ret;
112 }
113
114 [[nodiscard]] bool empty() const noexcept { return Bits.empty(); }
115 [[nodiscard]] size_t size() const noexcept { return Bits.count(); }
116
117 void clear() noexcept { Bits.clear(); }
118
119 [[nodiscard]] auto begin() const noexcept { return Bits.begin(); }
120 [[nodiscard]] auto end() const noexcept { return Bits.end(); }
121
122 [[nodiscard]] bool tryMergeWith(const LLVMRawAliasSet &Other) {
123 return Bits |= Other.Bits;
124 }
125
126 void erase(IdT Id) { Bits.reset(uint32_t(Id)); }
127
128 [[nodiscard]] bool operator==(const LLVMRawAliasSet &Other) const noexcept {
129 return Bits == Other.Bits;
130 }
131
132 bool mergeWithDiff(const LLVMRawAliasSet &Other,
133 std::invocable<IdT> auto WithNewElem,
134 LLVMRawAliasSet &IntoDiff) {
135 return mergeWithDiffImpl(Other, copyOrRef(WithNewElem), &IntoDiff);
136 }
137
138 bool mergeWithDiff(const LLVMRawAliasSet &Other,
139 std::invocable<IdT> auto WithNewElem) {
140 return mergeWithDiffImpl(Other, copyOrRef(WithNewElem), nullptr);
141 }
142
143private:
144 bool mergeWithDiffImpl(const LLVMRawAliasSet &Other,
145 std::invocable<IdT> auto WithNewElem,
146 LLVMRawAliasSet *IntoDiff) {
147 auto Diff = Other.Bits - Bits;
148 if (Diff.empty()) {
149 return false;
150 }
151
152 Bits |= Diff;
153 if (IntoDiff) {
154 IntoDiff->Bits |= Diff;
155 }
156 for (auto Elem : Diff) {
157 std::invoke(WithNewElem, IdT(Elem));
158 }
159 return true;
160 }
161
162 llvm::SparseBitVector<> Bits;
163};

165template <SmallIdType IdT> class RoaringAliasSet {
166public:
167 using value_type = IdT;
168
169 RoaringAliasSet() = default;
170
171 void insert(IdT Id) { Bits.add(uint32_t(Id)); }
172
173 [[nodiscard]] bool tryInsert(IdT Id) { return Bits.addChecked(uint32_t(Id)); }
174
175 [[nodiscard]] bool contains(IdT Id) const {
176 return Bits.contains(uint32_t(Id));
177 }
178
179 template <std::invocable<IdT> HandlerFn>
180 LLVM_ATTRIBUTE_ALWAYS_INLINE void foreach (HandlerFn Handler) const {
181 return Bits.iterate(
182 [](uint32_t Id, void *HandlerPtr) {
183 auto &Handler = *(HandlerFn *)HandlerPtr;
184 if constexpr (std::convertible_to<
185 std::invoke_result_t<HandlerFn &, IdT>, bool>) {
186 if (!std::invoke(Handler, IdT(Id))) {
187 return false;
188 }
189 } else {
190 std::invoke(Handler, IdT(Id));
191 }
192 return true;
193 },
194 &Handler);
195 }
196
197 void operator|=(const RoaringAliasSet &Other) { Bits |= Other.Bits; }
198 void operator&=(const RoaringAliasSet &Other) { Bits &= Other.Bits; }
199 void operator-=(const RoaringAliasSet &Other) { Bits -= Other.Bits; }
200 [[nodiscard]] RoaringAliasSet operator-(const RoaringAliasSet &Other) const {
201 return Bits - Other.Bits;
202 }
203
204 [[nodiscard]] bool empty() const noexcept { return Bits.isEmpty(); }
205 [[nodiscard]] size_t size() const noexcept { return Bits.cardinality(); }
206
207 void clear() noexcept { Bits.clear(); }
208
209 [[nodiscard]] auto begin() const noexcept { return Bits.begin(); }
210 [[nodiscard]] auto end() const noexcept { return Bits.end(); }
211
212 [[nodiscard]] bool tryMergeWith(const RoaringAliasSet &Other) {
213 auto OldSz = size();
214 Bits |= Other.Bits;
215 return size() != OldSz;
216 }
217
218 void erase(IdT Id) { Bits.remove(uint32_t(Id)); }
219
220 // Bulk-inserts from a sorted, deduplicated array.
221 // Roaring constructs containers in O(N) for sorted input.
222 void insertSorted(llvm::ArrayRef<uint32_t> Sorted) {
223 Bits.addMany(Sorted.size(), Sorted.data());
224 }
225
226 [[nodiscard]] bool operator==(const RoaringAliasSet &Other) const noexcept {
227 return Bits == Other.Bits;
228 }
229
230 bool mergeWithDiff(const RoaringAliasSet &Other,
231 std::invocable<IdT> auto WithNewElem) {
232 constexpr size_t DiffThreshold = 16;
233 // operator- is expensive, but it is definitely a lot faster than the
234 // foreach loop if UPending is large
235
236 if (Other.size() > DiffThreshold) {
237 RoaringAliasSet Diff = Other - *this;
238 if (Diff.empty()) {
239 return false;
240 }
241
242 *this |= Diff;
243
244 Diff.foreach (copyOrRef(WithNewElem));
245 return true;
246 }
247
248 bool Ret = false;
249 Other.foreach ([&](IdT Elem) {
250 if (tryInsert(Elem)) {
251 std::invoke(WithNewElem, Elem);
252 Ret = true;
253 }
254 });
255 return Ret;
256 }
257
258 bool mergeWithDiff(const RoaringAliasSet &Other,
259 std::invocable<IdT> auto WithNewElem,
260 RoaringAliasSet &IntoDiff) {
261 constexpr size_t DiffThreshold = 16;
262 // operator- is expensive, but it is definitely a lot faster than the
263 // foreach loop if Other is large
264
265 if (Other.size() > DiffThreshold) {
266 RoaringAliasSet Diff = Other - *this;
267 if (Diff.empty()) {
268 return false;
269 }
270
271 *this |= Diff;
272 IntoDiff |= Diff;
273
274 Diff.foreach (copyOrRef(WithNewElem));
275 return true;
276 }
277
278 bool Ret = false;
279 Other.foreach ([&](IdT Elem) {
280 if (tryInsert(Elem)) {
281 IntoDiff.insert(Elem);
282 std::invoke(WithNewElem, Elem);
283 Ret = true;
284 }
285 });
286
287 return Ret;
288 }
289
290private:
291 RoaringAliasSet(roaring::Roaring &&RR) : Bits(std::move(RR)) {}
292
293 roaring::Roaring Bits{};
294};
