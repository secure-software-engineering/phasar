20/// Concept for alias-set containers used inside the union-find analyses.
21/// The container must support set insertion, membership test, bitwise
22/// union/intersection/difference, a \c foreach() visitor, and comparison.
23template <typename ASet>
24concept IsRawAliasSet = requires(ASet &MutSet, const ASet &ConstSet,
25 typename ASet::value_type ValId) {
26 ASet();
27 ASet(ConstSet);
28 ASet(std::move(MutSet));
29 MutSet = ConstSet;
30 MutSet = std::move(MutSet);
31 MutSet.insert(ValId);
32 { MutSet.tryInsert(ValId) } -> std::convertible_to<bool>;
33 { ConstSet.contains(ValId) } -> std::convertible_to<bool>;
34 // ConstSet.begin();
35 // ConstSet.end();
36 ConstSet.foreach (DummyFn<typename ASet::value_type>{});
37 MutSet |= ConstSet;
38 MutSet &= ConstSet;
39 MutSet -= ConstSet;
40 { ConstSet == ConstSet } noexcept -> std::convertible_to<bool>;
41 { ConstSet != ConstSet } noexcept -> std::convertible_to<bool>;
42 { MutSet.tryMergeWith(ConstSet) } -> std::convertible_to<bool>;
43 { MutSet.clear() } noexcept;
44 { ConstSet.empty() } noexcept -> std::convertible_to<bool>;
45 { ConstSet.size() } noexcept -> std::convertible_to<size_t>;
46};

48/// Sparse bit-set used to represent alias sets in union-find analyses.
49///
50/// Currently backed by \c llvm::SparseBitVector for compact storage when ids
51/// are scattered in a large range and dense storage when ids are clustered.
52/// Satisfies \c IsRawAliasSet.
53///
54/// \tparam IdT Integer-like id type (e.g., \c ValueId).
55template <SmallIdType IdT> class RawAliasSet {
56public:
57 using value_type = IdT;
58
59 RawAliasSet() = default;
60
61 void insert(IdT Id) { Bits.set(uint32_t(Id)); }
62
63 [[nodiscard]] bool tryInsert(IdT Id) {
64 return Bits.test_and_set(uint32_t(Id));
65 }
66
67 [[nodiscard]] bool contains(IdT Id) const { return Bits.test(uint32_t(Id)); }
68
69 LLVM_ATTRIBUTE_ALWAYS_INLINE void foreach (
70 std::invocable<IdT> auto Handler) const {
71 for (auto Bit : Bits) {
72 std::invoke(Handler, IdT(Bit));
73 }
74 }
75
76 void operator|=(const RawAliasSet &Other) { Bits |= Other.Bits; }
77 void operator&=(const RawAliasSet &Other) { Bits &= Other.Bits; }
78 void operator-=(const RawAliasSet &Other) {
79 Bits.intersectWithComplement(Other.Bits);
80 }
81
82 [[nodiscard]] bool empty() const noexcept { return Bits.empty(); }
83 [[nodiscard]] size_t size() const noexcept { return Bits.count(); }
84
85 void clear() noexcept { Bits.clear(); }
86
87 [[nodiscard]] auto begin() const noexcept { return Bits.begin(); }
88 [[nodiscard]] auto end() const noexcept { return Bits.end(); }
89
90 [[nodiscard]] bool tryMergeWith(const RawAliasSet &Other) {
91 return Bits |= Other.Bits;
92 }
93
94 void erase(IdT Id) { Bits.reset(uint32_t(Id)); }
95
96 [[nodiscard]] bool operator==(const RawAliasSet &Other) const noexcept {
97 return Bits == Other.Bits;
98 }
99
100private:
101 llvm::SparseBitVector<> Bits;
102 // TODO: roaring::Roaring Bits;
103};
104} // namespace psr
