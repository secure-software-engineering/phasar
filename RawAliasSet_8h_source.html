19/// Concept for alias-set containers used inside the union-find analyses.
20/// The container must support set insertion, membership test, bitwise
21/// union/intersection/difference, a \c foreach() visitor, and comparison.
22template <typename ASet>
23concept IsRawAliasSet = requires(ASet &MutSet, const ASet &ConstSet,
24 typename ASet::value_type ValId) {
25 ASet();
26 ASet(ConstSet);
27 ASet(std::move(MutSet));
28 MutSet = ConstSet;
29 MutSet = std::move(MutSet);
30 MutSet.insert(ValId);
31 { MutSet.tryInsert(ValId) } -> std::convertible_to<bool>;
32 { ConstSet.contains(ValId) } -> std::convertible_to<bool>;
33 // ConstSet.begin();
34 // ConstSet.end();
35
36 /// Iteration must be in ascending order
37 ConstSet.foreach (DummyFn<typename ASet::value_type>{});
38 MutSet |= ConstSet;
39 MutSet &= ConstSet;
40 MutSet -= ConstSet;
41 { ConstSet - ConstSet } -> std::convertible_to<ASet>;
42 { ConstSet == ConstSet } noexcept -> std::convertible_to<bool>;
43 { ConstSet != ConstSet } noexcept -> std::convertible_to<bool>;
44 { MutSet.tryMergeWith(ConstSet) } -> std::convertible_to<bool>;
45 { MutSet.clear() } noexcept;
46 { ConstSet.empty() } noexcept -> std::convertible_to<bool>;
47 { ConstSet.size() } noexcept -> std::convertible_to<size_t>;
48 {
49 // Merges the ConstSet into MutSet, as with tryMergeWith, but invokes a
50 // callback for each element that was newly inserted.The Diff will be
51 // materialized and merged into that out-param
52 MutSet.mergeWithDiff(ConstSet, DummyFn<typename ASet::value_type>{}, MutSet)
53 } -> std::convertible_to<bool>;
54 {
55 // Merges the ConstSet into MutSet, as with tryMergeWith, but invokes a
56 // callback for each element that was newly inserted.
57 MutSet.mergeWithDiff(ConstSet, DummyFn<typename ASet::value_type>{})
58 } -> std::convertible_to<bool>;
59};
