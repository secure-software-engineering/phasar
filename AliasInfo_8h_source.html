45/// \brief A type-erased reference to any object implementing the IsAliasInfo
46/// interface. Use this, if your analysis is not tied to a specific alias info
47/// implementation.
48///
49/// This is a *non-owning* reference similar to std::string_view and
50/// llvm::ArrayRef. Pass values of this type by value.
51///
52/// Example:
53/// \code
54/// LLVMAliasSet ASet(...);
55/// LLVMAliasInfoRef AA = &ASet;
56/// \endcode
57///
58/// NOTE: AliasInfoRef::mergeWith() only works if supplied with a compatible
59/// other AliasInfo. Otherwise, it asserts out
60template <typename V, typename N>
61class PSR_POINTER() AliasInfoRef
62 : public AnalysisPropertiesMixin<AliasInfoRef<V, N>> {
63 friend class AliasInfo<V, N>;
64
65 using traits_t = AliasInfoTraits<AliasInfoRef<V, N>>;
66
67public:
68 using AliasSetPtrTy = typename traits_t::AliasSetPtrTy;
69 using AliasSetTy = typename traits_t::AliasSetTy;
70 using AllocationSiteSetPtrTy = typename traits_t::AllocationSiteSetPtrTy;
71 using n_t = typename traits_t::n_t;
72 using v_t = typename traits_t::v_t;
73
74 constexpr AliasInfoRef() noexcept = default;
75 constexpr AliasInfoRef(std::nullptr_t) noexcept : AliasInfoRef() {}
76 template <typename ConcreteAA>
77 requires(!std::is_base_of_v<AliasInfoRef, ConcreteAA> &&
78 std::is_same_v<v_t, typename ConcreteAA::v_t> &&
79 std::is_same_v<n_t, typename ConcreteAA::n_t>)
80 constexpr AliasInfoRef(ConcreteAA *AA) noexcept
81 : AA(AA), VT((std::is_empty_v<ConcreteAA> || AA) ? &VTableFor<ConcreteAA>
82 : nullptr) {}

215 template <typename ConcreteAA>
216 static constexpr VTable VTableFor = {
217 [](void *AA, ByConstRef<v_t> Of, ByConstRef<n_t> At,
218 llvm::function_ref<void(v_t)> WithAlias) {
219 if constexpr (IsAliasIterator<ConcreteAA>) {
220 return static_cast<ConcreteAA *>(AA)->forallAliasesOf(Of, At,
221 WithAlias);
222 } else {
223 auto AliasSetPtr = static_cast<ConcreteAA *>(AA)->getAliasSet(Of, At);
224 for (auto &&Alias : *AliasSetPtr) {
225 WithAlias(PSR_FWD(Alias));
226 }
227 }
228 },
229 [](void *AA, ByConstRef<v_t> Pointer1, ByConstRef<v_t> Pointer2,
230 ByConstRef<n_t> AtInstruction) {
231 return static_cast<ConcreteAA *>(AA)->alias(Pointer1, Pointer2,
232 AtInstruction);
233 },
234 [](const void *AA) noexcept -> bool {
235 return static_cast<const ConcreteAA *>(AA)->isInterProcedural();
236 },
237 [](const void *AA) noexcept -> AliasAnalysisType {
238 return static_cast<const ConcreteAA *>(AA)->getAliasAnalysisType();
239 },
240 [](void *AA, ByConstRef<v_t> Pointer, ByConstRef<n_t> AtInstruction) {
241 return static_cast<ConcreteAA *>(AA)->getAliasSet(Pointer,
242 AtInstruction);
243 },
244 [](void *AA, ByConstRef<v_t> Pointer, bool IntraProcOnly,
245 ByConstRef<n_t> AtInstruction) {
246 return static_cast<ConcreteAA *>(AA)->getReachableAllocationSites(
247 Pointer, IntraProcOnly, AtInstruction);
248 },
249 [](void *AA, ByConstRef<v_t> Pointer1, ByConstRef<v_t> Pointer2,
250 bool IntraProcOnly, ByConstRef<n_t> AtInstruction) {
251 return static_cast<ConcreteAA *>(AA)->isInReachableAllocationSites(
252 Pointer1, Pointer2, IntraProcOnly, AtInstruction);
253 },
254 [](const void *AA, llvm::raw_ostream &OS) {
255 static_cast<const ConcreteAA *>(AA)->print(OS);
256 },
257 [](const void *AA, llvm::raw_ostream &OS) {
258 static_cast<const ConcreteAA *>(AA)->printAsJson(OS);
259 },
260 [](void *AA, void *Other) {
261 if constexpr (requires(ConcreteAA &CAA) { CAA.mergeWith(CAA); }) {
262 static_cast<ConcreteAA *>(AA)->mergeWith(
263 *static_cast<ConcreteAA *>(Other));
264 } else {
265 llvm::report_fatal_error(llvm::getTypeName<ConcreteAA>() +
266 " does not support mergeWith()");
267 }
268 },
269 [](void *AA, ByConstRef<v_t> Pointer1, ByConstRef<v_t> Pointer2,
270 ByConstRef<n_t> AtInstruction, AliasResult Kind) {
271 if constexpr (requires(ConcreteAA &CAA) {
272 CAA.introduceAlias(Pointer1, Pointer2, AtInstruction,
273 Kind);
274 }) {
275 static_cast<ConcreteAA *>(AA)->introduceAlias(Pointer1, Pointer2,
276 AtInstruction, Kind);
277 } else {
278 llvm::report_fatal_error(llvm::getTypeName<ConcreteAA>() +
279 " does not support introduceAlias()");
280 }
281 },
282 [](const void *AA) noexcept -> AnalysisProperties {
283 return static_cast<const ConcreteAA *>(AA)->getAnalysisProperties();
284 },
285 []() noexcept { return llvm::getTypeName<ConcreteAA>(); },
286 [](const void *AA) noexcept {
287 delete static_cast<const ConcreteAA *>(AA);
288 },
289 }; // namespace psr

297/// \brief Similar to AliasInfoRef, but exclusively owns the held reference. Use
298/// this, if you need to decide dynamically, which alias info implementation to
299/// use.
300///
301/// Implicitly convertible to AliasInfoRef.
302///
303/// Example:
304/// \code
305/// AliasInfo AA = std::make_unique<LLVMAliasSet>(...);
306/// \endcode
307///
308template <typename V, typename N>
309class PSR_TRIVIAL_ABI PSR_OWNER() AliasInfo final : public AliasInfoRef<V, N> {
310 using base_t = AliasInfoRef<V, N>;
311
312public:
313 using typename base_t::AliasSetPtrTy;
314 using typename base_t::AliasSetTy;
315 using typename base_t::AllocationSiteSetPtrTy;
316 using typename base_t::n_t;
317 using typename base_t::v_t;
318
319 AliasInfo() noexcept = default;
320 AliasInfo(std::nullptr_t) noexcept {};
321 AliasInfo(const AliasInfo &) = delete;
322 AliasInfo &operator=(const AliasInfo &) = delete;
323 AliasInfo(AliasInfo &&Other) noexcept { swap(Other); }
324 AliasInfo &operator=(AliasInfo &&Other) noexcept {
325 auto Cpy{std::move(Other)};
326 swap(Cpy);
327 return *this;
328 }
329
330 void swap(AliasInfo &Other) noexcept {
331 std::swap(this->AA, Other.AA);
332 std::swap(this->VT, Other.VT);
333 }
334 friend void swap(AliasInfo &LHS, AliasInfo &RHS) noexcept { LHS.swap(RHS); }
335
336 template <typename ConcretePTA, typename... ArgTys>
337 explicit AliasInfo(std::in_place_type_t<ConcretePTA> /*unused*/,
338 ArgTys &&...Args)
339 : base_t(new ConcretePTA(std::forward<ArgTys>(Args)...)) {}
340
341 template <typename ConcretePTA>
342 AliasInfo(std::unique_ptr<ConcretePTA> AA) : base_t(AA.release()) {}
343
344 ~AliasInfo() noexcept {
345 if (*this) {
346 this->VT->Destroy(this->AA);
347 this->VT = nullptr;
348 this->AA = nullptr;
349 }
350 }
351
352 [[nodiscard]] base_t asRef() & noexcept { return *this; }
353 [[nodiscard]] AliasInfoRef<V, N> asRef() const & noexcept { return *this; }
354 [[nodiscard]] AliasInfoRef<V, N> asRef() && = delete;
355
356 /// For better interoperability with unique_ptr
357 [[nodiscard]] base_t get() & noexcept { return asRef(); }
358 [[nodiscard]] AliasInfoRef<V, N> get() const & noexcept { return asRef(); }
359 [[nodiscard]] AliasInfoRef<V, N> get() && = delete;
360
361 using base_t::operator psr::AliasIteratorRef<V, N>;
362 constexpr operator AliasIteratorRef<V, N>() && noexcept = delete;
363};
