28/// FIXME: This is not entirely correct: Does not skip ignored statements and
29/// does not work for backwards analyses.
30/// The right way would be to ask the ICFG, but we don't have a reference to it
31/// here yet (TODO!)
32[[nodiscard]] auto resultsAtInLLVMSSAImpl(const auto &Self, const auto &Stmt,
33 bool AllowOverapproximation) {
34 if (Stmt->getType()->isVoidTy()) {
35 return Self.row(Stmt);
36 }
37 if (!Stmt->getNextNode()) {
38 auto GetStartRow = [&Self](const llvm::BasicBlock *BB) -> decltype(auto) {
39 const auto *First = &BB->front();
40 if (llvm::isa<llvm::DbgInfoIntrinsic>(First)) {
41 First = First->getNextNonDebugInstruction();
42 }
43 return Self.row(First);
44 };
45
46 // We have reached the end of a BasicBlock. If there is a successor BB
47 // that only has one predecessor, we are lucky and can just take results
48 // from there
49 for (const llvm::BasicBlock *Succ : llvm::successors(Stmt)) {
50 if (Succ->hasNPredecessors(1)) {
51 return GetStartRow(Succ);
52 }
53 }
54
55 if (!AllowOverapproximation) {
56 llvm::report_fatal_error("[resultsAtInLLVMSSA]: Cannot precisely "
57 "collect the results at instruction " +
58 llvm::Twine(llvmIRToString(Stmt)));
59 }
60
61 // There is no successor with only one predecessor.
62 // All we can do is merge the results from all successors to get a sound
63 // overapproximation. This is not optimal and may be replaced in the
64 // future.
65 PHASAR_LOG_LEVEL(WARNING, "[resultsAtInLLVMSSA]: Cannot precisely "
66 "collect the results at instruction "
67 << llvmIRToString(Stmt)
68 << ". Use a sound, but potentially "
69 "imprecise overapproximation");
70 std::remove_cvref_t<decltype(Self.resultsAt(Stmt))> Ret;
71 using l_t = typename decltype(Ret)::mapped_type;
72 for (const llvm::BasicBlock *Succ : llvm::successors(Stmt)) {
73 const auto &Row = GetStartRow(Succ);
74 for (const auto &[Fact, Value] : Row) {
75 auto [It, Inserted] = Ret.try_emplace(Fact, Value);
76 if (!Inserted && Value != It->second) {
77 if constexpr (HasJoinLatticeTraits<l_t>) {
78 It->second = JoinLatticeTraits<l_t>::join(It->second, Value);
79 } else {
80 // We have no way of correctly merging, so set the value to the
81 // default constructed l_t hoping it marks BOTTOM.
82 It->second = l_t();
83 }
84 }
85 }
86 }
87 return Ret;
88 }
89 assert(Stmt->getNextNode() && "Expected to find a valid successor node!");
90 return Self.row(Stmt->getNextNode());
91}

93[[nodiscard]] auto resultAtInLLVMSSAImpl(const auto &Self, const auto &Stmt,
94 const auto &Fact,
95 bool AllowOverapproximation) {
96 if (Stmt->getType()->isVoidTy()) {
97 return Self.resultAt(Stmt, Fact);
98 }
99
100 if (auto Next = Stmt->getNextNode()) {
101 return Self.resultAt(Next, Fact);
102 }
103
104 auto GetStartVal = [&Self,
105 &Fact](const llvm::BasicBlock *BB) -> decltype(auto) {
106 const auto *First = &BB->front();
107 if (llvm::isa<llvm::DbgInfoIntrinsic>(First)) {
108 First = First->getNextNonDebugInstruction();
109 }
110 return Self.resultAt(First, Fact);
111 };
112
113 // We have reached the end of a BasicBlock. If there is a successor BB
114 // that only has one predecessor, we are lucky and can just take results
115 // from there
116 for (const llvm::BasicBlock *Succ : llvm::successors(Stmt)) {
117 if (Succ->hasNPredecessors(1)) {
118 return GetStartVal(Succ);
119 }
120 }
121
122 if (!AllowOverapproximation) {
123 llvm::report_fatal_error("[resultsAtInLLVMSSA]: Cannot precisely "
124 "collect the results at instruction " +
125 llvm::Twine(llvmIRToString(Stmt)));
126 }
127
128 // There is no successor with only one predecessor.
129 // All we can do is merge the results from all successors to get a sound
130 // overapproximation. This is not optimal and may be replaced in the
131 // future.
132 PHASAR_LOG_LEVEL(WARNING, "[resultAtInLLVMSSA]: Cannot precisely "
133 "collect the results at instruction "
134 << *Stmt
135 << ". Use a sound, but potentially "
136 "imprecise overapproximation");
137 auto It = llvm::succ_begin(Stmt);
138 auto End = llvm::succ_end(Stmt);
139
140 using l_t = std::remove_cvref_t<decltype(Self.resultAt(Stmt, Fact))>;
141
142 l_t Ret{};
143 if (It != End) {
144 Ret = GetStartVal(*It);
145 for (++It; It != End; ++It) {
146 const auto &Val = GetStartVal(*It);
147 if constexpr (HasJoinLatticeTraits<l_t>) {
148 Ret = JoinLatticeTraits<l_t>::join(Ret, Val);
149 if (Ret == JoinLatticeTraits<l_t>::bottom()) {
150 break;
151 }
152 } else {
153 if (Ret != Val) {
154 // We have no way of correctly merging, so set the value to the
155 // default constructed l_t hoping it marks BOTTOM.
156 return l_t{};
157 }
158 }
159 }
160 }
161 return Ret;
162}

164template <typename Derived, typename N, typename D, typename L>
165auto SolverResultsBase<Derived, N, D, L>::resultsAtInLLVMSSA(
166 ByConstRef<n_t> Stmt, bool AllowOverapproximation, bool StripZero) const
167 -> std::unordered_map<d_t, l_t>
168 requires same_as_decay<std::remove_pointer_t<n_t>, llvm::Instruction>
169{
170 std::unordered_map<d_t, l_t> Result =
171 resultsAtInLLVMSSAImpl(self(), Stmt, AllowOverapproximation);
172 if (StripZero) {
173 Result.erase(self().ZV);
174 }
175 return Result;
176}
