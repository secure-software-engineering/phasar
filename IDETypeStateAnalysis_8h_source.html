214 struct TSEdgeFunction {
215 using l_t = l_t;
216 const TypeStateDescriptionTy *TSD{};
217 // XXX: Do we really need a string here? Can't we just use an integer or sth
218 // else that is cheap?
219 std::string Token;

222 [[nodiscard]] l_t computeTarget(l_t Source) const {
223
224 // assert((Source != TSD->top()) && "Error: call computeTarget with
225 // TOP\n");
226
227 auto CurrentState = TSD->getNextState(
228 Token, Source == TSD->top() ? TSD->uninit() : Source, CallSite);
229 PHASAR_LOG_LEVEL(DEBUG, "State machine transition: ("
230 << Token << " , " << LToString(Source)
231 << ") -> " << LToString(CurrentState));
232 return CurrentState;
233 }
234
235 static EdgeFunction<l_t> compose(EdgeFunctionRef<TSEdgeFunction> This,
236 const EdgeFunction<l_t> &SecondFunction) {
237 if (auto Default = defaultComposeOrNull(This, SecondFunction)) {
238 return Default;
239 }
240
241 return TSEdgeFunctionComposer{This, SecondFunction, This->TSD};
242 }
243
244 static EdgeFunction<l_t> join(EdgeFunctionRef<TSEdgeFunction> This,
245 const EdgeFunction<l_t> &OtherFunction) {
246 if (auto Default = defaultJoinOrNull(This, OtherFunction)) {
247 return Default;
248 }
249
250 return makeAllBottom(This->TSD);
251 }

269 TSConstant(l_t Value, const TypeStateDescriptionTy *TSD) noexcept

276 template <typename LL = l_t,
277 typename = std::enable_if_t<HasJoinLatticeTraits<LL>>>
278 TSConstant(l_t Value, EmptyType /*unused*/ = {}) noexcept
279 : ConstantEdgeFunction<l_t>{Value} {

285 /// XXX: Cannot default compose() and join(), because l_t does not implement
286 /// JoinLatticeTraits (because bottom value is not constant)
287 template <typename ConcreteEF>
288 static EdgeFunction<l_t> compose(EdgeFunctionRef<ConcreteEF> This,
289 const EdgeFunction<l_t> &SecondFunction) {
290
291 if (auto Default = defaultComposeOrNull(This, SecondFunction)) {
292 return Default;
293 }
294
295 l_t Ret = SecondFunction.computeTarget(This->Value);
296 if (Ret == This->Value) {
297 return This;
298 }
299 if (isBottom(Ret, This->TSD)) {
300 return makeAllBottom(This->TSD);
301 }
302
303 return TSConstant{Ret, This->TSD};
304 }
305
306 template <typename ConcreteEF>
307 static EdgeFunction<l_t> join(EdgeFunctionRef<ConcreteEF> This,
308 const EdgeFunction<l_t> &OtherFunction) {
309 if (auto Default = defaultJoinOrNull(This, OtherFunction)) {
310 return Default;
311 }
312
313 auto Top = [TSD = This->TSD] {
314 if constexpr (HasJoinLatticeTraits<l_t>) {
315 return JoinLatticeTraits<l_t>::top();
316 } else {
317 return TSD->top();
318 }
319 }();
320 if (const auto *C = llvm::dyn_cast<TSConstant>(OtherFunction)) {
321 if (C->Value == This->Value || C->Value == Top) {
322 return This;
323 }
324 if (This->Value == Top) {
325 return OtherFunction;
326 }
327 }
328 return makeAllBottom(This->TSD);
329 }
330
331 bool operator==(const TSConstant &Other) const noexcept {
332 return this->Value == Other.Value;
333 }
334
335 friend llvm::raw_ostream &operator<<(llvm::raw_ostream &OS,
336 const TSConstant &EF) {
337 return OS << "TSConstant[" << LToString(EF.Value) << "]";
338 }

425 EdgeFunction<l_t>
426 getCallToRetEdgeFunction(n_t CallSite, d_t CallNode, n_t /*RetSite*/,
427 d_t RetSiteNode,
428 llvm::ArrayRef<f_t> Callees) override {
429 const auto *CS = llvm::cast<llvm::CallBase>(CallSite);
430 for (const auto *Callee : Callees) {
431 std::string DemangledFname = llvm::demangle(Callee->getName().str());
432
433 // For now we assume that we can only generate from the return value.
434 // We apply the same edge function for the return value, i.e. callsite.
435 if (TSD->isFactoryFunction(DemangledFname)) {
436 PHASAR_LOG_LEVEL(DEBUG, "Processing factory function");
437 if (isZeroValue(CallNode) && RetSiteNode == CS) {
438 return TSConstant{
439 TSD->getNextState(DemangledFname, TSD->uninit(), CS), TSD};
440 }
441 }
442
443 // For every consuming parameter and all its aliases and relevant alloca's
444 // we apply the same edge function.
445 if (TSD->isConsumingFunction(DemangledFname)) {
446 PHASAR_LOG_LEVEL(DEBUG, "Processing consuming function");
447 for (auto Idx : TSD->getConsumerParamIdx(DemangledFname)) {
448 const auto &AliasAndAllocas =
449 getWMAliasesAndAllocas(CS->getArgOperand(Idx));
450
451 if (CallNode == RetSiteNode && AliasAndAllocas.count(CallNode)) {
452 return TSEdgeFunction{TSD, DemangledFname, CS};
453 }
454 }
455 }
456 }
457 return EdgeIdentity<l_t>{};
458 }

470 /**
471 * We have a lattice with BOTTOM representing all information
472 * and TOP representing no information. The other lattice elements
473 * are defined by the type state description, i.e. represented by the
474 * states of the finite state machine.
475 *
476 * @note Only one-level lattice's are handled currently
477 */
478 l_t join(l_t Lhs, l_t Rhs) override {
479 if (Lhs == Rhs) {
480 return Lhs;
481 }
482 if (Lhs == TSD->top()) {
483 return Rhs;
484 }
485 if (Rhs == TSD->top()) {
486 return Lhs;
487 }
488 return TSD->bottom();
489 }
