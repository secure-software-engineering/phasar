31/// \brief A dense, partially-populated map keyed by a sequential integer-like
32/// id.
33///
34/// Models a subset of the std::unordered_map / llvm::DenseMap interface for
35/// keys that are dense unsigned integer ids (e.g. enum classes over uint32_t).
36/// Entries are stored in a flat array of uninitialized ValueT storage;
37/// only explicitly inserted entries are ever constructed — there is no upfront
38/// default-construction of the entire value array.
39///
40/// Reallocation moves all live entries via their move constructors and then
41/// destroys the originals in one pass. **Move constructors are assumed to be
42/// nothrow**.
43///
44/// \remarks Partially implemented by Claude Sonnet 4.6
45///
46/// \tparam IdT Key type. Must satisfy SmallIdType (fits in uint32_t,
47/// losslessly convertible to/from size_t).
48/// \tparam ValueT Mapped value type.
49/// \tparam Allocator Allocator whose value_type is ValueT.
50/// Defaults to std::allocator<ValueT>.
51template <SmallIdType IdT, typename ValueT,
52 typename Allocator = std::allocator<ValueT>>
53class ValueIdMap {
54 using AllocTraits = std::allocator_traits<Allocator>;

153 /// Constructs from an initializer list. \p V need not equal \p ValueT;
154 /// any type constructible to \p ValueT is accepted, avoiding an
155 /// intermediate copy (e.g. string literals when ValueT is std::string).
156 /// Duplicate keys are silently dropped (first occurrence wins).
157 template <typename V = ValueT>
158 requires std::is_constructible_v<ValueT, const V &>
159 ValueIdMap(std::initializer_list<std::pair<const IdT, V>> Init,
160 const allocator_type &A = allocator_type{})
161 : ValueIdMap(Init.begin(), Init.end(), A) {}

163 /// Constructs from an iterator range \p [First, Last).
164 /// Duplicate keys are silently dropped (first occurrence wins).
165 template <typename InputIt>
166 ValueIdMap(InputIt First, InputIt Last,
167 const allocator_type &A = allocator_type{})
168 : ValueIdMap(A) {
169 if constexpr (std::is_base_of_v<std::random_access_iterator_tag,
170 typename std::iterator_traits<
171 InputIt>::iterator_category>) {
172 reserve(static_cast<size_t>(std::distance(First, Last)));
173 }
174 for (; First != Last; ++First) {
175 try_emplace(First->first, First->second);
176 }
177 }

179 /// Copy constructor. The allocator is obtained via
180 /// allocator_traits::select_on_container_copy_construction.
181 ValueIdMap(const ValueIdMap &Other)
182 : ValueIdMap(Other, AllocTraits::select_on_container_copy_construction(
183 Other.Alloc)) {}
184
185 /// Copy constructor with explicit allocator.
186 ValueIdMap(const ValueIdMap &Other, const allocator_type &A)
187 : Alloc(A), IsSet(Other.IsSet), Capacity(Other.Capacity) {
188 if (!Capacity) {
189 return;
190 }
191 Slots = AllocTraits::allocate(Alloc, Capacity);
192 try {
193 Other.IsSet.foreach ([&](IdT Key) {
194 AllocTraits::construct(Alloc, slot(Key), *Other.slot(Key));
195 });
196 } catch (...) {
197 freeStorage();
198 IsSet.clear();
199 throw;
200 }
201 }
202
203 /// Move constructor. The allocator is move-constructed from \p Other.
204 ValueIdMap(ValueIdMap &&Other) noexcept
205 : Alloc(std::move(Other.Alloc)), IsSet(std::move(Other.IsSet)),
206 Capacity(std::exchange(Other.Capacity, 0)),
207 Slots(std::exchange(Other.Slots, nullptr)) {}
208
209 /// Move constructor with explicit allocator.
210 ///
211 /// If the supplied allocator compares equal to Other's allocator the
212 /// storage is stolen in O(1); otherwise each live element is move-constructed
213 /// individually using the supplied allocator.
214 ValueIdMap(ValueIdMap &&Other, const allocator_type &A) : Alloc(A) {
215 if (Alloc == Other.Alloc) {
216 stealFrom(Other);
217 } else {
218 // Allocators differ: move elements individually into fresh storage.
219 IsSet = Other.IsSet;
220 if (Other.Capacity) {
221 Capacity = Other.Capacity;
222 Slots = AllocTraits::allocate(Alloc, Capacity);
223 Other.IsSet.foreach ([&](IdT Key) {
224 ValueT *Src = Other.slot(Key);
225 AllocTraits::construct(Alloc, slot(Key), std::move(*Src));
226 AllocTraits::destroy(Other.Alloc, Src);
227 });
228 Other.IsSet.clear();
229 Other.freeStorage();
230 }
231 }
232 }
233
234 /// Copy assignment. Propagates the allocator when
235 /// allocator_traits::propagate_on_container_copy_assignment is true.
236 ValueIdMap &operator=(const ValueIdMap &Other) {
237 if (this == &Other) {
238 return *this;
239 }
240 clear();
241 if constexpr (AllocTraits::propagate_on_container_copy_assignment::value) {
242 if (Alloc != Other.Alloc) {
243 freeStorage();
244 Alloc = Other.Alloc;
245 }
246 }
247 reallocAtLeast(Other.Capacity);
248 IsSet = Other.IsSet;
249 Other.IsSet.foreach ([&](IdT Key) {
250 AllocTraits::construct(Alloc, slot(Key), *Other.slot(Key));
251 });
252 return *this;
253 }
254
255 /// Replaces the contents with those of \p Init.
256 /// Duplicate keys are silently dropped (first occurrence wins).
257 template <typename V = ValueT>
258 requires std::is_constructible_v<ValueT, const V &>
259 ValueIdMap &operator=(std::initializer_list<std::pair<const IdT, V>> Init) {
260 clear();
261 reserve(Init.size());
262 for (auto &&[Key, Val] : Init) {
263 try_emplace(Key, Val);
264 }
265 return *this;
266 }
267
268 /// Move assignment. Propagates the allocator when
269 /// allocator_traits::propagate_on_container_move_assignment is true.
270 ValueIdMap &operator=(ValueIdMap &&Other) noexcept(
271 AllocTraits::propagate_on_container_move_assignment::value ||
272 AllocTraits::is_always_equal::value) {
273 if (this == &Other) {
274 return *this;
275 }
276 clear();
277 if constexpr (AllocTraits::propagate_on_container_move_assignment::value) {
278 freeStorage();
279 Alloc = std::move(Other.Alloc);
280 stealFrom(Other);
281 } else if (Alloc == Other.Alloc) {
282 freeStorage();
283 stealFrom(Other);
284 } else {
285 // Unequal allocators, no propagation: move elements individually.
286 reallocAtLeast(Other.Capacity);
287 IsSet = Other.IsSet;
288 Other.IsSet.foreach ([&](IdT Key) {
289 ValueT *Src = Other.slot(Key);
290 AllocTraits::construct(Alloc, slot(Key), std::move(*Src));
291 AllocTraits::destroy(Other.Alloc, Src);
292 });
293 Other.IsSet.clear();
294 Other.freeStorage();
295 }
296 return *this;
297 }

403 // ── Modifiers ─────────────────────────────────────────────────────────────

405 /// Constructs a value from \p Args for \p Key if not already present.
406 /// Returns {iterator-to-entry, true} on insertion and
407 /// {iterator-to-existing-entry, false} otherwise.
408 template <typename... ArgsT>
409 std::pair<iterator, bool> try_emplace(IdT Key, ArgsT &&...Args) {
410 if (!inbounds(Key)) {
411 grow(std::max(size_t(Key) + 1, Capacity ? Capacity * 2 : 8));
412 } else if (IsSet.contains(Key)) {
413 return {iterator(this, Key), false};
414 }
415 AllocTraits::construct(Alloc, slot(Key), PSR_FWD(Args)...);
416 IsSet.insert(Key);
417 return {iterator(this, Key), true};
418 }
419
420 std::pair<iterator, bool> insert(value_type Val) {
421 return try_emplace(Val.first, std::move(Val.second));
422 }
423
424 /// Inserts or replaces the entry for \p Key with \p Obj.
425 template <typename M>
426 // NOLINTNEXTLINE(readability-identifier-naming) -- STL API name
427 std::pair<iterator, bool> insert_or_assign(IdT Key, M &&Obj) {
428 if (IsSet.contains(Key)) {
429 *slot(Key) = std::forward<M>(Obj);
430 return {iterator(this, Key), false};
431 }
432 return try_emplace(Key, std::forward<M>(Obj));
433 }

497 /// Frees current storage and allocates a fresh array of \p N elements.
498 /// No-op if N <= Capacity.
499 constexpr void reallocAtLeast(size_t N) {
500 if (N > Capacity) {
501 freeStorage();
502 Slots = AllocTraits::allocate(Alloc, N);
503 Capacity = N;
504 }
505 }
506
507 /// Deallocates the storage array if non-null and resets Slots/Capacity.
508 constexpr void freeStorage() noexcept {
509 if (Slots) {
510 AllocTraits::deallocate(Alloc, Slots, Capacity);
511 Slots = nullptr;
512 Capacity = 0;
513 }
514 }
515
516 /// Takes ownership of Other's storage (Other must have been cleared first).
517 void stealFrom(ValueIdMap &Other) noexcept {
518 IsSet = std::move(Other.IsSet);
519 Capacity = std::exchange(Other.Capacity, 0);
520 Slots = std::exchange(Other.Slots, nullptr);
521 }
522
523 void grow(size_t NewCap) {
524 assert(NewCap > Capacity);
525 ValueT *New = AllocTraits::allocate(Alloc, NewCap);
526
527 if constexpr (IsTriviallyRelocatable<ValueT>) {
528 // Trivially relocatable: a flat memcpy of all used slots suffices.
529 // Uninitialized slots contain indeterminate bytes, which is safe to copy.
530 // No per-element construction or destruction is needed.
531 if (Slots) {
532 memcpy(New, Slots, Capacity * sizeof(ValueT));
533 }
534 } else {
535 // Move-construct each live entry into new storage and destroy the
536 // original in one pass. Assumes nothrow move construction; types with
537 // throwing moves are not supported.
538 IsSet.foreach ([&](IdT Key) {
539 ValueT *Src = slot(Key);
540 AllocTraits::construct(Alloc, New + size_t(Key), // NOLINT
541 std::move(*Src));
542 AllocTraits::destroy(Alloc, Src);
543 });
544 }
545
546 if (Slots) {
547 AllocTraits::deallocate(Alloc, Slots, Capacity);
548 }
549 Slots = New;
550 Capacity = NewCap;
551 }
552
553 // ── Data members ───────────────────────────────────────────────────────────
554
555 // Alloc must be declared first so it is available for deallocation when
556 // other members are destroyed. [[no_unique_address]] enables the empty-base
557 // optimisation for stateless allocators such as std::allocator.
558 [[no_unique_address]] Allocator Alloc{};
559 // IsSet must be declared before Slots so that if its copy constructor throws
560 // during ValueIdMap's copy constructor, Slots has not yet been allocated.
561 BitSet<IdT> IsSet;
562 size_t Capacity = 0;
563 ValueT *Slots = nullptr;
564};
