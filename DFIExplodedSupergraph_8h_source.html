107 /// Computes the same-level reachability from the entries of Fun to its exit
108 /// vertices and calls Handler(CallVtx, Entry, Exit) for each new
109 /// (Entry, Exit) pair and each call vertex CallVtx calling Entry.
110 ///
111 /// With FollowReturnsPastSeeds, also calls UnbalancedHandler(Exit) for each
112 /// new exit vertex that is reachable without calling context, i.e., from a
113 /// seed without callers or from an unbalanced return. The solver must
114 /// return it to all callers of Fun.
115 void summarize(
116 FunctionId Fun,
117 std::invocable<dfi::VertexId, dfi::EntryId, dfi::VertexId> auto Handler,
118 std::invocable<dfi::VertexId> auto UnbalancedHandler) {
119 indexFunction(Fun);
120
121 for (auto Entry : FunState[Fun].Entries) {
122 auto NewExits = updateSummaryExits(Fun, Entry);
123 // Handler may add vertices and summary edges, but no entries or callers
124 for (size_t I = 0; I != EntryCallers[Entry].size(); ++I) {
125 for (auto Exit : NewExits) {
126 std::invoke(Handler, EntryCallers[Entry][I], Entry, Exit);
127 }
128 }
129 }
130
131 if (FollowReturnsPastSeeds) {
132 // UnbalancedHandler may add vertices, roots and unbalanced returns, but
133 // no entries or callers
134 for (auto Exit : updateUnbalancedExits(Fun)) {
135 std::invoke(UnbalancedHandler, Exit);
136 }
137 }
138 }
