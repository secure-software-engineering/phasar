28/// The PAG-node used for LLVM-based pointer-assignment graphs. It consists of
29/// regular LLVM values + special return-slots per non-void function. In that
30/// case, the stored pointer is the corresponding function.
31///
32/// This class is compatible with llvm::TinyPtrVector and other APIs relying on
33/// llvm::PointerLikeTypeTraits.
34struct PAGVariable : public llvm::PointerIntPair<const llvm::Value *, 1, bool> {
35 using Base = llvm::PointerIntPair<const llvm::Value *, 1, bool>;
36
37 struct Return {
38 const llvm::Function *Fun;
39 };
40
41 PAGVariable() noexcept = default; // To enable move-assign in TinyPtrVector
42 PAGVariable(const llvm::Value *Var) noexcept : Base(Var, false) {}
43 PAGVariable(Return RetVar) noexcept : Base(RetVar.Fun, true) {
44 assert(RetVar.Fun != nullptr);
45 }
46
47 [[nodiscard]] bool isReturnVariable() const noexcept {
48 return this->getInt();
49 }
50
51 [[nodiscard]] bool isInFunction() const noexcept {
52 return isReturnVariable() ||
53 llvm::isa<llvm::Argument, llvm::Instruction>(this->getPointer());
54 }
55
56 [[nodiscard]] const llvm::Function *getFunction() const noexcept {
57 const auto *Ptr = this->getPointer();
58 if (isReturnVariable()) {
59 return llvm::cast<llvm::Function>(Ptr);
60 }
61 if (const auto *Arg = llvm::dyn_cast<llvm::Argument>(Ptr)) {
62 return Arg->getParent();
63 }
64 if (const auto *Inst = llvm::dyn_cast<llvm::Instruction>(Ptr)) {
65 return Inst->getFunction();
66 }
67
68 return nullptr;
69 }
70
71 [[nodiscard]] const llvm::Value *get() const noexcept {
72 return this->getPointer();
73 }
74 explicit operator const llvm::Value *() const noexcept { return get(); }
75
76 [[nodiscard]] const llvm::Value *valueOrNull() const noexcept {
77 return isReturnVariable() ? nullptr : get();
78 }
79
80 friend bool operator==(PAGVariable V1, PAGVariable V2) noexcept {
81 return V1.getOpaqueValue() == V2.getOpaqueValue();
82 }
83 friend bool operator!=(PAGVariable V1, PAGVariable V2) noexcept {
84 return !(V1 == V2);
85 }
86
87 friend auto hash_value(PAGVariable V) noexcept {
88 return llvm::hash_value(V.getOpaqueValue());
89 }
90};
91
92[[nodiscard]] std::string to_string(PAGVariable Var);
