31/// The PAG-node used for LLVM-based pointer-assignment graphs. It consists of
32/// regular LLVM values + special return-slots per non-void function. In that
33/// case, the stored pointer is the corresponding function.
34///
35/// This class is compatible with llvm::TinyPtrVector and other APIs relying on
36/// llvm::PointerLikeTypeTraits.
37struct PAGVariable : public llvm::PointerIntPair<const llvm::Value *, 1, bool> {
38 using Base = llvm::PointerIntPair<const llvm::Value *, 1, bool>;
39
40 struct Return {
41 const llvm::Function *Fun;
42 };
43
44 PAGVariable() noexcept = default; // To enable move-assign in TinyPtrVector
45 PAGVariable(const llvm::Value *Var) noexcept : Base(Var, false) {}
46 PAGVariable(Return RetVar) noexcept : Base(RetVar.Fun, true) {
47 assert(RetVar.Fun != nullptr);
48 }
49
50 [[nodiscard]] bool isReturnVariable() const noexcept {
51 return this->getInt();
52 }
53
54 [[nodiscard]] bool isInFunction() const noexcept {
55 return isReturnVariable() ||
56 llvm::isa<llvm::Argument, llvm::Instruction>(this->getPointer());
57 }
58
59 [[nodiscard]] const llvm::Function *getFunction() const noexcept {
60 const auto *Ptr = this->getPointer();
61 if (isReturnVariable()) {
62 return llvm::cast<llvm::Function>(Ptr);
63 }
64 if (const auto *Arg = llvm::dyn_cast<llvm::Argument>(Ptr)) {
65 return Arg->getParent();
66 }
67 if (const auto *Inst = llvm::dyn_cast<llvm::Instruction>(Ptr)) {
68 return Inst->getFunction();
69 }
70
71 return nullptr;
72 }
73
74 [[nodiscard]] const llvm::Value *get() const noexcept {
75 return this->getPointer();
76 }
77 explicit operator const llvm::Value *() const noexcept { return get(); }
78
79 [[nodiscard]] const llvm::Value *valueOrNull() const noexcept {
80 return isReturnVariable() ? nullptr : get();
81 }
82
83 friend bool operator==(PAGVariable V1, PAGVariable V2) noexcept {
84 return V1.getOpaqueValue() == V2.getOpaqueValue();
85 }
86 friend bool operator!=(PAGVariable V1, PAGVariable V2) noexcept {
87 return !(V1 == V2);
88 }
89
90 friend auto hash_value(PAGVariable V) noexcept {
91 return llvm::hash_value(V.getOpaqueValue());
92 }
93};
94
95[[nodiscard]] std::string to_string(PAGVariable Var);

104/// Concrete \c PAGBuilder for LLVM IR.
105///
106/// Traverses all functions in the given \c LLVMProjectIRDB and emits PAG
107/// nodes and edges to the provided \c PBStrategyRef.
108///
109/// Achieves basic field sensitivity by handling constant GEPs in an ad-hoc
110/// manner.
111///
112/// An optional \c LLVMFunctionDataFlowFacts object can be supplied to apply
113/// pre-computed library summaries at potentially declaration-only calls.
114///
115/// An optional \c MemSSAProviderFn enables flow-sensitive load edges: for each
116/// load instruction the builder queries the provider for the function's
117/// \c MemorySSA and connects only the reaching stores rather than all stores to
118/// the same base pointer. Use \c withBuiltinMemSSA() when no external analysis
119/// manager is available.
120class LLVMPAGBuilder : public PAGBuilder<LLVMPAGDomain> {
121public:
122 /// Callback that maps a function to its \c MemorySSA. The returned pointer
123 /// must remain valid for the duration of \c buildPAG. Returning null
124 /// disables flow-sensitivity for that function.
125 using MemSSAProviderFn =
126 llvm::unique_function<llvm::MemorySSA *(const llvm::Function &)>;
127
128 constexpr LLVMPAGBuilder() noexcept = default;
129
130 /// \param MLSum Pre-computed library summary facts. If non-null, callee
131 /// analysis for matched library functions is replaced by these summaries.
132 /// \param MemSSAProvider Optional per-function MemorySSA provider for
133 /// flow-sensitive load handling.
134 LLVMPAGBuilder(const library_summary::LLVMFunctionDataFlowFacts *MLSum,
135 MemSSAProviderFn MemSSAProvider = nullptr) noexcept
136 : MLSum(MLSum), MemSSAProvider(std::move(MemSSAProvider)) {}
137
138 /// Returns a builder that constructs \c DominatorTree + \c BasicAA +
139 /// \c MemorySSA internally for each function on demand. No prior alias
140 /// analysis or \c FunctionAnalysisManager required.
141 [[nodiscard]] static LLVMPAGBuilder withBuiltinMemSSA(
142 const library_summary::LLVMFunctionDataFlowFacts *MLSum = nullptr);
143
144 void buildPAG(const LLVMProjectIRDB &IRDB, ValueCompressor<v_t> &VC,
145 pag::PBStrategyRef<LLVMPAGDomain> Strategy) override;
146
147private:
148 struct PAGBuildData;
149
150 const library_summary::LLVMFunctionDataFlowFacts *MLSum{};
151 MemSSAProviderFn MemSSAProvider{};
152};
