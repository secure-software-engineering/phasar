49/// \file
50/// PhASAR's performance measurement mechanism (PAMM).
51/// Provides counters, histograms, and timers for fine-grained performance
52/// tracking. Use the macros from PAMMMacros.h to declare static instances
53/// of these metrics. On construction, they are automatically registered in the
54/// static pamm::Registry, so that report-generation can be automated.
55/// The macros expect a valid C++ identifier as name, and a severity-level
56/// (Core/Full) under which the metric should be active.
57///
58/// Metrics are categorizable via a pamm::Category. The
59/// performance report groups by category; categories can be enabled/disabled at
60/// runtime. By convention, the macros register the metrics under the category
61/// with the identifier 'PAMMCategory'; create a new static PAMMCategory with
62/// the macro PAMM_CATEGORY(name).
63///
64/// Currently, PAMM can be run with all metrics (severity level 2 = Full) or
65/// only core metrics (severity level 1 = Core) enabled. The severity level can
66/// be changed when building PhASAR. The CMake option is
67/// -DPHASAR_ENABLE_PAMM=[Off/Core/Full].
68///
69/// Thread-safety:
70/// If PhASAR is configured with the cmake-option -DPHASAR_THREAD_SAFE_PAMM=ON,
71/// PAMM provides some thread-safety guarentees: Generally, construction and
72/// destruction of metrics is NOT thread-safe. The pamm::Registry is therefore
73/// also NOT thread-safe. Adding data, i.e, incrementing counters, adding to
74/// histograms, starting/stopping timers is protected by suitable
75/// synchronization primitives and can therefore be safely performed
76/// concurrently.
77

180struct MinMaxCounterBase : PAMMBase {
181 std::atomic_size_t Min = SIZE_MAX;
182 std::atomic_size_t Max = 0;
183 std::atomic_size_t Sum{};
184 std::atomic_size_t NumSamples{};
185
186 constexpr void add(size_t Offset) noexcept {
187 Sum.fetch_add(Offset, std::memory_order_relaxed);
188 NumSamples.fetch_add(1, std::memory_order_relaxed);
189
190 {
191 auto PrevMin = Min.load(std::memory_order_relaxed);
192 while (PrevMin > Offset &&
193 Min.compare_exchange_weak(PrevMin, Offset,
194 std::memory_order_relaxed)) {
195 }
196 }
197 {
198 auto PrevMax = Max.load(std::memory_order_relaxed);
199 while (PrevMax < Offset &&
200 Max.compare_exchange_weak(PrevMax, Offset,
201 std::memory_order_relaxed)) {
202 }
203 }
204 };
205
206 [[nodiscard]] double getAverage() const noexcept {
207 return double(Sum.load(std::memory_order_relaxed)) /
208 double(getNumSamples());
209 }
210 [[nodiscard]] size_t getNumSamples() const noexcept {
211 return NumSamples.load(std::memory_order_relaxed);
212 }
213
214 void clear() noexcept {
215 Min = SIZE_MAX;
216 Max = 0;
217 Sum = 0;
218 NumSamples = 0;
219 }
220};

222struct TimerBase : PAMMBase {
223
224 static constexpr std::chrono::steady_clock::time_point None =
225 std::chrono::steady_clock::time_point::max();
226
227 std::atomic<std::chrono::steady_clock::time_point> StartPoint = None;
228 // no atomic arithmetic with nanoseconds, so use int64_t here:
229 std::atomic<int64_t> Acc{};
230
231 [[nodiscard]] constexpr bool isStarted() const noexcept {
232 // XXX: Revisit the memory-order here: Can it be relaxed?
233 return StartPoint.load(std::memory_order_acquire) != None;
234 }
235
236 void start() noexcept {
237 assert(!isStarted() && "Starting an already running timer is not allowed");
238 StartPoint.store(std::chrono::steady_clock::now(),
239 std::memory_order_release);
240 }
241
242 void stop() noexcept {
243 auto EndPoint = std::chrono::steady_clock::now();
244 auto StartPoint =
245 this->StartPoint.exchange(None, std::memory_order_acq_rel);
246 assert(StartPoint != None && "Stopping a not-running timer is not allowed");
247 Acc.fetch_add((EndPoint - StartPoint).count(), std::memory_order_relaxed);
248 }
249
250 void reset() noexcept {
251 Acc = 0;
252 StartPoint = None;
253 }
254
255 [[nodiscard]] constexpr auto elapsedNanos() const noexcept {
256 return std::chrono::nanoseconds(Acc.load(std::memory_order_relaxed));
257 }
258
259 [[nodiscard]] constexpr std::chrono::nanoseconds
260 pendingNanos() const noexcept {
261 auto StartPoint = this->StartPoint.load(std::memory_order_relaxed);
262 if (StartPoint == None) {
263 return {};
264 }
265 auto EndPoint = std::chrono::steady_clock::now();
266 return EndPoint - StartPoint;
267 }
268
269 [[nodiscard]] hms elapsed() const noexcept { return elapsedNanos(); }
270};
271
272#else
273struct CounterBase : PAMMBase {
274 ptrdiff_t Ctr{};
275
276 constexpr void add(ptrdiff_t Offset) noexcept { Ctr += Offset; }
277
278 [[nodiscard]] constexpr ptrdiff_t value() const noexcept { return Ctr; }
279};

281struct MinMaxCounterBase : PAMMBase {
282 size_t Min = SIZE_MAX;
283 size_t Max = 0;
284 Sampler Avg{};
285
286 constexpr void add(size_t Offset) noexcept {
287 Avg.addSample(Offset);
288 if (Offset > Max) {
289 Max = Offset;
290 }
291 if (Offset < Min) {
292 Min = Offset;
293 }
294 };
295
296 [[nodiscard]] double getAverage() const noexcept { return Avg.getAverage(); }
297 [[nodiscard]] size_t getNumSamples() const noexcept {
298 return Avg.getNumSamples();
299 }
300
301 void clear() noexcept {
302 Min = SIZE_MAX;
303 Max = 0;
304 Avg = {};
305 }
306};

308struct TimerBase : PAMMBase {
309
310 static constexpr std::chrono::steady_clock::time_point None =
311 std::chrono::steady_clock::time_point::max();
312
313 std::chrono::steady_clock::time_point StartPoint = None;
314 std::chrono::nanoseconds Acc{};
315
316 [[nodiscard]] constexpr bool isStarted() const noexcept {
317 return StartPoint != None;
318 }
319
320 void start() noexcept {
321 assert(!isStarted() && "Starting an already running timer is not allowed");
322 StartPoint = std::chrono::steady_clock::now();
323 }
324
325 void stop() noexcept {
326 assert(isStarted() && "Stopping a not-running timer is not allowed");
327 auto EndPoint = std::chrono::steady_clock::now();
328 Acc += (EndPoint - StartPoint);
329 StartPoint = None;
330 }
331
332 void reset() noexcept {
333 Acc = {};
334 StartPoint = None;
335 }
336
337 [[nodiscard]] constexpr std::chrono::nanoseconds
338 elapsedNanos() const noexcept {
339 return Acc;
340 }
341
342 [[nodiscard]] constexpr std::chrono::nanoseconds
343 pendingNanos() const noexcept {
344 if (!isStarted()) {
345 return {};
346 }
347 auto EndPoint = std::chrono::steady_clock::now();
348 return EndPoint - StartPoint;
349 }
350
351 [[nodiscard]] hms elapsed() const noexcept { return Acc; }
352};

500
501 friend llvm::raw_ostream &operator<<(llvm::raw_ostream &OS,

503 OS << Cat->name() << "::" << Name << ":\n";
504 OS << "Value\t| #Occurrences\n";
505#ifdef PHASAR_THREAD_SAFE_PAMM

520 }
521 }

526 }

529 [[nodiscard]] const auto &internalRawData() const noexcept {
