25template <typename EntryRange, typename C, typename HandlerFn>
26void forallStartingPoints(const EntryRange &EntryPoints, const CFGBase<C> &CFG,
27 HandlerFn Handler) {
28 for (const auto &EntryPointFn : EntryPoints) {
29 if constexpr (std::is_convertible_v<std::decay_t<decltype(EntryPointFn)>,
30 bool>) {
31 if (!EntryPointFn) {
32 continue;
33 }
34 }
35 for (const auto &SP : CFG.getStartPointsOf(EntryPointFn)) {
36 std::invoke(Handler, SP);
37 }
38 }
39}

41template <typename EntryRange, typename C, typename ICFGorIRDB,
42 typename HandlerFn>
43void forallStartingPoints(const EntryRange &EntryPoints, const ICFGorIRDB *ICDB,
44 const CFGBase<C> &CFG, HandlerFn Handler) {
45
46 if (llvm::hasSingleElement(EntryPoints) &&
47 *llvm::adl_begin(EntryPoints) == "__ALL__") {
48 forallStartingPoints(ICDB->getAllFunctions(), CFG, std::move(Handler));
49 } else {
50 forallStartingPoints(llvm::map_range(EntryPoints,
51 [ICDB](llvm::StringRef Name) {
52 return ICDB->getFunction(Name);
53 }),
54 CFG, std::move(Handler));
55 }
56}
