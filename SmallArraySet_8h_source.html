30/// \brief Simple set container similar to the upcoming std::flat_set but only
31/// uses sorting+binary-search, once the size exceeds a fixed threshold. Below
32/// the threshold unique-insertion + lookup is handled via linear search.
33///
34/// Should work well, if the expected size is small.
35/// Use the static methods fromSorted() and fromSortedUniqued(), to speedup
36/// construction,. if you can make some assumptions about the incoming data.
37template <typename T,
38 unsigned N =
39 llvm::CalculateSmallVectorDefaultInlinedElements<T>::value>
40class SmallArraySet : private llvm::SmallVector<T, N> {
41public:

45 using typename llvm::SmallVector<T, N>::const_reference;
46 using typename llvm::SmallVector<T, N>::reference;
47 using typename llvm::SmallVector<T, N>::const_pointer;
48 using typename llvm::SmallVector<T, N>::pointer;
49 using typename llvm::SmallVector<T, N>::difference_type;

51 static constexpr size_t LinearThreshold = std::max<size_t>(1, 64 / sizeof(T));

64 [[nodiscard]] static SmallArraySet
65 fromSorted(llvm::SmallVectorImpl<T> &&Vec) {
66 assert(std::ranges::is_sorted(Vec) && "Vec is not sorted");
67 SmallArraySet Ret(std::move(Vec), /*PreSortedAndUniqued=*/std::true_type{});

76 assert(std::ranges::is_sorted(Vec) && "Vec is not sorted");
77 assert(std::ranges::adjacent_find(Vec) == Vec.end() &&

102 // Without inline, clang may not inline this, even in O3
103 // NOLINTNEXTLINE(readability-redundant-inline-specifier)
104 inline bool insert(ByConstRef<T> Val) {
105 const auto Size = this->size();
106 const auto Begin = this->begin();
107 const auto End = this->end();
108
109 if (Size < LinearThreshold) [[likely]] {
110 if (std::find(Begin, End, Val) != End) {
111 return false;
112 }
113
114 this->push_back(Val);
115
116 if (Size + 1 == LinearThreshold) [[unlikely]] {
117 std::ranges::sort(base());
118 }
119
120 return true;
121 }
122
123 return insertImpl(Val);
124 }
125
126 // Assume, the new Val is not in the set yet
127 void insertUnique(T Val) {
128 assert(!contains(Val));
129 if (this->size() < LinearThreshold || Val > this->back()) {
130 this->push_back(std::move(Val));
131 return;
132 }
133 insertImpl(Val);
134 }

136 void insert(const SmallArraySet &Other) {
137 const auto Size = size();
138 const auto OtherSize = Other.size();
139
140 llvm::ArrayRef<T> OtherArr = Other;
141
142 if (OtherSize <= Size && Size + OtherSize <= LinearThreshold) {
143 size_t I = 0;
144 for (; I != OtherSize; ++I) {
145 const auto End = this->end();
146 ByConstRef<T> Val = OtherArr[I];
147 if (std::find(this->begin(), End, Val) != End) {
148 continue;
149 }
150 if (this->size() == LinearThreshold) {
151 break;
152 }
153 this->push_back(Val);
154 }
155 if (I == OtherSize) {
156 if (this->size() == LinearThreshold) [[unlikely]] {
157 std::ranges::sort(base());
158 }
159 return;
160 }
161
162 OtherArr = OtherArr.slice(I);
163 }
164
165 insertImpl(OtherArr);
166 }

168 void insertAll(auto &&Range) {
169 size_t EstimatedRngSize = SIZE_MAX;
170 if constexpr (requires() {
171 { Range.size() } -> std::convertible_to<size_t>;
172 }) {
173 EstimatedRngSize = Range.size();
174 }
175
176 if (size() < LinearThreshold && EstimatedRngSize < LinearThreshold) {
177 for (auto &&Elem : Range) {
178 insert(PSR_FWD(Elem));
179 }
180 return;
181 }
182
183 this->append(llvm::adl_begin(Range), llvm::adl_end(Range));
184 psr::sortUnique(base());
185 }

187 [[nodiscard]] SmallArraySet setUnion(const SmallArraySet &Other) const {
188
189 const auto Size = size();
190 const auto OtherSize = Other.size();
191 if (std::min(Size, OtherSize) < LinearThreshold) {
192 const auto ThisSmaller = Size < OtherSize;
193 const auto &Smaller = ThisSmaller ? *this : Other;
194 const auto &Larger = ThisSmaller ? Other : *this;
195 auto Merged = Larger;
196 Merged.insert(Smaller);
197 return Merged;
198 }
199
200 SmallArraySet Merged{};
201 Merged.resize_for_overwrite(size() + Other.size());
202 auto LastOut = std::set_union(begin(), end(), Other.begin(), Other.end(),
203 Merged.begin());
204 Merged.base().erase(LastOut, Merged.end());
205 return Merged;
206 }

208 bool intersectWith(const SmallArraySet &Other) {
209 const auto Size = size();
210 const auto OtherSize = Other.size();
211
212 if (Size < LinearThreshold || OtherSize > Size) {
213 auto It = std::ranges::remove_if(
214 base(), [&Other](const auto &Elem) { return !Other.contains(Elem); });
215 base().erase(It.begin(), It.end());
216 return Size != size();
217 }
218
219 // TODO: Optimize

221 SmallArraySet Merged{};
222 Merged.resize_for_overwrite(std::min(Size, OtherSize));
223
224 auto [Unused1, Unused2, LastOut] =
225 std::ranges::set_intersection(*this, Other, Merged.begin());
226 Merged.base().erase(LastOut, Merged.end());
227 *this = std::move(Merged);

231 [[nodiscard]] SmallArraySet
232 setIntersection(const SmallArraySet &Other) const {
233 const auto Size = size();
234 const auto OtherSize = Other.size();
235
236 if (std::min(Size, OtherSize) < LinearThreshold) {
237 auto Ret = Size < OtherSize ? *this : Other;
238 const auto &Larger = Size < OtherSize ? Other : *this;
239 auto It = std::ranges::remove_if(
240 Ret, [&Larger](const auto &Elem) { return !Larger.contains(Elem); });
241 Ret.base().erase(It.begin(), It.end());
242 return Ret;
243 }
244
245 SmallArraySet Merged{};
246 Merged.resize_for_overwrite(std::min(Size, OtherSize));
247
248 auto [Unused1, Unused2, LastOut] =
249 std::ranges::set_intersection(*this, Other, Merged.begin());
250 Merged.base().erase(LastOut, Merged.end());
251 return Merged;

278 [[nodiscard]] friend auto hash_value(const SmallArraySet &Set) noexcept {
279 if (Set.size() < LinearThreshold) {
280 if constexpr (std::is_trivially_copyable_v<T>) {
281 T Arr[LinearThreshold]{};
282 memcpy(&Arr, Set.data(), Set.size_in_bytes());
283 std::ranges::sort(Arr, Arr + Set.size());
284 return llvm::hash_combine_range(Arr, Arr + Set.size());
285 } else {
286 // Some reduction that ignores the order
287 return std::transform_reduce(Set.begin(), Set.end(), 37,
288 std::bit_xor<>{}, [](ByConstRef<T> Val) {
289 using llvm::hash_value;
290 return hash_value(Val);
291 });
292 }
293 } else {
294 return llvm::hash_combine_range(Set.begin(), Set.end());
295 }
296 }

298 [[nodiscard]] bool operator==(const SmallArraySet &Other) const noexcept {
299 if (size() != Other.size()) {
300 return false;
301 }
302
303 if (size() < LinearThreshold) {
304 if (hash_value(*this) != hash_value(Other)) {
305 // Some pre-check to avoid the quadratic loop.
306 // XXX: Need to measure, whether this actually helps
307 return false;
308 }
309
310 for (const auto &Val : Other) {
311 if (!contains(Val)) {
312 return false;
313 }
314 }
315 return true;
316 }
317
318 return std::equal(begin(), end(), Other.begin());
319 }

326 bool insertImpl(ByConstRef<T> Val) {
327 const auto It = std::ranges::lower_bound(base(), Val);
328 if (It != this->end() && *It == Val) {
329 return false;
330 }
331
332 base().insert(It, Val);
333 return true;
334 }
335
336 void insertImpl(llvm::ArrayRef<T> OtherArr) {
337 if (OtherArr.size() < 3) {
338 // For small sizes, it is probably better to insert the elements
339 // one-by-one, instead of using std::sort.
340 // TODO: Find good threshold
341
342 for (const auto &Val : OtherArr) {
343 insert(Val);
344 }
345 return;
346 }
347
348 this->append(OtherArr.begin(), OtherArr.end());
349 psr::sortUnique(base());
350 }
