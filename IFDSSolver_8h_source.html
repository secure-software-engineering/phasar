31/// \brief Solves the given IFDSTabulationProblem as described in the 1995 paper
32/// by Reps, Horwitz and Sagiv. To solve the problem, call solve(). Results can
33/// then be queried by using resultAt() and resultsAt().
34///
35/// \note PhASAR implements IFDS in terms of IDE, so in case you do not need the
36/// raw SolverResults, for maximum performance you should use
37/// IFDSIDESolverConfig#setComputeValues(bool) to disable IDE's
38/// phase 2.
39template <typename AnalysisDomainTy,
40 typename Container = std::set<typename AnalysisDomainTy::d_t>,
41 ICFG ICFGTy = typename AnalysisDomainTy::i_t>

69 /// Returns the data-flow results at the given statement.
70 [[nodiscard]] virtual std::set<d_t> ifdsResultsAt(n_t Inst) {
71 std::set<d_t> KeySet;
72 std::unordered_map<d_t, BinaryDomain> ResultMap = this->resultsAt(Inst);
73 for (const auto &FlowFact : ResultMap) {
74 KeySet.insert(FlowFact.first);
75 }
76 return KeySet;
77 }

79 /// Returns the data-flow results at the given statement while respecting
80 /// LLVM's SSA semantics.
81 ///
82 /// An example: when a value is loaded and the location loaded from, here
83 /// variable 'i', is a data-flow fact that holds, then the loaded value '%0'
84 /// will usually be generated and also holds. However, due to the underlying
85 /// theory (and respective implementation) this load instruction causes the
86 /// loaded value to be generated and thus, it will be valid only AFTER the
87 /// load instruction, i.e., at the successor instruction.
88 ///
89 /// %0 = load i32, i32* %i, align 4
90 ///
91 /// This result accessor function returns the results at the successor
92 /// instruction(s) reflecting that the expression on the left-hand side holds
93 /// if the expression on the right-hand side holds.
94 [[nodiscard]]
95 std::set<d_t> ifdsResultsAtInLLVMSSA(n_t Inst)
96 requires same_as_decay<std::remove_pointer_t<n_t>, llvm::Instruction>
97 {
98 auto getResultMap // NOLINT
99 = [this, Inst]() {
100 if (Inst->getType()->isVoidTy()) {
101 return this->resultsAt(Inst);
102 }
103 // In this case we have a value on the left-hand side and must
104 // return the results at the successor instruction. Note that
105 // terminator instructions are always of void type.
106 assert(Inst->getNextNode() &&
107 "Expected to find a valid successor node!");
108 return this->resultsAt(Inst->getNextNode());
109 };
110 std::set<d_t> KeySet;
111 for (auto &[FlowFact, LatticeValue] : getResultMap()) {
112 KeySet.insert(FlowFact);
113 }
114 return KeySet;
115 }
116};
