59 /// Whether alias sets are reported globally or filtered to the function
60 /// containing the query instruction.
61 enum class AnalysisLocality : uint8_t {
62 /// All aliases across all functions are reported.
63 Global,
64 /// Aliases are intersected with variables visible in the querying function
65 /// (globals + function-local values).
66 FunctionLocal,
67 };

82/// Concrete \c IsAliasInfo implementation backed by a raw alias
83/// analysis. Provides convenience constructors to invoke union-find-based
84/// analyses, but can be instantiated with *any* analysis result that conforms
85/// to \c RawAAResult .
86///
87/// Alias sets are materialized lazily on first query and cached per \c ValueId
88/// in \c AliasSets.
89///
90/// \note When \c AnalysisLocality::FunctionLocal is selected, alias sets are
91/// filtered to variables visible in the function that contains the query
92/// instruction. The per-\c ValueId cache does **not** account for the
93/// instruction context, so the first caller's function wins — do not mix
94/// queries to local pointers from different functions for the same value in
95/// local mode.
96class LLVMRawAliasSet : public LLVMRawAliasSetBase,
97 public AnalysisPropertiesMixin<LLVMRawAliasSet> {
98public:

100 const LLVMBasedCallGraph &BaseCG, Config Cfg,
101 ValueCompressor<PAGVariable> *VC);
102 explicit LLVMRawAliasSet(const LLVMProjectIRDB *IRDB,
103 const LLVMBasedCallGraph &BaseCG, Config Cfg)
104 : LLVMRawAliasSet(IRDB, BaseCG, Cfg, nullptr) {}
105 explicit LLVMRawAliasSet(const LLVMProjectIRDB *IRDB,
106 const LLVMBasedCallGraph &BaseCG)
107 : LLVMRawAliasSet(IRDB, BaseCG, Config{}, nullptr) {}

149 [[nodiscard]] AliasSetPtrTy getAliasSet(v_t V, n_t I) {
150 assert(isValid());
151 auto ValId = AARes->VC->getOrNull(V);
152 if (!ValId) {
153 return getEmptyAliasSet();
154 }
155
156 assert(AliasSets.inbounds(*ValId));
157 if (!AliasSets[*ValId]) [[unlikely]] {
158 AliasSets[*ValId] = AARes->constructAliasSet(*ValId, I, Owner);

175 [[nodiscard]] bool isInReachableAllocationSites(
176 const llvm::Value *V, const llvm::Value *PotentialValue,
177 bool IntraProcOnly, const llvm::Instruction *I) const {
178 assert(isValid());
179 if (!psr::isInterestingPointer(V)) {
180 return false;
181 }
182
183 if (!psr::isInReachableAllocationSitesTy(V, PotentialValue,
184 IntraProcOnly)) {
185 return false;
186 }
187
188 return alias(V, PotentialValue, I) != AliasResult::NoAlias;
189 }

225 static bool isPotentialAllocSite(const llvm::Value *Val) {
226 if (!Val->getType()->isPointerTy()) {
227 return false;
228 }
229 if (llvm::isa<llvm::AllocaInst, llvm::Argument>(Val)) {
230 return true;
231 }
232 if (const auto *Call = llvm::dyn_cast<llvm::CallBase>(Val)) {
233 return Call->getCalledFunction() &&
234 psr::isHeapAllocatingFunction(Call->getCalledFunction());
235 }
236 return false;
237 }

239 template <template <typename, typename> typename AAResIterT, typename AAResT>
240 struct AAResultModel
241 : public AAResultConcept,
242 public AAResIterT<AAResultModel<AAResIterT, AAResT>, AAResT> {
243
244 using base_t = AAResIterT<AAResultModel<AAResIterT, AAResT>, AAResT>;
245
246 template <typename... ArgsT>
247 constexpr AAResultModel(
248 MaybeUniquePtr<const ValueCompressor<PAGVariable>> VC, ArgsT &&...Args)
249 : AAResultConcept{std::move(VC)}, base_t{PSR_FWD(Args)...} {}
250
251 void forallAliasesOf(v_t Ptr, n_t Inst,
252 llvm::function_ref<void(v_t)> Callback) override {
253 this->base_t::forallAliasesOf(Ptr, Inst, Callback);
254 }
255
256 AliasResult alias(v_t Ptr1, v_t Ptr2,
257 n_t AtInstruction = nullptr) override {
258 return this->base_t::alias(Ptr1, Ptr2, AtInstruction);
259 }
260
261 AliasSetPtrTy constructAliasSet(ValueId ValId, n_t Inst,
262 AliasSetOwner<AliasSetTy> &Owner) override {
263 auto ASet = Owner.acquire();
264 try {
265 this->base_t::forallAliasesOf(
266 ValId, Inst,
267 [ASet{ASet.get()}](v_t Alias) -> void { ASet->insert(Alias); });
268 return ASet;
269 } catch (...) {
270 Owner.release(ASet.get());
271 throw;
272 }
273 }
274
275 AllocationSiteSetPtrTy constructReachableAllocSites(v_t V, ValueId ValId,
276 bool IntraProcOnly,
277 n_t Inst) override {
278 if (!this->AllocationSites) [[unlikely]] {
279 this->AllocationSites.emplace();
280 for (const auto &[VId, Vars] : this->VC->id2vars().enumerate()) {
281 for (const PAGVariable &Var : Vars) {
282 if (const auto *LLVMVar = Var.valueOrNull();
283 LLVMVar && isPotentialAllocSite(LLVMVar)) {
284 this->AllocationSites->insert(VId);
285 break;
286 }
287 }
288 }
289 }
290
291 auto RawAliases = [&]() -> RawAliasSet<ValueId> {
292 if constexpr (requires() {
293 this->base_t::getRawAliasSet(ValId, Inst);
294 }) {
295 return this->base_t::getRawAliasSet(ValId, Inst);
296 } else {
297 return this->base_t::getRawAliasSet(ValId);
298 }
299 }();
300 RawAliases &= *this->AllocationSites;
301
302 auto Ret = std::make_unique<AliasSetTy>();
303
304 RawAliases.foreach (
305 llvmRawAliasHandler(*this->VC, [&Ret, V, IntraProcOnly](v_t Alias) {
306 if (psr::isInReachableAllocationSitesTy(V, Alias, IntraProcOnly)) {
307 Ret->insert(Alias);
308 }
309 }));
310
311 return Ret;
312 }
313
314 void print(llvm::raw_ostream &OS, Config Cfg) const override {
315 OS << "LLVMRawAliasSet(" << to_string(Cfg.ALocality) << ", "
316 << llvm::getTypeName<AAResT>() << ") {\n";
317
318 for (auto ValId : iota<ValueId>(VC->size())) {
319 OS << " #" << psr::to_underlying(ValId) << ": {";
320 bool First = true;
321 const auto &Aliases = this->base_t::getRawAliasSet(ValId);
322 Aliases.foreach ([&](auto AliasId) {
323 if (First) {
324 First = false;
325 } else {
326 OS << ", ";
327 }
328
329 OS << psr::to_underlying(AliasId);
330 });
331 OS << "}\n";
332 }
333 OS << "}\n";
334 };
335 };
