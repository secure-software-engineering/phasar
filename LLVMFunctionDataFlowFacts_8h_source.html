34/// @brief A LLVM-specific mapping of FunctionDataFlowFacts
35class LLVMFunctionDataFlowFacts {
36public:
37 LLVMFunctionDataFlowFacts() noexcept = default;
38 using ParameterMappingTy = FunctionDataFlowFacts::ParameterMappingTy;
39
40 [[nodiscard]] bool contains(const llvm::Function *Fn) const {
41 return LLVMFdff.count(Fn);
42 }
43
44 [[nodiscard]] const std::vector<DataFlowFact> &
45 getFacts(const llvm::Function *Fun, uint32_t Index) const {
46 if (const auto *Iter = getOrDefault(LLVMFdff, Fun)) {
47 return getOrDefault(*Iter, Index);
48 }
49 return default_value();
50 }
51 [[nodiscard]] const std::vector<DataFlowFact> &
52 getFacts(const llvm::Function *Fun, const llvm::Argument *Arg) const {
53 return getFacts(Fun, Arg->getArgNo());
54 }
55
56 [[nodiscard]] const ParameterMappingTy &
57 getFactsForFunction(const llvm::Function *Fun) const {
58 if (const auto *Iter = getOrDefault(LLVMFdff, Fun)) {
59 return *Iter;
60 }
61 return default_value();
62 }
63
64 [[nodiscard]] const ParameterMappingTy *
65 getFactsForFunctionOrNull(const llvm::Function *Fun) const {
66 return getOrDefault(LLVMFdff, Fun);
67 }
68
69 [[nodiscard]] friend LLVMFunctionDataFlowFacts
70 readFromFDFF(const FunctionDataFlowFacts &Fdff,
71 std::invocable<llvm::StringRef> auto GetFunctionByNameOrNull) {
72 LLVMFunctionDataFlowFacts Ret{};
73 Ret.LLVMFdff.reserve(Fdff.size());
74
75 for (const auto &Elem : Fdff) {
76 if (const auto *Fun =
77 std::invoke(GetFunctionByNameOrNull, Elem.getKey())) {
78 Ret.LLVMFdff.try_emplace(Fun, &Elem.getValue());
79 }
80 }
81
82 return Ret;
83 }
84
85 [[nodiscard]] friend LLVMFunctionDataFlowFacts
86 readFromFDFF(const FunctionDataFlowFacts &Fdff,
87 const ProjectIRDB auto &IRDB) {
88 return readFromFDFF(
89 Fdff, [&IRDB](llvm::StringRef Name) { return IRDB.getFunction(Name); });
90 }
91
92private:
93 llvm::DenseMap<const llvm::Function *, const ParameterMappingTy *> LLVMFdff;
94};
95} // namespace psr::library_summary
