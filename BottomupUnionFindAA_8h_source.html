43/// SCC-based, bottom-up union-find alias analysis.
44///
45/// Interprocedural Steensgaard-style analysis with summary propagation along
46/// the SCC DAG of the call graph.
47///
48/// When \p SoundnessFlag is \c Soundness::Soundy (default), a top-down pass
49/// propagates parameter aliasing information back to callers for soundness.
50/// Set it to \c Soundness::Unsound to skip this pass and trade precision for
51/// speed.
52///
53/// Intra-SCC Call/Return edges (mutual recursion) are handled
54/// context-insensitively.
55///
56/// Implements \c pag::PBStrategy. After \c buildPAG() completes, call
57/// \c consumeAAResults() to obtain a \c BasicUnionFindAAResult.
58///
59/// \tparam AnalysisDomainT The analysis domain, e.g., \c LLVMPAGDomain.
60/// \tparam SoundnessFlag Controls whether the top-down parameter-aliasing
61/// pass is performed.
62template <typename AnalysisDomainT, Soundness SoundnessFlag = Soundness::Soundy>
63class BottomupUnionFindAA : BottomupUnionFindAABase {
64 struct ParamInfo {
65 ValueId Arg;
66 ValueId Param;
67 pag::Call CS;
68 };
69 struct CallSiteInfo {
70 llvm::SmallVector<ParamInfo, 2> Params{};
71 llvm::SmallVector<ValueId, 1> Ret{};
72 std::optional<ValueId> RetSlot{};
73 };

112 void onAddEdge(ValueId From, ValueId To, pag::Edge E,
113 Nullable<n_t> CallSite) {
114 if (E.isa<pag::Call>()) {
115 auto FromSCC = SCCOfVal[From];
116 auto ToSCC = SCCOfVal[To];
117 if (FromSCC != ToSCC && FromSCC != InvalidSCC && ToSCC != InvalidSCC) {
118 Calls[FromSCC][CallSite].Params.emplace_back(From, To,
119 E.cast<pag::Call>());
120 IdsForSum[ToSCC].insert(To);
121 return;
122 }
123 // fallthrough -- intra-SCC edges are context-insensitive
124 } else if (E.isa<pag::Return>()) {
125 auto FromSCC = SCCOfVal[From];
126 auto ToSCC = SCCOfVal[To];
127 if (FromSCC != ToSCC && FromSCC != InvalidSCC && ToSCC != InvalidSCC) {
128 auto &RetInfo = Calls[ToSCC][CallSite];
129 assert(RetInfo.RetSlot == std::nullopt || RetInfo.RetSlot == To);
130 RetInfo.RetSlot = To;
131 RetInfo.Ret.push_back(From);
132
133 IdsForSum[FromSCC].insert(From);
134 return;
135 }
136 // fallthrough -- intra-SCC edges are context-insensitive
137 }
138
139 Base.onAddEdge(From, To, E, CallSite);
140 }
141
142 void onAddValue(ByConstRef<v_t> Var, ValueId VId) {
143 auto &SccPlace = SCCOfVal.emplace_back(InvalidSCC);
144 if (auto &&Fun = getFunction(Var)) {
145 if (auto FunVtx = RevCG.FC.getOrNull(Fun)) {
146 SccPlace = CGSCCs->SCCOfNode[*FunVtx];
147 }
148 }
149
150 Base.onAddValue(Var, VId);
151 }

153 template <std::predicate<ValueId> FilterFn = TrueFn>
154 [[nodiscard]] BasicUnionFindAAResult
155 consumeAAResults(size_t NumVars, FilterFn ShouldInclude = {}) && {
156
157 bottomupPropagation(NumVars);
158
159 if constexpr (SoundnessFlag != Soundness::Unsound) {
160 // For soundness, we need to handle parameter aliasing
161 topdownPropagation();
162 }
163
164 return std::move(Base).consumeAAResults(NumVars, IdentityFn{},
165 std::move(ShouldInclude));
166 }
167
168private:
169 void bottomupPropagation(size_t NumVars) {
170 TypedVector<ValueId, RawAliasSet<ValueId>> IntermediateBackView(NumVars);
171
172 for (SCCId<FunVtxTy> CurrSCC :
173 llvm::reverse(iota<SCCId<FunVtxTy>>(CGSCCs->size()))) {
174
175 for (const auto &[CS, CSInfo] : Calls[CurrSCC]) {
176 PHASAR_LOG_LEVEL_CAT(DEBUG, LogCategory, "At CS: " << NToString(CS));
177 llvm::SmallDenseMap<ValueId, llvm::SmallVector<ValueId, 2>> BackMap;
178
179 for (const auto &[Arg, Param, E] : CSInfo.Params) {
180 auto ParamRep = Base.AliasSets.find(Param);
181 BackMap[ParamRep].push_back(Arg);
182 }
183 if (CSInfo.RetSlot) {
184 for (auto Ret : CSInfo.Ret) {
185 auto RetRep = Base.AliasSets.find(Ret);
186 BackMap[RetRep].push_back(*CSInfo.RetSlot);
187 }
188 }
189
190 // --- Apply summaries:
191 for (const auto &[Param, Args] : BackMap) {
192 PHASAR_LOG_LEVEL_CAT(DEBUG, LogCategory,
193 " Param: " << psr::to_underlying(Param)
194 << "; Args: "
195 << PrettyPrinter{Args});
196
197 const auto &ToAliases = IntermediateBackView[Param];
198 ToAliases.foreach ([&](ValueId Alias) {
199 auto AliasRep = Base.AliasSets.find(Alias);
200 if (const auto *AliasFroms = getOrNull(BackMap, AliasRep)) {
201 for (auto Arg : Args) {
202 PHASAR_LOG_LEVEL_CAT(DEBUG, LogCategory,
203 " JOIN " << psr::to_underlying(Arg)
204 << " WITH "
205 << PrettyPrinter{*AliasFroms});
206
207 for (auto Alias : *AliasFroms) {
208 Base.AliasSets.join(Arg, Alias);
209 }
210 }
211 }
212 });
213 }
214 }
215 // --- Compute new summaries:
216
217 IdsForSum[CurrSCC].foreach ([&](ValueId SumId) {
218 auto SumIdRep = Base.AliasSets.find(SumId);
219 IntermediateBackView[SumIdRep].insert(SumId);
220 });
221 }
222 }
223
224 void topdownPropagation() {
225 for (SCCId<FunVtxTy> CurrSCC : iota<SCCId<FunVtxTy>>(CGSCCs->size())) {
226 for (const auto &[CS, CSInfo] : Calls[CurrSCC]) {
227 const size_t Sz = CSInfo.Params.size();
228 for (size_t I : psr::iota(Sz)) {
229 const ParamInfo FirstInfo = CSInfo.Params[I];
230 const auto FirstArgRep = Base.AliasSets.find(FirstInfo.Arg);
231 for (size_t J : psr::iota(I + 1, Sz)) {
232 const ParamInfo &SecondInfo = CSInfo.Params[J];
233 const auto SecondArgRep = Base.AliasSets.find(SecondInfo.Arg);
234
235 if (FirstArgRep == SecondArgRep) {
236 // Found parameter-aliasing!
237 Base.AliasSets.join(FirstInfo.Param, SecondInfo.Param);
238 }
239 }
240 }
241 }
242 }
243 }

245 [[nodiscard]] Nullable<f_t> getFunction(ByConstRef<v_t> Var) {
246 if constexpr (requires() { getPointerFrom(Var)->getFunction(); }) {
247 return getPointerFrom(Var)->getFunction();
248 } else {
249 return RevCG.IRDB->getFunctionOf(Var);
250 }
251 }

253 RevCGTy RevCG;
254 MaybeUniquePtr<SCCHolder<FunVtxTy>> CGSCCs;

256 TypedVector<ValueId, SCCId<FunVtxTy>> SCCOfVal;
257 TypedVector<SCCId<FunVtxTy>, std::unordered_map<n_t, CallSiteInfo>> Calls;
258 TypedVector<SCCId<FunVtxTy>, RawAliasSet<ValueId>> IdsForSum;
259 BasicUnionFindAA<AnalysisDomainT> Base{};
260};
