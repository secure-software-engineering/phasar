25namespace psr {
26/// A K-limited call-string context. Stores the \p K most recent call-sites
27/// that led to the current execution point (index 0 = most recent). The
28/// default \p K = 1 gives a 1-call-site context (CFA/k-CFA style).
29template <typename N, unsigned K = 1> struct CallingContext {
30 using n_t = N;
31
32 /// The K most recent call-sites; index 0 is the most recent.
33 std::array<n_t, K> Callers{};
34
35 [[nodiscard]] friend auto hash_value(const CallingContext &CC) {
36 return llvm::hash_combine_range(CC.Callers.begin(), CC.Callers.end());
37 }
38
39 [[nodiscard]] friend bool
40 operator==(const CallingContext &L,
41 const CallingContext &R) noexcept = default;
42
43 [[nodiscard]] CallingContext withPrefix(n_t FirstCS) const noexcept {
44 auto Ret = *this;
45 auto Tmp = FirstCS;
46 for (size_t I = 0; I != K; ++I) {
47 auto NextTmp = Ret.Callers[I];
48 Ret.Callers[I] = Tmp;
49 Tmp = NextTmp;
50 }
51
52 return Ret;
53 }
54};
55} // namespace psr

88/// Manages the set of calling contexts encountered for a given analysis and
89/// maps them to dense sequential \c CallingContextId values.
90///
91/// Used by \c CallingContextSensUnionFindAA to enumerate all call-string
92/// prefixes of length ≤ K for each function. Contexts are interned in a
93/// \c Compressor; \c CallingContextId::None (id 0) always represents the
94/// "no calling context" (top-level / uncalled) entry.
95///
96/// \tparam N Node type (call-site).
97/// \tparam F Function type.
98/// \tparam K Maximum call-string depth (must be in [1, 9]).
99template <typename N, typename F, unsigned K = 1>
100class CallingContextConstructor {
101public:
102 static_assert(K > 0);
103 static_assert(K < 10, "Do you really want a sooo large context-k-limit? "
104 "Reconsider your choices!");
105
106 using n_t = N;
107 using f_t = F;
108
109 CallingContextConstructor() {
110 // Assign Id 0 === ContextId::None
111 CC2Id.getOrInsert(CallingContext<n_t, K>{});
112 }
113
114 /// Enumerates all call-string prefixes of length ≤ K that lead to \p Fun
115 /// and invokes \p CCVisitor(firstCallSite, contextId) for each one.
116 /// If \p Fun has no callers (or K is exhausted), the "None" context is used.
117 template <ProjectIRDB DBTy>
118 requires(std::same_as<typename DBTy::f_t, f_t> &&
119 std::same_as<typename DBTy::n_t, n_t>)
120 void visitAllCallingContexts(
121 ByConstRef<f_t> Fun, const CallGraph<n_t, f_t> &CG, const DBTy &IRDB,
122 std::invocable<n_t, CallingContextId> auto CCVisitor) {
123 CallingContext<n_t, K> Ctx{};
124 visitAllCallingContextsImpl<0>(Fun, CG, IRDB, CCVisitor, Ctx);
125 }
126
127 /// Enumerates all calling contexts that include \p Call as their most recent
128 /// call-site and invokes \p CCVisitor(contextId) for each one.
129 template <ProjectIRDB DBTy>
130 requires(std::same_as<typename DBTy::f_t, f_t> &&
131 std::same_as<typename DBTy::n_t, n_t>)
132 void visitContextsAtCall(ByConstRef<n_t> Call, const CallGraph<n_t, f_t> &CG,
133 const DBTy &IRDB,
134 std::invocable<CallingContextId> auto CCVisitor) {
135 CallingContext<n_t, K> Ctx{};
136 Ctx.Callers[0] = Call;
137 visitAllCallingContextsImpl<1>(
138 IRDB.getFunctionOf(Call), CG, IRDB,
139 [CCVisitor{std::move(CCVisitor)}](ByConstRef<n_t>,
140 CallingContextId Ctx) {
141 std::invoke(CCVisitor, Ctx);
142 },
143 Ctx);
144 }
145
146 CallingContextId getOrInsert(ByConstRef<CallingContext<n_t, K>> CC) {
147 return CC2Id.getOrInsert(CC);
148 }
149
150 [[nodiscard]] ByConstRef<CallingContext<n_t, K>>
151 operator[](CallingContextId Ctx) {
152 return CC2Id[Ctx];
153 }
154
155private:
156 template <unsigned Idx, typename DBTy>
157 void visitAllCallingContextsImpl(
158 ByConstRef<f_t> Fun, const CallGraph<n_t, f_t> &CG, const DBTy &IRDB,
159 std::invocable<n_t, CallingContextId> auto &&CCVisitor,
160 CallingContext<n_t, K> &CurrCtx) {
161 // TODO: Improve this algorithm
162
163 if constexpr (Idx < K) {
164 auto &&CallersOfCS = CG.getCallersOf(Fun);
165
166 for (const auto &CS : CallersOfCS) {
167 CurrCtx.Callers[Idx] = CS;
168 visitAllCallingContextsImpl<(Idx + 1)>(IRDB.getFunctionOf(CS), CG, IRDB,
169 CCVisitor, CurrCtx);
170 }
171
172 if (!CallersOfCS.empty()) {
173 return;
174 }
175 }
176
177 auto CtxId = CC2Id.getOrInsert(CurrCtx);
178 std::invoke(CCVisitor, CurrCtx.Callers[0], CtxId);
179 }
180
181 Compressor<CallingContext<n_t, K>, CallingContextId> CC2Id{};
182};
183} // namespace psr
