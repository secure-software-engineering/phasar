21/// Minimum requirements of an analysis domain. Analyses may add extra
22/// requirements.

24concept IsAnalysisDomain = IRDomain<T> && requires() {
25 typename T::ir_t;

37/// AnalysisDomain - This class should be specialized by different static
38/// analyses types... which is why the default version declares all analysis
39/// domains as aliases of void.
40///
41/// Virtually all of PhASAR's internal analyses are implemented in a generic way
42/// using interfaces and template parameters. In order to specify concrete types
43/// for the template parameters such that an analysis can compute some useful
44/// information on some concrete target code, a configuration template parameter
45/// of type AnalysisDomain is passed around to make the necessary information
46/// available to the required analyses.
47///
48/// If a type is not meant to be used by an analysis it should be left as an
49/// alias to void. If any analysis detects that a parameter is required to
50/// conduct an analysis but not correctly set, it will statically report an
51/// error and ask for the missing piece of information.
52struct AnalysisDomain {
53 /// Program IR
54 using ir_t = void;
55 /// Data-flow fact --- Specifies the type of an individual data-flow fact that
56 /// is propagated through the program under analysis.
57 using d_t = void;
58 // (Control-flow) Node --- Specifies the type of a node in the
59 // (inter-procedural) control-flow graph and can be though of as an individual
60 // statement or instruction of the target program.
61 using n_t = void;
62 // Function --- Specifies the type of functions/procedures in the target
63 // program.
64 using f_t = void;
65 // (User-defined) type --- Specifies the type of a user-defined (i.e. struct
66 // or class) data type.
67 using t_t = void;
68 // (Pointer) value --- Specifies the type of pointers.
69 using v_t = void;
70 // Intra-procedural control flow --- Specifies the type of the
71 // control-flow graph to be used.
72 using c_t = void;
73 // Inter-procedural control flow --- Specifies the type of the
74 // inter-procedural control-flow graph to be used.
75 using i_t = void;
76 // The ProjectIRDB type to use. Must conform to the ProjectIRDB concept
77 // template
78 using db_t = void;
79 // Lattice element --- Specifies the type of the underlying lattice; the value
80 // computation domain IDE's edge functions or WPDS's weights operate on.
81 using l_t = void;
82 // Container type to be used for analyses run in the monotone framework.
83 using mono_container_t = void;
84};
