27/// This class owns the IR code of the target being analyzed and some
28/// very important information associated with the IR.
29/// When an object of this class is destroyed it will clean up all IR related
30/// stuff that is stored in it.
31///
32/// m_t - module type
33/// f_t - function type
34/// n_t - instruction type
35/// g_t - global variable type
36///
37/// \remark XXX Once we have upgraded to C++20, we might want to use a concept
38/// instead...
39template <typename Derived> class ProjectIRDBBase {
40public:
41 using n_t = typename ProjectIRDBTraits<Derived>::n_t;
42 using f_t = typename ProjectIRDBTraits<Derived>::f_t;
43 using m_t = typename ProjectIRDBTraits<Derived>::m_t;
44 using g_t = typename ProjectIRDBTraits<Derived>::g_t;
45
46 /// Returns the managed module
47 [[nodiscard]] m_t getModule() const noexcept {
48 assert(isValid());
49 return self().getModuleImpl();
50 }
51
52 /// Check if debug information are available.
53 [[nodiscard]] bool debugInfoAvailable() const {
54 assert(isValid());
55 return self().debugInfoAvailableImpl();
56 }
57
58 /// Returns a range of all function definitions and declarations available
59 [[nodiscard]] decltype(auto) getAllFunctions() const {
60 static_assert(
61 is_iterable_over_v<decltype(self().getAllFunctionsImpl()), f_t>);
62 assert(isValid());
63 return self().getAllFunctionsImpl();
64 }
65
66 // Returns the function's definition if available, its declaration otherwise.
67 [[nodiscard]] f_t getFunction(llvm::StringRef FunctionName) const {
68 assert(isValid());
69 return self().getFunctionImpl(FunctionName);
70 }
71 /// Returns the function's definition if available, null otherwise.
72 [[nodiscard]] f_t getFunctionDefinition(llvm::StringRef FunctionName) const {
73 assert(isValid());
74 return self().getFunctionDefinitionImpl(FunctionName);
75 }
76 /// Returns whether the IRDB contains a function with the given name.
77 [[nodiscard]] bool hasFunction(llvm::StringRef FunctionName) const noexcept {
78 assert(isValid());
79 return self().hasFunctionImpl(FunctionName);
80 }
81
82 /// Returns the global variable's definition if available, null otherwise.
83 [[nodiscard]] g_t
84 getGlobalVariableDefinition(llvm::StringRef GlobalVariableName) const {
85 assert(isValid());
86 return self().getGlobalVariableDefinitionImpl(GlobalVariableName);
87 }
88
89 /// Returns the number of instruction in the managed module.
90 [[nodiscard]] size_t getNumInstructions() const noexcept {
91 assert(isValid());
92 return self().getNumInstructionsImpl();
93 }
94 /// Returns the number of global variables in the managed module.
95 [[nodiscard]] size_t getNumGlobals() const noexcept {
96 assert(isValid());
97 return self().getNumGlobalsImpl();
98 }
99 /// Returns the number of functions in the managed module.
100 [[nodiscard]] size_t getNumFunctions() const noexcept {
101 assert(isValid());
102 return self().getNumFunctionsImpl();
103 }
104
105 /// Returns the instruction to the corresponding Id. Returns nullptr, if there
106 /// is no instruction for this Id
107 [[nodiscard]] n_t getInstruction(size_t Id) const {
108 assert(isValid());
109 return self().getInstructionImpl(Id);
110 }
111 /// Returns an instruction's ID. The instruction must belong to the managed
112 /// module for this function to work
113 [[nodiscard]] size_t getInstructionId(n_t Inst) const {
114 assert(isValid());
115 return self().getInstructionIdImpl(Inst);
116 }
117
118 [[nodiscard]] decltype(auto) getAllInstructions() const {
119 static_assert(
120 is_iterable_over_v<decltype(self().getAllInstructionsImpl()), n_t>);
121 assert(isValid());
122 return self().getAllInstructionsImpl();
123 }
124
125 /// Sanity check to verify that th IRDB really manages a Module and all
126 /// functions work properly.
127 /// All functions of this IRDB require isValid() to return true, if not stated
128 /// differently -- otherwise, they must not be called.
129 /// This function (obviously) does not require isValid() to return true.
130 [[nodiscard]] bool isValid() const noexcept { return self().isValidImpl(); }
131
132 /// Dumps the managed module to llvm::dbgs() if isValid(); otherwise prints
133 /// "<Invalid Module>".
134 /// This function can be invoked from a debugger.
135 LLVM_DUMP_METHOD void dump() const {
136 if (!isValid()) {
137 llvm::dbgs() << "<Invalid Module>\n";
138 llvm::dbgs().flush();
139 return;
140 }
141 self().dumpImpl();
142 }
143
144 ProjectIRDBBase(const ProjectIRDBBase &) = delete;
145 ProjectIRDBBase &operator=(const ProjectIRDBBase &) = delete;
146
147 ProjectIRDBBase(ProjectIRDBBase &&) noexcept = default;
148 ProjectIRDBBase &operator=(ProjectIRDBBase &&) noexcept = default;
149
150 ~ProjectIRDBBase() = default;
151
152private:
153 friend Derived;
154 ProjectIRDBBase() noexcept = default;
155
156 Derived &self() noexcept { return static_cast<Derived &>(*this); }
157 const Derived &self() const noexcept {
158 return static_cast<const Derived &>(*this);
159 }
160};
