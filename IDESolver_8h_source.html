105 };
106 }

120 nlohmann::json getAsJson() {
121 using TableCell = typename Table<n_t, d_t, l_t>::Cell;
122 const static std::string DataFlowID = "DataFlow";
123 nlohmann::json J;
124 auto Results = this->ValTab.cellSet();
125 if (Results.empty()) {
126 J[DataFlowID] = "EMPTY";
127 } else {
128 std::vector<TableCell> Cells(Results.begin(), Results.end());
129 sort(Cells.begin(), Cells.end(), [](TableCell Lhs, TableCell Rhs) {
130 return Lhs.getRowKey() < Rhs.getRowKey();
131 });
132 n_t Curr;
133 for (unsigned I = 0; I < Cells.size(); ++I) {
134 Curr = Cells[I].getRowKey();
135 auto NStr =
136 llvm::StringRef(NToString(Cells[I].getRowKey())).trim().str();
137
138 std::string NodeStr =
139 ICF->getFunctionName(ICF->getFunctionOf(Curr)).str() + "::" + NStr;
140 J[DataFlowID][NodeStr];
141 std::string FactStr =
142 llvm::StringRef(DToString(Cells[I].getColumnKey())).trim().str();
143 std::string ValueStr =
144 llvm::StringRef(LToString(Cells[I].getValue())).trim().str();
145 J[DataFlowID][NodeStr]["Facts"] += {FactStr, ValueStr};
146 }
147 }
148 return J;
149 }

156 /// Returns the L-type result at the given statement for the given data-flow
157 /// fact while respecting LLVM's SSA semantics.
158 ///
159 /// An example: when a value is loaded and the location loaded from, here
160 /// variable 'i', is a data-flow fact that holds, then the loaded value '%0'
161 /// will usually be generated and also holds. However, due to the underlying
162 /// theory (and respective implementation) this load instruction causes the
163 /// loaded value to be generated and thus, it will be valid only AFTER the
164 /// load instruction, i.e., at the successor instruction.
165 ///
166 /// %0 = load i32, i32* %i, align 4
167 ///
168 /// This result accessor function returns the results at the successor
169 /// instruction(s) reflecting that the expression on the left-hand side holds
170 /// if the expression on the right-hand side holds.
171 template <typename NTy = n_t>
172 [[nodiscard]] typename std::enable_if_t<
173 std::is_same_v<std::remove_reference_t<NTy>, llvm::Instruction *>, l_t>
174 resultAtInLLVMSSA(NTy Stmt, d_t Value) {
175 return getSolverResults().resultAtInLLVMSSA(Stmt, Value);
176 }

281 [[nodiscard]] EdgeFunctionStats getEdgeFunctionStatistics() const {
282 detail::EdgeFunctionStatsData Stats{};
283
284 // Cached Edge Functions
285 {
286 std::array<llvm::DenseSet<EdgeFunction<l_t>>, EdgeFunctionKindCount>
287 UniqueEFs{};
288 Sampler DepthSampler{};
289 Sampler UniqueDepthSampler{};
290 // TODO: Cache EFs
291 CachedFlowEdgeFunctions.foreachCachedEdgeFunction(
292 [&](EdgeFunction<l_t> EF, EdgeFunctionKind Kind) {
293 auto Depth = EF.depth();
294 DepthSampler.addSample(Depth);
295 if (Depth > Stats.MaxDepth) {
296 Stats.MaxDepth = Depth;
297 }
298
299 if (UniqueEFs[size_t(Kind)].insert(std::move(EF)).second) {
300 UniqueDepthSampler.addSample(Depth);
301 }
302 Stats.TotalEFCount[size_t(Kind)]++;
303 Stats.PerAllocCount[size_t(EF.getAllocationPolicy())]++;
304 });
305
306 size_t TotalUniqueNumEF = 0;
307 for (size_t I = 0, End = UniqueEFs.size(); I != End; ++I) {
308 Stats.UniqueEFCount[I] = UniqueEFs[I].size(); // NOLINT
309 TotalUniqueNumEF += UniqueEFs[I].size();
310 }
311
312 Stats.AvgDepth = DepthSampler.getAverage();
313 Stats.AvgUniqueDepth = UniqueDepthSampler.getAverage();
314 }
315
316 // Jump Functions
317 {
318 llvm::DenseSet<EdgeFunction<l_t>> UniqueJumpFns;
319 llvm::DenseSet<const void *> AllocatedJumpFns;
320 Sampler DepthSampler{};
321 Sampler UniqueDepthSampler{};
322 Sampler AllocDepthSampler{};
323
324 JumpFn->foreachEdgeFunction([&](EdgeFunction<l_t> JF) {
325 ++Stats.TotalNumJF;
326 auto Depth = JF.depth();
327 DepthSampler.addSample(Depth);
328 if (Depth > Stats.MaxJFDepth) {
329 Stats.MaxJFDepth = Depth;
330 }
331
332 if (AllocatedJumpFns.insert(JF.getOpaqueValue()).second) {
333 AllocDepthSampler.addSample(Depth);
334 }
335
336 Stats.PerAllocJFCount[size_t(JF.getAllocationPolicy())]++;
337
338 if (UniqueJumpFns.insert(std::move(JF)).second) {
339 UniqueDepthSampler.addSample(Depth);
340 }
341 });
342
343 Stats.UniqueNumJF = UniqueJumpFns.size();
344 Stats.NumJFObjects = AllocatedJumpFns.size();
345 Stats.AvgJFDepth = DepthSampler.getAverage();
346 Stats.AvgUniqueJFDepth = UniqueDepthSampler.getAverage();
347 Stats.AvgJFObjDepth = AllocDepthSampler.getAverage();
348 }
349 return Stats;
350 }

358 ByConstRef<n_t> n) {

363 return NextUserOrNullCB(SVFG, Fun, d3, n);

366 /// Lines 13-20 of the algorithm; processing a call site in the caller's
367 /// context.
368 ///
369 /// For each possible callee, registers incoming call edges.
370 /// Also propagates call-to-return flows and summarized callee flows within
371 /// the caller.
372 ///
373 /// The following cases must be considered and handled:
374 /// 1. Process as usual and just process the call
375 /// 2. Create a new summary for that function (which shall be done
376 /// by the problem)
377 /// 3. Just use an existing summary provided by the problem
378 /// 4. If a special function is called, use a special summary
379 /// function
380 ///
381 /// @param edge an edge whose target node resembles a method call
382 ///
383 virtual void processCall(const PathEdge<n_t, d_t> Edge) {
384 PAMM_GET_INSTANCE;
385 INC_COUNTER("Process Call", 1, Full);
386 PHASAR_LOG_LEVEL(DEBUG,
387 "Process call at target: " << NToString(Edge.getTarget()));
388 d_t d1 = Edge.factAtSource();
389 n_t n = Edge.getTarget();
390 // a call node; line 14...
391 d_t d2 = Edge.factAtTarget();
392 EdgeFunction<l_t> f = jumpFunction(Edge);
393 const auto &ReturnSiteNs = ICF->getReturnSitesOfCallAt(n);
394 const auto &Callees = ICF->getCalleesOfCallAt(n);
395
396 IF_LOG_LEVEL_ENABLED(DEBUG, {
397 PHASAR_LOG_LEVEL(DEBUG, "Possible callees:");
398 for (auto Callee : Callees) {
399 PHASAR_LOG_LEVEL(DEBUG, " " << Callee->getName());
400 }
401 PHASAR_LOG_LEVEL(DEBUG, "Possible return sites:");
402 for (auto ret : ReturnSiteNs) {
403 PHASAR_LOG_LEVEL(DEBUG, " " << NToString(ret));
404 }
405 });
406
407 bool HasNoCalleeInformation = true;
408
409 auto &&Fun = ICF->getFunctionOf(n);
410 auto GetNextUse = [this, &Fun, &n](n_t nPrime, ByConstRef<d_t> d3) {
411 if (auto &&NextUser = getNextUserOrNull(Fun, d3, n)) {
412 return psr::unwrapNullable(PSR_FWD(NextUser));
413 }
414
415 return nPrime;
416 };
417
418 // for each possible callee
419 for (f_t SCalledProcN : Callees) { // still line 14
420 // check if a special summary for the called procedure exists
421 FlowFunctionPtrType SpecialSum =
422 CachedFlowEdgeFunctions.getSummaryFlowFunction(n, SCalledProcN);
423
424 // if a special summary is available, treat this as a normal flow
425 // and use the summary flow and edge functions
426
427 if (SpecialSum) {
428 HasNoCalleeInformation = false;
429 PHASAR_LOG_LEVEL(DEBUG, "Found and process special summary");
430 for (n_t ReturnSiteN : ReturnSiteNs) {
431 container_type Res = computeSummaryFlowFunction(SpecialSum, d1, d2);
432 INC_COUNTER("SpecialSummary-FF Application", 1, Full);
433 ADD_TO_HISTOGRAM("Data-flow facts", Res.size(), 1, Full);
434 saveEdges(n, ReturnSiteN, d2, Res, ESGEdgeKind::Summary);
435 for (d_t d3 : Res) {
436 EdgeFunction<l_t> SumEdgFnE =
437 CachedFlowEdgeFunctions.getSummaryEdgeFunction(n, d2,
438 ReturnSiteN, d3);
439 INC_COUNTER("SpecialSummary-EF Queries", 1, Full);
440
441 PHASAR_LOG_LEVEL(DEBUG,
442 "Queried Summary Edge Function: " << SumEdgFnE);
443 PHASAR_LOG_LEVEL(DEBUG,
444 "Compose: " << SumEdgFnE << " * " << f << '\n');
445
446 auto DestN = GetNextUse(ReturnSiteN, d3);
447 WorkList.emplace_back(PathEdge(d1, DestN, std::move(d3)),
448 IDEProblem.extend(f, SumEdgFnE));
449 }
450 }
451 } else {
452 // compute the call-flow function
453 FlowFunctionPtrType Function =
454 CachedFlowEdgeFunctions.getCallFlowFunction(n, SCalledProcN);
455 INC_COUNTER("FF Queries", 1, Full);
456 container_type Res = computeCallFlowFunction(Function, d1, d2);
457 ADD_TO_HISTOGRAM("Data-flow facts", Res.size(), 1, Full);
458 // for each callee's start point(s)
459 auto StartPointsOf = ICF->getStartPointsOf(SCalledProcN);
460 if (StartPointsOf.empty()) {
461 PHASAR_LOG_LEVEL(DEBUG, "Start points of '" +
462 ICF->getFunctionName(SCalledProcN) +
463 "' currently not available!");
464 }
465 // if startPointsOf is empty, the called function is a declaration
466 for (n_t SP : StartPointsOf) {
467 HasNoCalleeInformation = false;
468 saveEdges(n, SP, d2, Res, ESGEdgeKind::Call);
469 // for each result node of the call-flow function
470 for (d_t d3 : Res) {
471 using TableCell = typename Table<n_t, d_t, EdgeFunction<l_t>>::Cell;
472 // create initial self-loop
473 PHASAR_LOG_LEVEL(
474 DEBUG, "Create initial self-loop with D: " << DToString(d3));
475 WorkList.emplace_back(PathEdge(d3, SP, d3),
476 EdgeIdentity<l_t>{}); // line 15
477 // register the fact that <sp,d3> has an incoming edge from <n,d2>
478 // line 15.1 of Naeem/Lhotak/Rodriguez
479 addIncoming(SP, d3, n, d2);
480 // line 15.2, copy to avoid concurrent modification exceptions by
481 // other threads
482 // const std::set<TableCell> endSumm(endSummary(sP, d3));
483 // llvm::outs() << "ENDSUMM" << '\n';
484 // llvm::outs() << "Size: " << endSumm.size() << '\n';
485 // llvm::outs() << "sP: " << NToString(sP)
486 // << "\nd3: " << DToString(d3)
487 // << '\n';
488 // printEndSummaryTab();
489 // still line 15.2 of Naeem/Lhotak/Rodriguez
490 // for each already-queried exit value <eP,d4> reachable from
491 // <sP,d3>, create new caller-side jump functions to the return
492 // sites because we have observed a potentially new incoming
493 // edge into <sP,d3>
494 for (const TableCell &Entry : endSummary(SP, d3)) {
495 n_t eP = Entry.getRowKey();
496 d_t d4 = Entry.getColumnKey();
497 EdgeFunction<l_t> fCalleeSummary = Entry.getValue();
498 // for each return site
499 for (n_t RetSiteN : ReturnSiteNs) {
500 // compute return-flow function
501 FlowFunctionPtrType RetFunction =
502 CachedFlowEdgeFunctions.getRetFlowFunction(n, SCalledProcN,
503 eP, RetSiteN);
504 INC_COUNTER("FF Queries", 1, Full);
505 const container_type ReturnedFacts = computeReturnFlowFunction(
506 RetFunction, d3, d4, n, Container{d2});
507 ADD_TO_HISTOGRAM("Data-flow facts", ReturnedFacts.size(), 1,
508 Full);
509 saveEdges(eP, RetSiteN, d4, ReturnedFacts, ESGEdgeKind::Ret);
510 // for each target value of the function
511 for (d_t d5 : ReturnedFacts) {
512 // update the caller-side summary function
513 // get call edge function
514 EdgeFunction<l_t> f4 =
515 CachedFlowEdgeFunctions.getCallEdgeFunction(
516 n, d2, SCalledProcN, d3);
517 PHASAR_LOG_LEVEL(DEBUG, "Queried Call Edge Function: " << f4);
518 // get return edge function
519 EdgeFunction<l_t> f5 =
520 CachedFlowEdgeFunctions.getReturnEdgeFunction(
521 n, SCalledProcN, eP, d4, RetSiteN, d5);
522 PHASAR_LOG_LEVEL(DEBUG,
523 "Queried Return Edge Function: " << f5);
524 if (SolverConfig.emitESG()) {
525 for (auto SP : ICF->getStartPointsOf(SCalledProcN)) {
526 IntermediateEdgeFunctions[std::make_tuple(n, d2, SP, d3)]
527 .push_back(f4);
528 }
529 IntermediateEdgeFunctions[std::make_tuple(eP, d4, RetSiteN,
530 d5)]
531 .push_back(f5);
532 }
533 INC_COUNTER("EF Queries", 2, Full);
534 // compose call * calleeSummary * return edge functions
535 PHASAR_LOG_LEVEL(DEBUG, "Compose: " << f5 << " * "
536 << fCalleeSummary << " * "
537 << f4);
538 PHASAR_LOG_LEVEL(DEBUG,
539 " (return * calleeSummary * call)");
540 EdgeFunction<l_t> fPrime = IDEProblem.extend(
541 IDEProblem.extend(f4, fCalleeSummary), f5);
542 PHASAR_LOG_LEVEL(DEBUG, " = " << fPrime);
543 d_t d5_restoredCtx = restoreContextOnReturnedFact(n, d2, d5);
544 // propagte the effects of the entire call
545 PHASAR_LOG_LEVEL(DEBUG, "Compose: " << fPrime << " * " << f);
546
547 auto DestN = GetNextUse(RetSiteN, d5_restoredCtx);
548 WorkList.emplace_back(
549 PathEdge(d1, DestN, std::move(d5_restoredCtx)),
550 IDEProblem.extend(f, fPrime));
551 }
552 }
553 }
554 }
555 }
556 }
557 }
558 // line 17-19 of Naeem/Lhotak/Rodriguez
559 // process intra-procedural flows along call-to-return flow functions
560 for (n_t ReturnSiteN : ReturnSiteNs) {
561 FlowFunctionPtrType CallToReturnFF =
562 CachedFlowEdgeFunctions.getCallToRetFlowFunction(n, ReturnSiteN,
563 Callees);
564 INC_COUNTER("FF Queries", 1, Full);
565 container_type ReturnFacts =
566 computeCallToReturnFlowFunction(CallToReturnFF, d1, d2);
567 ADD_TO_HISTOGRAM("Data-flow facts", ReturnFacts.size(), 1, Full);
568 saveEdges(n, ReturnSiteN, d2, ReturnFacts,
569 HasNoCalleeInformation ? ESGEdgeKind::SkipUnknownFn
570 : ESGEdgeKind::CallToRet);
571 for (d_t d3 : ReturnFacts) {
572 EdgeFunction<l_t> EdgeFnE =
573 CachedFlowEdgeFunctions.getCallToRetEdgeFunction(n, d2, ReturnSiteN,
574 d3, Callees);
575 PHASAR_LOG_LEVEL(DEBUG,

1181 /// Propagates the flow further down the exploded super graph, merging any
1182 /// edge function that might already have been computed for TargetVal at
1183 /// Target.
1184 ///
1185 /// @param SourceVal the source value of the propagated summary edge
1186 /// @param Target the target statement
1187 /// @param TargetVal the target value at the target statement
1188 /// @param f the new edge function computed from (s0,SourceVal) to
1189 /// (Target,TargetVal)
1190 /// @param relatedCallSite for call and return flows the related call
1191 /// statement, nullptr otherwise (this value is not used within this
1192 /// implementation but may be useful for subclasses of IDESolver)
1193 /// @param isUnbalancedReturn true if this edge is propagating an
1194 /// unbalanced return (this value is not used within this implementation
1195 /// but may be useful for subclasses of {@link IDESolver})
1196 ///
1197 void propagate(d_t SourceVal, n_t Target, d_t TargetVal,
1198 EdgeFunction<l_t> f) {
1199 PHASAR_LOG_LEVEL(DEBUG, "Propagate flow");
1200 PHASAR_LOG_LEVEL(DEBUG, "Source value : " << DToString(SourceVal));
1201 PHASAR_LOG_LEVEL(DEBUG, "Target : " << NToString(Target));
1202 PHASAR_LOG_LEVEL(DEBUG, "Target value : " << DToString(TargetVal));
1203 PHASAR_LOG_LEVEL(
1204 DEBUG, "Edge function : " << f << " (result of previous compose)");
1205
1206 EdgeFunction<l_t> JumpFnE = [&]() {
1207 const auto RevLookupResult = JumpFn->reverseLookup(Target, TargetVal);
1208 if (RevLookupResult) {
1209 const auto &JumpFnContainer = RevLookupResult->get();
1210 const auto Find = std::find_if(
1211 JumpFnContainer.begin(), JumpFnContainer.end(),
1212 [SourceVal](auto &KVpair) { return KVpair.first == SourceVal; });
1213 if (Find != JumpFnContainer.end()) {
1214 return Find->second;
1215 }
1216 }
1217 // jump function is initialized to all-top if no entry
1218 // was found
1219 return AllTop;
1220 }();
1221 EdgeFunction<l_t> fPrime = IDEProblem.combine(JumpFnE, f);
1222 bool NewFunction = fPrime != JumpFnE;
1223
1224 IF_LOG_LEVEL_ENABLED(DEBUG, {
1225 PHASAR_LOG_LEVEL(DEBUG,
1226 "Join: " << JumpFnE << " & " << f
1227 << (JumpFnE == f ? " (EF's are equal)" : " "));
1228 PHASAR_LOG_LEVEL(DEBUG, " = "
1229 << fPrime
1230 << (NewFunction ? " (new jump func)" : " "));
1231 PHASAR_LOG_LEVEL(DEBUG, ' ');
1232 });
1233 if (NewFunction) {
1234 JumpFn->addFunction(SourceVal, Target, TargetVal, fPrime);
1235 PathEdge Edge(SourceVal, Target, TargetVal);
1236 PathEdgeCount++;
1237 pathEdgeProcessingTask(std::move(Edge));
1238
1239 IF_LOG_LEVEL_ENABLED(DEBUG, {
1240 if (!IDEProblem.isZeroValue(TargetVal)) {
1241 PHASAR_LOG_LEVEL(
1242 DEBUG, "EDGE: <F: " << Target->getFunction()->getName()
1243 << ", D: " << DToString(SourceVal) << '>');
1244 PHASAR_LOG_LEVEL(DEBUG, " ---> <N: " << NToString(Target) << ',');
1245 PHASAR_LOG_LEVEL(DEBUG, " D: " << DToString(TargetVal) << ',');
1246 PHASAR_LOG_LEVEL(DEBUG, " EF: " << fPrime << '>');
1247 PHASAR_LOG_LEVEL(DEBUG, ' ');
1248 }
1249 });
1250 } else {
1251 PHASAR_LOG_LEVEL(DEBUG, "PROPAGATE: No new function!");
1252 }
1253 }

1380 /// The invariant for computing the number of generated (#gen) and killed
1381 /// (#kill) facts:
1382 /// (1) #Valid facts at the last statement <= #gen - #kill
1383 /// (2) #gen >= #kill
1384 ///
1385 /// The total number of valid facts can be smaller than the difference of
1386 /// generated and killed facts, due to set semantics, i.e., a fact can be
1387 /// generated multiple times but appears only once.
1388 ///
1389 /// Zero value is not counted!
1390 ///
1391 /// @brief Computes and prints statistics of the analysis run, e.g. number of
1392 /// generated/killed facts, number of summary-reuses etc.
1393 ///
1394 void computeAndPrintStatistics() {
1395 PAMM_GET_INSTANCE;
1396 // Stores all valid facts at return site in caller context; return-site is
1397 // key
1398 std::unordered_map<n_t, std::set<d_t>> ValidInCallerContext;
1399 size_t NumGenFacts = 0;
1400 size_t NumIntraPathEdges = 0;
1401 size_t NumInterPathEdges = 0;
1402 // --- Intra-procedural Path Edges ---
1403 // d1 --> d2-Set
1404 // Case 1: d1 in d2-Set
1405 // Case 2: d1 not in d2-Set, i.e., d1 was killed. d2-Set could be empty.
1406 for (const auto &Cell : ComputedIntraPathEdges.cellSet()) {
1407 auto Edge = std::make_pair(Cell.getRowKey(), Cell.getColumnKey());
1408 PHASAR_LOG_LEVEL(DEBUG, "N1: " << NToString(Edge.first));
1409 PHASAR_LOG_LEVEL(DEBUG, "N2: " << NToString(Edge.second));
1410 for (auto &[D1, D2s] : Cell.getValue()) {
1411 PHASAR_LOG_LEVEL(DEBUG, "d1: " << DToString(D1));
1412 NumIntraPathEdges += D2s.size();
1413 // Case 1
1414 if (D2s.find(D1) != D2s.end()) {
1415 NumGenFacts += D2s.size() - 1;
1416 }
1417 // Case 2
1418 else {
1419 NumGenFacts += D2s.size();
1420 }
1421 // Store all valid facts after call-to-return flow
1422 if (ICF->isCallSite(Edge.first)) {
1423 ValidInCallerContext[Edge.second].insert(D2s.begin(), D2s.end());
1424 }
1425 IF_LOG_LEVEL_ENABLED(DEBUG, [this](const auto &D2s) {
1426 for (auto D2 : D2s) {
1427 PHASAR_LOG_LEVEL(DEBUG, "d2: " << DToString(D2));
1428 }
1429 PHASAR_LOG_LEVEL(DEBUG, "----");
1430 }(D2s));
1431 }
1432 PHASAR_LOG_LEVEL(DEBUG, " ");
1433 }
1434 // Stores all pairs of (Startpoint, Fact) for which a summary was applied
1435 std::set<std::pair<n_t, d_t>> ProcessSummaryFacts;
1436 PHASAR_LOG_LEVEL(DEBUG, "==============================================");
1437 PHASAR_LOG_LEVEL(DEBUG, "INTER PATH EDGES");
1438 for (const auto &Cell : ComputedInterPathEdges.cellSet()) {
1439 auto Edge = std::make_pair(Cell.getRowKey(), Cell.getColumnKey());
1440 PHASAR_LOG_LEVEL(DEBUG, "N1: " << NToString(Edge.first));
1441 PHASAR_LOG_LEVEL(DEBUG, "N2: " << NToString(Edge.second));
1442 // --- Call-flow Path Edges ---
1443 // Case 1: d1 --> empty set
1444 // Can be ignored, since killing a fact in the caller context will
1445 // actually happen during call-to-return.
1446 //
1447 // Case 2: d1 --> d2-Set
1448 // Every fact d_i != ZeroValue in d2-set will be generated in the
1449 // callee context, thus counts as a new fact. Even if d1 is passed as it
1450 // is, it will count as a new fact. The reason for this is, that d1 can
1451 // be killed in the callee context, but still be valid in the caller
1452 // context.
1453 //
1454 // Special Case: Summary was applied for a particular call
1455 // Process the summary's #gen and #kill.
1456 if (ICF->isCallSite(Edge.first)) {
1457 for (auto &[D1, D2s] : Cell.getValue()) {
1458 PHASAR_LOG_LEVEL(DEBUG, "d1: " << DToString(D1));
1459 NumInterPathEdges += D2s.size();
1460 for (auto D2 : D2s) {
1461 if (!IDEProblem.isZeroValue(D2)) {
1462 NumGenFacts++;
1463 }
1464 // Special case
1465 if (ProcessSummaryFacts.find(std::make_pair(Edge.second, D2)) !=
1466 ProcessSummaryFacts.end()) {
1467
1468 std::set<d_t> SummaryDSet;
1469 EndsummaryTab.get(Edge.second, D2)
1470 .foreachCell([&SummaryDSet](const auto &Row, const auto &Col,
1471 const auto &Val) {
1472 SummaryDSet.insert(Col);
1473 });
1474
1475 // Process summary just as an intra-procedural edge
1476 if (SummaryDSet.find(D2) != SummaryDSet.end()) {
1477 NumGenFacts += SummaryDSet.size() - 1;
1478 } else {
1479 NumGenFacts += SummaryDSet.size();
1480 }
1481 } else {
1482 ProcessSummaryFacts.emplace(Edge.second, D2);
1483 }
1484 PHASAR_LOG_LEVEL(DEBUG, "d2: " << DToString(D2));
1485 }
1486 PHASAR_LOG_LEVEL(DEBUG, "----");
1487 }
1488 }
1489 // --- Return-flow Path Edges ---
1490 // Since every fact passed to the callee was counted as a new fact, we
1491 // have to count every fact propagated to the caller as a kill to
1492 // satisfy our invariant. Obviously, every fact not propagated to the
1493 // caller will count as a kill. If an actual new fact is propagated to
1494 // the caller, we have to increase the number of generated facts by one.
1495 // Zero value does not count towards generated/killed facts.
1496 if (ICF->isExitInst(Cell.getRowKey())) {
1497 for (auto &[D1, D2s] : Cell.getValue()) {
1498 PHASAR_LOG_LEVEL(DEBUG, "d1: " << DToString(D1));
1499 NumInterPathEdges += D2s.size();
1500 auto CallerFacts = ValidInCallerContext[Edge.second];
1501 for (auto D2 : D2s) {
1502 // d2 not valid in caller context
1503 if (CallerFacts.find(D2) == CallerFacts.end()) {
1504 NumGenFacts++;
1505 }
1506 PHASAR_LOG_LEVEL(DEBUG, "d2: " << DToString(D2));
1507 }
1508 PHASAR_LOG_LEVEL(DEBUG, "----");
1509 }
1510 }
1511 PHASAR_LOG_LEVEL(DEBUG, " ");
1512 }
1513 PHASAR_LOG_LEVEL(DEBUG, "SUMMARY REUSE");
1514 std::size_t TotalSummaryReuse = 0;
1515 for (const auto &Entry : FSummaryReuse) {
1516 PHASAR_LOG_LEVEL(DEBUG, "N1: " << NToString(Entry.first.first));
1517 PHASAR_LOG_LEVEL(DEBUG, "D1: " << DToString(Entry.first.second));
1518 PHASAR_LOG_LEVEL(DEBUG, "#Reuse: " << Entry.second);
1519 TotalSummaryReuse += Entry.second;
1520 }
1521 INC_COUNTER("Gen facts", NumGenFacts, Core);
1522 INC_COUNTER("Summary-reuse", TotalSummaryReuse, Core);
1523 INC_COUNTER("Intra Path Edges", NumIntraPathEdges, Core);
1524 INC_COUNTER("Inter Path Edges", NumInterPathEdges, Core);
1525
1526 PHASAR_LOG_LEVEL(INFO, "----------------------------------------------");
1527 PHASAR_LOG_LEVEL(INFO, "=== Solver Statistics ===");
1528 PHASAR_LOG_LEVEL(INFO, "#Facts generated : " << GET_COUNTER("Gen facts"));
1529 PHASAR_LOG_LEVEL(INFO, "#Facts killed : " << GET_COUNTER("Kill facts"));
1530 PHASAR_LOG_LEVEL(INFO,
1531 "#Summary-reuse : " << GET_COUNTER("Summary-reuse"));
1532 PHASAR_LOG_LEVEL(INFO,
1533 "#Intra Path Edges: " << GET_COUNTER("Intra Path Edges"));
1534 PHASAR_LOG_LEVEL(INFO,
1535 "#Inter Path Edges: " << GET_COUNTER("Inter Path Edges"));
1536 if constexpr (PAMM_CURR_SEV_LEVEL >= PAMM_SEVERITY_LEVEL::Full) {
1537 PHASAR_LOG_LEVEL(
1538 INFO, "Flow function query count: " << GET_COUNTER("FF Queries"));
1539 PHASAR_LOG_LEVEL(
1540 INFO, "Edge function query count: " << GET_COUNTER("EF Queries"));
1541 PHASAR_LOG_LEVEL(INFO, "Data-flow value propagation count: "
1542 << GET_COUNTER("Value Propagation"));
1543 PHASAR_LOG_LEVEL(INFO, "Data-flow value computation count: "
1544 << GET_COUNTER("Value Computation"));
1545 PHASAR_LOG_LEVEL(INFO,
1546 "Special flow function usage count: "
1547 << GET_COUNTER("SpecialSummary-FF Application"));
1548 PHASAR_LOG_LEVEL(INFO, "Jump function construciton count: "
1549 << GET_COUNTER("JumpFn Construction"));
1550 PHASAR_LOG_LEVEL(INFO,
1551 "Phase I duration: " << PRINT_TIMER("DFA Phase I"));
1552 PHASAR_LOG_LEVEL(INFO,
1553 "Phase II duration: " << PRINT_TIMER("DFA Phase II"));
1554 PHASAR_LOG_LEVEL(INFO, "----------------------------------------------");
1555 CachedFlowEdgeFunctions.print();
1556 }
1557 }

1671 for (const auto &Cell : Cells) {
1672 auto Edge = std::make_pair(Cell.getRowKey(), Cell.getColumnKey());
1673 std::string N1Label = NToString(Edge.first);
1674 std::string N2Label = NToString(Edge.second);
1675 std::string FNameOfN1 = ICF->getFunctionOf(Edge.first)->getName().str();
1676 std::string FNameOfN2 = ICF->getFunctionOf(Edge.second)->getName().str();
1677 std::string N1StmtId = ICF->getStatementId(Edge.first);
1678 std::string N2StmtId = ICF->getStatementId(Edge.second);
1679 PHASAR_LOG_LEVEL(DEBUG, "N1: " << N1Label);
1680 PHASAR_LOG_LEVEL(DEBUG, "N2: " << N2Label);
1681
1682 // Add inter-procedural control flow edge
1683 DOTNode N1(FNameOfN1, N1Label, N1StmtId);
1684 DOTNode N2(FNameOfN2, N2Label, N2StmtId);
1685
1686 // Handle recursion control flow as intra-procedural control flow
1687 // since those eges never leave the function subgraph
1688 FG = nullptr;
1689 if (FNameOfN1 == FNameOfN2) {
1690 // This function subgraph is guaranteed to exist
1691 FG = &G.Functions[FNameOfN1];
1692 FG->IntraCFEdges.emplace(N1, N2);
1693 } else {
1694 // Check the case where the callee is a single statement function,
1695 // thus does not contain intra-procedural path edges. We have to
1696 // generate the function sub graph here!
1697 if (!G.Functions.count(FNameOfN1)) {
1698 FG = &G.Functions[FNameOfN1];
1699 FG->Id = FNameOfN1;
1700 FG->Stmts.insert(N1);
1701 } else if (!G.Functions.count(FNameOfN2)) {
1702 FG = &G.Functions[FNameOfN2];
1703 FG->Id = FNameOfN2;
1704 FG->Stmts.insert(N2);
1705 }
1706 G.InterCFEdges.emplace(N1, N2);
1707 }
1708
1709 // Create D1 and D2, if D1 == D2 == lambda then add Edge(D1, D2) to
1710 // interLambdaEges otherwise add Edge(D1, D2) to interFactEdges
1711 unsigned D1FactId = 0;
1712 unsigned D2FactId = 0;
1713 for (const auto &D1ToD2Set : Cell.getValue()) {
1714 auto D1Fact = D1ToD2Set.first;
1715 PHASAR_LOG_LEVEL(DEBUG, "d1: " << DToString(D1Fact));
1716 DOTNode D1;
1717 if (IDEProblem.isZeroValue(D1Fact)) {
1718 D1 = {FNameOfN1, "Λ", N1StmtId, 0, false, true};
1719 } else {
1720 // Get the fact-ID
1721 D1FactId = G.getFactID(D1Fact);
1722 std::string D1Label{DToString(D1Fact)};
1723 D1 = {FNameOfN1, D1Label, N1StmtId, D1FactId, false, true};
1724 // FG should already exist even for single statement functions
1725 if (!G.containsFactSG(FNameOfN1, D1FactId)) {
1726 FG = &G.Functions[FNameOfN1];
1727 auto *D1FSG = FG->getOrCreateFactSG(D1FactId, D1Label);
1728 D1FSG->Nodes.insert(std::make_pair(N1StmtId, D1));
1729 }
1730 }
1731
1732 auto D2Set = D1ToD2Set.second;
1733 for (const auto &D2Fact : D2Set) {
1734 PHASAR_LOG_LEVEL(DEBUG, "d2: " << DToString(D2Fact));
1735 DOTNode D2;
1736 if (IDEProblem.isZeroValue(D2Fact)) {
1737 D2 = {FNameOfN2, "Λ", N2StmtId, 0, false, true};
1738 } else {
1739 // Get the fact-ID
1740 D2FactId = G.getFactID(D2Fact);
1741 std::string D2Label{DToString(D2Fact)};
1742 D2 = {FNameOfN2, D2Label, N2StmtId, D2FactId, false, true};
1743 // FG should already exist even for single statement functions
1744 if (!G.containsFactSG(FNameOfN2, D2FactId)) {
1745 FG = &G.Functions[FNameOfN2];
1746 auto *D2FSG = FG->getOrCreateFactSG(D2FactId, D2Label);
1747 D2FSG->Nodes.insert(std::make_pair(N2StmtId, D2));
1748 }
1749 }
1750
1751 if (IDEProblem.isZeroValue(D1Fact) &&
1752 IDEProblem.isZeroValue(D2Fact)) {
1753 // Do not add lambda recursion edges as inter-procedural edges
1754 if (D1.FuncName != D2.FuncName) {
1755 G.InterLambdaEdges.emplace(D1, D2, true, "AllBottom", "BOT");
1756 }
1757 } else {
1758 // std::string EFLabel = EF ? EF->str() : " ";
1759 std::string EFLabel;
1760 auto EFVec = IntermediateEdgeFunctions[std::make_tuple(
1761 Edge.first, D1Fact, Edge.second, D2Fact)];
1762 for (const auto &EF : EFVec) {
1763 PHASAR_LOG_LEVEL(DEBUG, "Partial EF Label: " << EF);
1764 EFLabel.append(to_string(EF) + ", ");
1765 }
1766 PHASAR_LOG_LEVEL(DEBUG, "EF LABEL: " << EFLabel);
1767 G.InterFactEdges.emplace(D1, D2, true, EFLabel);
1768 }
1769 }
1770 PHASAR_LOG_LEVEL(DEBUG, "----------");
1771 }
1772 PHASAR_LOG_LEVEL(DEBUG, " ");
1773 }
1774 OS << G;
1775 }
