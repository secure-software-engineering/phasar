178 nlohmann::json getAsJson() {
179 using TableCell = typename Table<n_t, d_t, l_t>::Cell;
180 const static std::string DataFlowID = "DataFlow";
181 nlohmann::json J;
182 auto Results = this->ValTab.cellSet();
183 if (Results.empty()) {
184 J[DataFlowID] = "EMPTY";
185 } else {
186 std::vector<TableCell> Cells(Results.begin(), Results.end());
187 sort(Cells.begin(), Cells.end(), [](TableCell Lhs, TableCell Rhs) {
188 return Lhs.getRowKey() < Rhs.getRowKey();
189 });
190 n_t Curr;
191 for (unsigned I = 0; I < Cells.size(); ++I) {
192 Curr = Cells[I].getRowKey();
193 auto NStr =
194 llvm::StringRef(NToString(Cells[I].getRowKey())).trim().str();
195
196 std::string NodeStr =
197 ICF->getFunctionName(ICF->getFunctionOf(Curr)).str() + "::" + NStr;
198 J[DataFlowID][NodeStr];
199 std::string FactStr =
200 llvm::StringRef(DToString(Cells[I].getColumnKey())).trim().str();
201 std::string ValueStr =
202 llvm::StringRef(LToString(Cells[I].getValue())).trim().str();
203 J[DataFlowID][NodeStr]["Facts"] += {FactStr, ValueStr};
204 }
205 }
206 return J;
207 }

214 /// Returns the L-type result at the given statement for the given data-flow
215 /// fact while respecting LLVM's SSA semantics.
216 ///
217 /// An example: when a value is loaded and the location loaded from, here
218 /// variable 'i', is a data-flow fact that holds, then the loaded value '%0'
219 /// will usually be generated and also holds. However, due to the underlying
220 /// theory (and respective implementation) this load instruction causes the
221 /// loaded value to be generated and thus, it will be valid only AFTER the
222 /// load instruction, i.e., at the successor instruction.
223 ///
224 /// %0 = load i32, i32* %i, align 4
225 ///
226 /// This result accessor function returns the results at the successor
227 /// instruction(s) reflecting that the expression on the left-hand side holds
228 /// if the expression on the right-hand side holds.
229 [[nodiscard]] l_t resultAtInLLVMSSA(n_t Stmt, d_t Value)
230 requires same_as_decay<std::remove_pointer_t<n_t>, llvm::Instruction>
231 {
232 return getSolverResults().resultAtInLLVMSSA(Stmt, Value);
233 }

344 [[nodiscard]] EdgeFunctionStats getEdgeFunctionStatistics() const {
345 detail::EdgeFunctionStatsData Stats{};
346
347 // Cached Edge Functions
348 {
349 std::array<llvm::DenseSet<EdgeFunction<l_t>>, EdgeFunctionKindCount>
350 UniqueEFs{};
351 Sampler DepthSampler{};
352 Sampler UniqueDepthSampler{};
353 // TODO: Cache EFs
354 CachedFlowEdgeFunctions.foreachCachedEdgeFunction(
355 [&](EdgeFunction<l_t> EF, EdgeFunctionKind Kind) {
356 auto Depth = EF.depth();
357 DepthSampler.addSample(Depth);
358 if (Depth > Stats.MaxDepth) {
359 Stats.MaxDepth = Depth;
360 }
361
362 if (UniqueEFs[size_t(Kind)].insert(std::move(EF)).second) {
363 UniqueDepthSampler.addSample(Depth);
364 }
365 Stats.TotalEFCount[size_t(Kind)]++;
366 Stats.PerAllocCount[size_t(EF.getAllocationPolicy())]++;
367 });
368
369 size_t TotalUniqueNumEF = 0;
370 for (size_t I = 0, End = UniqueEFs.size(); I != End; ++I) {
371 Stats.UniqueEFCount[I] = UniqueEFs[I].size(); // NOLINT
372 TotalUniqueNumEF += UniqueEFs[I].size();
373 }
374
375 Stats.AvgDepth = DepthSampler.getAverage();
376 Stats.AvgUniqueDepth = UniqueDepthSampler.getAverage();
377 }
378
379 // Jump Functions
380 {
381 llvm::DenseSet<EdgeFunction<l_t>> UniqueJumpFns;
382 llvm::DenseSet<const void *> AllocatedJumpFns;
383 Sampler DepthSampler{};
384 Sampler UniqueDepthSampler{};
385 Sampler AllocDepthSampler{};
386
387 JumpFn->foreachEdgeFunction([&](EdgeFunction<l_t> JF) {
388 ++Stats.TotalNumJF;
389 auto Depth = JF.depth();
390 DepthSampler.addSample(Depth);
391 if (Depth > Stats.MaxJFDepth) {
392 Stats.MaxJFDepth = Depth;
393 }
394
395 if (AllocatedJumpFns.insert(JF.getOpaqueValue()).second) {
396 AllocDepthSampler.addSample(Depth);
397 }
398
399 Stats.PerAllocJFCount[size_t(JF.getAllocationPolicy())]++;
400
401 if (UniqueJumpFns.insert(std::move(JF)).second) {
402 UniqueDepthSampler.addSample(Depth);
403 }
404 });
405
406 Stats.UniqueNumJF = UniqueJumpFns.size();
407 Stats.NumJFObjects = AllocatedJumpFns.size();
408 Stats.AvgJFDepth = DepthSampler.getAverage();
409 Stats.AvgUniqueJFDepth = UniqueDepthSampler.getAverage();
410 Stats.AvgJFObjDepth = AllocDepthSampler.getAverage();
411 }
412 return Stats;
413 }

419protected:
420 [[nodiscard]] Nullable<n_t> getNextUserOrNull(ByConstRef<f_t> Fun,
421 ByConstRef<d_t> d3,
422 ByConstRef<n_t> n) {
423 if constexpr (has_getSparseCFG_v<ICFGTy, d_t>) {
424 if (IDEProblem.isZeroValue(d3)) {
425 return {};
426 }
427
428 auto &&SCFG = ICF->getSparseCFG(Fun, d3);
429 return SCFG.nextUserOrNull(n);
430 } else {
431 return {};
432 }
433 }
434
435 /// Lines 13-20 of the algorithm; processing a call site in the caller's
436 /// context.
437 ///
438 /// For each possible callee, registers incoming call edges.
439 /// Also propagates call-to-return flows and summarized callee flows within
440 /// the caller.
441 ///
442 /// The following cases must be considered and handled:
443 /// 1. Process as usual and just process the call
444 /// 2. Create a new summary for that function (which shall be done
445 /// by the problem)
446 /// 3. Just use an existing summary provided by the problem
447 /// 4. If a special function is called, use a special summary
448 /// function
449 ///
450 /// @param edge an edge whose target node resembles a method call
451 ///
452 virtual void processCall(const PathEdge<n_t, d_t> Edge) {
453 ProcessCall++;
454 PHASAR_LOG_LEVEL(DEBUG,
455 "Process call at target: " << NToString(Edge.getTarget()));
456 d_t d1 = Edge.factAtSource();
457 n_t n = Edge.getTarget();
458 // a call node; line 14...
459 d_t d2 = Edge.factAtTarget();
460 auto f = jumpFunction(Edge);
461 const auto &ReturnSiteNs = ICF->getReturnSitesOfCallAt(n);
462 const auto &Callees = ICF->getCalleesOfCallAt(n);
463
464 IF_LOG_LEVEL_ENABLED(DEBUG, {
465 PHASAR_LOG_LEVEL(DEBUG, "Possible callees:");
466 for (auto Callee : Callees) {
467 PHASAR_LOG_LEVEL(DEBUG, " " << Callee->getName());
468 }
469 PHASAR_LOG_LEVEL(DEBUG, "Possible return sites:");
470 for (auto Ret : ReturnSiteNs) {
471 PHASAR_LOG_LEVEL(DEBUG, " " << NToString(Ret));
472 }
473 });
474
475 bool HasNoCalleeInformation = true;
476
477 auto &&Fun = ICF->getFunctionOf(n);
478 auto GetNextUse = [this, &Fun, &n](n_t NPrime, ByConstRef<d_t> d3) {
479 if (auto &&NextUser = getNextUserOrNull(Fun, d3, n)) {
480 return psr::unwrapNullable(PSR_FWD(NextUser));
481 }
482
483 return NPrime;
484 };
485
486 // for each possible callee
487 for (f_t SCalledProcN : Callees) { // still line 14
488 // check if a special summary for the called procedure exists
489 auto SpecialSum =
490 CachedFlowEdgeFunctions.getSummaryFlowFunction(n, SCalledProcN);
491
492 // if a special summary is available, treat this as a normal flow
493 // and use the summary flow and edge functions
494
495 if (SpecialSum) {
496 HasNoCalleeInformation = false;
497 PHASAR_LOG_LEVEL(DEBUG, "Found and process special summary");
498 for (n_t ReturnSiteN : ReturnSiteNs) {
499 container_type Res = computeSummaryFlowFunction(SpecialSum, d1, d2);
500 SpecialSummaryFF_Application++;
501 DataFlowFacts.add(Res.size(), 1);
502 saveEdges(n, ReturnSiteN, d2, Res, ESGEdgeKind::Summary);
503 for (d_t d3 : Res) {
504 auto SumEdgFnE = CachedFlowEdgeFunctions.getSummaryEdgeFunction(
505 n, d2, ReturnSiteN, d3);
506 SpecialSummaryEF_Queries++;
507
508 PHASAR_LOG_LEVEL(DEBUG,
509 "Queried Summary Edge Function: " << SumEdgFnE);
510 PHASAR_LOG_LEVEL(DEBUG,
511 "Compose: " << SumEdgFnE << " * " << f << '\n');
512
513 auto DestN = GetNextUse(ReturnSiteN, d3);
514 WorkList.emplace_back(PathEdge(d1, DestN, std::move(d3)),
515 IDEProblem.extend(f, SumEdgFnE));
516 }
517 }
518 } else {
519 // compute the call-flow function
520 auto Function =
521 CachedFlowEdgeFunctions.getCallFlowFunction(n, SCalledProcN);
522 FFQueries++;
523 container_type Res = computeCallFlowFunction(Function, d1, d2);
524 DataFlowFacts.add(Res.size(), 1);
525 // for each callee's start point(s)
526 auto StartPointsOf = ICF->getStartPointsOf(SCalledProcN);
527 if (StartPointsOf.empty()) {
528 PHASAR_LOG_LEVEL(DEBUG, "Start points of '" +
529 ICF->getFunctionName(SCalledProcN) +
530 "' currently not available!");
531 }
532 // if startPointsOf is empty, the called function is a declaration
533 for (n_t SP : StartPointsOf) {
534 HasNoCalleeInformation = false;
535 saveEdges(n, SP, d2, Res, ESGEdgeKind::Call);
536 // for each result node of the call-flow function
537 for (d_t d3 : Res) {
538 using TableCell = typename Table<n_t, d_t, EdgeFunction<l_t>>::Cell;
539 // create initial self-loop
540 PHASAR_LOG_LEVEL(
541 DEBUG, "Create initial self-loop with D: " << DToString(d3));
542 WorkList.emplace_back(PathEdge(d3, SP, d3),
543 EdgeIdentity<l_t>{}); // line 15
544 // register the fact that <sp,d3> has an incoming edge from <n,d2>
545 // line 15.1 of Naeem/Lhotak/Rodriguez
546 addIncoming(SP, d3, n, d2);
547 // line 15.2, copy to avoid concurrent modification exceptions by
548 // other threads
549 // const std::set<TableCell> endSumm(endSummary(sP, d3));
550 // llvm::outs() << "ENDSUMM" << '\n';
551 // llvm::outs() << "Size: " << endSumm.size() << '\n';
552 // llvm::outs() << "sP: " << NToString(sP)
553 // << "\nd3: " << DToString(d3)
554 // << '\n';
555 // printEndSummaryTab();
556 // still line 15.2 of Naeem/Lhotak/Rodriguez
557 // for each already-queried exit value <eP,d4> reachable from
558 // <sP,d3>, create new caller-side jump functions to the return
559 // sites because we have observed a potentially new incoming
560 // edge into <sP,d3>
561 for (const TableCell &Entry : endSummary(SP, d3)) {
562 n_t eP = Entry.getRowKey();
563 d_t d4 = Entry.getColumnKey();
564 auto fCalleeSummary = Entry.getValue();
565 // for each return site
566 for (n_t RetSiteN : ReturnSiteNs) {
567 // compute return-flow function
568 auto RetFunction = CachedFlowEdgeFunctions.getRetFlowFunction(
569 n, SCalledProcN, eP, RetSiteN);
570 FFQueries++;
571 const container_type ReturnedFacts = computeReturnFlowFunction(
572 RetFunction, d3, d4, n, Container{d2});
573 DataFlowFacts.add(ReturnedFacts.size(), 1);
574 saveEdges(eP, RetSiteN, d4, ReturnedFacts, ESGEdgeKind::Ret);
575 // for each target value of the function
576 for (d_t d5 : ReturnedFacts) {
577 // update the caller-side summary function
578 // get call edge function
579 auto f4 = CachedFlowEdgeFunctions.getCallEdgeFunction(
580 n, d2, SCalledProcN, d3);
581 PHASAR_LOG_LEVEL(DEBUG, "Queried Call Edge Function: " << f4);
582 // get return edge function
583 auto f5 = CachedFlowEdgeFunctions.getReturnEdgeFunction(
584 n, SCalledProcN, eP, d4, RetSiteN, d5);
585 PHASAR_LOG_LEVEL(DEBUG,
586 "Queried Return Edge Function: " << f5);
587 if (SolverConfig.emitESG()) {
588 for (auto SP : ICF->getStartPointsOf(SCalledProcN)) {
589 IntermediateEdgeFunctions[std::make_tuple(n, d2, SP, d3)]
590 .push_back(f4);
591 }
592 IntermediateEdgeFunctions[std::make_tuple(eP, d4, RetSiteN,
593 d5)]
594 .push_back(f5);
595 }
596 EFQueries++;
597 // compose call * calleeSummary * return edge functions

1228 /// Propagates the flow further down the exploded super graph, merging any
1229 /// edge function that might already have been computed for TargetVal at
1230 /// Target.
1231 ///
1232 /// @param SourceVal the source value of the propagated summary edge
1233 /// @param Target the target statement
1234 /// @param TargetVal the target value at the target statement
1235 /// @param f the new edge function computed from (s0,SourceVal) to
1236 /// (Target,TargetVal)
1237 /// @param relatedCallSite for call and return flows the related call
1238 /// statement, nullptr otherwise (this value is not used within this
1239 /// implementation but may be useful for subclasses of IDESolver)
1240 /// @param isUnbalancedReturn true if this edge is propagating an
1241 /// unbalanced return (this value is not used within this implementation
1242 /// but may be useful for subclasses of {@link IDESolver})
1243 ///
1244 void propagate(d_t SourceVal, n_t Target, d_t TargetVal,
1245 EdgeFunction<l_t> f) {
1246 PHASAR_LOG_LEVEL(DEBUG, "Propagate flow");
1247 PHASAR_LOG_LEVEL(DEBUG, "Source value : " << DToString(SourceVal));
1248 PHASAR_LOG_LEVEL(DEBUG, "Target : " << NToString(Target));
1249 PHASAR_LOG_LEVEL(DEBUG, "Target value : " << DToString(TargetVal));
1250 PHASAR_LOG_LEVEL(
1251 DEBUG, "Edge function : " << f << " (result of previous compose)");
1252
1253 auto JumpFnE = [&]() {
1254 const auto RevLookupResult = JumpFn->reverseLookup(Target, TargetVal);
1255 if (RevLookupResult) {
1256 const auto &JumpFnContainer = RevLookupResult->get();
1257 const auto Find = std::find_if(
1258 JumpFnContainer.begin(), JumpFnContainer.end(),
1259 [SourceVal](auto &KVpair) { return KVpair.first == SourceVal; });
1260 if (Find != JumpFnContainer.end()) {
1261 return Find->second;
1262 }
1263 }
1264 // jump function is initialized to all-top if no entry
1265 // was found
1266 return AllTop;
1267 }();
1268 auto fPrime = IDEProblem.combine(JumpFnE, f);
1269 bool NewFunction = fPrime != JumpFnE;
1270
1271 IF_LOG_LEVEL_ENABLED(DEBUG, {
1272 PHASAR_LOG_LEVEL(DEBUG,
1273 "Join: " << JumpFnE << " & " << f
1274 << (JumpFnE == f ? " (EF's are equal)" : " "));
1275 PHASAR_LOG_LEVEL(DEBUG, " = "
1276 << fPrime
1277 << (NewFunction ? " (new jump func)" : " "));
1278 PHASAR_LOG_LEVEL(DEBUG, ' ');
1279 });
1280 if (NewFunction) {
1281 JumpFn->addFunction(SourceVal, Target, TargetVal, fPrime);
1282 PathEdge Edge(SourceVal, Target, TargetVal);
1283 PathEdgeCount++;
1284 pathEdgeProcessingTask(std::move(Edge));
1285
1286 IF_LOG_LEVEL_ENABLED(DEBUG, {
1287 if (!IDEProblem.isZeroValue(TargetVal)) {
1288 PHASAR_LOG_LEVEL(
1289 DEBUG, "EDGE: <F: " << Target->getFunction()->getName()
1290 << ", D: " << DToString(SourceVal) << '>');
1291 PHASAR_LOG_LEVEL(DEBUG, " ---> <N: " << NToString(Target) << ',');
1292 PHASAR_LOG_LEVEL(DEBUG, " D: " << DToString(TargetVal) << ',');
1293 PHASAR_LOG_LEVEL(DEBUG, " EF: " << fPrime << '>');
1294 PHASAR_LOG_LEVEL(DEBUG, ' ');
1295 }
1296 });
1297 } else {
1298 PHASAR_LOG_LEVEL(DEBUG, "PROPAGATE: No new function!");
1299 }
1300 }

1427 /// The invariant for computing the number of generated (#gen) and killed
1428 /// (#kill) facts:
1429 /// (1) #Valid facts at the last statement <= #gen - #kill
1430 /// (2) #gen >= #kill
1431 ///
1432 /// The total number of valid facts can be smaller than the difference of
1433 /// generated and killed facts, due to set semantics, i.e., a fact can be
1434 /// generated multiple times but appears only once.
1435 ///
1436 /// Zero value is not counted!
1437 ///
1438 /// @brief Computes and prints statistics of the analysis run, e.g. number of
1439 /// generated/killed facts, number of summary-reuses etc.
1440 ///
1441 void computeAndPrintStatistics() {
1442 // Stores all valid facts at return site in caller context; return-site is
1443 // key
1444 std::unordered_map<n_t, std::set<d_t>> ValidInCallerContext;
1445 ptrdiff_t NumGenFacts = 0;
1446 ptrdiff_t NumIntraPathEdges = 0;
1447 ptrdiff_t NumInterPathEdges = 0;
1448 // --- Intra-procedural Path Edges ---
1449 // d1 --> d2-Set
1450 // Case 1: d1 in d2-Set
1451 // Case 2: d1 not in d2-Set, i.e., d1 was killed. d2-Set could be empty.
1452 for (const auto &Cell : ComputedIntraPathEdges.cellSet()) {
1453 auto Edge = std::make_pair(Cell.getRowKey(), Cell.getColumnKey());
1454 PHASAR_LOG_LEVEL(DEBUG, "N1: " << NToString(Edge.first));
1455 PHASAR_LOG_LEVEL(DEBUG, "N2: " << NToString(Edge.second));
1456 for (auto &[D1, D2s] : Cell.getValue()) {
1457 PHASAR_LOG_LEVEL(DEBUG, "d1: " << DToString(D1));
1458 NumIntraPathEdges += D2s.size();
1459 // Case 1
1460 if (D2s.find(D1) != D2s.end()) {
1461 NumGenFacts += D2s.size() - 1;
1462 }
1463 // Case 2
1464 else {
1465 NumGenFacts += D2s.size();
1466 }
1467 // Store all valid facts after call-to-return flow
1468 if (ICF->isCallSite(Edge.first)) {
1469 ValidInCallerContext[Edge.second].insert(D2s.begin(), D2s.end());
1470 }
1471 IF_LOG_LEVEL_ENABLED(DEBUG, [](const auto &D2s) {
1472 for (auto D2 : D2s) {
1473 PHASAR_LOG_LEVEL(DEBUG, "d2: " << DToString(D2));
1474 }
1475 PHASAR_LOG_LEVEL(DEBUG, "----");
1476 }(D2s));
1477 }
1478 PHASAR_LOG_LEVEL(DEBUG, " ");
1479 }
1480 // Stores all pairs of (Startpoint, Fact) for which a summary was applied
1481 std::set<std::pair<n_t, d_t>> ProcessSummaryFacts;
1482 PHASAR_LOG_LEVEL(DEBUG, "==============================================");
1483 PHASAR_LOG_LEVEL(DEBUG, "INTER PATH EDGES");
1484 for (const auto &Cell : ComputedInterPathEdges.cellSet()) {
1485 auto Edge = std::make_pair(Cell.getRowKey(), Cell.getColumnKey());
1486 PHASAR_LOG_LEVEL(DEBUG, "N1: " << NToString(Edge.first));
1487 PHASAR_LOG_LEVEL(DEBUG, "N2: " << NToString(Edge.second));
1488 // --- Call-flow Path Edges ---
1489 // Case 1: d1 --> empty set
1490 // Can be ignored, since killing a fact in the caller context will
1491 // actually happen during call-to-return.
1492 //
1493 // Case 2: d1 --> d2-Set
1494 // Every fact d_i != ZeroValue in d2-set will be generated in the
1495 // callee context, thus counts as a new fact. Even if d1 is passed as it
1496 // is, it will count as a new fact. The reason for this is, that d1 can
1497 // be killed in the callee context, but still be valid in the caller
1498 // context.
1499 //
1500 // Special Case: Summary was applied for a particular call
1501 // Process the summary's #gen and #kill.
1502 if (ICF->isCallSite(Edge.first)) {
1503 for (auto &[D1, D2s] : Cell.getValue()) {
1504 PHASAR_LOG_LEVEL(DEBUG, "d1: " << DToString(D1));
1505 NumInterPathEdges += D2s.size();
1506 for (auto D2 : D2s) {
1507 if (!IDEProblem.isZeroValue(D2)) {
1508 NumGenFacts++;
1509 }
1510 // Special case
1511 if (ProcessSummaryFacts.find(std::make_pair(Edge.second, D2)) !=
1512 ProcessSummaryFacts.end()) {
1513
1514 std::set<d_t> SummaryDSet;
1515 EndsummaryTab.get(Edge.second, D2)
1516 .foreachCell([&SummaryDSet](const auto & /*Row*/,
1517 const auto &Col,
1518 const auto & /*Val*/) {
1519 SummaryDSet.insert(Col);
1520 });
1521
1522 // Process summary just as an intra-procedural edge
1523 if (SummaryDSet.find(D2) != SummaryDSet.end()) {
1524 NumGenFacts += SummaryDSet.size() - 1;
1525 } else {
1526 NumGenFacts += SummaryDSet.size();
1527 }
1528 } else {
1529 ProcessSummaryFacts.emplace(Edge.second, D2);
1530 }
1531 PHASAR_LOG_LEVEL(DEBUG, "d2: " << DToString(D2));
1532 }
1533 PHASAR_LOG_LEVEL(DEBUG, "----");
1534 }
1535 }
1536 // --- Return-flow Path Edges ---
1537 // Since every fact passed to the callee was counted as a new fact, we
1538 // have to count every fact propagated to the caller as a kill to
1539 // satisfy our invariant. Obviously, every fact not propagated to the
1540 // caller will count as a kill. If an actual new fact is propagated to
1541 // the caller, we have to increase the number of generated facts by one.
1542 // Zero value does not count towards generated/killed facts.
1543 if (ICF->isExitInst(Cell.getRowKey())) {
1544 for (auto &[D1, D2s] : Cell.getValue()) {
1545 PHASAR_LOG_LEVEL(DEBUG, "d1: " << DToString(D1));
1546 NumInterPathEdges += D2s.size();
1547 auto CallerFacts = ValidInCallerContext[Edge.second];
1548 for (auto D2 : D2s) {
1549 // d2 not valid in caller context
1550 if (CallerFacts.find(D2) == CallerFacts.end()) {
1551 NumGenFacts++;
1552 }
1553 PHASAR_LOG_LEVEL(DEBUG, "d2: " << DToString(D2));
1554 }
1555 PHASAR_LOG_LEVEL(DEBUG, "----");
1556 }
1557 }
1558 PHASAR_LOG_LEVEL(DEBUG, " ");
1559 }
1560 PHASAR_LOG_LEVEL(DEBUG, "SUMMARY REUSE");
1561 ptrdiff_t TotalSummaryReuse = 0;
1562 for (const auto &Entry : FSummaryReuse) {
1563 PHASAR_LOG_LEVEL(DEBUG, "N1: " << NToString(Entry.first.first));
1564 PHASAR_LOG_LEVEL(DEBUG, "D1: " << DToString(Entry.first.second));
1565 PHASAR_LOG_LEVEL(DEBUG, "#Reuse: " << Entry.second);
1566 TotalSummaryReuse += Entry.second;
1567 }
1568 Genfacts += NumGenFacts;
1569 Summaryreuse += TotalSummaryReuse;
1570 IntraPathEdges += NumIntraPathEdges;
1571 InterPathEdges += NumInterPathEdges;
1572
1573 PHASAR_LOG_LEVEL(INFO, "----------------------------------------------");
1574 PHASAR_LOG_LEVEL(INFO, "=== Solver Statistics ===");
1575 PHASAR_LOG_LEVEL(INFO, "#Facts generated : " << Genfacts.value());
1576 PHASAR_LOG_LEVEL(INFO, "#Facts killed : " << Killfacts.value());
1577 PHASAR_LOG_LEVEL(INFO, "#Summary-reuse : " << Summaryreuse.value());
1578 PHASAR_LOG_LEVEL(INFO, "#Intra Path Edges: " << IntraPathEdges.value());
1579 PHASAR_LOG_LEVEL(INFO, "#Inter Path Edges: " << InterPathEdges.value());
1580 if constexpr (PAMM_CURR_SEV_LEVEL >= PAMM_SEVERITY_LEVEL::Full) {
1581 PHASAR_LOG_LEVEL(INFO,
1582 "Flow function query count: " << FFQueries.value());
1583 PHASAR_LOG_LEVEL(INFO,
1584 "Edge function query count: " << EFQueries.value());
1585 PHASAR_LOG_LEVEL(INFO, "Data-flow value propagation count: "
1586 << ValuePropagation.value());
1587 PHASAR_LOG_LEVEL(INFO, "Data-flow value computation count: "
1588 << ValueComputation.value());
1589 PHASAR_LOG_LEVEL(INFO, "Special flow function usage count: "
1590 << SpecialSummaryFF_Application.value());
1591 PHASAR_LOG_LEVEL(INFO, "Jump function construciton count: "
1592 << JumpFnConstruction.value());
1593 PHASAR_LOG_LEVEL(INFO, "Phase I duration: " << DFAPhase1.elapsed());
1594 PHASAR_LOG_LEVEL(INFO, "Phase II duration: " << DFAPhase2.elapsed());
1595 PHASAR_LOG_LEVEL(INFO, "----------------------------------------------");
1596 CachedFlowEdgeFunctions.print();
1597 }

1705 PHASAR_LOG_LEVEL(DEBUG, "=============================================");
1706 PHASAR_LOG_LEVEL(DEBUG, "Process inter-procedural path edges");
1707 PHASAR_LOG_LEVEL(DEBUG, "=============================================");
1708 Cells = ComputedInterPathEdges.cellVec();
1709 sort(Cells.begin(), Cells.end(), [&Stmtless](auto Lhs, auto Rhs) {
1710 return Stmtless(Lhs.getRowKey(), Rhs.getRowKey());
1711 });
1712 for (const auto &Cell : Cells) {
1713 auto Edge = std::make_pair(Cell.getRowKey(), Cell.getColumnKey());
1714 std::string N1Label = NToString(Edge.first);
1715 std::string N2Label = NToString(Edge.second);
1716 std::string FNameOfN1 = ICF->getFunctionOf(Edge.first)->getName().str();
1717 std::string FNameOfN2 = ICF->getFunctionOf(Edge.second)->getName().str();
1718 std::string N1StmtId = ICF->getStatementId(Edge.first);
1719 std::string N2StmtId = ICF->getStatementId(Edge.second);
1720 PHASAR_LOG_LEVEL(DEBUG, "N1: " << N1Label);
1721 PHASAR_LOG_LEVEL(DEBUG, "N2: " << N2Label);
1722
1723 // Add inter-procedural control flow edge
1724 DOTNode N1(FNameOfN1, N1Label, N1StmtId);
1725 DOTNode N2(FNameOfN2, N2Label, N2StmtId);
1726
1727 // Handle recursion control flow as intra-procedural control flow
1728 // since those eges never leave the function subgraph
1729 FG = nullptr;
1730 if (FNameOfN1 == FNameOfN2) {
1731 // This function subgraph is guaranteed to exist
1732 FG = &G.Functions[FNameOfN1];
1733 FG->IntraCFEdges.emplace(N1, N2);
1734 } else {
1735 // Check the case where the callee is a single statement function,
1736 // thus does not contain intra-procedural path edges. We have to
1737 // generate the function sub graph here!
1738 if (!G.Functions.count(FNameOfN1)) {
1739 FG = &G.Functions[FNameOfN1];
1740 FG->Id = FNameOfN1;
1741 FG->Stmts.insert(N1);
1742 } else if (!G.Functions.count(FNameOfN2)) {
1743 FG = &G.Functions[FNameOfN2];
1744 FG->Id = FNameOfN2;
1745 FG->Stmts.insert(N2);
1746 }
1747 G.InterCFEdges.emplace(N1, N2);
1748 }
1749
1750 // Create D1 and D2, if D1 == D2 == lambda then add Edge(D1, D2) to
1751 // interLambdaEges otherwise add Edge(D1, D2) to interFactEdges
1752 unsigned D1FactId = 0;
1753 unsigned D2FactId = 0;
1754 for (const auto &D1ToD2Set : Cell.getValue()) {
1755 auto D1Fact = D1ToD2Set.first;
1756 PHASAR_LOG_LEVEL(DEBUG, "d1: " << DToString(D1Fact));
1757 DOTNode D1;
1758 if (IDEProblem.isZeroValue(D1Fact)) {
1759 D1 = {FNameOfN1, "Λ", N1StmtId, 0, false, true};
1760 } else {
1761 // Get the fact-ID
1762 D1FactId = G.getFactID(D1Fact);
1763 std::string D1Label{DToString(D1Fact)};
1764 D1 = {FNameOfN1, D1Label, N1StmtId, D1FactId, false, true};
1765 // FG should already exist even for single statement functions
1766 if (!G.containsFactSG(FNameOfN1, D1FactId)) {
1767 FG = &G.Functions[FNameOfN1];
1768 auto *D1FSG = FG->getOrCreateFactSG(D1FactId, D1Label);
1769 D1FSG->Nodes.insert(std::make_pair(N1StmtId, D1));
1770 }
1771 }
1772
1773 auto D2Set = D1ToD2Set.second;
1774 for (const auto &D2Fact : D2Set) {
1775 PHASAR_LOG_LEVEL(DEBUG, "d2: " << DToString(D2Fact));
1776 DOTNode D2;
1777 if (IDEProblem.isZeroValue(D2Fact)) {
1778 D2 = {FNameOfN2, "Λ", N2StmtId, 0, false, true};
1779 } else {
1780 // Get the fact-ID
1781 D2FactId = G.getFactID(D2Fact);
1782 std::string D2Label{DToString(D2Fact)};
1783 D2 = {FNameOfN2, D2Label, N2StmtId, D2FactId, false, true};
1784 // FG should already exist even for single statement functions
1785 if (!G.containsFactSG(FNameOfN2, D2FactId)) {
1786 FG = &G.Functions[FNameOfN2];
1787 auto *D2FSG = FG->getOrCreateFactSG(D2FactId, D2Label);
1788 D2FSG->Nodes.insert(std::make_pair(N2StmtId, D2));
1789 }
1790 }
1791
1792 if (IDEProblem.isZeroValue(D1Fact) &&
1793 IDEProblem.isZeroValue(D2Fact)) {
1794 // Do not add lambda recursion edges as inter-procedural edges
1795 if (D1.FuncName != D2.FuncName) {
1796 G.InterLambdaEdges.emplace(D1, D2, true, "AllBottom", "BOT");
1797 }
1798 } else {
1799 // std::string EFLabel = EF ? EF->str() : " ";
1800 std::string EFLabel;
1801 auto EFVec = IntermediateEdgeFunctions[std::make_tuple(
1802 Edge.first, D1Fact, Edge.second, D2Fact)];
1803 for (const auto &EF : EFVec) {
1804 PHASAR_LOG_LEVEL(DEBUG, "Partial EF Label: " << EF);
1805 EFLabel.append(to_string(EF) + ", ");
1806 }
1807 PHASAR_LOG_LEVEL(DEBUG, "EF LABEL: " << EFLabel);
1808 G.InterFactEdges.emplace(D1, D2, true, EFLabel);
1809 }
1810 }
1811 PHASAR_LOG_LEVEL(DEBUG, "----------");
1812 }
1813 PHASAR_LOG_LEVEL(DEBUG, " ");
1814 }
1815 OS << G;
1816 }
