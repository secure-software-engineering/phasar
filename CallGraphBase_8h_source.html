28template <typename T>
29concept IsCallGraph =
30 requires(const T &CG, typename T::n_t Inst, typename T::f_t Fun) {
31 typename T::n_t;
32 typename T::f_t;
33
34 /// Returns an iterable range of all possible callee candidates at the
35 /// given call-site induced by the used call-graph.
36 ///
37 /// NOTE: This function is typically called in a hot part of the analysis
38 /// and should therefore be very fast
39 {
40 CG.getCalleesOfCallAt(Inst)
41 } -> psr::is_iterable_over_v<typename T::f_t>;
42
43 /// Returns an iterable range of all possible call-site candidates that
44 /// may call the given function induced by the used call-graph.
45 { CG.getCallersOf(Fun) } -> psr::is_iterable_over_v<typename T::n_t>;
46
47 /// A range of all functions that are vertices in the call-graph. The
48 /// number of vertex functions can be retrieved by
49 /// getNumVertexFunctions().
50 {
51 CG.getAllVertexFunctions()
52 } -> psr::is_iterable_over_v<typename T::f_t>;
53
54 /// A range of all call-sites that are vertices in the call-graph. The
55 /// number of vertex-callsites can be retrived by getNumVertexCallSites().
56 {
57 CG.getAllVertexCallSites()
58 } -> psr::is_iterable_over_v<typename T::n_t>;
59
60 { CG.getNumVertexFunctions() } -> std::convertible_to<size_t>;
61 { CG.getNumVertexCallSites() } -> std::convertible_to<size_t>;
62
63 /// Same as getNumVertexFunctions()
64 { CG.size() } noexcept -> std::convertible_to<size_t>;
65 { CG.empty() } noexcept -> std::convertible_to<bool>;
66 };

76/// Base class of all CallGraph implementations within phasar (currently only
77/// CallGraph<N, F>).
78/// Only represents the data, not how to create it.
79template <typename Derived> class CallGraphBase : public CRTPBase<Derived> {
80 friend Derived;
81 using CRTPBase<Derived>::self;
82
83public:
84 using n_t = typename CGTraits<Derived>::n_t;
85 using f_t = typename CGTraits<Derived>::f_t;
86
87 /// Returns an iterable range of all possible callee candidates at the given
88 /// call-site induced by the used call-graph.
89 ///
90 /// NOTE: This function is typically called in a hot part of the analysis and
91 /// should therefore be very fast
92 [[nodiscard]] constexpr decltype(auto)
93 getCalleesOfCallAt(ByConstRef<n_t> Inst) const
94 noexcept(noexcept(this->self().getCalleesOfCallAtImpl(Inst))) {
95 static_assert(
96 is_iterable_over_v<decltype(self().getCalleesOfCallAtImpl(Inst)), f_t>);
97 return self().getCalleesOfCallAtImpl(Inst);
98 }
99
100 /// Returns an iterable range of all possible call-site candidates that may
101 /// call the given function induced by the used call-graph.
102 [[nodiscard]] constexpr decltype(auto)
103 getCallersOf(ByConstRef<f_t> Fun) const {
104 static_assert(
105 is_iterable_over_v<decltype(this->self().getCallersOfImpl(Fun)), n_t>);
106 return self().getCallersOfImpl(Fun);
107 }
108
109 /// A range of all functions that are vertices in the call-graph. The number
110 /// of vertex functions can be retrieved by getNumVertexFunctions().
111 [[nodiscard]] constexpr decltype(auto)
112 getAllVertexFunctions() const noexcept {
113 return self().getAllVertexFunctionsImpl();
114 }
115
116 /// A range of all call-sites that are vertices in the call-graph. The number
117 /// of vertex-callsites can be retrived by getNumVertexCallSites().
118 [[nodiscard]] constexpr auto getAllVertexCallSites() const noexcept {
119 return self().getAllVertexCallSitesImpl();
120 }
121
122 [[nodiscard]] constexpr size_t getNumVertexFunctions() const noexcept {
123 return self().getNumVertexFunctionsImpl();
124 }
125 [[nodiscard]] constexpr size_t getNumVertexCallSites() const noexcept {
126 return self().getNumVertexCallSitesImpl();
127 }
128
129 /// The number of functions within this call-graph
130 [[nodiscard]] constexpr size_t size() const noexcept {
131 return getNumVertexFunctions();
132 }
133
134 [[nodiscard]] constexpr bool empty() const noexcept { return size() == 0; }
135};

137/// A view over a call-graph that implements psr::GraphTraits on the
138/// getCallersOf() relation.
139/// Useful to compute CG-SCCs.
140template <typename CallGraphTy, typename DB> class ReverseCGGraph {
141 static constexpr bool NeedsMapping = !IdType<typename CallGraphTy::f_t>;
142
143 enum class FunctionIdImpl : uint32_t {};
144
145public:
146 using FunctionId = std::conditional_t<NeedsMapping, FunctionIdImpl,
147 typename CallGraphTy::f_t>;
148
149 constexpr ReverseCGGraph(
150 NonNullPtr<const CallGraphTy> CGView, NonNullPtr<const DB> IRDB,
151 Compressor<typename CallGraphTy::f_t, FunctionId> FC) noexcept
152 requires(NeedsMapping)
153 : CGView(CGView), IRDB(IRDB), FC(std::move(FC)) {}
154
155 constexpr ReverseCGGraph(NonNullPtr<const CallGraphTy> CGView,
156 NonNullPtr<const DB> IRDB) noexcept
157 : CGView(CGView), IRDB(IRDB) {
158 FC.reserve(CGView->getNumVertexFunctions());
159 for (const auto &Fun : CGView->getAllVertexFunctions()) {
160 FC.insert(Fun);
161 }
162 }
163
164 NonNullPtr<const CallGraphTy> CGView;
165 NonNullPtr<const DB> IRDB;
166 [[no_unique_address]] std::conditional_t<
167 NeedsMapping, Compressor<typename CallGraphTy::f_t, FunctionId>,
168 NoneCompressor> FC{};
169};

171template <typename CallGraphTy, typename DB>
172ReverseCGGraph(NonNullPtr<const CallGraphTy>, NonNullPtr<const DB>)
173 -> ReverseCGGraph<CallGraphTy, DB>;
174template <typename CallGraphTy, typename DB>
175ReverseCGGraph(const CallGraphTy *, const DB *)
176 -> ReverseCGGraph<CallGraphTy, DB>;

178template <typename CallGraphTy, typename DB>
179struct GraphTraits<ReverseCGGraph<CallGraphTy, DB>> {
180 using graph_type = ReverseCGGraph<CallGraphTy, DB>;
181 using value_type = typename CallGraphTy::f_t;
182 using vertex_t = typename graph_type::FunctionId;
183 using edge_t = vertex_t;
184
185 static constexpr vertex_t Invalid = vertex_t(UINT32_MAX);
186
187 static constexpr auto mapToFunction(const graph_type &G) {
188 return [&G](ByConstRef<typename CallGraphTy::n_t> Inst) {
189 const auto &Fun = G.IRDB->getFunctionOf(Inst);
190 return G.FC.getOrNull(Fun).value();
191 };
192 }
193
194 static constexpr auto outEdges(const graph_type &G, vertex_t Vtx) {
195 return llvm::map_range(G.CGView->getCallersOf(G.FC[Vtx]), mapToFunction(G));
196 }
197
198 static constexpr decltype(auto) nodes(const graph_type &G) {
199 return G.CGView->getAllVertexFunctions();
200 }
201
202 // TODO: Roots
203
204 static constexpr auto vertices(const graph_type &G) {
205 return iota<vertex_t>(G.CGView->getNumVertexFunctions());
206 }
207
208 static constexpr decltype(auto) node(const graph_type &G, vertex_t Vtx) {
209 return G.FC[Vtx];
210 }
211
212 static constexpr size_t size(const graph_type &G) {
213 return G.CGView->getNumVertexFunctions();
214 }
215
216 static constexpr vertex_t target(edge_t Edge) { return Edge; }
217 static constexpr vertex_t withEdgeTarget(edge_t /*Edge*/, vertex_t Vtx) {
218 return Vtx;
219 }
220};
