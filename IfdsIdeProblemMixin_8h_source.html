26/// \brief A template that lets you inherit recurring APIs into IFDS/IDE
27/// problems (solvable by the IterativeIDESolver or the legacy
28/// IFDSSolver/IDESolver). Create a subclass from this and implement all
29/// missing functions to create your own IFDS/IDE analysis.
30///
31/// For more information on how to write an IFDS/IDE analysis see
32/// - [Writing an IFDS
33/// Analysis](https://github.com/secure-software-engineering/phasar/wiki/Writing-an-IFDS-analysis).
34/// - [Writing an IDE
35/// Analysis](https://github.com/secure-software-engineering/phasar/wiki/Writing-an-IDE-analysis)
36template <IfdsAnalysisDomain AnalysisDomainTy,
37 typename Container = std::set<typename AnalysisDomainTy::d_t>>
38class IfdsIdeProblemMixin
39 : protected FlowFunctionTemplates<typename AnalysisDomainTy::d_t,
40 Container>,
41 public DerivedJoinLattice<
42 typename detail::ValueDomainAdder<AnalysisDomainTy>::l_t>,
43 public DefaultSemiRing<
44 typename detail::ValueDomainAdder<AnalysisDomainTy>::l_t>,
45 public detail::ValueDomainAdder<AnalysisDomainTy> {
46protected:
47 using FFTemplates =
48 FlowFunctionTemplates<typename AnalysisDomainTy::d_t, Container>;
49
50public:
51 using ProblemAnalysisDomain = detail::ValueDomainAdder<AnalysisDomainTy>;
52 using typename FFTemplates::container_type;
53 using typename FFTemplates::FlowFunctionPtrType;
54 using typename ProblemAnalysisDomain::d_t;
55 using typename ProblemAnalysisDomain::db_t;
56 using typename ProblemAnalysisDomain::f_t;
57 using typename ProblemAnalysisDomain::i_t;
58 using typename ProblemAnalysisDomain::l_t;
59 using typename ProblemAnalysisDomain::n_t;
60 using typename ProblemAnalysisDomain::t_t;
61 using typename ProblemAnalysisDomain::v_t;
62 using EdgeFunctionType = EdgeFunction<l_t>;
63
64 [[nodiscard]] constexpr ByConstRef<d_t> getZeroValue() const noexcept {
65 return ZeroValue;
66 }
67
68 [[nodiscard]] constexpr bool
69 isZeroValue(ByConstRef<d_t> Fact) const noexcept {
70 return Fact == ZeroValue;
71 }

105 /// Seeds that just start with ZeroValue and bottomElement() at the starting
106 /// points of each EntryPoint function.
107 /// Takes the __ALL__ EntryPoint into account.
108 [[nodiscard]] static InitialSeeds<n_t, d_t, l_t>
109 createDefaultSeeds(auto &&Self)
110 requires std::is_nothrow_default_constructible_v<
111 typename AnalysisDomainTy::c_t>
112 {
113 static_assert(std::derived_from<std::remove_cvref_t<decltype(Self)>,
114 IfdsIdeProblemMixin>);
115 InitialSeeds<n_t, d_t, l_t> Seeds;
116 typename AnalysisDomainTy::c_t C{};
117
118 addSeedsForStartingPoints(Self.getEntryPoints(), Self.getProjectIRDB(), C,
119 Seeds, Self.getZeroValue(), Self.bottomElement());
120
121 return Seeds;
122 }
123
124#if __cpp_explicit_this_parameter >= 202110L

126 createDefaultSeeds(this auto &&Self) {

128 }
129#endif

131 NonNullPtr<const db_t> IRDB{};
132 d_t ZeroValue{};
133 std::vector<std::string> EntryPoints;
134};
