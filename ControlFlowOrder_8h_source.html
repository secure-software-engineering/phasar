37template <typename N, typename F, CFGOf<N, F> CFGTy>
38void computeCFGOrder(ControlFlowOrder<N> &Into, const CFGTy &CF, const F &Fun) {
39 llvm::SmallDenseSet<N> Seen;
40
41 const auto Visit = [&](auto &Visit, ByConstRef<N> Inst) {
42 if (!Seen.insert(Inst).second) {
43 return;
44 }
45
46 scope_exit Push = [&]() { Into.Order.insert(Inst); };
47
48 if constexpr (IsBlockAwareControlFlow<CFGTy>) {
49 if (auto Next = CF.getUniqueSuccessor(Inst)) {
50 Visit(Visit, unwrapNullable(Next));
51 return;
52 }
53 }
54 for (const auto &Succ : CF.getSuccsOf(Inst)) {
55 Visit(Visit, Succ);
56 }
57 };
58
59 for (const auto &SP : CF.getStartPointsOf(Fun)) {
60 Visit(Visit, SP);
61 }
62}

64template <typename N, typename F, CFGOf<N, F> CFGTy>
65[[nodiscard]] inline ControlFlowOrder<N> computeCFGOrder(const CFGTy &CF,
66 const F &Fun) {
67 ControlFlowOrder<N> Ret;
68 computeCFGOrder(Ret, CF, Fun);
69 return Ret;
70}
