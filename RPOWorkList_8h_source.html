20template <typename ItemT> class TopoFixpointDriver {
21public:
22 TopoFixpointDriver() noexcept = default;
23
24 TopoFixpointDriver(ControlFlowOrder<ItemT> &&CFO) : CFO(std::move(CFO)) {
25 WorkList.resize(this->CFO.Order.size());
26 }
27
28 void push(ByConstRef<ItemT> Item) {
29 auto IId = CFO.Order.get(Item);
30 WorkList.set(uint32_t(IId));
31
32 if (int(IId) > Max) {
33 Max = int(IId);
34 }
35 }
36
37 std::optional<ItemT> pop() {
38 if (Max < 0) {
39 return std::nullopt;
40 }
41
42 auto IId = typename ControlFlowOrder<ItemT>::CFGOrderId(Max);
43 Max = WorkList.find_prev(Max);
44 return CFO.Order[IId];
45 }
46
47 [[nodiscard]] bool empty() const noexcept { return Max < 0; }
48
49 LLVM_ATTRIBUTE_ALWAYS_INLINE void run(std::invocable<ItemT> auto Handler) {
50 while (auto Inst = pop()) {
51 std::invoke(Handler, *Inst);
52 }
53 }
54
55 [[nodiscard]] constexpr const auto &getCFO() const noexcept { return CFO; }
56
57private:
58 ControlFlowOrder<ItemT> CFO;
59 llvm::BitVector WorkList;
60 int Max = -1;
61};
