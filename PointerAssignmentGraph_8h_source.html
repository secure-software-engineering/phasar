32/// Direct pointer assignment: `To = From`.
33struct Assign {};
34/// GEP (field-access) edge. \p Offset holds the byte offset when statically
35/// known; \c std::nullopt for non-constant GEPs.
36struct Gep {
37 std::optional<int16_t> Offset;
38};
39/// Passing an argument at a call site. \p ArgNo is the 0-based parameter index.
40struct Call {
41 uint16_t ArgNo;
42};
43/// Return value flows from callee to caller call-site.
44struct Return {};
45/// Load of a pointer: `To = *From`.
46struct Load {};
47/// Store of a non-aliased pointer: `*To = From`.
48struct Store {};
49/// Store of a potentially aliased pointer: `*To = From`.
50struct StorePOI {};
51/// Block-copy of aggregate memory (e.g., memcpy/memmove): all pointer fields
52/// inside the source are assumed to flow to the corresponding fields of the
53/// destination.
54struct Copy {};

56/// Tagged union of all PAG edge kinds, see above.
57struct Edge : public std::variant<Assign, Gep, Call, Return, Load, Store,
58 StorePOI, Copy> {
59 using Base =

62 Copy>::variant;

117template <typename T, typename N>
118concept CanOnAddEdgeN =
119 requires(T &Strategy, ValueId From, ValueId To, pag::Edge E) {

126template <typename T>
127concept CanWithCalleesOfCallAt =

132/// Minimal requirements for a PAG-builder strategy: analysis-domain type
133/// members plus \c withCalleesOfCallAt. Does not require \c onAddEdge; use
134/// \c PBStrategy for the full requirement set.
135template <typename T>
136concept PBStrategyBase =
137 CanWithCalleesOfCallAt<T> &&
138 requires(T &Strategy, const T &CStrategy, ValueId From, ValueId To,
139 const typename T::n_t &Inst, pag::Edge E) {
140 typename T::v_t;
141 typename T::db_t;
142 typename T::f_t;
143 typename T::n_t;
144 };

146template <typename T>
147concept PBStrategy = CanOnAddEdge<T> && PBStrategyBase<T>;

150concept CanOnAddValue = requires(T &Strategy, const T::v_t &Var) {

168/// Fans out every PAG-builder callback to two independent strategies.
169/// \p FirstT must be a full \c PBStrategy; \p SecondT only needs to satisfy
170/// \c CanOnAddEdge (it borrows the \c withCalleesOfCallAt and analysis-domain
171/// types from \p FirstT).
172///
173/// Useful for intersecting alias analyses.
174template <PBStrategy FirstT, CanOnAddEdgeN<typename FirstT::n_t> SecondT>
175// requires(std::same_as<typename FirstT::n_t, typename SecondT::n_t> &&
176// std::same_as<typename FirstT::v_t, typename SecondT::v_t> &&
177// std::same_as<typename FirstT::f_t, typename SecondT::f_t> &&
178// std::same_as<typename FirstT::db_t, typename SecondT::db_t>)
179struct PBStrategyCombinator {
180 using n_t = typename FirstT::n_t;
181 using v_t = typename FirstT::v_t;
182 using f_t = typename FirstT::f_t;
183 using db_t = typename FirstT::db_t;
184
185 [[no_unique_address]] FirstT First;
186 [[no_unique_address]] SecondT Second;
187
188 constexpr void onAddEdge(ValueId From, ValueId To, Edge E,
189 Nullable<n_t> AtInstruction) {
190 First.onAddEdge(From, To, E, AtInstruction);
191 Second.onAddEdge(From, To, E, AtInstruction);
192 }
193
194 constexpr void onAddValue(ByConstRef<v_t> Variable, ValueId VId)
195 requires(CanOnAddValue<FirstT> || CanOnAddValue<SecondT>)
196 {
197 if constexpr (CanOnAddValue<FirstT>) {
198 First.onAddValue(Variable, VId);
199 }
200 if constexpr (CanOnAddValue<SecondT>) {
201 Second.onAddValue(Variable, VId);
202 }
203 }
204
205 [[nodiscard]] constexpr size_t
206 getNumPossibleValues(const db_t &IRDB) const noexcept
207 requires(CanGetNumPossibleValues<FirstT> ||
208 CanGetNumPossibleValues<SecondT>)
209 {
210 const auto Retrieve = [&IRDB]<typename S>(const S &Strategy) {
211 if constexpr (CanGetNumPossibleValues<S>) {
212 return Strategy.getNumPossibleValues(IRDB);
213 }
214 return 0;
215 };
216
217 return std::max(Retrieve(First), Retrieve(Second));
218 }
219
220 constexpr void
221 withCalleesOfCallAt(ByConstRef<n_t> CS,
222 llvm::function_ref<void(f_t)> WithCallee) const {
223 // We assume, both are on the same call-graph.
224 if constexpr (CanWithCalleesOfCallAt<SecondT>) {
225 Second.withCalleesOfCallAt(CS, WithCallee);
226 } else {
227 First.withCalleesOfCallAt(CS, WithCallee);
228 }
229 }
230
231 constexpr auto consumeAAResults(size_t NumVars) &&
232 requires(CanConsumeAAResults<FirstT> && !CanConsumeAAResults<SecondT>) {
233 return std::move(First).consumeAAResults(NumVars);
234 }
235};

237/// Upgrades a \c CanOnAddEdge type to a full \c PBStrategy by mixing in the
238/// analysis-domain types and \c withCalleesOfCallAt from a separate
239/// \c PBStrategyBase. Typical use: wrap an edge-listener (e.g., a union-find
240/// accumulator) that does not own a call-graph with a call-graph provider.
241///
242/// \tparam FirstT The edge-listener. Must satisfy \c CanOnAddEdge.
243/// \tparam SecondT The call-graph / domain provider. Must satisfy
244/// \c PBStrategyBase.
245template <CanOnAddEdge FirstT, PBStrategyBase SecondT>
246struct PBMixin : public FirstT {
247 using n_t = typename SecondT::n_t;
248 using v_t = typename SecondT::v_t;
249 using f_t = typename SecondT::f_t;
250 using db_t = typename SecondT::db_t;
251
252 [[no_unique_address]] SecondT Second;
253
254 [[nodiscard]] constexpr size_t
255 getNumPossibleValues(const db_t &IRDB) const noexcept
256 requires(!CanGetNumPossibleValues<FirstT> &&
257 CanGetNumPossibleValues<SecondT>)
258 {
259 return Second.getNumPossibleValues(IRDB);
260 }
261
262 constexpr void
263 withCalleesOfCallAt(ByConstRef<n_t> CS,
264 llvm::function_ref<void(f_t)> WithCallee) const {
265 Second.withCalleesOfCallAt(CS, WithCallee);
266 }
267};

298/// The main customization point for PAG building.
299/// Implement this interface to provide several callbacks to the PAGBuilder,
300/// specifying how the PAG should be built.
301template <typename AnalysisDomainT> class PBStrategyRef final {
302public:
303 using n_t = typename AnalysisDomainT::n_t;
304 using v_t = typename AnalysisDomainT::v_t;
305 using f_t = typename AnalysisDomainT::f_t;
306 using db_t = typename AnalysisDomainT::db_t;
307
308 /// Creates a type-erased PBStrategyRef from a non-null pointer to a concrete
309 /// Strategy-object.
310 template <PBStrategy ConcreteStrategyT>
311 requires(!same_as_decay<PBStrategyRef, ConcreteStrategyT>)
312 constexpr PBStrategyRef(ConcreteStrategyT *Strategy) noexcept
313 : VT(&VTableFor<ConcreteStrategyT>), Ctx(Strategy) {
314 assert(Strategy != nullptr);
315 }
316
317 /// Called by buildPAG() for every (unique) edge that should be added to the
318 /// PAG
319 ///
320 /// \param From The source-node.
321 /// \param To The destination-node.
322 /// \param E The edge-label that specifies the kind of data-flow that is
323 /// denoted by this edge.
324 /// \param AtInstruction Optionally holds the instruction that caused this
325 /// edge to be created.
326 void onAddEdge(ValueId From, ValueId To, pag::Edge E,
327 Nullable<n_t> AtInstruction) {
328 VT->OnAddEdge(Ctx, From, To, E, AtInstruction);
329 }
330
331 /// Called by buildPAG() for every (unique) node that should be added to the
332 /// PAG, *excluding* nodes that have already been registered in the used
333 /// ValueCompressor before buildPAG() was called.
334 ///
335 /// \param Variable The IR-specific variable/value for which the new node
336 /// has been created.
337 /// \param VId The value-id of the newly created node.
338 void onAddValue(ByConstRef<v_t> Variable, ValueId VId) {
339 VT->OnAddValue(Ctx, Variable, VId);
340 }
341
342 /// Estimates the maximum number of values created in the ValueCompressor.
343 /// Must not be exact, this is just for pre-allocating buffers for
344 /// optimization purposes.
345 [[nodiscard]] size_t getNumPossibleValues(const db_t &IRDB) const noexcept {
346 return VT->GetNumPossibleValues(Ctx, IRDB);
347 }
348
349 /// Invokes a call-back with each function that may be called at the given
350 /// call-site. Usually, you want to use a CallGraph or a Resolver to implement
351 /// this function.
352 ///
353 /// \param CS The call-site.
354 /// \param WithCallee The callback to be invoked for each callee.
355 void withCalleesOfCallAt(ByConstRef<n_t> CS,
356 llvm::function_ref<void(f_t)> WithCallee) const {
357 VT->WithCalleesOfCallAt(Ctx, CS, WithCallee);
358 }
359
360private:
361 struct VTable {
362 void (*OnAddEdge)(void *Ctx, ValueId From, ValueId To, pag::Edge E,
363 Nullable<n_t> AtInstruction);
364 void (*OnAddValue)(void *Ctx, ByConstRef<v_t> Variable, ValueId VId);
365 size_t (*GetNumPossibleValues)(const void *Ctx, const db_t &IRDB) noexcept;
366 void (*WithCalleesOfCallAt)(const void *Ctx, ByConstRef<n_t> CS,
367 llvm::function_ref<void(f_t)> WithCallee);
368 };
369
370 template <typename ConcreteStrategyT>
371 static void onAddEdgeThunk(void *Ctx, ValueId From, ValueId To, pag::Edge E,
372 Nullable<n_t> AtInstruction) {
373 // Do things like allocating a new slot in the adjacency-list here
374 static_cast<ConcreteStrategyT *>(Ctx)->onAddEdge(From, To, E,
375 AtInstruction);
376 }
377
378 template <typename ConcreteStrategyT>
379 static void onAddValueThunk(void *Ctx, ByConstRef<v_t> Variable,
380 ValueId VId) {
381 if constexpr (CanOnAddValue<ConcreteStrategyT>) {
382 static_cast<ConcreteStrategyT *>(Ctx)->onAddValue(Variable, VId);
383 }
384 }
385
386 template <typename ConcreteStrategyT>
387 static size_t getNumPossibleValuesThunk(const void *Ctx,
388 const db_t &IRDB) noexcept {
389 if constexpr (CanGetNumPossibleValues<ConcreteStrategyT>) {
390 return static_cast<const ConcreteStrategyT *>(Ctx)->getNumPossibleValues(
391 IRDB);
392 } else {
393 // Fallback-heuristic
394 return IRDB.getNumInstructions();
395 }
396 }
397
398 template <typename ConcreteStrategyT>
399 static void
400 withCalleesOfCallAtThunk(const void *Ctx, ByConstRef<n_t> CS,
401 llvm::function_ref<void(f_t)> WithCallee) {
402
403 static_cast<const ConcreteStrategyT *>(Ctx)->withCalleesOfCallAt(
404 CS, WithCallee);
405 }
406
407 template <typename ConcreteStrategyT>
408 static constexpr VTable VTableFor = {
409 &onAddEdgeThunk<ConcreteStrategyT>,
410 &onAddValueThunk<ConcreteStrategyT>,
411 &getNumPossibleValuesThunk<ConcreteStrategyT>,
412 &withCalleesOfCallAtThunk<ConcreteStrategyT>,
413 };
414
415 const VTable *VT{};
416 void *Ctx{};
417};

420/// A utility-class that can be used to build a pointer-assignment graph (PAG).
421///
422/// This class does not enforce a specifiy graph-layout -- it does not even
423/// require that you explicitly store a graph in memory. Instead, you get
424/// notified about every node and edge that should be created in the PAG.
425///
426/// \tparam AnalysisDomainT The analysis domain to use.
427template <typename AnalysisDomainT> class PAGBuilder {
428public:
429 using n_t = typename AnalysisDomainT::n_t;
430 using v_t = typename AnalysisDomainT::v_t;
431 using f_t = typename AnalysisDomainT::f_t;
432 using db_t = typename AnalysisDomainT::db_t;
433
434 constexpr PAGBuilder() noexcept = default;
435 virtual ~PAGBuilder() = default;
436
437 /// Iterates the passed IRDB to build a PAG, based on the given Strategy.
438 ///
439 /// \param IRDB The IR program to analyze.
440 /// \param VC The value-compressor to use for assigning unique sequential
441 /// integer-ids to each PAG-node. This VC may be pre-populated. You can expect
442 /// that buildPAG() will respect the pre-population in that case. Otherwise,
443 /// ids will be assigned in visitation order (which is not stable and should
444 /// not be relied on!).
445 /// \param Strategy The customization-point for this function. See the docs of
446 /// PBStrategy for more information.
447 virtual void buildPAG(const db_t &IRDB, ValueCompressor<v_t> &VC,
448 pag::PBStrategyRef<AnalysisDomainT> Strategy) = 0;
449};
