36/// Whether a value of the pointer-free type \p Ty is laid out such that it
37/// could hold pointer bit patterns.
38[[nodiscard]] inline bool mayHidePointer(const llvm::DataLayout &DL,
39 const llvm::Type *Ty) {
40 if (Ty->isIntegerTy(DL.getPointerSizeInBits())) {
41 return true;
42 }
43 if (const auto *Arr = llvm::dyn_cast<llvm::ArrayType>(Ty)) {
44 return mayHidePointer(DL, Arr->getElementType());
45 }
46 const auto *Struct = llvm::dyn_cast<llvm::StructType>(Ty);
47 return Struct && !Struct->isOpaque() && !Struct->elements().empty() &&
48 llvm::all_of(Struct->elements(), [&DL](const llvm::Type *ElemTy) {
49 return mayHidePointer(DL, ElemTy);
50 });
51}

53/// Whether Ptr is a memory-location (alloca or global), accessed as an
54/// integer.
55///
56/// Useful for handling atomicrmw of pointers, which clang punns to i64, and
57/// for ABI-coerced aggregates (see PunnedABICache).
58[[nodiscard]] inline bool isPunnedPointerAccess(const llvm::DataLayout &DL,
59 const llvm::Value *Ptr,
60 const llvm::Type *AccessedTy) {
61 if (!mayHidePointer(DL, AccessedTy)) {
62 return false;
63 }
64 const llvm::Value *Base = Ptr->stripPointerCastsAndAliases();
65 if (const auto *A = llvm::dyn_cast<llvm::AllocaInst>(Base)) {
66 return !definitelyContainsNoPointer(A->getAllocatedType());
67 }
68 if (const auto *G = llvm::dyn_cast<llvm::GlobalVariable>(Base)) {
69 return !definitelyContainsNoPointer(G->getValueType());
70 }
71 return false;
72}

74/// The memory access to model for a load, store, atomicrmw or cmpxchg.
75struct LLVMMemoryAccess {
76 const llvm::Instruction *Instr{};
77 const llvm::Value *Pointer{};
78 /// Null if the access only reads.
79 const llvm::Value *StoredValue{};
80 /// Null if the access only writes; \c Instr itself for an atomic.
81 const llvm::Instruction *LoadedInto{};
82 bool Punned{};
83
84 /// The value whose type decides whether a pointer is transferred. For a
85 /// cmpxchg that is the new value, not the { ty, i1 } result.
86 [[nodiscard]] const llvm::Value *transferredValue() const noexcept {
87 return StoredValue ? StoredValue : LoadedInto;
88 }
89
90 /// Whether the access has to be modeled at all. Gating on
91 /// definitelyContainsNoPointer alone would drop punned accesses.
92 [[nodiscard]] bool mayTransferPointer() const {
93 return Punned || !definitelyContainsNoPointer(transferredValue());
94 }
95};

97/// Decomposes \p I, or returns nullopt if it is not a memory access. Field-
98/// insensitively an atomicrmw is a store of the new value plus a load of the
99/// old one; cmpxchg likewise, into its { ty, i1 } result.
100///
101/// The result still has to pass mayTransferPointer().
102[[nodiscard]] inline std::optional<LLVMMemoryAccess>
103asMemoryAccess(const llvm::Instruction &I, const llvm::DataLayout &DL) {
104 const auto Make = [&DL, &I](const llvm::Value *Ptr,
105 const llvm::Value *StoredValue,
106 const llvm::Instruction *LoadedInto) {
107 const auto *Transferred = StoredValue ? StoredValue : LoadedInto;
108 return LLVMMemoryAccess{
109 .Instr = &I,
110 .Pointer = Ptr,
111 .StoredValue = StoredValue,
112 .LoadedInto = LoadedInto,
113 .Punned = isPunnedPointerAccess(DL, Ptr, Transferred->getType()),
114 };
115 };
116
117 if (const auto *S = llvm::dyn_cast<llvm::StoreInst>(&I)) {
118 return Make(S->getPointerOperand(), S->getValueOperand(), nullptr);
119 }
120 if (const auto *L = llvm::dyn_cast<llvm::LoadInst>(&I)) {
121 return Make(L->getPointerOperand(), nullptr, L);
122 }
123 if (const auto *RMW = llvm::dyn_cast<llvm::AtomicRMWInst>(&I)) {
124 return Make(RMW->getPointerOperand(), RMW->getValOperand(), RMW);
125 }
126 if (const auto *CX = llvm::dyn_cast<llvm::AtomicCmpXchgInst>(&I)) {
127 return Make(CX->getPointerOperand(), CX->getNewValOperand(), CX);
128 }
129 return std::nullopt;
130}

132/// Recognizes ABI-coerced boundary values: values that carry a pointer
133/// although their type has none.
134///
135/// A small pointer-carrying struct is passed and returned in registers:
136///
137/// For whatever reason, on AArch64 clang punns such nested pointers as i64
138/// instead of ptr. This here is a best-effort approcach to keep pointer
139/// data-flows in such situations.
140class PunnedABICache {
141public:
142 explicit PunnedABICache(const llvm::DataLayout *DL) noexcept
143 : DL(&assertNotNull(DL)) {}
144
145 /// Whether \p V -- an argument, returned value or call result of \p F --
146 /// carries a pointer that its type does not reveal.
147 [[nodiscard]] bool isCoercedPointer(const llvm::Value *V,
148 const llvm::Function *F) {
149 return !llvm::isa<llvm::ConstantData>(V) &&
150 definitelyContainsNoPointer(V->getType()) &&
151 mayHidePointer(*DL, V->getType()) && punsPointers(F);
152 }
153
154private:
155 bool punsPointers(const llvm::Function *F) {
156 auto [It, Inserted] = Cache.try_emplace(F, false);
157 if (Inserted) {
158 It->second = llvm::any_of(llvm::instructions(*F), [this](const auto &I) {
159 const auto Access = asMemoryAccess(I, *DL);
160 return Access && Access->Punned;
161 });
162 }
163 return It->second;
164 }
165
166 const llvm::DataLayout *DL;
167 llvm::DenseMap<const llvm::Function *, bool> Cache;
168};
