33/// \brief This class may be used to represent the special zero value (aka. Λ)
34/// for IFDS and IDE problems. The LLVMZeroValue is implemented as a singleton.
35///
36class LLVMZeroValue : public llvm::GlobalVariable {
37private:
38 LLVMZeroValue(llvm::Module &Mod);
39
40 static constexpr llvm::StringLiteral LLVMZeroValueInternalName = "zero_value";
41 static bool isZeroValueImpl(const llvm::Value *V) noexcept {
42 return V == getInstance();
43 }
44
45public:
46 LLVMZeroValue(const LLVMZeroValue &Z) = delete;
47 LLVMZeroValue &operator=(const LLVMZeroValue &Z) = delete;
48 LLVMZeroValue(LLVMZeroValue &&Z) = delete;
49 LLVMZeroValue &operator=(LLVMZeroValue &&Z) = delete;
50 ~LLVMZeroValue() = default;
51
52 [[nodiscard]] llvm::StringRef getName() const noexcept {
53 return LLVMZeroValueInternalName;
54 }
55
56 /// Gets the singleton instance of the special zero value (aka. Λ).
57 [[nodiscard]] static const LLVMZeroValue *getInstance();
58
59 /// Checks, whether the given llvm::Value * is the special zero-value (aka.
60 /// Λ).
61 ///
62 /// You can use this as follows:
63 /// \code
64 /// return strongUpdateStore(Store, LLVMZeroValue::isLLVMZeroValue);
65 /// \endcode
66 // NOLINTNEXTLINE(readability-identifier-naming)
67 static constexpr auto isLLVMZeroValue = fn<isZeroValueImpl>;
68};
69} // namespace psr
