22/// \file This file contains useful functions for handling errors, by using
23/// std::system_error, or returning null or a default value.

25template <typename T> T getOrThrow(llvm::ErrorOr<T> ValOrErr) {
26 if (ValOrErr) {
27 return std::move(*ValOrErr);
28 }
29 throw std::system_error(ValOrErr.getError());
30}

32template <typename T>
33std::optional<T> getOrNull(llvm::ErrorOr<T> ValOrErr) noexcept(
34 std::is_nothrow_move_constructible_v<T>) {
35 if (ValOrErr) {
36 return std::move(*ValOrErr);
37 }
38 return std::nullopt;

41template <typename T>
42 requires std::is_default_constructible_v<T>
43T getOrEmpty(llvm::ErrorOr<T> ValOrErr) noexcept(
44 std::is_nothrow_move_constructible_v<T>) {
45 if (ValOrErr) {
46 return std::move(*ValOrErr);

51template <typename T, typename MapFn>
52llvm::ErrorOr<std::invoke_result_t<MapFn, const T &>>
53mapValue(const llvm::ErrorOr<T> &ValOrErr,
54 MapFn Mapper) noexcept(std::is_nothrow_invocable_v<MapFn, const T &>) {
55 if (ValOrErr) {
56 return std::invoke(std::move(Mapper), *ValOrErr);
57 }
58 return ValOrErr.getError();
59}
