22/// \file
23/// Defines requirements for an analysis problem that can be solved by the
24/// MonoIFDSSolver.
25///
26/// Since MonoIFDS solves analyses bottom-up, each CG-SCC is analyzed in
27/// isolation. The solver will call Problem.localAnalysis(...) once per SCC and
28/// use the returned LocalMonoIFDSProblem to drive the solving process.
29/// Although the solver currently is only single-threaded, you can expect that
30/// (also in a multi-threaded future) LocalMonoIFDSProblem instances are not
31/// accessed by multiple threads at a time, while different instances may be
32/// solved on different threads in parallel.
33

40/// \brief CG-SCC-local analysis
41template <typename T, typename Dom>
42concept LocalMonoIFDSProblem = requires(
43 T &Problem, DataFlowEnvironment<typename Dom::d_t> &InOut,
44 const DataFlowEnvironment<typename Dom::d_t> &In, typename Dom::n_t Inst,
45 const typename Dom::n_t &Fact, const typename Dom::f_t &Fun,
46 Compressor<typename Dom::d_t, SourceFactId> &SeedCompressor) {
47 /// Intra-procedural data-flow. Input facts are passed-in as InOut;
48 /// modifications are performed in-place.
49 ///
50 /// Corresponds to the $flow()$ function in the paper.
51 Problem.normalFlow(InOut, Inst);
52
53 /// Intra-procedural data-flow at call-sites. Input facts are passed-in as
54 /// InOut; modifications are performed in-place. Kills facts that may be
55 /// strongly updated by the callee. Don't use it to *generate* facts.
56 ///
57 /// Corresponds to the $callFlow()$ function in the paper.
58 Problem.callToRetFlow(InOut, Inst);
59
60 /// Inter-procedural data-flow at exit-statements; Maps callee-facts back
61 /// to the return-site in the caller. As with normal IFDS, this function
62 /// will be called for each incoming Fact, that should be mapped back;
63 /// return-site facts are returned by this function.
64 ///
65 /// Corresponds to the $returnVal()$ function in the paper.
66 {
67 Problem.returnFlow(Inst, Fact)
68 } -> psr::is_iterable_over_v<typename Dom::d_t>;
69
70 /// Inter-procedural data-flow at entry-statements; Maps
71 /// callee-source-facts back to the call-site in the caller. This function
72 /// will be called for each source Fact; call-site facts are returned by
73 /// this function.
74 ///
75 /// Corresponds to the $passArgs^{-1}()$ function in the paper.
76 {
77 Problem.invCallFlow(Inst, Fact)
78 } -> psr::is_iterable_over_v<typename Dom::d_t>;
79
80 /// Applies a pre-computed summary of Fun at Inst into InOut, if
81 /// applicable.
82 ///
83 /// Useful for pre-known taint-propagators and declaration-only library
84 /// functions.
85 ///
86 /// \returns True, iff a summary was applied. This will take precedence
87 /// over a summary that the solver may have computed for Fun!
88 { Problem.summaryFlow(In, InOut, Inst, Fun) } -> std::convertible_to<bool>;
89
90 /// The special zero-value, aka. $\Lambda$. Always holds. Facts that are
91 /// generated unconditionally, originate from zero.
92 ///
93 /// Note: The solver guarantees that the zero-value always has
94 /// SourceFactId 0.
95 { Problem.getZeroValue() } -> std::convertible_to<typename Dom::d_t>;
96
97 /// Approximates the source-facts that should hold at the entry of Fun.
98 /// Input the facts in the InOut map as
99 /// `SeedState[Fact].insert(SeedCompressor.getOrInsert(Fact))`
100 ///
101 /// Note: This is assumed to be a (conservative) over-approximation!
102 Problem.initialSeeds(InOut, SeedCompressor, Fun);
103
104 /// At a call-site Inst calling Fun, invokes the given callback for each
105 /// fact that should be generated from zero there.
106 ///
107 /// Useful for taint sources.
108 Problem.generateFactsAtCall(Inst, Fun, DummyFn<const typename Dom::d_t &>{});
109
110 /// At a non-call-site Inst, invokes the given callback for each
111 /// fact that should be generated from zero there.
112 ///
113 /// Useful for taint sources.
114 Problem.generateFacts(Inst, DummyFn<const typename Dom::d_t &>{});
115
116 /// Invokes the given callback for each LeakFact for which the solver
117 /// should later call onResult(Inst, LeakFact), if LeakFacts holds at
118 /// Inst. Here, Inst is assumed to be a call-site that may call Fun.
119 ///
120 /// Useful for taint sinks.
121 Problem.requestResultCallbackAtCallSite(Inst, Fun,
122 DummyFn<const typename Dom::d_t &>{});
123
124 /// Invokes the given callback for each LeakFact for which the solver
125 /// should later call onResult(Inst, LeakFact), if LeakFacts holds at
126 /// Inst.
127 ///
128 /// Useful for taint sinks.
129 Problem.requestResultCallback(Inst, DummyFn<const typename Dom::d_t &>{});
130
131 /// Notifies the problem that a previously requested leak-Fact now is
132 /// known to hold at Inst.
133 ///
134 /// Useful for reporting taint leaks.
135 Problem.onResult(Inst, Fact);
136};

138/// \brief Defines requirements for an analysis problem that can be solved by
139/// the MonoIFDSSolver.
140template <typename T>
141concept MonoIFDSProblem =
142 requires(T &Problem, SCCId<FunctionId> CurrSCC,
143 std::pmr::memory_resource *MRes, llvm::raw_ostream &OS) {
144 /// The analysis domain. Defines the type of data-flow facts, and the IR
145 /// on which the analysis can be run.
146 typename T::ProblemAnalysisDomain;
147 requires MonoIFDSAnalysisDomain<typename T::ProblemAnalysisDomain>;
148
149 /// Create a local analysis for the given SCC.
150 /// Use the given std::memory_resource to allocate node-based containers,
151 /// if you have any.
152 {
153 Problem.localAnalysis(CurrSCC, MRes)
154 } -> LocalMonoIFDSProblem<typename T::ProblemAnalysisDomain>;
155
156 /// Pretty-print the analysis results into the given llvm::raw_ostream.
157 Problem.emitTextReport(OS);
158 };
