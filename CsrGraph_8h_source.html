49/// Immutable directed graph in compressed-sparse-row format: the successors of
50/// vertex V are Targets[Offsets[V] .. Offsets[V+1]).
51///
52/// Build it with fromEdges() or assignEdges(). assignEdges() reuses the
53/// allocated storage, so a single CsrGraph object can serve as buffer for many
54/// short-lived graphs.
55template <IdType VtxId> class CsrGraph {
56public:
57 /// Builds the graph from (Source, Target) pairs. The successors of each
58 /// vertex keep the relative order of Edges.
59 [[nodiscard]] static CsrGraph
60 fromEdges(size_t NumVertices, llvm::ArrayRef<std::pair<VtxId, VtxId>> Edges,
61 llvm::ArrayRef<VtxId> RootVtxs = {}) {
62 CsrGraph Ret;
63 Ret.assignEdges(NumVertices, Edges, RootVtxs);
64 return Ret;
65 }
66
67 /// Returns G with all edges inverted. The predecessors of each vertex are
68 /// sorted ascending. The roots of the result are the vertices without
69 /// successors in G.
70 [[nodiscard]] static CsrGraph reversed(const CsrGraph &G) {
71 CsrGraph Ret;
72 Ret.countEdges(G.numVertices(), G.Targets);
73 Ret.Targets.resize_for_overwrite(G.numEdges());
74 for (auto From : llvm::reverse(iota<VtxId>(G.numVertices()))) {
75 for (auto To : G.succsOf(From)) {
76 Ret.Targets[--Ret.Offsets[To]] = From;
77 }
78 }
79 for (auto Vtx : iota<VtxId>(G.numVertices())) {
80 if (G.succsOf(Vtx).empty()) {
81 Ret.Roots.push_back(Vtx);
82 }
83 }
84 return Ret;
85 }
86
87 /// Replaces the contents of this graph by NumVertices vertices and the
88 /// given (Source, Target) edges. Clears the roots.
89 void assignEdges(size_t NumVertices,
90 llvm::ArrayRef<std::pair<VtxId, VtxId>> Edges,
91 llvm::ArrayRef<VtxId> RootVtxs = {}) {
92 countEdges(NumVertices, llvm::make_first_range(Edges));
93 Targets.resize_for_overwrite(Edges.size());
94 for (auto [From, To] : llvm::reverse(Edges)) {
95 assert(size_t(To) < NumVertices);
96 Targets[--Offsets[From]] = To;
97 }
98 Roots.assign(RootVtxs.begin(), RootVtxs.end());
99 }
100
101 [[nodiscard]] size_t numVertices() const noexcept {
102 assert(!Offsets.empty());
103 return Offsets.size() - 1;
104 }
105
106 [[nodiscard]] size_t numEdges() const noexcept { return Targets.size(); }
107
108 [[nodiscard]] llvm::ArrayRef<VtxId> succsOf(VtxId Vtx) const noexcept {
109 return csrSuccs(Offsets, Targets, Vtx);
110 }
111
112 [[nodiscard]] llvm::ArrayRef<VtxId> roots() const & noexcept { return Roots; }
113 [[nodiscard]] auto roots() && noexcept { return Roots; }
114
115private:
116 /// Sets Offsets[V] to the end of the successors of V. Inserting the edges
117 /// at --Offsets[From] in reverse order then leaves Offsets[V] at the begin.
118 void countEdges(size_t NumVertices, auto &&Sources) {
119 Offsets.assign(NumVertices + 1, 0);
120 for (auto From : Sources) {
121 ++Offsets[From];
122 }
123 std::partial_sum(Offsets.begin(), Offsets.end(), Offsets.begin());
124 }
125
126 /// One entry per vertex plus a sentinel holding numEdges()
127 TypedVector<VtxId, uint32_t> Offsets{0};
128 llvm::SmallVector<VtxId, 0> Targets;
129 llvm::SmallVector<VtxId, 1> Roots;
130};
