74 /**
75 * Records a jump function. The source statement is implicit.
76 * @see PathEdge
77 */
78 void addFunction(d_t SourceVal, n_t Target, d_t TargetVal,
79 EdgeFunction<l_t> EdgeFunc) {
80 PHASAR_LOG_LEVEL(DEBUG, "Start adding new jump function");
81 PHASAR_LOG_LEVEL(DEBUG, "Fact at source : " << DToString(SourceVal));
82 PHASAR_LOG_LEVEL(DEBUG, "Fact at target : " << DToString(TargetVal));
83 PHASAR_LOG_LEVEL(DEBUG, "Destination : " << NToString(Target));
84 PHASAR_LOG_LEVEL(DEBUG, "Edge Function : " << EdgeFunc);
85 // we do not store the default function (all-top)
86 if (llvm::isa<AllTop<l_t>>(EdgeFunc)) {
87 return;
88 }
89
90 auto &SourceValToFunc = NonEmptyReverseLookup.get(Target, TargetVal);
91 if (auto Find = std::find_if(
92 SourceValToFunc.begin(), SourceValToFunc.end(),
93 [SourceVal](const std::pair<d_t, EdgeFunction<l_t>> &Entry) {
94 return SourceVal == Entry.first;
95 });
96 Find != SourceValToFunc.end()) {
97 // it is important that existing values in JumpFunctions
98 // are overwritten
99 Find->second = EdgeFunc;
100 } else {
101 SourceValToFunc.emplace_back(SourceVal, EdgeFunc);
102 }
103
104 auto &TargetValToFunc = NonEmptyForwardLookup.get(SourceVal, Target);
105 if (auto Find = std::find_if(
106 TargetValToFunc.begin(), TargetValToFunc.end(),
107 [TargetVal](const std::pair<d_t, EdgeFunction<l_t>> &Entry) {
108 return TargetVal == Entry.first;
109 });
110 Find != TargetValToFunc.end()) {
111 // it is important that existing values in JumpFunctions
112 // are overwritten
113 Find->second = EdgeFunc;
114 } else {
115 TargetValToFunc.emplace_back(TargetVal, EdgeFunc);
116 }
117
118 // V Table::insert(R r, C c, V v) always overrides (see
119 // comments above)
120 NonEmptyLookupByTargetNode[Target].insert(SourceVal, TargetVal, EdgeFunc);
121 PHASAR_LOG_LEVEL(DEBUG, "End adding new jump function");
122 }

124 /**
125 * Returns, for a given target statement and value all associated
126 * source values, and for each the associated edge function.
127 * The return value is a mapping from source value to function.
128 */
129 std::optional<std::reference_wrapper<
130 llvm::SmallVectorImpl<std::pair<d_t, EdgeFunction<l_t>>>>>
131 reverseLookup(n_t Target, d_t TargetVal) {
132 if (!NonEmptyReverseLookup.contains(Target, TargetVal)) {
133 return std::nullopt;
134 }
135 return {NonEmptyReverseLookup.get(Target, TargetVal)};
136 }
137
138 /**
139 * Returns, for a given source value and target statement all
140 * associated target values, and for each the associated edge function.
141 * The return value is a mapping from target value to function.
142 */
143 std::optional<std::reference_wrapper<
144 llvm::SmallVectorImpl<std::pair<d_t, EdgeFunction<l_t>>>>>
145 forwardLookup(d_t SourceVal, n_t Target) {
146 if (!NonEmptyForwardLookup.contains(SourceVal, Target)) {
147 return std::nullopt;
148 }
149 return {NonEmptyForwardLookup.get(SourceVal, Target)};
150 }

152 /**
153 * Returns for a given target statement all jump function records with this
154 * target.
155 * The return value is a set of records of the form
156 * (sourceVal,targetVal,edgeFunction).
157 */
158 Table<d_t, d_t, EdgeFunction<l_t>> &lookupByTarget(n_t Target) {
159 return NonEmptyLookupByTargetNode[Target];
160 }

162 template <typename HandlerFn>
163 void foreachEdgeFunction(HandlerFn Handler) const {
164 NonEmptyForwardLookup.foreachCell(
165 [Handler = std::move(Handler)](ByConstRef<d_t> /*Row*/,
166 ByConstRef<n_t> /*Col*/,
167 const auto &TargetFactAndEF) {
168 for (const auto &[TargetFact, EF] : TargetFactAndEF) {
169 std::invoke(Handler, EF);
170 }
171 });
172 }

174 /**
175 * Removes a jump function. The source statement is implicit.
176 * @see PathEdge
177 * @return True if the function has actually been removed. False if it was not
178 * there anyway.
179 */
180 bool removeFunction(d_t SourceVal, n_t Target, d_t TargetVal) {
181 auto &SourceValToFunc = NonEmptyReverseLookup.get(Target, TargetVal);
182 if (auto Find = std::find_if(
183 SourceValToFunc.begin(), SourceValToFunc.end(),
184 [SourceVal](const std::pair<d_t, EdgeFunction<l_t>> &Entry) {
185 return SourceVal == Entry.first;
186 });
187 Find != SourceValToFunc.end()) {
188 SourceValToFunc.erase(Find);
189 }
190 auto &TargetValToFunc = NonEmptyForwardLookup.get(SourceVal, Target);
191 if (auto Find = std::find_if(
192 TargetValToFunc.begin(), TargetValToFunc.end(),
193 [TargetVal](const std::pair<d_t, EdgeFunction<l_t>> &Entry) {
194 return TargetVal == Entry.first;
195 });
196 Find != TargetValToFunc.end()) {
197 TargetValToFunc.erase(Find);
198 }
199 return NonEmptyLookupByTargetNode.erase(Target);
200 }

211 void printJumpFunctions(llvm::raw_ostream &OS) {
212 OS << "\n******************************************************";
213 OS << "\n* Print all Jump Functions *";
214 OS << "\n******************************************************\n";
215 for (auto &Entry : NonEmptyLookupByTargetNode) {
216 std::string NLabel = NToString(Entry.first);
217 OS << "\nN: " << NLabel << "\n---" << std::string(NLabel.size(), '-')
218 << '\n';
219 for (auto Cell : Entry.second.cellSet()) {
220 OS << "D1: " << DToString(Cell.r) << '\n'
221 << "\tD2: " << DToString(Cell.c) << '\n'
222 << "\tEF: " << Cell.v << "\n\n";
223 }
224 }
