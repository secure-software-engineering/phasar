43/// Utility class to make pag::PBMixin<IndirectionSensUnionFindAA,
44/// LLVMCGProvider> implement PBStrategy.
45class LLVMCGProvider : public LLVMPAGDomain {
46public:
47 constexpr LLVMCGProvider(
48 NonNullPtr<const LLVMBasedCallGraph> CG PSR_LIFETIMEBOUND) noexcept
49 : CG(CG) {}

51 void withCalleesOfCallAt(n_t Inst,
52 std::invocable<f_t> auto WithCallee) const {
53 for (const auto *Callee : CG->getCalleesOfCallAt(Inst)) {

63/// Low-level entry point: builds the PAG for \p IRDB using the given
64/// \p AnalysisT strategy, and returns the raw alias-analysis result (not
65/// wrapped in an \c LLVMUnionFindAliasIterator).
66///
67/// \param IRDB The project to analyze.
68/// \param Ana A \c PBStrategy instance (e.g., \c BasicUnionFindAA,
69/// \c BottomupUnionFindAA). Taken by forwarding reference and
70/// consumed when results are extracted.
71/// \param VC Optional pre-populated \c ValueCompressor. A new one is
72/// allocated if null.
73/// \param Impl PAG builder implementation (default: \c LLVMPAGBuilder).
74template <pag::PBStrategy AnalysisT,
75 std::derived_from<PAGBuilder<LLVMPAGDomain>> PAGBuilderImpl =
76 LLVMPAGBuilder>
77[[nodiscard]] inline RawAAResult auto computeUnionFindAARaw(
78 const LLVMProjectIRDB &IRDB, AnalysisT &&Ana,
79 MaybeUniquePtr<ValueCompressor<PAGVariable>> VC = nullptr,
80 PAGBuilderImpl Impl = LLVMPAGBuilder::withBuiltinMemSSA()) {
81 if (!VC) {
82 VC = std::make_unique<ValueCompressor<PAGVariable>>();
83 }
84
85 Impl.buildPAG(IRDB, *VC, &Ana);
86
87 const auto NumVars = VC->size();
88 return PSR_FWD(Ana).consumeAAResults(NumVars, [&VC](ValueId VId) {
89 return llvm::any_of(VC->id2vars(VId),
90 [](PAGVariable V) { return !V.isReturnVariable(); });
91 });
92}

128/// Builds the PAG and returns a \c LLVMUnionFindAliasIterator (owns the \c
129/// ValueCompressor and the raw result) that implements the \c

131///
132/// \param IRDB The project to analyze.
133/// \param Ana A \c PBStrategy instance. Consumed when results are extracted.
134/// \param VC Optional pre-populated \c ValueCompressor. A new one is
135/// allocated if null; the iterator takes ownership.
136/// \param Impl PAG builder implementation (default: \c LLVMPAGBuilder).
137template <pag::PBStrategy AnalysisT,
138 std::derived_from<PAGBuilder<LLVMPAGDomain>> PAGBuilderImpl =
139 LLVMPAGBuilder>
140[[nodiscard]] inline IsLLVMAliasIterator auto computeUnionFindAA(
141 const LLVMProjectIRDB &IRDB, AnalysisT &&Ana,
142 MaybeUniquePtr<ValueCompressor<PAGVariable>> VC PSR_LIFETIMEBOUND = nullptr,
143 PAGBuilderImpl Impl = LLVMPAGBuilder::withBuiltinMemSSA()) {
144 if (!VC) {
145 VC = std::make_unique<ValueCompressor<PAGVariable>>();

147
148 auto AARes =
