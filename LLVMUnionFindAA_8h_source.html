38/// Returns a \c ValueId handler suitable for \c RawAliasSet::foreach() that
39/// maps each alias \c ValueId back to all of its underlying \c llvm::Value*
40/// (via \p VC), forwarding non-null values to \p Callback.
41constexpr std::invocable<ValueId> auto
42llvmUnionFindAliasHandler(const ValueCompressor<PAGVariable> &VC,
43 std::invocable<const llvm::Value *> auto Callback) {
44 return [&VC, Callback{copyOrRef(Callback)}](ValueId Alias) {
45 for (auto V : VC.id2vars(Alias)) {
46 if (const auto *LLVMVar = V.valueOrNull()) [[likely]] {
47 std::invoke(Callback, LLVMVar);
48 }
49 }
50 };
51}

73/// CRTP mixin that adds the LLVM alias-iterator interface to a class that
74/// holds a \c UnionFindAAResult and a \c ValueCompressor<PAGVariable>.
75///
76/// Provides \c forallAliasesOf(), \c mayAlias(), and \c alias() overloads
77/// accepting both \c llvm::Value* and \c ValueId arguments. Results are
78/// reported as \c llvm::Value* via the stored \c ValueCompressor.
79///
80/// The derived class must expose a \c VC member (pointer to a
81/// \c ValueCompressor<PAGVariable>).
82///
83/// \tparam Derived The CRTP derived class.
84/// \tparam AAResT A type satisfying \c UnionFindAAResult.
85template <typename Derived, typename AAResT>
86 requires UnionFindAAResult<std::remove_cvref_t<AAResT>>
87// NOLINTNEXTLINE(cppcoreguidelines-pro-type-member-init)
88struct LLVMUnionFindAliasIteratorMixin {
89 [[no_unique_address]] AAResT AARes;
90
91 using v_t = const llvm::Value *;
92 using n_t = const llvm::Instruction *;
93
94 [[nodiscard]] decltype(auto) getRawAliasSet(ValueId ValId) const {
95 return AARes.getRawAliasSet(ValId);
96 }
97
98 [[nodiscard]] const auto &base() const noexcept { return AARes; }
99
100 void
101 forallAliasesOf(ValueId VId, const auto & /*Inst*/,
102 std::invocable<const llvm::Value *> auto Callback) const {
103 const auto &RawAliases = AARes.getRawAliasSet(VId);
104 RawAliases.foreach (
105 llvmUnionFindAliasHandler(*self().VC, copyOrRef(Callback)));
106 }
107
108 void
109 forallAliasesOf(const llvm::Value *Ptr, const auto &Inst,
110 std::invocable<const llvm::Value *> auto Callback) const {
111 if (auto ValId = self().VC->getOrNull(Ptr)) {
112 forallAliasesOf(*ValId, Inst, copyOrRef(Callback));
113 }
114 }
115
116 [[nodiscard]] bool mayAlias(ValueId Ptr1, ValueId Ptr2) const {
117 return AARes.mayAlias(Ptr1, Ptr2);
118 }
119
120 [[nodiscard]] bool mayAlias(ValueId Ptr1, ValueId Ptr2,
121 const auto & /*AtInstruction*/) const {
122 return AARes.mayAlias(Ptr1, Ptr2);
123 }
124
125 [[nodiscard]] bool mayAlias(const llvm::Value *Ptr1,
126 const llvm::Value *Ptr2) const {
127 auto ValId1 = self().VC->getOrNull(Ptr1);
128 auto ValId2 = self().VC->getOrNull(Ptr2);
129
130 return ValId1 && ValId2 && mayAlias(*ValId1, *ValId2);
131 }
132
133 [[nodiscard]] bool mayAlias(const llvm::Value *Ptr1, const llvm::Value *Ptr2,
134 const auto & /*AtInstruction*/) const {
135 return mayAlias(Ptr1, Ptr2);
136 }
137
138 [[nodiscard]] AliasResult alias(const llvm::Value *Ptr1,
139 const llvm::Value *Ptr2,
140 const auto &AtInstruction) const {
141 auto ValId1 = self().VC->getOrNull(Ptr1);
142 auto ValId2 = self().VC->getOrNull(Ptr2);
143 if (!ValId1 || !ValId2) {
144 return AliasResult::NoAlias;
145 }
146 if (*ValId1 == *ValId2) {
147 if (Ptr1 == Ptr2) {
148 return AliasResult::MustAlias;
149 }
150 return !llvm::isa<llvm::LoadInst>(Ptr1) &&
151 !llvm::isa<llvm::LoadInst>(Ptr2)
152 ? AliasResult::MustAlias
153 : AliasResult::MayAlias;
154 }
155 return mayAlias(*ValId1, *ValId2, AtInstruction) ? AliasResult::MayAlias
156 : AliasResult::NoAlias;
157 }
158
159 [[nodiscard]] constexpr const Derived &self() const noexcept {
160 return *static_cast<const Derived *>(this);
161 }
162};

340/// Low-level entry point: builds the PAG for \p IRDB using the given
341/// \p AnalysisT strategy, and returns the raw alias-analysis result (not
342/// wrapped in an \c LLVMUnionFindAliasIterator).
343///
344/// \param IRDB The project to analyze.
345/// \param Ana A \c PBStrategy instance (e.g., \c BasicUnionFindAA,
346/// \c BottomupUnionFindAA). Taken by forwarding reference and
347/// consumed when results are extracted.
348/// \param VC Optional pre-populated \c ValueCompressor. A new one is
349/// allocated if null.
350/// \param Impl PAG builder implementation (default: \c LLVMPAGBuilder).
351template <pag::PBStrategy AnalysisT,
352 std::derived_from<PAGBuilder<LLVMPAGDomain>> PAGBuilderImpl =
353 LLVMPAGBuilder>
354[[nodiscard]] inline UnionFindAAResult auto computeUnionFindAARaw(
355 const LLVMProjectIRDB &IRDB, AnalysisT &&Ana,
356 MaybeUniquePtr<ValueCompressor<PAGVariable>> VC = nullptr,
357 PAGBuilderImpl Impl = LLVMPAGBuilder::withBuiltinMemSSA()) {
358 if (!VC) {
359 VC = std::make_unique<ValueCompressor<PAGVariable>>();
360 }
361
362 Impl.buildPAG(IRDB, *VC, &Ana);
363
364 const auto NumVars = VC->size();
365 return PSR_FWD(Ana).consumeAAResults(NumVars, [&VC](ValueId VId) {
366 return llvm::any_of(VC->id2vars(VId),
367 [](PAGVariable V) { return !V.isReturnVariable(); });
368 });
369}
