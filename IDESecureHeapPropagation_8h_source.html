22enum class SecureHeapFact { ZERO, INITIALIZED };
23enum class SecureHeapValue { TOP, INITIALIZED, BOT };
24
25template <> struct JoinLatticeTraits<SecureHeapValue> {
26 static constexpr auto bottom() noexcept { return SecureHeapValue::BOT; }
27 static constexpr auto top() noexcept { return SecureHeapValue::TOP; }
28 static constexpr auto join(SecureHeapValue LHS,
29 SecureHeapValue RHS) noexcept {
30 if (LHS == RHS || RHS == top()) {
31 return LHS;
32 }
33 if (LHS == top()) {
34 return RHS;
35 }
36 return bottom();
37 }
38};
