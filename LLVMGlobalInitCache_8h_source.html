26/// Memoised walker for global-variable pointer initializers.
27///
28/// Traverses a \c llvm::Constant initializer and collects the \c ValueId of
29/// every pointer-typed sub-constant it contains (direct pointer, GEP base,
30/// or pointer elements of an aggregate). Results are cached so shared
31/// sub-expressions are not revisited.
32///
33/// Create one instance per analysis run; it is tied to a single
34/// \c ValueCompressor via the \p GetVar callback.
35struct GlobalInitCache {
36 std::unordered_map<const llvm::Constant *, llvm::SmallVector<ValueId, 1>>
37 Cache;
38
39 /// Returns the \c ValueId slice for all pointer-typed constants reachable
40 /// from \p Const. \p GetVar maps an \c llvm::Value* to a \c ValueId
41 /// (typically \c getOrInsertVar).
42 template <std::invocable<const llvm::Value *> GetVarFn>
43 [[nodiscard]] llvm::ArrayRef<ValueId> getOrCreate(const llvm::Constant *Const,
44 GetVarFn &&GetVar) {
45 if (definitelyContainsNoPointer(Const)) {
46 return {};
47 }
48
49 auto [It, Inserted] = Cache.try_emplace(Const);
50 if (!Inserted) {
51 return It->second;
52 }
53 auto &Vec = It->second;
54
55 if (llvm::isa<llvm::ConstantPointerNull>(Const)) {
56 return {};
57 }
58
59 if (const auto *CGep = llvm::dyn_cast<llvm::GEPOperator>(Const)) {
60 // TODO: Properly handle constant GEPs
61 return getOrCreate(llvm::cast<llvm::Constant>(CGep->getPointerOperand()),
62 GetVar);
63 }
64
65 if (Const->getType()->isPointerTy()) {
66 Vec.push_back(std::invoke(GetVar, Const));
67 return Vec;
68 }
69
70 // TODO: Get rid of the recursion
71
72 if (const auto *Agg = llvm::dyn_cast<llvm::ConstantAggregate>(Const)) {
73 if (Agg->getType()->isArrayTy() &&
74 definitelyContainsNoPointer(Agg->getType()->getArrayElementType())) {
75 return {};
76 }
77 for (size_t I = 0, N = Agg->getNumOperands(); I < N; ++I) {
78 const auto *Elem = llvm::cast<llvm::Constant>(
79 Agg->getAggregateElement(I)->stripPointerCastsAndAliases());
80 auto Sub = getOrCreate(Elem, GetVar);
81 Vec.append(Sub.begin(), Sub.end());
82 }
83 }
84
85 // TODO: more
86
87 return Vec;
88 }
89};
