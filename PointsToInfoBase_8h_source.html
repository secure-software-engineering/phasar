53/// Base class of all points-to analysis implementations. Don't use this class
54/// directly. For a type-erased variant, use PointsToInfoRef or PointsToInfo.
55template <typename Derived> class PointsToInfoBase {
56public:
57 using v_t = typename PointsToTraits<Derived>::v_t;
58 using n_t = typename PointsToTraits<Derived>::n_t;
59 using o_t = typename PointsToTraits<Derived>::o_t;
60 using PointsToSetTy = typename PointsToTraits<Derived>::PointsToSetTy;
61 using PointsToSetPtrTy = typename PointsToTraits<Derived>::PointsToSetPtrTy;
62
63 explicit PointsToInfoBase() noexcept {
64 static_assert(std::is_base_of_v<PointsToInfoBase, Derived>,
65 "Invalid CRTP instantiation: Derived must inherit from "
66 "PointsToInfoBase<Derived>!");
67 }
