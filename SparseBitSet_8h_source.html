23/// Thin wrapper over a Roaring bitmap, providing an API conforming to the
24/// IsRawAliasSet concept.
25template <SmallIdType IdT> class SparseBitSet {
26public:
27 using value_type = IdT;
28
29 SparseBitSet() = default;
30
31 void insert(IdT Id) { Bits.add(uint32_t(Id)); }
32
33 [[nodiscard]] bool tryInsert(IdT Id) { return Bits.addChecked(uint32_t(Id)); }
34
35 [[nodiscard]] bool contains(IdT Id) const {
36 return Bits.contains(uint32_t(Id));

39 template <std::invocable<IdT> HandlerFn>
40 LLVM_ATTRIBUTE_ALWAYS_INLINE void foreach (HandlerFn Handler) const {
41 return Bits.iterate(
42 [](uint32_t Id, void *HandlerPtr) {
43 auto &Handler = *(HandlerFn *)HandlerPtr;
44 if constexpr (std::convertible_to<
45 std::invoke_result_t<HandlerFn &, IdT>, bool>) {
46 if (!std::invoke(Handler, IdT(Id))) {
47 return false;
48 }
49 } else {
50 std::invoke(Handler, IdT(Id));
51 }
52 return true;
53 },
54 &Handler);
55 }

90 bool mergeWithDiff(const SparseBitSet &Other,
91 std::invocable<IdT> auto WithNewElem) {
92 constexpr size_t DiffThreshold = 16;
93 // operator- is expensive, but it is definitely a lot faster than the
94 // foreach loop if Other is large
95
96 if (Other.size() > DiffThreshold) {
97 SparseBitSet Diff = Other - *this;
98 if (Diff.empty()) {
99 return false;
100 }
101
102 *this |= Diff;
103
104 Diff.foreach (copyOrRef(WithNewElem));
105 return true;
106 }
107
108 bool Ret = false;
109 Other.foreach ([&](IdT Elem) {
110 if (tryInsert(Elem)) {
111 std::invoke(WithNewElem, Elem);
112 Ret = true;
113 }
114 });
115 return Ret;
116 }
117
118 bool mergeWithDiff(const SparseBitSet &Other,
119 std::invocable<IdT> auto WithNewElem,
120 SparseBitSet &IntoDiff) {
121 constexpr size_t DiffThreshold = 16;
122 // operator- is expensive, but it is definitely a lot faster than the
123 // foreach loop if Other is large
124
125 if (Other.size() > DiffThreshold) {
126 SparseBitSet Diff = Other - *this;
127 if (Diff.empty()) {
128 return false;
129 }
130
131 *this |= Diff;
132 IntoDiff |= Diff;
133
134 Diff.foreach (copyOrRef(WithNewElem));
135 return true;
136 }
137
138 bool Ret = false;
139 Other.foreach ([&](IdT Elem) {
140 if (tryInsert(Elem)) {
141 IntoDiff.insert(Elem);
142 std::invoke(WithNewElem, Elem);
143 Ret = true;
144 }
145 });
146
147 return Ret;
148 }
