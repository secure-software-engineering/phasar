38/// Opt-in call-string context-sensitivity for \c AndersenOTFSolver.
39///
40/// The call-string k-limit is fixed at 1: a selected function gets one set of
41/// PAG nodes per call-site that reaches it. Non-selected functions keep a
42/// single set of nodes shared by all callers, exactly as before.
43struct ContextSensitivityOptions {
44 enum class Mode : uint8_t {
45 Off, ///< Root context only; identical to the insensitive solver.
46 Manual, ///< Only functions matching \c AllowList.
47 Dynamic, ///< \c AllowList plus functions observed as precision-critical.
48 All, ///< Every function, until \c MaxContextualNodes is reached.
49 };
50
51 Mode SelectionMode = Mode::Off;
52 /// Function-name globs (\c llvm::GlobPattern). \c DenyList wins over
53 /// \c AllowList.
54 std::vector<std::string> AllowList{};
55 std::vector<std::string> DenyList{};
56 /// Hard cap on context-qualified PAG nodes. Once reached, no function is
57 /// newly selected and no already-selected function gets a further context:
58 /// sound, just less precise. This is the knob that bounds run time; solve
59 /// time grows super-linearly in the node count, so raising it is not a
60 /// proportional trade.
61 size_t MaxContextualNodes = 20'000;
62 /// Cap on distinct calling contexts per function; further call sites fall
63 /// back to the shared root context. A selected function costs one clone of
64 /// its whole body per context, so without this a single hot function can
65 /// consume \c MaxContextualNodes on its own.
66 unsigned MaxContextsPerFunction = 32;
67 /// \c Mode::Dynamic only: functions with more LLVM instructions than this
68 /// are never selected. Cloning a large body per context is expensive, and
69 /// large functions are rarely the point where callers merge.
70 unsigned MaxContextualFunctionSize = 256;
71 /// \c Mode::Dynamic only: tighter size limit for the weaker signal where
72 /// the merged parameters never leave the function body. Off by default:
73 /// that signal's only payoff is formal-vs-formal aliasing inside the body,
74 /// which \c buildResult unions back together across contexts anyway.
75 unsigned MaxLocalMergeFunctionSize = 0;
76
77 [[nodiscard]] constexpr bool isOff() const noexcept {
78 return SelectionMode == Mode::Off;
79 }
80};

82/// Alias-analysis result for the Andersen-style OTF points-to analysis.
83///
84/// Two values may-alias iff their points-to sets share at least one abstract
85/// object. Satisfies \c RawAAResult so it can be wrapped by
86/// \c LLVMUnionFindAliasIterator.
87struct AndersenOTFResult {
88 TypedVector<ValueId, RawAliasSet<ValueId>> AliasSets;
89 LLVMBasedCallGraph CG;
90
91 [[nodiscard]] static constexpr bool isCached() noexcept { return true; }
92 [[nodiscard]] constexpr size_t size() const noexcept {
93 return AliasSets.size();
94 }
95
96 [[nodiscard]] RawAliasSet<ValueId>
97 getRawAliasSet(ValueId Var) const noexcept {
98 if (!AliasSets.inbounds(Var)) {
99 return {};
100 }
101 return AliasSets[Var];
102 }
103
104 [[nodiscard]] bool mayAlias(ValueId Var1, ValueId Var2) const noexcept {
105 if (Var1 == Var2) {
106 return true;
107 }
108 if (!AliasSets.inbounds(Var1)) {
109 return false;
110 }
111 return AliasSets[Var1].contains(Var2);
112 }
113};

115static_assert(RawAAResult<AndersenOTFResult>);

117/// Andersen-style inclusion-based points-to analysis that co-refines the call
118/// graph and points-to sets in a single fixpoint.
119///
120/// Unlike the staged pipeline (resolver → PA), this solver owns its own
121/// function-worklist loop: direct calls add callees immediately; indirect
122/// calls are resolved as \c pts(fp) grows.
123///
124/// Context-sensitivity is opt-in via \c ContextSensitivityOptions and off by
125/// default.
126class AndersenOTFSolver {
127public:
128 explicit AndersenOTFSolver(const LLVMProjectIRDB &IRDB PSR_LIFETIMEBOUND,
129 llvm::ArrayRef<const llvm::Function *> Entries
130 PSR_LIFETIMEBOUND,
131 ValueCompressor<PAGVariable> &VC PSR_LIFETIMEBOUND,
132 Soundness S = Soundness::Soundy,
133 ContextSensitivityOptions CSOpts = {}) noexcept;
134
135 /// Run the full OTF fixpoint and return the alias-analysis result.
136 [[nodiscard]] AndersenOTFResult solve();
137
138private:
139 struct SolverData;
140
141 NonNullPtr<const LLVMProjectIRDB> IRDB;
142 llvm::ArrayRef<const llvm::Function *> Entries;
143 NonNullPtr<ValueCompressor<PAGVariable>> VC;
144 Soundness S;
145 ContextSensitivityOptions CSOpts;
146};
